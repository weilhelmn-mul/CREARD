  2533→  const calculateHours = (start: string, end: string): number => {
  2534→    const [sh, sm] = start.split(':').map(Number)
  2535→    const [eh, em] = end.split(':').map(Number)

  2540→  const calculatePriceForTimeSlot = (schedule: PricingScheduleItem[], startTime: string, endTime: string): { total: number; breakdown: Array<{ label: string; hours: number; pricePerHour: number; subtotal: number }> } => {
  2541→    const [startH, startM] = startTime.split(':').map(Number)
  2542→    const [endH, endM] = endTime.split(':').map(Number)
  2543→    const startDecimal = startH + startM / 60
  2544→    const endDecimal = endH + endM / 60
  2545→
  2546→    if (schedule.length === 0) return { total: 0, breakdown: [] }
  2547→
  2548→    const sorted = [...schedule].sort((a, b) => a.startHour - b.startHour)
  2549→    let total = 0
  2550→    let cursor = startDecimal
  2551→    const breakdown: Array<{ label: string; hours: number; pricePerHour: number; subtotal: number }> = []
  2552→
  2553→    for (const slot of sorted) {
  2554→      if (cursor >= slot.endHour) continue
  2555→      const overlapStart = Math.max(cursor, slot.startHour)
  2556→      const overlapEnd = Math.min(endDecimal, slot.endHour)
  2557→      if (overlapEnd > overlapStart) {
  2558→        const hours = overlapEnd - overlapStart
  2559→        const subtotal = Math.round(hours * slot.pricePerHour * 100) / 100
  2560→        breakdown.push({ label: slot.label, hours: Math.round(hours * 100) / 100, pricePerHour: slot.pricePerHour, subtotal })
  2561→        total += subtotal
  2562→        cursor = overlapEnd
  2563→      }
  2564→    }
  2565→    return { total: Math.round(total * 100) / 100, breakdown }
  2566→  }
