   111→function formatDateISO(date: Date): string {
   112→  const y = date.getFullYear()
   113→  const m = String(date.getMonth() + 1).padStart(2, '0')
   114→  const d = String(date.getDate()).padStart(2, '0')
   115→  return `${y}-${m}-${d}`
   116→}

   137→function isToday(date: Date): boolean {
   138→  const today = new Date()

   140→    date.getMonth() === today.getMonth() &&
   141→    date.getDate() === today.getDate()

   148→/** Minimum minutes of advance required to book a slot */
   149→const MIN_ADVANCE_MINUTES = 30
   150→
   151→/** Returns the latest hour that is still within the 30-min restriction.
   152→ *  For example, if it's 14:35, hours 14 and earlier are too soon.
   153→ *  Returns -1 if no hours are restricted (i.e. more than 30 min until SLOT_START). */
   154→function getRestrictedHour(): number {
   155→  const now = new Date()
   156→  const curH = now.getHours()
   157→  const curM = now.getMinutes()
   158→  // If current minute >= 30, then the current hour + 1 is still within 30 min
   159→  // e.g. 14:45 → 15:00 is in 15 min → 15 is restricted too
   160→  // e.g. 14:20 → 15:00 is in 40 min → 15 is NOT restricted, but 14 IS restricted
   161→  if (curM >= 30) {
   162→    return curH + 1
   163→  }
   164→  return curH
   165→}

   167→function getAdminSlotInfo(
   168→  hour: number,
   169→  bookings: BookingInfo[],
   170→  isTodayDate: boolean,
   171→  currentHour: number
   172→): SlotInfo {
   173→  const booking = bookings.find((b) => {
   174→    const sh = parseInt(b.startTime.split(':')[0], 10)
   175→    const eh = parseInt(b.endTime.split(':')[0], 10)
   176→    return hour >= sh && hour < eh
   177→  })
   178→
   179→  if (!booking) {
   180→    // Admin: NO time restrictions — all unbooked slots are available (including past hours)
   181→    // This allows admin to register walk-in bookings for in-progress or past slots
   182→    return { status: 'available', booking: null }
   183→  }
   184→
   185→  if (booking.status === 'cancelled') return { status: 'blocked', booking }
   186→  if (booking.status === 'completed') return { status: 'completed', booking }
   187→
   188→  // Active: reserved
   189→  if (isTodayDate && hour === currentHour) return { status: 'in_play', booking }
   190→  return { status: 'reserved', booking }
   191→}

   247→function normalizeSchedule(schedule: Court['pricingSchedule']): PricingScheduleItem[] {
   248→  if (!schedule) return []
   249→  if (Array.isArray(schedule)) return schedule
   250→  // Legacy { morning, night } object format
   251→  const { morning, night } = schedule as PricingSchedule
   252→  return [
   253→    { label: 'Mañana', startHour: morning.startHour, endHour: morning.endHour, pricePerHour: morning.price },
   254→    { label: 'Noche', startHour: night.startHour, endHour: night.endHour, pricePerHour: night.price },
   255→  ]
   256→}
   257→
   258→function getMinPrice(court: Court): number {
   259→  const items = normalizeSchedule(court.pricingSchedule)
   260→  if (items.length > 0) {
   261→    return Math.min(...items.map(s => s.pricePerHour))
   262→  }
   263→  return court.pricePerHour
   264→}
   265→
   266→function getPriceForHour(court: Court, hour: number): number {
   267→  const items = normalizeSchedule(court.pricingSchedule)
   268→  for (const slot of items) {
   269→    if (hour >= slot.startHour && hour < slot.endHour) return slot.pricePerHour
   270→  }
   271→  return court.pricePerHour
   272→}
