    81→/* ─── helpers ─── */
    82→// Parse date string as local date (avoids UTC timezone issues for Peru, UTC-5)
    83→function parseLocalDate(dateStr: string): Date {
    84→  const parts = dateStr.split('-')
    85→  if (parts.length === 3) {
    86→    return new Date(parseInt(parts[0], 10), parseInt(parts[1], 10) - 1, parseInt(parts[2], 10))
    87→  }
    88→  return new Date(dateStr + 'T00:00:00')
    89→}
    90→
    91→const fmt = (dateStr: string) => {
    92→  const d = parseLocalDate(dateStr)
    93→  return d.toLocaleDateString('es-PE', { day: 'numeric', month: 'short', year: 'numeric' })
    94→}
