    63→
    64→type ClaimType = 'queja' | 'reclamo'
    65→type ClaimStatus = 'received' | 'in_process' | 'responded' | 'closed' | 'archived'
    66→
    67→interface Claim {
    68→  id: string
    69→  claimNumber: string
    70→  type: ClaimType
    71→  status: ClaimStatus
    72→  consumerName: string
    73→  docType: string
    74→  docNumber: string
    75→  address: string
    76→  phone: string
    77→  email: string
    78→  description: string
    79→  request: string
    80→  bookingId: string | null
    81→  bookingInfo: {
    82→    courtName: string
    83→    date: string
    84→    time: string
    85→    amount: number
    86→    paymentMethod: string
    87→  } | null
    88→  attachments: { name: string; size: number; type: string; base64: string }[]
    89→  providerResponse: string | null
    90→  qrCode: string | null
    91→  pdfUrl: string | null
    92→  deadlineDate: string | null
    93→  createdAt: string
    94→  updatedAt: string
    95→  statusHistory: { status: ClaimStatus; date: string; note?: string }[]
    96→}
    97→
    98→/* ═══════════════════════════════════════════
    99→   Constants
   100→   ═══════════════════════════════════════════ */
   101→
   102→const ITEMS_PER_PAGE = 10
   103→
   104→const statusConfig: Record<ClaimStatus, { label: string; color: string }> = {
   105→  received:   { label: 'Recibido',     color: 'bg-blue-500/20 text-blue-400 border-blue-500/30' },
   106→  in_process: { label: 'En Proceso',  color: 'bg-amber-500/20 text-amber-400 border-amber-500/30' },
   107→  responded:  { label: 'Respondido',  color: 'bg-green-500/20 text-green-400 border-green-500/30' },
   108→  closed:     { label: 'Atendido',    color: 'bg-emerald-500/20 text-emerald-400 border-emerald-500/30' },
   109→  archived:   { label: 'Archivado',   color: 'bg-gray-500/20 text-gray-400 border-gray-500/30' },
   110→}
   111→
   112→const typeConfig: Record<ClaimType, { label: string; color: string }> = {
   113→  queja:  { label: 'Queja',  color: 'bg-orange-500/20 text-orange-400 border-orange-500/30' },
   114→  reclamo: { label: 'Reclamo', color: 'bg-red-500/20 text-red-400 border-red-500/30' },
   115→}
   116→
   117→const statusLabel: Record<ClaimStatus, string> = {
   118→  received: 'Recibido',
   119→  in_process: 'En Proceso',
   120→  responded: 'Respondido',
   121→  closed: 'Atendido',
   122→  archived: 'Archivado',
   123→}
   124→
   125→function daysUntilDeadline(deadline: string | null): number {
   126→  if (!deadline) return Infinity
   127→  const diff = new Date(deadline).getTime() - Date.now()
   128→  return Math.ceil(diff / (1000 * 60 * 60 * 24))
   129→}
   130→
   131→function formatDate(iso: string): string {
   132→  return new Date(iso).toLocaleDateString('es-PE', {
   133→    day: '2-digit',
   134→    month: 'short',
   135→    year: 'numeric',
   136→  })
   137→}
   138→
   139→function formatDateTime(iso: string): string {
   140→  return new Date(iso).toLocaleString('es-PE', {
   141→    day: '2-digit',
   142→    month: 'short',
   143→    year: 'numeric',
   144→    hour: '2-digit',
   145→    minute: '2-digit',
   146→  })
   147→}

   149→/* ═══════════════════════════════════════════
   150→   Component

   152→
   153→export default function ClaimsAdminTab() {
   154→  /* state */
   155→  const [claims, setClaims] = useState<Claim[]>([])
   156→  const [loading, setLoading] = useState(true)
   157→  const [search, setSearch] = useState('')
   158→  const [filterStatus, setFilterStatus] = useState<string>('all')
   159→  const [filterType, setFilterType] = useState<string>('all')
   160→  const [page, setPage] = useState(1)
   161→
   162→  /* dialogs */
   163→  const [detailClaim, setDetailClaim] = useState<Claim | null>(null)
   164→  const [responseClaim, setResponseClaim] = useState<Claim | null>(null)
   165→  const [responseText, setResponseText] = useState('')
   166→  const [submitting, setSubmitting] = useState(false)
   167→
   168→  /* fetch */
   169→  const fetchClaims = useCallback(async () => {
   170→    try {
   171→      setLoading(true)
   172→      const res = await fetch('/api/claims', {
   173→        headers: { ...getAuthHeaders(), 'Content-Type': 'application/json' },
   174→      })
   175→      if (!res.ok) throw new Error('Error al cargar reclamos')
   176→      const data = await res.json()
   177→      setClaims(Array.isArray(data) ? data : data.claims ?? [])
   178→    } catch (err) {
   179→      console.error(err)
   180→      toast({ title: 'Error', description: 'No se pudieron cargar los reclamos', variant: 'destructive' })
   181→    } finally {
   182→      setLoading(false)
   183→    }
   184→  }, [])
   185→
   186→  useEffect(() => { fetchClaims() }, [fetchClaims])
   187→
   188→  /* filtered data */
   189→  const filtered = useMemo(() => {
   190→    let result = claims
   191→    if (filterStatus !== 'all') result = result.filter(c => c.status === filterStatus)
   192→    if (filterType !== 'all') result = result.filter(c => c.type === filterType)
   193→    if (search.trim()) {
   194→      const q = search.toLowerCase()
   195→      result = result.filter(
   196→        c =>
   197→          c.claimNumber.toLowerCase().includes(q) ||
   198→          c.consumerName.toLowerCase().includes(q) ||
   199→          c.description.toLowerCase().includes(q) ||
   200→          c.email.toLowerCase().includes(q)
   201→      )
   202→    }
   203→    return result
   204→  }, [claims, filterStatus, filterType, search])
   205→
   206→  /* pagination */
   207→  const totalPages = Math.max(1, Math.ceil(filtered.length / ITEMS_PER_PAGE))
   208→  const paginated = filtered.slice((page - 1) * ITEMS_PER_PAGE, page * ITEMS_PER_PAGE)
   209→
   210→  useEffect(() => { setPage(1) }, [filterStatus, filterType, search])
   211→
   212→  /* KPIs */
   213→  const kpis = useMemo(() => {
   214→    const total = claims.length
   215→    const inProcess = claims.filter(c => c.status === 'in_process' || c.status === 'received').length
   216→    const expiring = claims.filter(c => {
   217→      if (c.status === 'closed' || c.status === 'archived') return false
   218→      return daysUntilDeadline(c.deadlineDate) <= 3 && daysUntilDeadline(c.deadlineDate) >= 0
   219→    }).length
   220→    const attended = claims.filter(c => c.status === 'closed').length
   221→    return { total, inProcess, expiring, attended }
   222→  }, [claims])
   223→
   224→  /* actions */
   225→  const handleStatusChange = async (claimId: string, newStatus: ClaimStatus) => {
   226→    try {
   227→      const res = await fetch('/api/claims', {
   228→        method: 'PUT',
   229→        headers: { ...getAuthHeaders(), 'Content-Type': 'application/json' },
   230→        body: JSON.stringify({ claimId, action: 'update_status', status: newStatus }),
   231→      })
   232→      if (!res.ok) throw new Error()
   233→      toast({ title: 'Estado actualizado', description: 'Reclamo \u2192 ' + statusLabel[newStatus] })
   234→      fetchClaims()
   235→    } catch {
   236→      toast({ title: 'Error', description: 'No se pudo cambiar el estado', variant: 'destructive' })
   237→    }
   238→  }
   239→
   240→  const handleArchive = async (claimId: string) => {
   241→    await handleStatusChange(claimId, 'archived')
   242→    setDetailClaim(null)
   243→  }
   244→
   245→  const openResponse = (claim: Claim) => {
   246→    setResponseClaim(claim)
   247→    setResponseText(claim.providerResponse ?? '')
   248→  }
   249→
   250→  const submitResponse = async () => {
   251→    if (!responseClaim || !responseText.trim()) return
   252→    try {
   253→      setSubmitting(true)
   254→      const res = await fetch('/api/claims', {
   255→        method: 'PUT',
   256→        headers: { ...getAuthHeaders(), 'Content-Type': 'application/json' },
   257→        body: JSON.stringify({
   258→          claimId: responseClaim.id,
   259→          action: 'respond',
   260→          response: responseText.trim(),
   261→        }),
   262→      })
   263→      if (!res.ok) throw new Error()
   264→      toast({ title: 'Respuesta enviada', description: 'Se registr\u00f3 la respuesta del reclamo' })
   265→      setResponseClaim(null)
   266→      setResponseText('')
   267→      fetchClaims()
   268→    } catch {
   269→      toast({ title: 'Error', description: 'No se pudo enviar la respuesta', variant: 'destructive' })
   270→    } finally {
   271→      setSubmitting(false)
   272→    }
   273→  }
   274→
   275→  /* export */
   276→  const exportCSV = () => {
   277→    const header = 'N\u00b0 Reclamo,Tipo,Consumidor,Documento,Estado,Fecha,Plazo,Descripci\u00f3n\n'
   278→    const rows = filtered.map(c =>
   279→      [c.claimNumber, c.type, c.consumerName, c.docType + ' ' + c.docNumber, statusLabel[c.status], formatDate(c.createdAt), c.deadlineDate ?? '', '"' + c.description.replace(/"/g, '""') + '"'].join(',')
   280→    ).join('\n')
   281→    downloadBlob(header + rows, 'reclamos.csv', 'text/csv')
   282→  }
   283→
   284→  const exportExcel = () => {
   285→    const header = '<html xmlns:o="urn:schemas-microsoft-com:office:office" xmlns:x="urn:schemas-microsoft-com:office:excel"><head><meta charset="utf-8"></head><body><table>'
   286→    const th = '<tr><th>N\u00b0 Reclamo</th><th>Tipo</th><th>Consumidor</th><th>Documento</th><th>Estado</th><th>Fecha</th><th>Plazo</th></tr>'
   287→    const rows = filtered.map(c =>
   288→      '<tr><td>' + c.claimNumber + '</td><td>' + typeConfig[c.type].label + '</td><td>' + c.consumerName + '</td><td>' + c.docType + ' ' + c.docNumber + '</td><td>' + statusLabel[c.status] + '</td><td>' + formatDate(c.createdAt) + '</td><td>' + (c.deadlineDate ? formatDate(c.deadlineDate) : '') + '</td></tr>'
   289→    ).join('')
   290→    downloadBlob(header + th + rows + '</table></body></html>', 'reclamos.xls', 'application/vnd.ms-excel')
   291→  }
   292→
   293→  const downloadBlob = (content: string, filename: string, mime: string) => {
   294→    const blob = new Blob([content], { type: mime + ';charset=utf-8;' })
   295→    const url = URL.createObjectURL(blob)
   296→    const a = document.createElement('a')
   297→    a.href = url
   298→    a.download = filename
   299→    a.click()
   300→    URL.revokeObjectURL(url)
   301→  }
   302→
   303→  /* KPI cards data */
   304→  const kpCards = [
   305→    { label: 'Total Reclamos', value: kpis.total, icon: FileText, color: 'text-[#00ff41]' },
   306→    { label: 'En Proceso', value: kpis.inProcess, icon: Clock, color: 'text-amber-400' },
   307→    { label: 'Por Vencer', value: kpis.expiring, icon: AlertTriangle, color: 'text-red-400' },
   308→    { label: 'Atendidos', value: kpis.attended, icon: CheckCircle2, color: 'text-green-400' },
   309→  ]
   310→
   311→  /* ═══════════════════════════════════════════
   312→     Render
   313→     ═══════════════════════════════════════════ */
   314→  return (
   315→    <motion.div
   316→      initial={{ opacity: 0, y: 10 }}
   317→      animate={{ opacity: 1, y: 0 }}
   318→      transition={{ duration: 0.2 }}
   319→      className="space-y-5"
   320→    >
   321→      {/* KPI Row */}
   322→      <div className="grid grid-cols-2 lg:grid-cols-4 gap-3">
   323→        {kpCards.map(kp => (
   324→          <div
   325→            key={kp.label}
   326→            className="bg-[#141e12] border border-[#2a3a25] rounded-xl p-4 flex items-center gap-3"
   327→          >
   328→            <div className="p-2.5 rounded-lg bg-[#00ff41]/10">
   329→              <kp.icon className={'w-5 h-5 ' + kp.color} />
   330→            </div>
   331→            <div>
   332→              <p className="text-[11px] uppercase tracking-wider text-[#dae6d2]/50 font-medium">{kp.label}</p>
   333→              <p className="text-2xl font-bold text-[#dae6d2] mt-0.5">{kp.value}</p>
   334→            </div>
   335→          </div>
   336→        ))}
   337→      </div>
   338→
   339→      {/* Filters */}
   340→      <div className="bg-[#141e12] border border-[#2a3a25] rounded-xl p-4">
   341→        <div className="flex flex-col sm:flex-row gap-3 items-start sm:items-center">
   342→          <div className="relative flex-1 w-full">
   343→            <Search className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-[#dae6d2]/40" />
   344→            <Input
   345→              placeholder="Buscar por N\u00b0, nombre, descripci\u00f3n..."
   346→              value={search}
   347→              onChange={e => setSearch(e.target.value)}
   348→              className="pl-9 bg-[#1a2a16] border-[#2a3a25] text-[#dae6d2] placeholder:text-[#dae6d2]/40 h-9 text-sm focus-visible:ring-[#00ff41]/30"
   349→            />
   350→          </div>
   351→
   352→          <Select value={filterStatus} onValueChange={setFilterStatus}>
   353→            <SelectTrigger className="w-[150px] bg-[#1a2a16] border-[#2a3a25] text-[#dae6d2] h-9 text-sm">
   354→              <Filter className="w-4 h-4 mr-1.5 text-[#dae6d2]/50" />
   355→              <SelectValue placeholder="Estado" />
   356→            </SelectTrigger>
   357→            <SelectContent className="bg-[#141e12] border-[#2a3a25]">
   358→              <SelectItem value="all">Todos</SelectItem>
   359→              <SelectItem value="received">Recibido</SelectItem>
   360→              <SelectItem value="in_process">En Proceso</SelectItem>
   361→              <SelectItem value="responded">Respondido</SelectItem>
   362→              <SelectItem value="closed">Atendido</SelectItem>
   363→              <SelectItem value="archived">Archivado</SelectItem>
   364→            </SelectContent>
   365→          </Select>
   366→
   367→          <Select value={filterType} onValueChange={setFilterType}>
   368→            <SelectTrigger className="w-[140px] bg-[#1a2a16] border-[#2a3a25] text-[#dae6d2] h-9 text-sm">
   369→              <SelectValue placeholder="Tipo" />
   370→            </SelectTrigger>
   371→            <SelectContent className="bg-[#141e12] border-[#2a3a25]">
   372→              <SelectItem value="all">Todos</SelectItem>
   373→              <SelectItem value="queja">Queja</SelectItem>
   374→              <SelectItem value="reclamo">Reclamo</SelectItem>
   375→            </SelectContent>
   376→          </Select>
   377→
   378→          <div className="flex gap-2">
   379→            <Button variant="outline" size="sm" onClick={exportCSV}
   380→              className="bg-[#1a2a16] border-[#2a3a25] text-[#dae6d2] hover:bg-[#2a3a25] hover:text-[#dae6d2] h-9 gap-1.5">
   381→              <Download className="w-3.5 h-3.5" /> CSV
   382→            </Button>
   383→            <Button variant="outline" size="sm" onClick={exportExcel}
   384→              className="bg-[#1a2a16] border-[#2a3a25] text-[#dae6d2] hover:bg-[#2a3a25] hover:text-[#dae6d2] h-9 gap-1.5">
   385→              <FileSpreadsheet className="w-3.5 h-3.5" /> Excel
   386→            </Button>
   387→          </div>
   388→        </div>
   389→      </div>
   390→
   391→      {/* Table */}
   392→      <div className="bg-[#141e12] border border-[#2a3a25] rounded-xl overflow-hidden">
   393→        {loading ? (
   394→          <div className="flex items-center justify-center py-16">
   395→            <Loader2 className="w-6 h-6 text-[#00ff41] animate-spin" />
   396→          </div>
   397→        ) : filtered.length === 0 ? (
   398→          <div className="flex flex-col items-center justify-center py-16 text-[#dae6d2]/50">
   399→            <Inbox className="w-12 h-12 mb-3 opacity-40" />
   400→            <p className="text-sm font-medium">No se encontraron reclamos</p>
   401→            <p className="text-xs mt-1">Ajusta los filtros o espera nuevas solicitudes</p>
   402→          </div>
   403→        ) : (
   404→          <>
   405→            <div className="overflow-x-auto">
   406→              <Table>
   407→                <TableHeader>
   408→                  <TableRow className="border-[#2a3a25] hover:bg-transparent">
   409→                    <TableHead className="text-[#dae6d2]/60 text-xs font-semibold uppercase tracking-wider">N Reclamo</TableHead>
   410→                    <TableHead className="text-[#dae6d2]/60 text-xs font-semibold uppercase tracking-wider">Tipo</TableHead>
   411→                    <TableHead className="text-[#dae6d2]/60 text-xs font-semibold uppercase tracking-wider">Consumidor</TableHead>
   412→                    <TableHead className="text-[#dae6d2]/60 text-xs font-semibold uppercase tracking-wider">Estado</TableHead>
   413→                    <TableHead className="text-[#dae6d2]/60 text-xs font-semibold uppercase tracking-wider">Fecha</TableHead>
   414→                    <TableHead className="text-[#dae6d2]/60 text-xs font-semibold uppercase tracking-wider">Plazo</TableHead>
   415→                    <TableHead className="text-[#dae6d2]/60 text-xs font-semibold uppercase tracking-wider text-right">Acciones</TableHead>
   416→                  </TableRow>
   417→                </TableHeader>
   418→                <TableBody>
   419→                  {paginated.map(claim => {
   420→                    const days = daysUntilDeadline(claim.deadlineDate)
   421→                    const isUrgent = days <= 3 && days >= 0 && claim.status !== 'closed' && claim.status !== 'archived'
   422→                    return (
   423→                      <TableRow key={claim.id} className="border-[#2a3a25] hover:bg-[#00ff41]/5 transition-colors">
   424→                        <TableCell className="text-[#dae6d2] font-mono text-sm font-medium">{claim.claimNumber}</TableCell>
   425→                        <TableCell>
   426→                          <Badge variant="outline" className={'text-[10px] font-semibold border ' + typeConfig[claim.type].color + ' px-2 py-0.5'}>
   427→                            {typeConfig[claim.type].label}
   428→                          </Badge>
   429→                        </TableCell>
   430→                        <TableCell>
   431→                          <div className="text-[#dae6d2] text-sm font-medium">{claim.consumerName}</div>
   432→                          <div className="text-[#dae6d2]/50 text-xs">{claim.email}</div>
   433→                        </TableCell>
   434→                        <TableCell>
   435→                          <Badge variant="outline" className={'text-[10px] font-semibold border ' + statusConfig[claim.status].color + ' px-2 py-0.5'}>
   436→                            {statusConfig[claim.status].label}
   437→                          </Badge>
   438→                        </TableCell>
   439→                        <TableCell className="text-[#dae6d2]/70 text-sm">{formatDate(claim.createdAt)}</TableCell>
   440→                        <TableCell>
   441→                          {claim.deadlineDate ? (
   442→                            <span className={'text-sm font-medium ' + (isUrgent ? 'text-red-400' : 'text-[#dae6d2]/70')}>
   443→                              {isUrgent && <AlertTriangle className="w-3 h-3 inline mr-1" />}
   444→                              {formatDate(claim.deadlineDate)}
   445→                              {isUrgent && <span className="ml-1 text-[10px]">{days}d</span>}
   446→                            </span>
   447→                          ) : (
   448→                            <span className="text-[#dae6d2]/40 text-sm">--</span>
   449→                          )}
   450→                        </TableCell>
   451→                        <TableCell>
   452→                          <div className="flex items-center justify-end gap-1">
   453→                            <Button variant="ghost" size="sm" onClick={() => setDetailClaim(claim)}
   454→                              className="h-8 w-8 p-0 text-[#dae6d2]/60 hover:text-[#00ff41] hover:bg-[#00ff41]/10">
   455→                              <Eye className="w-4 h-4" />
   456→                            </Button>
   457→                            {claim.status !== 'closed' && claim.status !== 'archived' && (
   458→                              <Button variant="ghost" size="sm" onClick={() => openResponse(claim)}
   459→                                className="h-8 w-8 p-0 text-[#dae6d2]/60 hover:text-[#00ff41] hover:bg-[#00ff41]/10">
   460→                                <MessageSquare className="w-4 h-4" />
   461→                              </Button>
   462→                            )}
   463→                            {claim.status !== 'archived' && (
   464→                              <Button variant="ghost" size="sm" onClick={() => handleArchive(claim.id)}
   465→                                className="h-8 w-8 p-0 text-[#dae6d2]/60 hover:text-amber-400 hover:bg-amber-400/10">
   466→                                <Archive className="w-4 h-4" />
   467→                              </Button>
   468→                            )}
   469→                          </div>
   470→                        </TableCell>
   471→                      </TableRow>
   472→                    )
   473→                  })}
   474→                </TableBody>
   475→              </Table>
   476→            </div>
   477→
   478→            {/* Pagination */}
