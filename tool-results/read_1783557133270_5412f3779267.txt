    16→// Migrate old status values to the new 3-status system
    17→function migrateStatus(s: string): string {
    18→  switch (s) {
    19→    case 'confirmed':
    20→    case 'pending':
    21→    case 'partially_paid':
    22→      return 'reserved';
    23→    case 'fully_paid':
    24→    case 'completed':
    25→      return 'completed';
    26→    case 'no_show':
    27→    case 'expired':
    28→      return 'cancelled';
    29→    default:
    30→      return s; // 'reserved', 'completed', 'cancelled' pass through
    31→  }
    32→}

    34→// Transformar snake_case (Firestore) a camelCase (frontend)
    35→function toCamelBooking(b: Record<string, unknown>) {
    36→  // Primary court (backward compat)
    37→  const courtRaw = b._court as Record<string, unknown> | null;
    38→  const court = courtRaw ? {
    39→    id: courtRaw.id || courtRaw.court_id,
    40→    name: courtRaw.name || courtRaw.court_name,
    41→    sport: courtRaw.sport,
    42→    branch: (courtRaw.branch as Record<string, unknown>) || {
    43→      id: courtRaw.branch_id || 'branch-1',
    44→      name: 'CREARD',
    45→    },
    46→  } : null;
    47→
    48→  // Additional courts (multi-court bookings)
    49→  const courtsRaw = b._courts as Array<Record<string, unknown>> | null;
    50→  const courts = courtsRaw ? courtsRaw.map((cr) => ({
    51→    id: cr.id || cr.court_id,
    52→    name: cr.name || cr.court_name,
    53→    sport: cr.sport,
    54→    branch: (cr.branch as Record<string, unknown>) || { id: 'branch-1', name: 'CREARD' },
    55→  })) : [];
    56→
    57→  const userRaw = b._user as Record<string, unknown> | null;
    58→  const user = userRaw ? {
    59→    id: userRaw.id,
    60→    name: userRaw.name,
    61→    email: userRaw.email,
    62→    phone: userRaw.phone,
    63→  } : null;
    64→
    65→  // Court IDs array
    66→  const courtIds: string[] = Array.isArray(b.court_ids) ? b.court_ids : (b.court_id ? [b.court_id] : []);
    67→
    68→  // Equipment items
    69→  const equipmentItems = Array.isArray(b.equipment_items) ? b.equipment_items.map((eq: Record<string, unknown>) => ({
    70→    equipmentId: eq.equipment_id || eq.id,
    71→    name: eq.name,
    72→    quantity: eq.quantity || 1,
    73→    unitPrice: eq.unit_price || eq.price_per_rental || 0,
    74→    subtotal: (eq.quantity || 1) * (eq.unit_price || eq.price_per_rental || 0),
    75→  })) : [];
    76→
    77→  // Normalize: recalc equipment subtotal from items (more reliable than stored value)
    78→  const computedEquipSubtotal = equipmentItems.reduce((s: number, eq: { subtotal: number }) => s + eq.subtotal, 0);
    79→  const equipmentSubtotal = computedEquipSubtotal;
    80→
    81→  // Normalize court_subtotal: if stored value is wrong (old double-counting bug),
    82→  // derive it from total_price - equipment_subtotal
    83→  const totalPrice = b.total_price || 0;
    84→  const storedCourtSubtotal = b.court_subtotal || 0;
    85→  const courtSubtotal = (storedCourtSubtotal + equipmentSubtotal > totalPrice + 0.01)
    86→    ? Math.max(0, Math.round((totalPrice - equipmentSubtotal) * 100) / 100)
    87→    : storedCourtSubtotal;
    88→
    89→  // Normalize remaining_amount: ensure it equals total - advance
    90→  const advanceAmount = b.advance_amount || 0;
    91→  const storedRemaining = b.remaining_amount || 0;
    92→  const remainingAmount = (Math.abs(storedRemaining - (totalPrice - advanceAmount)) > 0.01)
    93→    ? Math.max(0, Math.round((totalPrice - advanceAmount) * 100) / 100)
    94→    : storedRemaining;
    95→
    96→  return {
    97→    id: b.id,
    98→    courtId: b.court_id,
    99→    courtIds,
   100→    userId: b.user_id,
   101→    date: typeof b.date === 'string' ? b.date : (b.date ? String(b.date) : ''),
   102→    startTime: b.start_time,
   103→    endTime: b.end_time,
   104→    totalPrice,
   105→    courtSubtotal,
   106→    equipmentSubtotal,
   107→    equipmentItems,
   108→    advanceAmount,
   109→    remainingAmount,
   110→    status: migrateStatus(b.status || 'reserved'),
   111→    slotStatus: b.slot_status,
   112→    paymentMethod: b.payment_method,
   113→    notes: b.notes,
   114→    createdAt: b.created_at,
   115→    updatedAt: b.updated_at,
   116→    recurringGroupId: b.recurring_group_id,
   117→    recurringIndex: b.recurring_index,
   118→    equipmentDelivered: b.equipment_delivered || false,
   119→    equipmentReturned: b.equipment_returned || false,
   120→    court,
   121→    courts,
   122→    user,
   123→  };
   124→}

   126→// Enrich a single booking (all courts + user), safe — never throws
   127→async function safeEnrichBooking(b: Record<string, unknown>): Promise<Record<string, unknown>> {
   128→  let court: Record<string, unknown> | null = null;
   129→  let courts: Array<Record<string, unknown>> = [];
   130→  let user: Record<string, unknown> | null = null;
   131→
   132→  // Get all court IDs (from court_ids array or single court_id)
   133→  const allCourtIds: string[] = Array.isArray(b.court_ids)
   134→    ? b.court_ids as string[]
   135→    : (b.court_id ? [b.court_id as string] : []);
   136→
   137→  // Enrich primary court (first one)
   138→  try {
   139→    if (allCourtIds.length > 0) {
   140→      const c = await getCourtById(allCourtIds[0]);
   141→      if (c) court = c as Record<string, unknown>;
   142→    }
   143→  } catch (e) {
   144→    console.warn('[BOOKINGS] Failed to load primary court for booking:', b.id, e);
   145→  }
   146→
   147→  // Enrich all courts for multi-court bookings
   148→  if (allCourtIds.length > 1) {
   149→    const enrichedCourts: Array<Record<string, unknown>> = [];
   150→    for (const cid of allCourtIds) {
   151→      try {
   152→        const c = await getCourtById(cid);
   153→        if (c) enrichedCourts.push(c as Record<string, unknown>);
   154→      } catch { /* skip failed court enrichment */ }
   155→    }
   156→    courts = enrichedCourts;
   157→  }
   158→
   159→  try {
   160→    if (b.user_id) {
   161→      const u = await getUserById(b.user_id as string);
   162→      if (u) user = u as Record<string, unknown>;
   163→    }
   164→  } catch (e) {
   165→    console.warn('[BOOKINGS] Failed to load user for booking:', b.id, e);
   166→  }
   167→
   168→  return { ...b, _court: court, _courts: courts, _user: user };
   169→}

   171→// Search bookings by userId OR userEmail, deduplicate
   172→async function searchBookingsForUser(userId: string, userEmail: string | null) {
   173→  const results = new Map<string, Record<string, unknown>>();
   174→
   175→  // Always search by userId
   176→  try {
   177→    const byId = await getBookings({ userId });
   178→    for (const b of byId) {
   179→      if (b.id && !results.has(b.id)) {
   180→        results.set(b.id, b as Record<string, unknown>);
   181→      }
   182→    }
   183→  } catch (e) {
   184→    console.error('[BOOKINGS] Error searching by userId:', userId, e);
   185→  }
   186→
   187→  // Also search by email to handle demo/Firebase ID mismatches
   188→  if (userEmail && userEmail !== userId) {
   189→    try {
   190→      const byEmail = await getBookings({ userEmail });
   191→      for (const b of byEmail) {
   192→        if (b.id && !results.has(b.id)) {
   193→          results.set(b.id, b as Record<string, unknown>);
   194→        }
   195→      }
   196→    } catch (e) {
   197→      console.error('[BOOKINGS] Error searching by userEmail:', userEmail, e);
   198→    }
   199→  }
   200→
   201→  return Array.from(results.values());
   202→}

   204→/**
   205→ * Calculate the total price for a time slot using a pricing schedule.
   206→ * Supports slots that span multiple schedules (e.g., 16:00-19:00 = 1hr morning + 2hr night).
   207→ */
   208→function calculatePriceForTimeSlot(
   209→  schedule: Array<{ label: string; startHour: number; endHour: number; pricePerHour: number }>,
   210→  startTime: string,
   211→  endTime: string,
   212→): number {
   213→  const [startH, startM] = startTime.split(':').map(Number);
   214→  const [endH, endM] = endTime.split(':').map(Number);
   215→  const startDecimal = startH + startM / 60;
   216→  const endDecimal = endH + endM / 60;
   217→
   218→  // Sort schedule by startHour
   219→  const sorted = [...schedule].sort((a, b) => a.startHour - b.startHour);
   220→
   221→  let total = 0;
   222→  let cursor = startDecimal;
   223→
   224→  for (const slot of sorted) {
   225→    const slotStart = slot.startHour;
   226→    const slotEnd = slot.endHour;
   227→
   228→    if (cursor >= slotEnd) continue; // cursor already past this slot
   229→
   230→    // Overlap start/end
   231→    const overlapStart = Math.max(cursor, slotStart);
   232→    const overlapEnd = Math.min(endDecimal, slotEnd);
   233→
   234→    if (overlapEnd > overlapStart) {
   235→      const hours = overlapEnd - overlapStart;
   236→      total += hours * slot.pricePerHour;
   237→      cursor = overlapEnd;
   238→    }
   239→  }
   240→
   241→  // If cursor didn't reach endDecimal, the remaining time is uncovered by schedule
   242→  // (fall back to 0 or caller's default)
   243→  return Math.round(total * 100) / 100;
   244→}

   246→export async function GET(request: NextRequest) {
   247→  const startTime = Date.now();
   248→  const firebaseOk = isFirebaseAvailable();
   249→
   250→  try {
   251→    const { searchParams } = new URL(request.url);
   252→    const courtId = searchParams.get('courtId');
   253→    const userId = searchParams.get('userId');
   254→    const date = searchParams.get('date');
   255→    const dateFrom = searchParams.get('dateFrom');
   256→    const dateTo = searchParams.get('dateTo');
   257→    const status = searchParams.get('status');
   258→
   259→    // Court availability check (courtId + date) is public — no auth required
   260→    if (courtId && date && !userId) {
   261→      if (!firebaseOk) {
   262→        return NextResponse.json({ error: 'Firebase no configurado', code: 'NO_FIREBASE' }, { status: 503 });
   263→      }
   264→      const bookings = await getBookings({ courtId, date });
   265→      return NextResponse.json(bookings.map(toCamelBooking));
   266→    }
   267→
   268→    // All other queries require authentication
   269→    const authResult = await requireAnyAuth(request);
   270→    if (authResult instanceof NextResponse) return authResult;
   271→    const authUser = authResult.user;
   272→
   273→    if (!firebaseOk) {
   274→      return NextResponse.json({ error: 'Firebase no configurado', code: 'NO_FIREBASE' }, { status: 503 });
   275→    }
   276→
   277→    let bookings: Record<string, unknown>[];
   278→
   279→    // Non-admin users: only their own bookings
   280→    if (authUser.role !== 'admin' && authUser.role !== 'super_admin') {
   281→      if (userId && userId !== authUser.id) {
   282→        return NextResponse.json({ error: 'No puedes ver reservas de otros usuarios.' }, { status: 403 });
   283→      }
   284→      const effectiveUserId = userId || authUser.id;
   285→      bookings = await searchBookingsForUser(effectiveUserId, authUser.email);
   286→    } else {
   287→      // Admin: fetch with optional filters (date range for performance)
   288→      bookings = (await getBookings({
   289→        courtId: courtId || undefined,
   290→        userId: userId || undefined,
   291→        date: date || undefined,
   292→        dateFrom: dateFrom || undefined,
   293→        dateTo: dateTo || undefined,
   294→        status: status || undefined,
   295→      })) as Record<string, unknown>[];
   296→      console.log(`[BOOKINGS] Firestore returned ${bookings.length} raw docs (dateFrom=${dateFrom}, dateTo=${dateTo})`);
   297→      if (bookings.length > 0) {
   298→        console.log(`[BOOKINGS] Sample raw date field: id=${bookings[0].id}, date=${JSON.stringify(bookings[0].date)}, type=${typeof bookings[0].date}`);
   299→      }
   300→    }
   301→
   302→    // Enrich each booking with cached court/user lookups (avoid N+1 Firestore reads)
   303→    const courtCache = new Map<string, Record<string, unknown> | null>();
   304→    const userCache = new Map<string, Record<string, unknown> | null>();
   305→    const getCourtCached = async (cid: string) => {
   306→      if (courtCache.has(cid)) return courtCache.get(cid)!;
   307→      try {
   308→        const c = await getCourtById(cid);
   309→        courtCache.set(cid, c as Record<string, unknown> | null);
   310→        return c as Record<string, unknown> | null;
   311→      } catch { courtCache.set(cid, null); return null; }
   312→    };
   313→    const getUserCached = async (uid: string) => {
   314→      if (userCache.has(uid)) return userCache.get(uid)!;
   315→      try {
   316→        const u = await getUserById(uid);
   317→        userCache.set(uid, u as Record<string, unknown> | null);
   318→        return u as Record<string, unknown> | null;
   319→      } catch { userCache.set(uid, null); return null; }
   320→    };
   321→
   322→    const enriched: Record<string, unknown>[] = [];
   323→    // Process in batches of 10 to avoid overwhelming Firestore
   324→    const BATCH_SIZE = 10;
   325→    for (let i = 0; i < bookings.length; i += BATCH_SIZE) {
   326→      const batch = bookings.slice(i, i + BATCH_SIZE);
   327→      const results = await Promise.allSettled(
   328→        batch.map(async (b) => {
   329→          const allCourtIds: string[] = Array.isArray(b.court_ids)
   330→            ? b.court_ids as string[] : (b.court_id ? [b.court_id as string] : []);
   331→          let court: Record<string, unknown> | null = null;
   332→          let courts: Array<Record<string, unknown>> = [];
   333→          let user: Record<string, unknown> | null = null;
   334→          try {
   335→            if (allCourtIds.length > 0) court = await getCourtCached(allCourtIds[0]);
   336→          } catch { /* skip */ }
   337→          if (allCourtIds.length > 1) {
   338→            const enrichedCourts: Array<Record<string, unknown>> = [];
   339→            for (const cid of allCourtIds) {
   340→              try {
   341→                const c = await getCourtCached(cid);
   342→                if (c) enrichedCourts.push(c);
   343→              } catch { /* skip */ }
   344→            }
   345→            courts = enrichedCourts;
   346→          }
   347→          try {
   348→            if (b.user_id) user = await getUserCached(b.user_id as string);
   349→          } catch { /* skip */ }
   350→          return toCamelBooking({ ...b, _court: court, _courts: courts, _user: user });
   351→        })
   352→      );
   353→      for (const r of results) {
   354→        if (r.status === 'fulfilled') {
   355→          enriched.push(r.value);
   356→        } else {
   357→          console.error('[BOOKINGS] Batch enrichment error:', r.reason);
   358→        }
   359→      }
   360→    }
   361→
   362→    // Default server-side sort: today first, future ascending, past descending
   363→    // Within each date: chronological time
   364→    const today = new Date().toLocaleDateString('en-CA', { timeZone: 'America/Lima' });
   365→    enriched.sort((a, b) => {
   366→      const dA = String(a.date || ''), dB = String(b.date || '');
   367→      const aIsToday = dA === today, bIsToday = dB === today;
   368→      const aFuture = dA > today, bFuture = dB > today;
   369→      const aPast = dA < today, bPast = dB < today;
   370→      // 1) Today always first
   371→      if (aIsToday && !bIsToday) return -1;
   372→      if (!aIsToday && bIsToday) return 1;
   373→      // 2) Both on same side of today
   374→      if (aFuture && bFuture) {
   375→        // Future dates: ascending (correlativo)
   376→        const dc = dA.localeCompare(dB);
   377→        if (dc !== 0) return dc;
   378→      } else if (aPast && bPast) {
   379→        // Past dates: descending (most recent past first)
   380→        const dc = dB.localeCompare(dA);
   381→        if (dc !== 0) return dc;
   382→      }
   383→      // Same date: chronological by start time
   384→      return String(a.startTime || '').localeCompare(String(b.startTime || ''));
   385→    });
   386→
   387→    console.log(`[BOOKINGS] GET: ${enriched.length} bookings returned in ${Date.now() - startTime}ms (auth=${authUser.role}, filters: courtId=${courtId||'all'}, date=${date||'all'}, status=${status||'all'})`);
   388→
   389→    return NextResponse.json(enriched);
   390→  } catch (error) {
   391→    console.error('[BOOKINGS] GET error:', error);
   392→    const msg = error instanceof Error ? error.message : 'Unknown error';
   393→    return NextResponse.json({ error: 'Error al obtener reservas', detail: msg }, { status: 500 });
   394→  }
   395→}

   397→export async function POST(request: NextRequest) {
   398→  try {
   399→    const authResult = await requireAnyAuth(request);
   400→    if (authResult instanceof NextResponse) return authResult;
   401→    const authUser = authResult.user;
   402→
   403→    if (!isFirebaseAvailable()) {
   404→      return NextResponse.json({ error: 'Firebase no configurado' }, { status: 503 });
   405→    }
   406→
   407→    const body = await request.json();
   408→    const {
   409→      courtId,
   410→      courtIds: bodyCourtIds,
   411→      userId,
   412→      date,
   413→      startTime,
   414→      endTime,
   415→      totalPrice,
   416→      advanceAmount,
   417→      remainingAmount,
   418→      status,
   419→      paymentMethod,
   420→      notes,
   421→      equipmentItems,
   422→    } = body;
   423→
   424→    // ── Resolve court IDs (support both single courtId and multi-court courtIds) ──
   425→    const allCourtIds: string[] = Array.isArray(bodyCourtIds) && bodyCourtIds.length > 0
   426→      ? bodyCourtIds
   427→      : courtId
   428→        ? [courtId]
   429→        : [];
   430→
   431→    // ── Validate payment method (restricted set, UPPERCASE) ──
   432→    const VALID_PAYMENT_METHODS = ['EFECTIVO', 'YAPE', 'PLIN'];
   433→    const normalizedPaymentMethod = VALID_PAYMENT_METHODS.includes(paymentMethod)
   434→      ? paymentMethod
   435→      : 'EFECTIVO';
   436→
   437→    if (allCourtIds.length === 0 || !userId || !date || !startTime || !endTime) {
   438→      return NextResponse.json(
   439→        { error: 'Faltan campos requeridos: courtId/courtIds, userId, date, startTime, endTime' },
   440→        { status: 400 }
   441→      );
   442→    }
   443→
   444→    // ── Minimum advance validation ──
   445→    // Admin/super_admin: NO time restrictions — can book any slot (walk-in, past, current)
   446→    // Regular users: require 30 min minimum advance
   447→    const isAdmin = authUser.role === 'admin' || authUser.role === 'super_admin';
   448→
   449→    if (!isAdmin) {
   450→      const now = new Date();
   451→      const slotDateTime = new Date(`${date}T${startTime}:00`);
   452→      const diffMs = slotDateTime.getTime() - now.getTime();
   453→      const thirtyMinMs = 30 * 60 * 1000;
   454→      if (diffMs < thirtyMinMs) {
   455→        const minsLeft = Math.floor(diffMs / 60000);
   456→        if (minsLeft < 0) {
   457→          return NextResponse.json(
   458→            { error: 'No se puede reservar un horario que ya ha pasado.' },
   459→            { status: 422 }
   460→          );
   461→        }
   462→        return NextResponse.json(
   463→          { error: `La reserva requiere al menos 30 minutos de anticipación. El horario seleccionado comienza en ${minsLeft} minuto${minsLeft === 1 ? '' : 's'}. Selecciona un horario a partir de las ${new Date(now.getTime() + thirtyMinMs).toLocaleTimeString('es-PE', { hour: '2-digit', minute: '2-digit', hour12: false })} o posterior.` },
   464→          { status: 422 }
   465→        );
   466→      }
   467→    }
   468→
   469→    // Non-admin users can only create bookings for themselves
   470→    if (!isAdmin) {
   471→      if (userId !== authUser.id) {
   472→        return NextResponse.json(
   473→          { error: 'No puedes crear reservas para otros usuarios.' },
   474→          { status: 403 }
   475→        );
   476→      }
   477→    }
   478→
   479→    // ── Check overlap for ALL courts ──
   480→    const conflicts: Array<{ courtId: string; courtName: string }> = [];
   481→    for (const cId of allCourtIds) {
   482→      const existing = await getBookings({ courtId: cId, date });
   483→      const overlapping = existing.filter(
   484→        (b) => {
   485→          const bCourtIds: string[] = Array.isArray(b.court_ids)
   486→            ? b.court_ids as string[]
   487→            : [b.court_id as string];
   488→          // Skip if this existing booking doesn't include the court we're checking
   489→          if (!bCourtIds.includes(cId)) return false;
   490→          return (
   491→            !['cancelled'].includes(migrateStatus(b.status || '')) &&
   492→            (b.start_time || '') < endTime &&
   493→            (b.end_time || '') > startTime
   494→          );
   495→        }
   496→      );
   497→      if (overlapping.length > 0) {
   498→        let courtName = cId;
   499→        try {
   500→          const c = await getCourtById(cId);
   501→          if (c?.name) courtName = c.name as string;
   502→        } catch { /* use ID */ }
   503→        conflicts.push({ courtId: cId, courtName });
   504→      }
   505→    }
   506→
   507→    if (conflicts.length > 0) {
   508→      const conflictNames = conflicts.map((c) => c.courtName).join(', ');
   509→      return NextResponse.json(
   510→        {
   511→          error: `Cancha${conflicts.length > 1 ? 's' : ''} no disponible${conflicts.length > 1 ? 's' : ''}: ${conflictNames}. Por favor selecciona otro horario o cancha.`,
   512→          conflicts,
   513→        },
   514→        { status: 409 }
   515→      );
   516→    }
   517→
   518→    // ── Calculate price ──
   519→    // Frontend sends totalPrice as grand total (court + equipment combined).
   520→    // We compute courtSubtotal and equipmentSubtotal separately for storage/display.
   521→    const eqItems = Array.isArray(equipmentItems) ? equipmentItems : [];
   522→    let equipmentSubtotal = 0;
   523→    for (const eq of eqItems) {
   524→      equipmentSubtotal += (eq.quantity || 0) * (eq.unit_price || eq.unitPrice || 0);
   525→    }
   526→
   527→    let courtPriceTotal: number;
   528→    const providedTotal = parseFloat(totalPrice) || 0;
   529→    if (providedTotal > 0) {
   530→      // Frontend already included equipment in totalPrice — extract court portion
   531→      courtPriceTotal = Math.max(0, providedTotal - equipmentSubtotal);
   532→    } else {
   533→      // No price from frontend — calculate from court rates
   534→      courtPriceTotal = 0;
   535→      for (const cId of allCourtIds) {
   536→        const court = await getCourtById(cId);
   537→        if (court) {
   538→          const schedule = court.pricing_schedule as Array<{ label: string; startHour: number; endHour: number; pricePerHour: number }> | undefined;
   539→          let courtPrice = 0;
   540→          if (schedule && schedule.length > 0) {
   541→            courtPrice = calculatePriceForTimeSlot(schedule, startTime, endTime);
   542→            if (courtPrice <= 0) courtPrice = (court.price_per_hour as number) || 0;
   543→          } else {
   544→            courtPrice = (court.price_per_hour as number) || 0;
   545→          }
   546→          courtPriceTotal += courtPrice;
   547→        }
   548→      }
   549→    }
   550→
   551→    const price = courtPriceTotal + equipmentSubtotal;
   552→
   553→    // B4 FIX: Validate advance + remaining === total consistency
   554→    const adv = parseFloat(advanceAmount) || price * 0.5;
   555→    let rem = parseFloat(remainingAmount) || price - adv;
   556→    // Force consistency: remaining must equal total - advance
   557→    rem = Math.max(0, Math.round((price - adv) * 100) / 100);
   558→    const bookingStatus = migrateStatus(status || 'reserved');
   559→
   560→    // Resolve client email for denormalized search
   561→    let clientEmail = authUser.email;
   562→    try {
   563→      const clientUser = await getUserById(userId);
   564→      if (clientUser?.email) clientEmail = clientUser.email;
   565→    } catch { /* fallback to admin email */ }
   566→
   567→    // Save to Firestore — single booking with all courts
   568→    const id = await createBooking({
   569→      court_ids: allCourtIds,
   570→      user_id: userId,
   571→      user_email: clientEmail,
   572→      date,
   573→      start_time: startTime,
   574→      end_time: endTime,
   575→      total_price: price,
   576→      court_subtotal: courtPriceTotal,
   577→      equipment_subtotal: equipmentSubtotal,
   578→      equipment_items: eqItems,
   579→      advance_amount: adv,
   580→      remaining_amount: rem,
   581→      status: bookingStatus,
   582→      slot_status: 'available',
   583→      payment_method: normalizedPaymentMethod || null,
   584→      notes: notes || null,
   585→    });
   586→
   587→    console.log(`[BOOKINGS] POST: Created booking ${id} for user=${userId} date=${date} ${startTime}-${endTime} courts=${allCourtIds.join(',')} total=${price} advance=${adv} remaining=${rem} method=${normalizedPaymentMethod} email=${clientEmail}`);
   588→
   589→    // B5 FIX: Only create payment record if advance > 0
   590→    if (adv > 0.01) {
   591→      try {
   592→        await createPayment(id, {
   593→          user_id: userId,
   594→          amount: adv,
   595→          type: 'advance',
   596→          method: normalizedPaymentMethod || 'EFECTIVO',
   597→          status: 'completed',
   598→        });
   599→      } catch (payErr) {
   600→        console.error('[BOOKINGS] Warning: could not create payment record:', payErr);
   601→      }
   602→    }
   603→
   604→    return NextResponse.json({
   605→      id,
   606→      courtId: allCourtIds[0],
   607→      courtIds: allCourtIds,
   608→      userId,
   609→      date,
   610→      startTime,
   611→      endTime,
   612→      totalPrice: price,
   613→      advanceAmount: adv,
   614→      remainingAmount: rem,
   615→      status: bookingStatus,
   616→      paymentMethod: paymentMethod || null,
   617→      success: true,
   618→    }, { status: 201 });
   619→  } catch (error: unknown) {
   620→    const err = error as { message?: string; code?: string; stack?: string };
   621→    console.error('[BOOKINGS] POST error:', err?.message || err);
   622→    console.error('[BOOKINGS] Stack:', err?.stack || 'no stack');
   623→    return NextResponse.json({
   624→      error: 'Failed to create booking',
   625→      detail: err?.message || 'Unknown error'
   626→    }, { status: 500 });
   627→  }
   628→}

   669→    // (editBooking handles status via its own branch to avoid double-processing)

   705→            if (!obCourtIds.some((cid: string) => allCourtIds.includes(cid))) return false;
   706→            return (ob.start_time as string) < reqEndTime && (ob.end_time as string) > (booking.start_time as string);
   707→          });
   708→          // Filter to only real conflicts (overlapping with the NEW extended period)
   709→          const realConflicts = conflicts.filter((ob: Record<string, unknown>) => {

   737→          }

   744→      if (authUser.role !== 'super_admin') {
   745→        return NextResponse.json({ error: 'Solo el super administrador puede editar reservas.' }, { status: 403 });

   885→          const existing = await getRetainedAdvances({ bookingId: id });
   886→          if (existing.length > 0) {

   890→          const isRetained = advanceAction === 'retain';
   891→          const statusAdvance = isRetained ? 'retained' : 'refunded';

   897→            if (cId) {
   898→              const court = await getCourtById(cId);

   900→            }
   901→          } catch { /* non-critical */ }

   904→          let userName = '';
   905→          try {
   906→            const uId = bookingForCancel.user_id as string;
   907→            if (uId) {
   908→              const user = await getUserById(uId);
   909→              userName = (user?.name as string) || '';
   910→            }
   911→          } catch { /* non-critical */ }
   912→
   913→          await createRetainedAdvance({
   914→            booking_id: id,
   915→            user_id: (bookingForCancel.user_id as string) || '',
   916→            user_name: userName,
   917→            user_email: (bookingForCancel.user_email as string) || null,
   918→            court_name: courtName,
   919→            booking_date: (bookingForCancel.date as string) || '',
   920→            amount: advAmount,
   921→            original_total: (bookingForCancel.total_price as number) || 0,
   922→            payment_method: (bookingForCancel.payment_method as string) || 'EFECTIVO',
   923→            reason: cancelReason || 'Cancelación de reserva',
   924→            status: statusAdvance,
   925→          });
   926→
   927→          // Create a payment record for the refund/retention
   928→          try {
   929→            await createPayment(id, {
   930→              user_id: (bookingForCancel.user_id as string) || '',
   931→              amount: advAmount,
   932→              type: isRetained ? 'retained' : 'refund',
   933→              method: (bookingForCancel.payment_method as string) || 'EFECTIVO',
   934→              status: 'completed',
   935→            });
   936→          } catch (payErr) {
   937→            console.error('[BOOKINGS] Warning: could not create payment record for cancellation:', payErr);
   938→          }
   939→          }
   940→        }
   941→      } catch (err) {
   942→        console.error('[BOOKINGS] Warning: could not process retained advance on cancellation:', err);
   943→      }
   944→      } // end isAdminCancellation
   945→    }
   946→
   947→    // B11 FIX: Skip advance payment record when cancelling or editBooking (handled above / no record needed)
   948→    if (typeof reqAdvance === 'number' && typeof reqRemaining === 'number' && !isCancelling && !editBooking) {
   949→      try {
   950→        const booking = await getBookingById(id);
   951→        if (booking) {
   952→          const prevAdvance = (booking.advance_amount as number) || 0;
   953→          const paymentIncrement = reqAdvance - prevAdvance;
   954→          if (paymentIncrement > 0.01) {
   955→            try {
   956→              await createPayment(id, {
   957→                user_id: booking.user_id as string,
   958→                amount: paymentIncrement,
   959→                type: reqRemaining <= 0 ? 'full_payment' : 'advance',
   960→                method: (updateData.payment_method as string) || 'EFECTIVO',
   961→                status: 'completed',
   962→              });
   963→            } catch (payErr) {
   964→              console.error('[BOOKINGS] Warning: could not create payment record for advance update:', payErr);
   965→            }
   966→          }
   967→        }
   968→      } catch { /* non-critical: proceed with booking update */ }
   969→    }
   970→
   971→    await updateBooking(id, updateData);
   972→
   973→    return NextResponse.json({ success: true });
   974→  } catch (error) {
   975→    console.error('[BOOKINGS] PUT error:', error);
   976→    const msg = error instanceof Error ? error.message : 'Unknown error';
   977→    return NextResponse.json({ error: 'Error al actualizar reserva', detail: msg }, { status: 500 });
   978→  }
