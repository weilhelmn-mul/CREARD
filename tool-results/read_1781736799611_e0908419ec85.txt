   186→/** Compare Firestore Timestamps (seconds) or ISO strings for sorting */
   187→function compareTimestamps(a: unknown, b: unknown): number {
   188→  const toMs = (t: unknown): number => {
   189→    if (!t) return 0
   190→    if (typeof t === 'number') return t * 1000 // Firestore seconds
   191→    if (typeof t === 'object' && t !== null && '_seconds' in (t as Record<string, unknown>)) return (t as Record<string, unknown>)._seconds as number * 1000
   192→    if (typeof t === 'string') return new Date(t).getTime()
   193→    return 0
   194→  }
   195→  return toMs(a) - toMs(b)
   196→}

   202→/* ─── Sortable Section Card ─── */
   203→function SortableSectionCard({
   204→  id,
   205→  label,
   206→  icon,
   207→  color,
   208→  isCustom,
   209→  visible,
   210→  onToggle,
   211→  onEdit,
   212→  onDelete,
   213→}: {
   214→  id: string
   215→  label: string
   216→  icon: string
   217→  color: string
   218→  isCustom: boolean
   219→  visible: boolean
   220→  onToggle: () => void
   221→  onEdit: () => void
   222→  onDelete?: () => void
   223→}) {
   224→  const { attributes, listeners, setNodeRef, transform, transition, isDragging } = useSortable({ id })
   225→  const style = { transform: CSS.Transform.toString(transform), transition }
   226→
   227→  return (
   228→    <div
   229→      ref={setNodeRef}
   230→      style={style}
   231→      className={`glass-card rounded-xl p-3 flex items-center gap-3 transition-all ${isDragging ? 'opacity-60 shadow-xl z-50' : ''} ${!visible ? 'opacity-40' : ''}`}
   232→    >
   233→      {/* Drag handle */}
   234→      <button type="button" {...attributes} {...listeners} className="cursor-grab active:cursor-grabbing text-cm-on-surface-variant/40 hover:text-cm-on-surface-variant p-1">
   235→        <span className="material-symbols-outlined text-[18px]">drag_indicator</span>
   236→      </button>
   237→
   238→      {/* Icon */}
   239→      <div className={`w-9 h-9 rounded-lg flex items-center justify-center flex-shrink-0 ${color}`}>
   240→        <span className="material-symbols-outlined text-[18px] text-white" style={{ fontVariationSettings: '"FILL" 1' }}>
   241→          {icon}
   242→        </span>
   243→      </div>
   244→
   245→      {/* Label */}
   246→      <span className="flex-1 text-sm font-semibold text-cm-on-surface truncate font-[family-name:var(--font-sora)]">
   247→        {label}
   248→      </span>
   249→
   250→      {/* Actions */}
   251→      <div className="flex items-center gap-1.5 flex-shrink-0">
   252→        <button type="button" onClick={onToggle} className={`p-1.5 rounded-lg transition-colors ${visible ? 'text-cm-primary hover:bg-cm-primary/10' : 'text-cm-on-surface-variant/40 hover:bg-white/5'}`} title={visible ? 'Ocultar' : 'Mostrar'}>
   253→          <span className="material-symbols-outlined text-[16px]">{visible ? 'visibility' : 'visibility_off'}</span>
   254→        </button>
   255→        <button type="button" onClick={onEdit} className="p-1.5 rounded-lg text-cm-on-surface-variant hover:text-cm-primary hover:bg-cm-primary/10 transition-colors" title="Editar">
   256→          <span className="material-symbols-outlined text-[16px]">edit</span>
   257→        </button>
   258→        {isCustom && onDelete && (
   259→          <button type="button" onClick={onDelete} className="p-1.5 rounded-lg text-cm-on-surface-variant hover:text-red-400 hover:bg-red-500/10 transition-colors" title="Eliminar">
   260→            <span className="material-symbols-outlined text-[16px]">delete</span>
   261→          </button>
   262→        )}
   263→      </div>
   264→    </div>
   265→  )
   266→}
