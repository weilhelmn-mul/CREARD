  2613→  const calculatePriceForTimeSlot = (schedule: PricingScheduleItem[], startTime: string, endTime: string): { total: number; breakdown: Array<{ label: string; hours: number; pricePerHour: number; subtotal: number }> } => {

  2617→    const endDecimal = endH + endM / 60

  2619→    if (schedule.length === 0) return { total: 0, breakdown: [] }
  2620→
  2621→    const sorted = [...schedule].sort((a, b) => a.startHour - b.startHour)

  2625→

  2627→      if (cursor >= slot.endHour) continue
  2628→      const overlapStart = Math.max(cursor, slot.startHour)

  2631→        const hours = overlapEnd - overlapStart

  2633→        breakdown.push({ label: slot.label, hours: Math.round(hours * 100) / 100, pricePerHour: slot.pricePerHour, subtotal })
  2634→        total += subtotal
  2635→        cursor = overlapEnd
  2636→      }
  2637→    }
  2638→    return { total: Math.round(total * 100) / 100, breakdown }
  2639→  }
