   254→   254→function buildQuery(
   255→   255→  collectionName: string,
   256→   256→  constraints: Array<{ field: string; op: string; value: unknown }>
   257→   257→): Query {
   258→   258→  let q: Query = getAdminDb().collection(collectionName);
   259→   259→  for (const c of constraints) {
   260→   260→    switch (c.op) {
   261→   261→      case '==': q = q.where(c.field, '==', c.value); break;
   262→   262→      case '!=': q = q.where(c.field, '!=', c.value); break;
   263→   263→      case '<': q = q.where(c.field, '<', c.value); break;
   264→   264→      case '<=': q = q.where(c.field, '<=', c.value); break;
   265→   265→      case '>': q = q.where(c.field, '>', c.value); break;
   266→   266→      case '>=': q = q.where(c.field, '>=', c.value); break;
   267→   267→      case 'in': q = q.where(c.field, 'in', c.value as unknown[]); break;
   268→   268→      case 'not-in': q = q.where(c.field, 'not-in', c.value as unknown[]); break;
   269→   269→      case 'array-contains': q = q.where(c.field, 'array-contains', c.value); break;
   270→   270→    }
   271→   271→  }
   272→   272→  return q;
   273→   273→}
   274→   274→
   275→   275→async function queryDocs(
   276→   276→  collectionName: string,
   277→   277→  constraints: Array<{ field: string; op: string; value: unknown }> = [],
   278→   278→  orderField?: string,
   279→   279→  orderDir: 'asc' | 'desc' = 'asc'
   280→   280→): Promise<DocumentData[]> {
   281→   281→  let q = buildQuery(collectionName, constraints);
   282→   282→  // NEVER use Firestore orderBy — always sort client-side
   283→   283→  // This avoids needing composite indexes (user_id+start_time, user_email+start_time, etc.)
   284→   284→  const snapshot: QuerySnapshot = await q.get();
   285→   285→  let docs = snapshot.docs.map((d) => ({ id: d.id, ...d.data() }));
   286→   286→  // Client-side sort for all queries
   287→   287→  if (orderField) {
   288→   288→    docs.sort((a, b) => {
   289→   289→      const va = a[orderField] ?? '';
   290→   290→      const vb = b[orderField] ?? '';
   291→   291→      if (va < vb) return orderDir === 'asc' ? -1 : 1;
   292→   292→      if (va > vb) return orderDir === 'asc' ? 1 : -1;
   293→   293→      return 0;
   294→   294→    });
   295→   295→  }
   296→   296→  return docs;
   297→   297→}

   405→   405→export async function getBookings(filters?: {
   406→   406→  courtId?: string;
   407→   407→  userId?: string;
   408→   408→  userEmail?: string;
   409→   409→  date?: string;
   410→   410→  dateFrom?: string;
   411→   411→  dateTo?: string;
   412→   412→  status?: string;
   413→   413→}): Promise<Partial<Booking>[]> {
   414→   414→  // Build non-court constraints (shared by both queries)
   415→   415→  const baseConstraints: Array<{ field: string; op: string; value: unknown }> = [];
   416→   416→  if (filters?.userId) baseConstraints.push({ field: 'user_id', op: '==', value: filters.userId });
   417→   417→  if (filters?.userEmail) baseConstraints.push({ field: 'user_email', op: '==', value: filters.userEmail });
   418→   418→  if (filters?.date) baseConstraints.push({ field: 'date', op: '==', value: filters.date });
   419→   419→  if (filters?.dateFrom) baseConstraints.push({ field: 'date', op: '>=', value: filters.dateFrom });
   420→   420→  if (filters?.dateTo) baseConstraints.push({ field: 'date', op: '<=', value: filters.dateTo });
   421→   421→  if (filters?.status) baseConstraints.push({ field: 'status', op: '==', value: filters.status });
   422→   422→
   423→   423→  // No court filter — single query
   424→   424→  if (!filters?.courtId) {
   425→   425→    return queryDocs('bookings', baseConstraints, 'start_time');
   426→   426→  }
   427→   427→
   428→   428→  // Court filter present — need TWO queries:
   429→   429→  //   1) court_id == X  (primary court)
   430→   430→  //   2) court_ids array-contains X  (secondary court in multi-court booking)
   431→   431→  const [byPrimary, byArray] = await Promise.all([
   432→   432→    queryDocs('bookings', [
   433→   433→      ...baseConstraints,
   434→   434→      { field: 'court_id', op: '==', value: filters.courtId },
   435→   435→    ], 'start_time'),
   436→   436→    queryDocs('bookings', [
   437→   437→      ...baseConstraints,
   438→   438→      { field: 'court_ids', op: 'array-contains', value: filters.courtId },
   439→   439→    ], 'start_time'),
   440→   440→  ]);
   441→   441→
   442→   442→  // Merge & deduplicate by document ID
   443→   443→  const seen = new Set<string>();
   444→   444→  const merged: DocumentData[] = [];
   445→   445→  for (const doc of [...byPrimary, ...byArray]) {
   446→   446→    if (!seen.has(doc.id)) {
   447→   447→      seen.add(doc.id);
   448→   448→      merged.push(doc);
   449→   449→    }
   450→   450→  }
   451→   451→
   452→   452→  // Client-side sort (already sorted individually, but merge breaks order)
   453→   453→  merged.sort((a, b) => {
   454→   454→    const va = a.start_time ?? '';
   455→   455→    const vb = b.start_time ?? '';
   456→   456→    if (va < vb) return -1;
   457→   457→    if (va > vb) return 1;
   458→   458→    return 0;
   459→   459→  });
   460→   460→
   461→   461→  return merged as Partial<Booking>[];
   462→   462→}

   464→   464→export async function createBooking(data: Record<string, unknown>): Promise<string> {
   465→   465→  const courtIds: string[] = Array.isArray(data.court_ids) ? data.court_ids : [data.court_id];
   466→   466→  return addDoc('bookings', {
   467→   467→    court_id: courtIds[0] || data.court_id, // Primary court (backward compat)
   468→   468→    court_ids: courtIds, // All courts in this booking
   469→   469→    user_id: data.user_id,
   470→   470→    user_email: data.user_email || null, // Denormalized for fallback search
   471→   471→    date: data.date,
   472→   472→    start_time: data.start_time,
   473→   473→    end_time: data.end_time,
   474→   474→    total_price: data.total_price || 0,
   475→   475→    court_subtotal: data.court_subtotal || 0,
   476→   476→    equipment_subtotal: data.equipment_subtotal || 0,
   477→   477→    equipment_items: data.equipment_items || [],
   478→   478→    equipment_delivered: data.equipment_delivered || false,
   479→   479→    equipment_returned: data.equipment_returned || false,
   480→   480→    advance_amount: data.advance_amount || 0,
   481→   481→    remaining_amount: data.remaining_amount || 0,
   482→   482→    status: data.status || 'reserved',
   483→   483→    slot_status: data.slot_status || 'available',

   594→   594→/**
   595→   595→ * Get the next payment number from the counters collection (atomic increment via transaction)
   596→   596→ */
   597→   597→export async function getNextPaymentNumber(): Promise<number> {
   598→   598→  const db = await getAdminDb();
   599→   599→  const counterRef = db.collection('counters').doc('payments');
   600→   600→
   601→   601→  const result = await db.runTransaction(async (transaction) => {
   602→   602→    const counterSnap = await transaction.get(counterRef);
   603→   603→    if (!counterSnap.exists) {
   604→   604→      // Counter doesn't exist yet, create it starting at 1
   605→   605→      const newCounter = { value: 1, updated_at: Timestamp.now() };
   606→   606→      transaction.set(counterRef, newCounter);
   607→   607→      return 1;
   608→   608→    }
   609→   609→    const current = counterSnap.data()?.value ?? 0;
   610→   610→    const next = current + 1;
   611→   611→    transaction.update(counterRef, {
   612→   612→      value: next,
   613→   613→      updated_at: Timestamp.now(),
   614→   614→    });
   615→   615→    return next;
   616→   616→  });
   617→   617→
   618→   618→  return result;
   619→   619→}
