     7→// Helper: add auth headers to fetch calls
     8→function authFetch(url: string, options: RequestInit = {}): Promise<Response> {
     9→  const headers = new Headers(options.headers || {})
    10→  const authHeaders = getAuthHeaders()
    11→  for (const [key, value] of Object.entries(authHeaders)) {
    12→    if (value) headers.set(key, value)
    13→  }
    14→  if (!headers.has('Content-Type')) {
    15→    headers.set('Content-Type', 'application/json')
    16→  }
    17→  return fetch(url, { ...options, headers })
    18→}
