     8→// Helper: add auth headers to fetch calls
     9→function authFetch(url: string, options: RequestInit = {}): Promise<Response> {
    10→  const headers = new Headers(options.headers || {})
    11→  const authHeaders = getAuthHeaders()
    12→  for (const [key, value] of Object.entries(authHeaders)) {
    13→    if (value) headers.set(key, value)
    14→  }
    15→  if (!headers.has('Content-Type')) {
    16→    headers.set('Content-Type', 'application/json')
    17→  }
    18→  return fetch(url, { ...options, headers })
    19→}
