    87→/* ─── helpers ─── */
    88→// Parse date string as local date (avoids UTC timezone issues for Peru, UTC-5)
    89→function parseLocalDate(dateStr: string): Date {
    90→  const parts = dateStr.split('-')
    91→  if (parts.length === 3) {
    92→    return new Date(parseInt(parts[0], 10), parseInt(parts[1], 10) - 1, parseInt(parts[2], 10))
    93→  }
    94→  return new Date(dateStr + 'T00:00:00')
    95→}
    96→
    97→const fmt = (dateStr: string) => {
    98→  const d = parseLocalDate(dateStr)
    99→  return d.toLocaleDateString('es-PE', { day: 'numeric', month: 'short', year: 'numeric' })
   100→}
