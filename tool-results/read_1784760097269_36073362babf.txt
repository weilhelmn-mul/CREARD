   230→function buildQuery(
   231→  collectionName: string,
   232→  constraints: Array<{ field: string; op: string; value: unknown }>
   233→): Query {
   234→  let q: Query = getAdminDb().collection(collectionName);
   235→  for (const c of constraints) {
   236→    switch (c.op) {
   237→      case '==': q = q.where(c.field, '==', c.value); break;
   238→      case '!=': q = q.where(c.field, '!=', c.value); break;
   239→      case '<': q = q.where(c.field, '<', c.value); break;
   240→      case '<=': q = q.where(c.field, '<=', c.value); break;
   241→      case '>': q = q.where(c.field, '>', c.value); break;
   242→      case '>=': q = q.where(c.field, '>=', c.value); break;
   243→      case 'in': q = q.where(c.field, 'in', c.value as unknown[]); break;
   244→      case 'not-in': q = q.where(c.field, 'not-in', c.value as unknown[]); break;
   245→      case 'array-contains': q = q.where(c.field, 'array-contains', c.value); break;
   246→    }
   247→  }
   248→  return q;
   249→}
   250→
   251→async function queryDocs(
   252→  collectionName: string,
   253→  constraints: Array<{ field: string; op: string; value: unknown }> = [],
   254→  orderField?: string,
   255→  orderDir: 'asc' | 'desc' = 'asc'
   256→): Promise<DocumentData[]> {
   257→  let q = buildQuery(collectionName, constraints);
   258→  // NEVER use Firestore orderBy — always sort client-side
   259→  // This avoids needing composite indexes (user_id+start_time, user_email+start_time, etc.)
   260→  const snapshot: QuerySnapshot = await q.get();
   261→  let docs = snapshot.docs.map((d) => ({ id: d.id, ...d.data() }));
   262→  // Client-side sort for all queries
   263→  if (orderField) {
   264→    docs.sort((a, b) => {
   265→      const va = a[orderField] ?? '';
   266→      const vb = b[orderField] ?? '';
   267→      if (va < vb) return orderDir === 'asc' ? -1 : 1;
   268→      if (va > vb) return orderDir === 'asc' ? 1 : -1;
   269→      return 0;
   270→    });
   271→  }
   272→  return docs;
   273→}

   380→// --- Bookings ---
   381→export async function getBookings(filters?: {
   382→  courtId?: string;
   383→  userId?: string;
   384→  userEmail?: string;
   385→  date?: string;
   386→  dateFrom?: string;
   387→  dateTo?: string;
   388→  status?: string;
   389→}): Promise<Partial<Booking>[]> {
   390→  // Build non-court constraints (shared by both queries)
   391→  const baseConstraints: Array<{ field: string; op: string; value: unknown }> = [];
   392→  if (filters?.userId) baseConstraints.push({ field: 'user_id', op: '==', value: filters.userId });
   393→  if (filters?.userEmail) baseConstraints.push({ field: 'user_email', op: '==', value: filters.userEmail });
   394→  if (filters?.date) baseConstraints.push({ field: 'date', op: '==', value: filters.date });
   395→  if (filters?.dateFrom) baseConstraints.push({ field: 'date', op: '>=', value: filters.dateFrom });
   396→  if (filters?.dateTo) baseConstraints.push({ field: 'date', op: '<=', value: filters.dateTo });
   397→  if (filters?.status) baseConstraints.push({ field: 'status', op: '==', value: filters.status });
   398→
   399→  // No court filter — single query
   400→  if (!filters?.courtId) {
   401→    return queryDocs('bookings', baseConstraints, 'start_time');
   402→  }
   403→
   404→  // Court filter present — need TWO queries:
   405→  //   1) court_id == X  (primary court)
   406→  //   2) court_ids array-contains X  (secondary court in multi-court booking)
   407→  const [byPrimary, byArray] = await Promise.all([
   408→    queryDocs('bookings', [
   409→      ...baseConstraints,
   410→      { field: 'court_id', op: '==', value: filters.courtId },
   411→    ], 'start_time'),
   412→    queryDocs('bookings', [
   413→      ...baseConstraints,
   414→      { field: 'court_ids', op: 'array-contains', value: filters.courtId },
   415→    ], 'start_time'),
   416→  ]);
   417→
   418→  // Merge & deduplicate by document ID
   419→  const seen = new Set<string>();
   420→  const merged: DocumentData[] = [];
   421→  for (const doc of [...byPrimary, ...byArray]) {
   422→    if (!seen.has(doc.id)) {
   423→      seen.add(doc.id);
   424→      merged.push(doc);
   425→    }
   426→  }
   427→
   428→  // Client-side sort (already sorted individually, but merge breaks order)
   429→  merged.sort((a, b) => {
   430→    const va = a.start_time ?? '';
   431→    const vb = b.start_time ?? '';
   432→    if (va < vb) return -1;
   433→    if (va > vb) return 1;
   434→    return 0;
   435→  });
   436→
   437→  return merged as Partial<Booking>[];
   438→}

   900→/** Get claims that are approaching or past deadline */
   901→export async function getClaimsNearDeadline(daysThreshold: number = 2): Promise<Partial<Claim>[]> {
   902→  const allClaims = await getClaims();
   903→  const now = new Date();
   904→  const threshold = new Date(now.getTime() + daysThreshold * 24 * 60 * 60 * 1000);
   905→  return allClaims.filter((c) => {
   906→    if (c.status === 'closed' || c.status === 'archived' || !c.deadline_date) return false;
   907→    const deadline = c.deadline_date instanceof Date ? c.deadline_date : new Date(c.deadline_date);
   908→    return deadline <= threshold;
   909→  });
   910→}
