   191→/** Compare Firestore Timestamps (seconds) or ISO strings for sorting */
   192→function compareTimestamps(a: unknown, b: unknown): number {
   193→  const toMs = (t: unknown): number => {
   194→    if (!t) return 0
   195→    if (typeof t === 'number') return t * 1000 // Firestore seconds
   196→    if (typeof t === 'object' && t !== null && '_seconds' in (t as Record<string, unknown>)) return (t as Record<string, unknown>)._seconds as number * 1000
   197→    if (typeof t === 'string') return new Date(t).getTime()
   198→    return 0
   199→  }
   200→  return toMs(a) - toMs(b)
   201→}

   207→/* ─── Sortable Section Card ─── */
   208→function SortableSectionCard({
   209→  id,
   210→  label,
   211→  icon,
   212→  color,
   213→  isCustom,
   214→  visible,
   215→  onToggle,
   216→  onEdit,
   217→  onDelete,
   218→}: {
   219→  id: string
   220→  label: string
   221→  icon: string
   222→  color: string
   223→  isCustom: boolean
   224→  visible: boolean
   225→  onToggle: () => void
   226→  onEdit: () => void
   227→  onDelete?: () => void
   228→}) {
   229→  const { attributes, listeners, setNodeRef, transform, transition, isDragging } = useSortable({ id })
   230→  const style = { transform: CSS.Transform.toString(transform), transition }
   231→
   232→  return (
   233→    <div
   234→      ref={setNodeRef}
   235→      style={style}
   236→      className={`glass-card rounded-xl p-3 flex items-center gap-3 transition-all ${isDragging ? 'opacity-60 shadow-xl z-50' : ''} ${!visible ? 'opacity-40' : ''}`}
   237→    >
   238→      {/* Drag handle */}
   239→      <button type="button" {...attributes} {...listeners} className="cursor-grab active:cursor-grabbing text-cm-on-surface-variant/40 hover:text-cm-on-surface-variant p-1">
   240→        <span className="material-symbols-outlined text-[18px]">drag_indicator</span>
   241→      </button>
   242→
   243→      {/* Icon */}
   244→      <div className={`w-9 h-9 rounded-lg flex items-center justify-center flex-shrink-0 ${color}`}>
   245→        <span className="material-symbols-outlined text-[18px] text-white" style={{ fontVariationSettings: '"FILL" 1' }}>
   246→          {icon}
   247→        </span>
   248→      </div>
   249→
   250→      {/* Label */}
   251→      <span className="flex-1 text-sm font-semibold text-cm-on-surface truncate font-[family-name:var(--font-sora)]">
   252→        {label}
   253→      </span>
   254→
   255→      {/* Actions */}
   256→      <div className="flex items-center gap-1.5 flex-shrink-0">
   257→        <button type="button" onClick={onToggle} className={`p-1.5 rounded-lg transition-colors ${visible ? 'text-cm-primary hover:bg-cm-primary/10' : 'text-cm-on-surface-variant/40 hover:bg-white/5'}`} title={visible ? 'Ocultar' : 'Mostrar'}>
   258→          <span className="material-symbols-outlined text-[16px]">{visible ? 'visibility' : 'visibility_off'}</span>
   259→        </button>
   260→        <button type="button" onClick={onEdit} className="p-1.5 rounded-lg text-cm-on-surface-variant hover:text-cm-primary hover:bg-cm-primary/10 transition-colors" title="Editar">
   261→          <span className="material-symbols-outlined text-[16px]">edit</span>
   262→        </button>
   263→        {isCustom && onDelete && (
   264→          <button type="button" onClick={onDelete} className="p-1.5 rounded-lg text-cm-on-surface-variant hover:text-red-400 hover:bg-red-500/10 transition-colors" title="Eliminar">
   265→            <span className="material-symbols-outlined text-[16px]">delete</span>
   266→          </button>
   267→        )}
   268→      </div>
   269→    </div>
   270→  )
   271→}

   531→      if (!target[keys[i]] || typeof target[keys[i]] !== 'object') target[keys[i]] = {}
   532→      target = target[keys[i]] as Record<string, unknown>
   533→    }
   534→    target[keys[keys.length - 1]] = value
   535→    setEditForm(copy)

   545→    }
   546→    return String(target ?? '')
