   483→    }
   484→
   485→    // ── FIX P1-5: Check overlap using Firestore transaction to prevent race conditions ──
   486→    const db = await getAdminDb();
   487→    let bookingId: string;
   488→
   489→    try {
   490→      bookingId = await db.runTransaction(async (transaction) => {
   491→        // Re-check overlap inside transaction for each court
   492→        for (const cId of allCourtIds) {
   493→          const bookingsRef = db.collection('bookings')
   494→            .where('date', '==', date)
   495→            .where('court_ids', 'array-contains', cId);
   496→          const snapshot = await transaction.get(bookingsRef);
   497→
   498→          for (const doc of snapshot.docs) {
   499→            const b = doc.data();
   500→            const bCourtIds: string[] = Array.isArray(b.court_ids) ? b.court_ids : [b.court_id];
   501→            if (!bCourtIds.includes(cId)) continue;
   502→            if (migrateStatus(b.status || '') === 'cancelled') continue;
   503→            if ((b.start_time || '') < endTime && (b.end_time || '') > startTime) {
   504→              let courtName = cId;
   505→              try {
   506→                const c = await getCourtById(cId);
   507→                if (c?.name) courtName = c.name as string;
   508→              } catch { /* use ID */ }
   509→              throw new Error(`Cancha no disponible: ${courtName}. Por favor selecciona otro horario o cancha.`);
   510→            }
   511→          }
   512→        }
   513→
   514→        // All clear — create the booking inside the transaction
   515→        const now = Timestamp.now();
   516→        const docRef = db.collection('bookings').doc();
   517→        const courtIds: string[] = allCourtIds;
   518→        transaction.set(docRef, {
   519→          court_id: courtIds[0],
   520→          court_ids: courtIds,
   521→          user_id: userId,
   522→          user_email: clientEmail || null,
   523→          date,
   524→          start_time: startTime,
   525→          end_time: endTime,
   526→          total_price: price,
   527→          court_subtotal: courtPriceTotal,
   528→          equipment_subtotal: equipmentSubtotal,
   529→          equipment_items: eqItems,
   530→          equipment_delivered: false,
   531→          equipment_returned: false,
   532→          advance_amount: adv,
   533→          remaining_amount: rem,
   534→          status: bookingStatus,
   535→          slot_status: 'available',
   536→          payment_method: normalizedPaymentMethod || null,
   537→          notes: notes || null,
   538→          recurring_group_id: null,
   539→          recurring_index: null,
   540→          selected_slots: Array.isArray(selectedSlots) ? selectedSlots : [],
   541→          created_at: now,
   542→          updated_at: now,
   543→          // FIX P0-4: Auto-expire unpaid bookings after 15 minutes
   544→          expires_at: Timestamp.fromMillis(now.toMillis() + 15 * 60 * 1000),
   545→        });
   546→        return docRef.id;
   547→      });
   548→    } catch (txError: any) {
   549→      if (txError?.message?.includes('Cancha no disponible')) {
   550→        return NextResponse.json({ error: txError.message }, { status: 409 });
   551→      }
   552→      throw txError;
   553→    }
   554→
   555→    // ── Calculate price ──
   556→    // Frontend sends totalPrice as grand total (court + equipment combined).
   557→    // We compute courtSubtotal and equipmentSubtotal separately for storage/display.
   558→    const eqItems = Array.isArray(equipmentItems) ? equipmentItems : [];
   559→    let equipmentSubtotal = 0;
   560→    for (const eq of eqItems) {
   561→      equipmentSubtotal += (eq.quantity || 0) * (eq.unit_price || eq.unitPrice || 0);
   562→    }
   563→
   564→    // FIX P1-6: ALWAYS recalculate price server-side from court rates.
   565→    // Client-sent totalPrice is ignored for storage — used only as a UI hint.
   566→    let courtPriceTotal = 0;
   567→    for (const cId of allCourtIds) {
   568→      const court = await getCourtById(cId);
   569→      if (court) {
   570→        const schedule = court.pricing_schedule as Array<{ label: string; startHour: number; endHour: number; pricePerHour: number }> | undefined;
   571→        let courtPrice = 0;
   572→        if (schedule && schedule.length > 0) {
   573→          courtPrice = calculatePriceForTimeSlot(schedule, startTime, endTime);
   574→          if (courtPrice <= 0) courtPrice = (court.price_per_hour as number) || 0;
   575→        } else {
   576→          courtPrice = (court.price_per_hour as number) || 0;
   577→        }
   578→        courtPriceTotal += courtPrice;
   579→      }
   580→    }
   581→
   582→    const price = courtPriceTotal + equipmentSubtotal;
   583→
   584→    // B4 FIX: Validate advance + remaining === total consistency
   585→    const adv = parseFloat(advanceAmount) || price * 0.5;
   586→    let rem = parseFloat(remainingAmount) || price - adv;
   587→    // Force consistency: remaining must equal total - advance
   588→    rem = Math.max(0, Math.round((price - adv) * 100) / 100);
   589→    const bookingStatus = migrateStatus(status || 'reserved');
   590→
   591→    // Resolve client email for denormalized search
   592→    let clientEmail = authUser.email;
   593→    try {
   594→      const clientUser = await getUserById(userId);
   595→      if (clientUser?.email) clientEmail = clientUser.email;
   596→    } catch { /* fallback to admin email */ }
   597→
   598→    // Booking was already created inside the transaction (P1-5)
   599→    const id = bookingId;
   600→
   601→    // B5 FIX: Only create payment record if advance > 0
   602→    if (adv > 0.01) {
   603→      try {
   604→        await createPayment(id, {
   605→          user_id: userId,
   606→          amount: adv,
   607→          type: 'advance',
   608→          method: normalizedPaymentMethod || 'EFECTIVO',
   609→          status: 'completed',
   610→        });
   611→      } catch (payErr) {
   612→        console.error('[BOOKINGS] Warning: could not create payment record:', payErr);
   613→      }
   614→    }
   615→
   616→    return NextResponse.json({
   617→      id,
   618→      courtId: allCourtIds[0],
   619→      courtIds: allCourtIds,
   620→      userId,
   621→      date,
   622→      startTime,
   623→      endTime,
   624→      totalPrice: price,
   625→      advanceAmount: adv,
   626→      remainingAmount: rem,
   627→      status: bookingStatus,
   628→      paymentMethod: normalizedPaymentMethod || paymentMethod || null,
   629→      success: true,
   630→    }, { status: 201 });
   631→  } catch (error: unknown) {
   632→    const err = error as { message?: string; code?: string; stack?: string };
   633→    console.error('[BOOKINGS] POST error:', err?.message || err);
   634→    console.error('[BOOKINGS] Stack:', err?.stack || 'no stack');
   635→    return NextResponse.json({
   636→      error: 'Failed to create booking',
   637→      detail: err?.message || 'Unknown error'
   638→    }, { status: 500 });
   639→  }
   640→}

   885→

   890→    if (isCancelling) {

   900→      }
   901→

   903→        try {
   904→          if (bookingForCancel && ((bookingForCancel.advance_amount as number) || 0) > 0) {
   905→            // FIX Bug A: Only create retained advance when admin explicitly chose an action
   906→            // If advanceAction is missing (unexpected code path), skip creation to avoid
   907→            // incorrectly defaulting to 'refunded'
   908→            if (!advanceAction) {

   910→              warningFlags.push('no_advance_action');
   911→            } else {
   912→            const advAmount = (bookingForCancel.advance_amount as number) || 0;
   913→            const isRetained = advanceAction === 'retain';
   914→            const statusAdvance = isRetained ? 'retained' : 'refunded';
   915→

   934→              let userEmail = (bookingForCancel.user_email as string) || null;
   935→              try {

   937→                if (uId) {
   938→                  const user = await getUserById(uId);
   939→                  userName = (user?.name as string) || '';
   940→                  // FIX Bug #8: Fallback userEmail from user doc if missing in booking
   941→                  if (!userEmail) {
   942→                    userEmail = (user?.email as string) || null;
   943→                  }
   944→                }
   945→              } catch { /* non-critical */ }
   946→
   947→              // Create retained advance record
   948→              try {
   949→                await createRetainedAdvance({
   950→                  booking_id: id,
   951→                  user_id: (bookingForCancel.user_id as string) || '',
   952→                  user_name: userName,

   955→                  booking_date: (bookingForCancel.date as string) || '',
   956→                  amount: advAmount,
   957→                  original_total: (bookingForCancel.total_price as number) || 0,
   958→                  payment_method: (bookingForCancel.payment_method as string) || 'EFECTIVO',
   959→                  reason: cancelReason || 'Cancelación de reserva',
   960→                  status: statusAdvance,
   961→                });
   962→              } catch (raErr) {
   963→                console.error('[BOOKINGS] Error: could not create retained advance record:', raErr);
   964→                warningFlags.push('retained_advance_failed');
   965→              }
   966→
   967→              // Create a payment record for the refund/retention
   968→              try {
   969→                await createPayment(id, {
   970→                  user_id: (bookingForCancel.user_id as string) || '',
   971→                  amount: advAmount,
   972→                  type: isRetained ? 'retained' : 'refund',
   973→                  method: (bookingForCancel.payment_method as string) || 'EFECTIVO',
   974→                  status: 'completed',
   975→                });
   976→              } catch (payErr) {
   977→                console.error('[BOOKINGS] Warning: could not create payment record for cancellation:', payErr);
   978→                warningFlags.push('payment_record_failed');
   979→              }
   980→            }
   981→            }
   982→          }
   983→        } catch (err) {
   984→          console.error('[BOOKINGS] Warning: could not process retained advance on cancellation:', err);
   985→          warningFlags.push('retained_advance_failed');
   986→        }
   987→      }
   988→    }
   989→
   990→    // B11 FIX: Skip advance payment record when cancelling or editBooking (handled above / no record needed)
   991→    if (typeof reqAdvance === 'number' && typeof reqRemaining === 'number' && !isCancelling && !editBooking) {
   992→      try {
   993→        const booking = await getBookingById(id);
   994→        if (booking) {
   995→          const prevAdvance = (booking.advance_amount as number) || 0;
   996→          const paymentIncrement = reqAdvance - prevAdvance;
   997→          if (paymentIncrement > 0.01) {
   998→            try {
   999→              await createPayment(id, {
  1000→                user_id: booking.user_id as string,
  1001→                amount: paymentIncrement,
  1002→                type: reqRemaining <= 0 ? 'full_payment' : 'advance',
  1003→                method: (updateData.payment_method as string) || 'EFECTIVO',
  1004→                status: 'completed',
  1005→              });
  1006→            } catch (payErr) {
  1007→              console.error('[BOOKINGS] Warning: could not create payment record for advance update:', payErr);
  1008→            }
  1009→          }
  1010→        }
  1011→      } catch { /* non-critical: proceed with booking update */ }
  1012→    }
  1013→
  1014→    await updateBooking(id, updateData);
  1015→
  1016→    // FIX Bug #9: Return warnings if any
  1017→    if (warningFlags.length > 0) {
  1018→      return NextResponse.json({ success: true, warnings: warningFlags });
  1019→    }
  1020→    return NextResponse.json({ success: true });
  1021→  } catch (error) {
  1022→    console.error('[BOOKINGS] PUT error:', error);
  1023→    const msg = error instanceof Error ? error.message : 'Unknown error';
  1024→    return NextResponse.json({ error: 'Error al actualizar reserva', detail: msg }, { status: 500 });
  1025→  }
  1026→}

  1028→// DELETE /api/bookings — permanently delete a booking (super_admin only)
  1029→export async function DELETE(request: NextRequest) {
  1030→  try {
  1031→    const authResult = await requireAuth(request, 'super_admin');
  1032→    if (authResult instanceof NextResponse) return authResult;
  1033→

  1035→      return NextResponse.json({ error: 'Firebase no configurado' }, { status: 503 });

  1037→

  1039→    const id = searchParams.get('id');
  1040→
  1041→    if (!id) {
  1042→      return NextResponse.json({ error: 'Booking ID is required' }, { status: 400 });
  1043→    }
  1044→

  1046→    const { getAdminDb } = await import('@/lib/firebase-admin');
  1047→    const db = await getAdminDb();

  1049→    // Delete payments subcollection
  1050→    try {
  1051→      const paymentsSnapshot = await db.collection('bookings').doc(id).collection('payments').get();
  1052→      for (const doc of paymentsSnapshot.docs) {
  1053→        await doc.ref.delete();
  1054→      }
  1055→    } catch (err) {
  1056→      console.error('[BOOKINGS] Warning: could not delete payments subcollection:', err);

  1058→

  1060→    try {
  1061→      const { getRetainedAdvances, deleteRetainedAdvance } = await import('@/lib/db');
  1062→      const retained = await getRetainedAdvances({ bookingId: id });

  1071→    }
  1072→

  1074→    await deleteDocById('bookings', id);
  1075→
  1076→    console.log('[BOOKINGS] Booking permanently deleted:', id);
  1077→    return NextResponse.json({ success: true });
  1078→  } catch (error) {
