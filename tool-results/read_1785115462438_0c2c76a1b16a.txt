    62→    62→function parseDate(dateStr: string | null): Date {
    63→    63→  if (!dateStr) return new Date()
    64→    64→  const parts = dateStr.split('-')
    65→    65→  if (parts.length === 3) {
    66→    66→    return new Date(parseInt(parts[0], 10), parseInt(parts[1], 10) - 1, parseInt(parts[2], 10))
    67→    67→  }
    68→    68→  // Fallback in case a Date object was stored
    69→    69→  return new Date(dateStr)
    70→    70→}

    72→    72→function formatDateES(date: Date): string {
    73→    73→  return date.toLocaleDateString('es', { weekday: 'long', day: 'numeric', month: 'long', year: 'numeric' })
    74→    74→}

    76→    76→function formatDateShort(date: Date): string {
    77→    77→  return date.toLocaleDateString('es', { day: 'numeric', month: 'short' })

    80→    80→function getRefCode(bookingId: string): string {
    81→    81→  const hash = bookingId.slice(-8).toUpperCase()

    85→    85→/**
    86→    86→ * Calculate price for a time slot using a pricing schedule.
    87→    87→ * Supports slots that span multiple schedules (e.g., 16:00-19:00).
    88→    88→ * Returns { total, breakdown } where breakdown shows per-schedule detail.
    89→    89→ */
    90→    90→function calculatePriceForTimeSlot(
    91→    91→  schedule: PricingScheduleItem[],
    92→    92→  startTime: string,
    93→    93→  endTime: string,
    94→    94→): { total: number; breakdown: Array<{ label: string; hours: number; pricePerHour: number; subtotal: number }> } {
    95→    95→  const [startH, startM] = startTime.split(':').map(Number)
    96→    96→  const [endH, endM] = endTime.split(':').map(Number)
    97→    97→  const startDecimal = startH + startM / 60
    98→    98→  const endDecimal = endH + endM / 60
    99→    99→
   100→   100→  if (schedule.length === 0) return { total: 0, breakdown: [] }
   101→   101→
   102→   102→  const sorted = [...schedule].sort((a, b) => a.startHour - b.startHour)
   103→   103→  let total = 0
   104→   104→  let cursor = startDecimal
   105→   105→  const breakdown: Array<{ label: string; hours: number; pricePerHour: number; subtotal: number }> = []
   106→   106→
   107→   107→  for (const slot of sorted) {
   108→   108→    if (cursor >= slot.endHour) continue
   109→   109→    const overlapStart = Math.max(cursor, slot.startHour)
   110→   110→    const overlapEnd = Math.min(endDecimal, slot.endHour)
   111→   111→    if (overlapEnd > overlapStart) {
   112→   112→      const hours = overlapEnd - overlapStart
   113→   113→      const subtotal = Math.round(hours * slot.pricePerHour * 100) / 100
   114→   114→      breakdown.push({ label: slot.label, hours: Math.round(hours * 100) / 100, pricePerHour: slot.pricePerHour, subtotal })
   115→   115→      total += subtotal
   116→   116→      cursor = overlapEnd
   117→   117→    }
   118→   118→  }
   119→   119→  return { total: Math.round(total * 100) / 100, breakdown }
   120→   120→}
