     5→
     6→interface OccupiedSlot {
     7→  startTime: string
     8→  endTime: string
     9→  label?: string
    10→}
    11→
    12→interface TimeSlotPickerProps {
    13→  /** Currently selected start time (HH:MM) */
    14→  startTime: string
    15→  /** Currently selected end time (HH:MM) */
    16→  endTime: string
    17→  /** Called when user selects a complete range */
    18→  onChange: (startTime: string, endTime: string) => void
    19→  /** Occupied/booking ranges to display */
    20→  occupied?: OccupiedSlot[]
    21→  /** 12-hour format */
    22→  use12hFormat?: boolean
    23→  /** Accent color theme: 'primary' (teal) | 'purple' (edit) */
    24→  theme?: 'primary' | 'purple'
    25→  /** Disabled state */
    26→  disabled?: boolean
    27→  /** Hour range */
    28→  startHour?: number
    29→  endHour?: number
    30→}
    31→
    32→/* ──────────────────────────────────────────────
    33→   Period definitions — each groups a range of hours
    34→   ────────────────────────────────────────────── */
    35→const PERIODS = [
    36→  { key: 'morning',   label: 'MAÑANA', icon: 'light_mode', startH: 6,  endH: 11 },
    37→  { key: 'afternoon', label: 'TARDE',   icon: 'wb_sunny',   startH: 12, endH: 17 },
    38→  { key: 'evening',   label: 'NOCHE',   icon: 'dark_mode',  startH: 18, endH: 23 },
    39→]
    40→
    41→/** Normalize a raw input string to "HH:MM" or return empty */
    42→function normalizeTime(raw: string, startH: number, endH: number): string {
    43→  const trimmed = raw.trim()
    44→  if (!trimmed) return ''
    45→
    46→  let h = -1
    47→  let m = 0
    48→
    49→  // AM/PM format: "6 PM", "6:30pm", "12:00 AM", "12am"
    50→  const ampmMatch = trimmed.match(/^(\d{1,2})(?::(\d{2}))?\s*(am|pm)$/i)
    51→  if (ampmMatch) {
    52→    h = parseInt(ampmMatch[1], 10)
    53→    m = ampmMatch[2] ? parseInt(ampmMatch[2], 10) : 0
    54→    const isPm = ampmMatch[3].toLowerCase() === 'pm'
    55→    if (isPm && h !== 12) h += 12
    56→    if (!isPm && h === 12) h = 0
    57→  }
    58→  // Already in HH:MM
    59→  else if (/^\d{1,2}:\d{2}$/.test(trimmed)) {
    60→    ;[h, m] = trimmed.split(':').map(Number)
    61→  }
    62→  // HHMM (no colon)
    63→  else if (/^\d{3,4}$/.test(trimmed)) {
    64→    const padded = trimmed.padStart(4, '0')
    65→    h = parseInt(padded.slice(0, 2), 10)
    66→    m = parseInt(padded.slice(2, 4), 10)
    67→  }
    68→  // H or HH (hour only → :00)
    69→  else if (/^\d{1,2}$/.test(trimmed)) {
    70→    h = parseInt(trimmed, 10)
    71→    m = 0
    72→  }
    73→
    74→  if (h < startH || h > endH || m < 0 || m > 59) return ''
    75→
    76→  // Snap minutes to 30-min boundary
    77→  const roundedM = m < 15 ? 0 : m < 45 ? 30 : 60
    78→  let adjustedH = roundedM === 60 ? h + 1 : h
    79→  const adjustedM = roundedM === 60 ? 0 : roundedM
    80→  if (adjustedH > endH) return ''
    81→
    82→  return `${String(adjustedH).padStart(2, '0')}:${String(adjustedM).padStart(2, '0')}`
    83→}
    84→
    85→export default function TimeSlotPicker({
    86→  startTime,
    87→  endTime,
    88→  onChange,
    89→  occupied = [],
    90→  use12hFormat = true,
    91→  theme = 'primary',
    92→  disabled = false,
    93→  startHour = 6,
    94→  endHour = 23,
    95→}: TimeSlotPickerProps) {
    96→  const [hoveredSlot, setHoveredSlot] = useState<string | null>(null)
    97→  const [showManual, setShowManual] = useState(false)
    98→  const [rawStart, setRawStart] = useState('')
    99→  const [rawEnd, setRawEnd] = useState('')
   100→  const startInputRef = useRef<HTMLInputElement>(null)
   101→  const endInputRef = useRef<HTMLInputElement>(null)
   102→
   103→  const allSlots = useMemo(
   104→    () => generateTimeSlots(startHour, endHour, [0, 30]),
   105→    [startHour, endHour],
   106→  )
   107→
   108→  const validSlots = useMemo(
   109→    () => new Set(allSlots.map(s => s.value)),
   110→    [allSlots],
   111→  )
   112→
   113→  /* ── occupied set ── */
   114→  const occupiedSet = useMemo(() => {
   115→    const set = new Set<string>()
   116→    for (const occ of occupied) {
   117→      for (const slot of allSlots) {
   118→        if (slot.value >= occ.startTime && slot.value < occ.endTime) {
   119→          set.add(slot.value)
   120→        }
   121→      }
   122→    }
   123→    return set
   124→  }, [occupied, allSlots])
   125→
   126→  /* ── derived booleans ── */
   127→  const hasStart = !!startTime
   128→  const hasEnd   = !!endTime && endTime > startTime
   129→  const fmt      = (v: string) => (use12hFormat ? formatTime12(v) : formatTime24(v))
   130→  const isOcc    = (v: string) => occupiedSet.has(v)
   131→  const isSel    = (v: string) => hasStart && hasEnd && v >= startTime && v < endTime
   132→  const isSrt    = (v: string) => v === startTime && hasEnd
   133→  const isEndM   = (v: string) => v === endTime   && hasEnd
   134→  const isRng    = (v: string) => hasStart && !hasEnd && v >= startTime
   135→
   136→  /* ── click handler (two-tap range) ── */
   137→  const handleClick = useCallback(
   138→    (v: string) => {
   139→      if (disabled || isOcc(v)) return
   140→      if (!hasStart || (hasStart && hasEnd)) {
   141→        onChange(v, '')
   142→      } else if (v <= startTime) {
   143→        onChange(v, '')
   144→      } else {
   145→        onChange(startTime, v)
   146→      }
   147→    },
   148→    [disabled, hasStart, hasEnd, startTime, isOcc, onChange],
   149→  )
   150→
   151→  /* ── manual input handlers ── */
   152→  const handleManualApply = useCallback(() => {
   153→    const s = normalizeTime(rawStart, startHour, endHour)
   154→    const e = normalizeTime(rawEnd, startHour, endHour)
   155→    if (s && e && e > s) {
   156→      onChange(s, e)
   157→    } else if (s) {
   158→      onChange(s, '')
   159→    }
   160→  }, [rawStart, rawEnd, startHour, endHour, onChange])
   161→
   162→  const handleStartKeyDown = (e: React.KeyboardEvent) => {
   163→    if (e.key === 'Tab' && !e.shiftKey) {
   164→      // default tab → go to end input
   165→    } else if (e.key === 'Enter') {
   166→      e.preventDefault()
   167→      if (rawStart) {
   168→        const s = normalizeTime(rawStart, startHour, endHour)
   169→        if (s) {
   170→          onChange(s, '')
   171→          setRawStart('')
   172→          endInputRef.current?.focus()
   173→        }
   174→      }
   175→    }
   176→  }
   177→
   178→  const handleEndKeyDown = (e: React.KeyboardEvent) => {
   179→    if (e.key === 'Enter') {
   180→      e.preventDefault()
   181→      handleManualApply()
   182→    }
   183→  }
   184→
   185→  // Sync manual inputs when selection changes externally (e.g. grid click)
   186→  useEffect(() => {
   187→    if (showManual) {
   188→      setRawStart(hasStart ? fmt(startTime) : '')
   189→      setRawEnd(hasEnd ? fmt(endTime) : '')
   190→    }
   191→  }, [startTime, endTime, showManual, hasStart, hasEnd, fmt])
   192→
   193→  /* ── theme tokens ── */
   194→  const isPurple = theme === 'purple'
   195→  const accent   = isPurple ? '#a855f7' : '#00ff41'
   196→  const rgb      = isPurple ? '168,85,247' : '0,255,65'
   197→
   198→  /* ── duration helpers ── */
   199→  const duration = useMemo(() => {
   200→    if (!hasStart || !hasEnd) return null
   201→    const [sh, sm] = startTime.split(':').map(Number)
   202→    const [eh, em] = endTime.split(':').map(Number)
   203→    return (eh * 60 + em) - (sh * 60 + sm)
   204→  }, [startTime, endTime, hasStart, hasEnd])
   205→
   206→  const durationLabel = duration !== null
   207→    ? duration >= 60
   208→      ? `${Math.floor(duration / 60)}h${duration % 60 ? ` ${duration % 60}min` : ''}`
   209→      : `${duration} min`
   210→    : null
   211→
   212→  const selectedCount = hasStart && hasEnd
   213→    ? allSlots.filter(s => s.value >= startTime && s.value < endTime).length
   214→    : 0
   215→
   216→  /* ── group slots by period ── */
   217→  const periods = useMemo(
   218→    () =>
   219→      PERIODS.filter(p => p.endH >= startHour && p.startH <= endHour).map(p => ({
   220→        ...p,
   221→        slots: allSlots.filter(s => {
   222→          const h = parseInt(s.value.split(':')[0], 10)
   223→          return h >= p.startH && h <= p.endH
   224→        }),
   225→      })),
   226→    [allSlots, startHour, endHour],
   227→  )
   228→
   229→  /* ══════════════ common input style ══════════════ */
   230→  const inputBaseStyle: React.CSSProperties = {
   231→    background: `rgba(${rgb},0.04)`,
   232→    border: `1px solid rgba(${rgb},0.15)`,
   233→    color: accent,
   234→  }
   235→  const inputFocusClass = `focus:outline-none focus:border-[rgba(${rgb},0.5)] focus:shadow-[0_0_12px_rgba(${rgb},0.08)]`
   236→
   237→  /* ─────────────────── RENDER ─────────────────── */
   238→  return (
   239→    <div className="space-y-3">
   240→      {/* ═══ Selection info bar ═══ */}
   241→      <div
   242→        className="relative overflow-hidden rounded-xl px-3 py-2.5"
   243→        style={{
   244→          background: `linear-gradient(135deg, rgba(${rgb},0.07), rgba(${rgb},0.02))`,
   245→          border: `1px solid rgba(${rgb},0.12)`,
   246→        }}
   247→      >
   248→        {/* scanline decoration */}
   249→        <div className="pointer-events-none absolute inset-0 overflow-hidden">
   250→          <div
   251→            className="absolute inset-y-0 w-10"
   252→            style={{
   253→              background: `linear-gradient(90deg, transparent, rgba(${rgb},0.07), transparent)`,
   254→              animation: 'tsp-scan 4s linear infinite',
   255→            }}
   256→          />
   257→        </div>
   258→
   259→        <div className="relative flex items-center justify-between gap-2">
   260→          <div className="flex items-center gap-2 min-w-0">
   261→            <span
   262→              className="material-symbols-outlined text-[16px] flex-shrink-0"
   263→              style={{ color: accent, fontVariationSettings: '"FILL" 1' }}
   264→            >
   265→              schedule
   266→            </span>
   267→
   268→            {hasStart && hasEnd ? (
   269→              <div className="flex items-center gap-2 flex-wrap">
   270→                <span
   271→                  className="text-sm font-bold font-[family-name:var(--font-sora)] whitespace-nowrap"
   272→                  style={{ color: accent }}
   273→                >
   274→                  {fmt(startTime)} — {fmt(endTime)}
   275→                </span>
   276→                {durationLabel && (
   277→                  <span
   278→                    className="text-[10px] font-semibold font-[family-name:var(--font-inter)] px-2 py-0.5 rounded-full whitespace-nowrap"
   279→                    style={{ background: `rgba(${rgb},0.15)`, color: accent }}
   280→                  >
   281→                    {durationLabel}
   282→                  </span>
   283→                )}
   284→                <span className="text-[10px] text-cm-on-surface-variant/30 font-[family-name:var(--font-inter)]">
   285→                  {selectedCount} bloque{selectedCount !== 1 ? 's' : ''}
   286→                </span>
   287→              </div>
   288→            ) : hasStart ? (
   289→              <span className="text-xs text-cm-on-surface-variant font-[family-name:var(--font-inter)] truncate">
   290→                Inicio:{' '}
   291→                <span style={{ color: accent }}>{fmt(startTime)}</span>
   292→                <span className="text-cm-on-surface-variant/40"> → toca la hora de fin</span>
   293→              </span>
   294→            ) : (
   295→              <span className="text-xs text-cm-on-surface-variant/40 font-[family-name:var(--font-inter)]">
   296→                Selecciona un horario
   297→              </span>
   298→            )}
   299→          </div>
   300→
   301→          <div className="flex items-center gap-1.5 flex-shrink-0">
   302→            {/* Manual toggle */}
   303→            <button
   304→              type="button"
   305→              onClick={() => {
   306→                setShowManual(prev => !prev)
   307→                if (!showManual) {
   308→                  // opening manual → pre-fill
   309→                  setRawStart(hasStart ? fmt(startTime) : '')
   310→                  setRawEnd(hasEnd ? fmt(endTime) : '')
   311→                  setTimeout(() => startInputRef.current?.focus(), 50)
   312→                }
   313→              }}
   314→              title="Ingreso manual de horario"
   315→              className="p-1 rounded-lg transition-all duration-200 hover:bg-white/5"
   316→              style={{
   317→                color: showManual ? accent : `rgba(${rgb},0.5)`,
   318→                background: showManual ? `rgba(${rgb},0.1)` : undefined,
   319→                border: showManual ? `1px solid rgba(${rgb},0.2)` : '1px solid transparent',
   320→              }}
   321→            >
   322→              <span className="material-symbols-outlined text-[16px]">edit</span>
   323→            </button>
   324→
   325→            {hasStart && hasEnd && (
   326→              <button
   327→                type="button"
   328→                onClick={() => onChange('', '')}
   329→                className="text-cm-on-surface-variant/40 hover:text-red-400 transition-colors flex items-center gap-0.5"
   330→              >
   331→                <span className="material-symbols-outlined text-[14px]">close</span>
   332→                <span className="text-[10px] font-[family-name:var(--font-inter)] hidden sm:inline">
   333→                  Limpiar
   334→                </span>
   335→              </button>
   336→            )}
   337→          </div>
   338→        </div>
   339→      </div>
   340→
   341→      {/* ═══ Manual input panel ═══ */}
   342→      {showManual && (
   343→        <div
   344→          className="rounded-xl p-3 space-y-2.5 transition-all duration-300"
   345→          style={{
   346→            background: `linear-gradient(135deg, rgba(${rgb},0.04), rgba(${rgb},0.01))`,
   347→            border: `1px solid rgba(${rgb},0.1)`,
   348→          }}
   349→        >
   350→          <div className="flex items-center gap-2 mb-1">
   351→            <span
   352→              className="material-symbols-outlined text-[12px]"
   353→              style={{ color: `rgba(${rgb},0.4)`, fontVariationSettings: '"FILL" 1' }}
   354→            >
   355→              keyboard
   356→            </span>
   357→            <span
   358→              className="text-[9px] font-bold font-[family-name:var(--font-inter)] uppercase tracking-[0.2em]"
   359→              style={{ color: `rgba(${rgb},0.4)` }}
   360→            >
   361→              Ingreso manual
   362→            </span>
   363→            <div
   364→              className="flex-1 h-px"
   365→              style={{ background: `linear-gradient(to right, rgba(${rgb},0.08), transparent)` }}
   366→            />
   367→            <span className="text-[8px] text-cm-on-surface-variant/25 font-[family-name:var(--font-inter)]">
   368→              Enter para confirmar
   369→            </span>
   370→          </div>
   371→
   372→          <div className="grid grid-cols-2 gap-2">
   373→            {/* Start time input */}
   374→            <div className="space-y-1">
   375→              <label
   376→                className="text-[9px] font-bold font-[family-name:var(--font-inter)] uppercase tracking-[0.15em] block px-0.5"
   377→                style={{ color: `rgba(${rgb},0.5)` }}
   378→              >
   379→                Inicio
   380→              </label>
   381→              <input
   382→                ref={startInputRef}
   383→                type="text"
   384→                inputMode="text"
   385→                placeholder={use12hFormat ? 'Ej: 6 PM o 18:00' : 'Ej: 18:00 o 1800'}
   386→                value={rawStart}
   387→                onChange={e => setRawStart(e.target.value)}
   388→                onKeyDown={handleStartKeyDown}
   389→                disabled={disabled}
   390→                className={[
   391→                  'w-full px-3 py-2 rounded-lg text-sm font-semibold font-[family-name:var(--font-inter)]',
   392→                  'placeholder:text-cm-on-surface-variant/20 placeholder:font-normal placeholder:text-[11px]',
   393→                  'transition-all duration-200',
   394→                  inputFocusClass,
   395→                ].join(' ')}
   396→                style={inputBaseStyle}
   397→              />
   398→            </div>
   399→
   400→            {/* End time input */}
   401→            <div className="space-y-1">
   402→              <label
   403→                className="text-[9px] font-bold font-[family-name:var(--font-inter)] uppercase tracking-[0.15em] block px-0.5"
   404→                style={{ color: `rgba(${rgb},0.5)` }}
   405→              >
   406→                Fin
   407→              </label>
   408→              <input
   409→                ref={endInputRef}
   410→                type="text"
   411→                inputMode="text"
   412→                placeholder={use12hFormat ? 'Ej: 8 PM o 20:00' : 'Ej: 20:00 o 2000'}
   413→                value={rawEnd}
   414→                onChange={e => setRawEnd(e.target.value)}
   415→                onKeyDown={handleEndKeyDown}
   416→                disabled={disabled}
   417→                className={[
   418→                  'w-full px-3 py-2 rounded-lg text-sm font-semibold font-[family-name:var(--font-inter)]',
   419→                  'placeholder:text-cm-on-surface-variant/20 placeholder:font-normal placeholder:text-[11px]',
   420→                  'transition-all duration-200',
   421→                  inputFocusClass,
   422→                ].join(' ')}
   423→                style={inputBaseStyle}
   424→              />
   425→            </div>
   426→          </div>
   427→
   428→          {/* Apply button */}
   429→          <div className="flex items-center justify-between pt-0.5">
   430→            <p className="text-[9px] text-cm-on-surface-variant/25 font-[family-name:var(--font-inter)] leading-tight">
   431→              Formatos: 18:00, 1800, 6 PM, 18
   432→            </p>
   433→            <button
   434→              type="button"
   435→              onClick={handleManualApply}
   436→              disabled={disabled || (!rawStart && !rawEnd)}
   437→              className="flex items-center gap-1.5 px-3 py-1.5 rounded-lg text-[11px] font-semibold font-[family-name:var(--font-inter)] transition-all duration-200 disabled:opacity-30 disabled:cursor-not-allowed active:scale-[0.97]"
   438→              style={{
   439→                background: `rgba(${rgb},0.12)`,
   440→                border: `1px solid rgba(${rgb},0.25)`,
   441→                color: accent,
   442→              }}
   443→            >
   444→              <span className="material-symbols-outlined text-[14px]" style={{ fontVariationSettings: '"FILL" 1' }}>
   445→                check_circle
   446→              </span>
   447→              Aplicar
   448→            </button>
   449→          </div>
   450→        </div>
   451→      )}
   452→
   453→      {/* ═══ Period grids ═══ */}
   454→      <div className="space-y-4">
   455→        {periods.map((period) => (
   456→          <div key={period.key}>
   457→            {/* period header */}
   458→            <div className="flex items-center gap-2 mb-2 px-0.5">
   459→              <span
   460→                className="material-symbols-outlined text-[12px]"
   461→                style={{ color: `rgba(${rgb},0.35)`, fontVariationSettings: '"FILL" 1' }}
   462→              >
   463→                {period.icon}
   464→              </span>
   465→              <span
   466→                className="text-[9px] font-bold font-[family-name:var(--font-inter)] uppercase tracking-[0.2em]"
   467→                style={{ color: `rgba(${rgb},0.35)` }}
   468→              >
   469→                {period.label}
   470→              </span>
   471→              <div
   472→                className="flex-1 h-px"
   473→                style={{
   474→                  background: `linear-gradient(to right, rgba(${rgb},0.1), transparent)`,
   475→                }}
   476→              />
   477→            </div>
   478→
   479→            {/* slot grid — 3 cols mobile, 4 sm, 6 md+ */}
   480→            <div className="grid grid-cols-3 sm:grid-cols-4 md:grid-cols-6 gap-1.5">
   481→              {period.slots.map((ts) => {
   482→                const occ = isOcc(ts.value)
   483→                const sel = isSel(ts.value)
   484→                const srt = isSrt(ts.value)
   485→                const end = isEndM(ts.value)
   486→                const inR = isRng(ts.value)
   487→                const hov = hoveredSlot === ts.value && !disabled
   488→
   489→                const hour   = parseInt(ts.value.split(':')[0], 10)
   490→                const minute = ts.value.split(':')[1]
   491→                const isHourSlot = minute === '00'
   492→
   493→                /* display label */
   494→                let label: string
   495→                if (use12hFormat) {
   496→                  label = isHourSlot
   497→                    ? formatTime12(ts.value).replace(':00', '')
   498→                    : `:${minute}`
   499→                } else {
   500→                  label = isHourSlot ? `${hour}:00` : `:${minute}`
   501→                }
   502→
   503→                const ampm =
   504→                  use12hFormat && isHourSlot ? (hour < 12 ? 'AM' : 'PM') : ''
   505→
   506→                /* ── cell styles ── */
   507→                let cellStyle: React.CSSProperties
   508→                let labelColor: string | undefined
   509→
   510→                if (occ) {
   511→                  cellStyle = {
   512→                    background: 'rgba(239,68,68,0.05)',
   513→                    border: '1px solid rgba(239,68,68,0.1)',
   514→                  }
   515→                  labelColor = undefined
   516→                } else if (sel) {
   517→                  cellStyle = {
   518→                    background: `linear-gradient(135deg, rgba(${rgb},0.22), rgba(${rgb},0.08))`,
   519→                    border: `1px solid rgba(${rgb},0.45)`,
   520→                    boxShadow: `0 0 20px rgba(${rgb},0.1), inset 0 0 20px rgba(${rgb},0.05)`,
   521→                  }
   522→                  labelColor = accent
   523→                } else if (inR) {
   524→                  cellStyle = {
   525→                    background: `rgba(${rgb},0.06)`,
   526→                    border: `1px dashed rgba(${rgb},0.22)`,
   527→                  }
   528→                  labelColor = accent
   529→                } else if (hov) {
   530→                  cellStyle = {
   531→                    background: `rgba(${rgb},0.08)`,
   532→                    border: `1px solid rgba(${rgb},0.2)`,
   533→                  }
   534→                  labelColor = accent
   535→                } else {
   536→                  cellStyle = {
   537→                    background: 'rgba(255,255,255,0.015)',
   538→                    border: '1px solid rgba(255,255,255,0.04)',
   539→                  }
   540→                  labelColor = undefined
   541→                }
   542→
   543→                return (
   544→                  <button
   545→                    key={ts.value}
   546→                    type="button"
   547→                    disabled={disabled || occ}
   548→                    onClick={() => handleClick(ts.value)}
   549→                    onMouseEnter={() => setHoveredSlot(ts.value)}
   550→                    onMouseLeave={() => setHoveredSlot(null)}
   551→                    className={[
   552→                      'relative rounded-xl text-center py-2.5 sm:py-3',
   553→                      'transition-all duration-200 font-[family-name:var(--font-inter)]',
   554→                      occ ? 'cursor-not-allowed' : 'cursor-pointer active:scale-[0.96]',
   555→                    ].join(' ')}
   556→                    style={cellStyle}
   557→                    title={occ ? 'Ocupado' : fmt(ts.value)}
   558→                  >
   559→                    {/* primary label */}
   560→                    <span
   561→                      className={[
   562→                        'block text-[13px] sm:text-sm font-bold leading-none',
   563→                        occ ? 'text-red-400/40' : labelColor ? '' : 'text-cm-on-surface/60',
   564→                      ].join(' ')}
   565→                      style={!occ && labelColor ? { color: labelColor } : undefined}
   566→                    >
   567→                      {occ ? (
   568→                        <span
   569→                          className="material-symbols-outlined text-[16px]"
   570→                          style={{ fontVariationSettings: '"FILL" 1' }}
   571→                        >
   572→                          block
   573→                        </span>
   574→                      ) : (
   575→                        label
   576→                      )}
   577→                    </span>
   578→
   579→                    {/* AM/PM sub-label for hour slots */}
   580→                    {!occ && ampm && (
   581→                      <span
   582→                        className="block text-[7px] mt-1 font-[family-name:var(--font-inter)] font-semibold"
   583→                        style={{ color: `rgba(${rgb},0.45)` }}
   584→                      >
   585→                        {ampm}
   586→                      </span>
   587→                    )}
   588→
   589→                    {/* ":30" sub-label for half-hour slots */}
   590→                    {!occ && !isHourSlot && (
   591→                      <span className="block text-[7px] mt-1 font-[family-name:var(--font-inter)] text-cm-on-surface-variant/20">
   592→                        {minute}
   593→                      </span>
   594→                    )}
   595→
   596→                    {/* start marker */}
   597→                    {srt && (
   598→                      <div
   599→                        className="absolute left-0 top-1/2 -translate-y-1/2 w-[3px] h-5 rounded-r-full"
   600→                        style={{
   601→                          background: accent,
   602→                          boxShadow: `0 0 10px ${accent}`,
   603→                          animation: 'tsp-pulse 2s ease-in-out infinite',
   604→                        }}
   605→                      />
   606→                    )}
   607→
   608→                    {/* end marker */}
   609→                    {end && (
   610→                      <div
   611→                        className="absolute right-0 top-1/2 -translate-y-1/2 w-[3px] h-5 rounded-l-full"
   612→                        style={{
   613→                          background: accent,
   614→                          boxShadow: `0 0 10px ${accent}`,
   615→                          animation: 'tsp-pulse 2s ease-in-out infinite',
   616→                        }}
   617→                      />
   618→                    )}
   619→
   620→                    {/* selected glow overlay */}
   621→                    {sel && !occ && (
   622→                      <div
   623→                        className="absolute inset-0 rounded-xl pointer-events-none"
   624→                        style={{
   625→                          background: `radial-gradient(ellipse at center, rgba(${rgb},0.08), transparent 70%)`,
   626→                        }}
   627→                      />
   628→                    )}
   629→                  </button>
   630→                )
   631→              })}
   632→            </div>
   633→          </div>
   634→        ))}
   635→      </div>
   636→
   637→      {/* ═══ Legend ═══ */}
   638→      <div className="flex items-center gap-4 pt-1 flex-wrap">
   639→        <div className="flex items-center gap-1.5">
   640→          <div
   641→            className="w-3 h-3 rounded"
   642→            style={{
   643→              background: 'rgba(255,255,255,0.015)',
   644→              border: '1px solid rgba(255,255,255,0.04)',
   645→            }}
   646→          />
   647→          <span className="text-[9px] text-cm-on-surface-variant/30 font-[family-name:var(--font-inter)]">
   648→            Disponible
   649→          </span>
   650→        </div>
   651→        <div className="flex items-center gap-1.5">
   652→          <div
   653→            className="w-3 h-3 rounded"
   654→            style={{
   655→              background: `linear-gradient(135deg, rgba(${rgb},0.22), rgba(${rgb},0.08))`,
   656→              border: `1px solid rgba(${rgb},0.45)`,
   657→              boxShadow: `0 0 6px rgba(${rgb},0.15)`,
   658→            }}
   659→          />
   660→          <span className="text-[9px] text-cm-on-surface-variant/30 font-[family-name:var(--font-inter)]">
   661→            Seleccionado
   662→          </span>
   663→        </div>
   664→        <div className="flex items-center gap-1.5">
   665→          <div
   666→            className="w-3 h-3 rounded"
   667→            style={{
   668→              background: 'rgba(239,68,68,0.05)',
   669→              border: '1px solid rgba(239,68,68,0.1)',
   670→            }}
   671→          />
   672→          <span className="text-[9px] text-cm-on-surface-variant/30 font-[family-name:var(--font-inter)]">
   673→            Ocupado
   674→          </span>
   675→        </div>
   676→      </div>
   677→    </div>
   678→  )
   679→}
