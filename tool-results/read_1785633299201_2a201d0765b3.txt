    19→// Migrate old status values to the new 3-status system
    20→function migrateStatus(s: string): string {
    21→  switch (s) {
    22→    case 'confirmed':
    23→    case 'pending':
    24→    case 'partially_paid':
    25→      return 'reserved';
    26→    case 'fully_paid':
    27→    case 'completed':
    28→      return 'completed';
    29→    case 'no_show':
    30→    case 'expired':
    31→      return 'cancelled';
    32→    default:
    33→      return s; // 'reserved', 'completed', 'cancelled' pass through
    34→  }
    35→}

    37→// Transformar snake_case (Firestore) a camelCase (frontend)
    38→function toCamelBooking(b: Record<string, unknown>) {
    39→  // Primary court (backward compat)
    40→  const courtRaw = b._court as Record<string, unknown> | null;
    41→  const court = courtRaw ? {
    42→    id: courtRaw.id || courtRaw.court_id,
    43→    name: courtRaw.name || courtRaw.court_name,
    44→    sport: courtRaw.sport,
    45→    branch: (courtRaw.branch as Record<string, unknown>) || {
    46→      id: courtRaw.branch_id || 'branch-1',
    47→      name: 'CREARD',
    48→    },
    49→  } : null;
    50→
    51→  // Additional courts (multi-court bookings)
    52→  const courtsRaw = b._courts as Array<Record<string, unknown>> | null;
    53→  const courts = courtsRaw ? courtsRaw.map((cr) => ({
    54→    id: cr.id || cr.court_id,
    55→    name: cr.name || cr.court_name,
    56→    sport: cr.sport,
    57→    branch: (cr.branch as Record<string, unknown>) || { id: 'branch-1', name: 'CREARD' },
    58→  })) : [];
    59→
    60→  const userRaw = b._user as Record<string, unknown> | null;
    61→  const user = userRaw ? {
    62→    id: userRaw.id,
    63→    name: userRaw.name,
    64→    email: userRaw.email,
    65→    phone: userRaw.phone,
    66→  } : null;
    67→
    68→  // Court IDs array
    69→  const courtIds: string[] = Array.isArray(b.court_ids) ? b.court_ids : (b.court_id ? [b.court_id] : []);
    70→
    71→  // Equipment items
    72→  const equipmentItems = Array.isArray(b.equipment_items) ? b.equipment_items.map((eq: Record<string, unknown>) => ({
    73→    equipmentId: eq.equipment_id || eq.id,
    74→    name: eq.name,
    75→    quantity: eq.quantity || 1,
    76→    unitPrice: eq.unit_price || eq.price_per_rental || 0,
    77→    subtotal: (eq.quantity || 1) * (eq.unit_price || eq.price_per_rental || 0),
    78→  })) : [];
    79→
    80→  // Normalize: recalc equipment subtotal from items (more reliable than stored value)
    81→  const computedEquipSubtotal = equipmentItems.reduce((s: number, eq: { subtotal: number }) => s + eq.subtotal, 0);
    82→  const equipmentSubtotal = computedEquipSubtotal;
    83→
    84→  // Normalize court_subtotal: if stored value is wrong (old double-counting bug),
    85→  // derive it from total_price - equipment_subtotal
    86→  const totalPrice = b.total_price || 0;
    87→  const storedCourtSubtotal = b.court_subtotal || 0;
    88→  const courtSubtotal = (storedCourtSubtotal + equipmentSubtotal > totalPrice + 0.01)
    89→    ? Math.max(0, Math.round((totalPrice - equipmentSubtotal) * 100) / 100)
    90→    : storedCourtSubtotal;
    91→
    92→  // Normalize remaining_amount: ensure it equals total - advance
    93→  const advanceAmount = b.advance_amount || 0;
    94→  const storedRemaining = b.remaining_amount || 0;
    95→  const remainingAmount = (Math.abs(storedRemaining - (totalPrice - advanceAmount)) > 0.01)
    96→    ? Math.max(0, Math.round((totalPrice - advanceAmount) * 100) / 100)
    97→    : storedRemaining;
    98→
    99→  return {
   100→    id: b.id,
   101→    courtId: b.court_id,
   102→    courtIds,
   103→    userId: b.user_id,
   104→    date: typeof b.date === 'string' ? b.date : (b.date ? String(b.date) : ''),
   105→    startTime: b.start_time,
   106→    endTime: b.end_time,
   107→    totalPrice,
   108→    courtSubtotal,
   109→    equipmentSubtotal,
   110→    equipmentItems,
   111→    advanceAmount,
   112→    remainingAmount,
   113→    status: migrateStatus(b.status || 'reserved'),
   114→    slotStatus: b.slot_status,
   115→    paymentMethod: b.payment_method,
   116→    remainingPaymentStatus: b.remaining_payment_status || null,
   117→    notes: b.notes,
   118→    createdAt: b.created_at,
   119→    updatedAt: b.updated_at,
   120→    recurringGroupId: b.recurring_group_id,
   121→    recurringIndex: b.recurring_index,
   122→    equipmentDelivered: b.equipment_delivered || false,
   123→    equipmentReturned: b.equipment_returned || false,
   124→    court,
   125→    courts,
   126→    user,
   127→  };
   128→}

   130→// Enrich a single booking (all courts + user), safe — never throws
   131→async function safeEnrichBooking(b: Record<string, unknown>): Promise<Record<string, unknown>> {
   132→  let court: Record<string, unknown> | null = null;
   133→  let courts: Array<Record<string, unknown>> = [];
   134→  let user: Record<string, unknown> | null = null;
   135→
   136→  // Get all court IDs (from court_ids array or single court_id)
   137→  const allCourtIds: string[] = Array.isArray(b.court_ids)
   138→    ? b.court_ids as string[]
   139→    : (b.court_id ? [b.court_id as string] : []);
   140→
   141→  // Enrich primary court (first one)
   142→  try {
   143→    if (allCourtIds.length > 0) {
   144→      const c = await getCourtById(allCourtIds[0]);
   145→      if (c) court = c as Record<string, unknown>;
   146→    }
   147→  } catch { /* court enrichment failed */ }
   148→
   149→
   150→  // Enrich all courts for multi-court bookings
   151→  if (allCourtIds.length > 1) {
   152→    const enrichedCourts: Array<Record<string, unknown>> = [];
   153→    for (const cid of allCourtIds) {
   154→      try {
   155→        const c = await getCourtById(cid);
   156→        if (c) enrichedCourts.push(c as Record<string, unknown>);
   157→      } catch { /* skip failed court enrichment */ }
   158→    }
   159→    courts = enrichedCourts;
   160→  }
   161→
   162→  try {
   163→    if (b.user_id) {
   164→      const u = await getUserById(b.user_id as string);
   165→      if (u) user = u as Record<string, unknown>;
   166→    }
   167→  } catch { /* user enrichment failed */ }
   168→
   169→
   170→  return { ...b, _court: court, _courts: courts, _user: user };
   171→}

   173→// Search bookings by userId OR userEmail, deduplicate
   174→async function searchBookingsForUser(userId: string, userEmail: string | null) {
   175→  const results = new Map<string, Record<string, unknown>>();
   176→
   177→  // Always search by userId
   178→  try {
   179→    const byId = await getBookings({ userId });
   180→    for (const b of byId) {
   181→      if (b.id && !results.has(b.id)) {
   182→        results.set(b.id, b as Record<string, unknown>);
   183→      }
   184→    }
   185→  } catch (e) {
   186→    console.error('[BOOKINGS] Error searching by userId:', userId, e);
   187→  }
   188→
   189→  // Also search by email to handle demo/Firebase ID mismatches
   190→  if (userEmail && userEmail !== userId) {
   191→    try {
   192→      const byEmail = await getBookings({ userEmail });
   193→      for (const b of byEmail) {
   194→        if (b.id && !results.has(b.id)) {
   195→          results.set(b.id, b as Record<string, unknown>);
   196→        }
   197→      }
   198→    } catch (e) {
   199→      console.error('[BOOKINGS] Error searching by userEmail:', userEmail, e);
   200→    }
   201→  }
   202→
   203→  return Array.from(results.values());
   204→}

   206→/**
   207→ * Calculate the total price for a time slot using a pricing schedule.
   208→ * Supports slots that span multiple schedules (e.g., 16:00-19:00 = 1hr morning + 2hr night).
   209→ */
   210→function calculatePriceForTimeSlot(
   211→  schedule: Array<{ label: string; startHour: number; endHour: number; pricePerHour: number }>,
   212→  startTime: string,
   213→  endTime: string,
   214→): number {
   215→  const [startH, startM] = startTime.split(':').map(Number);
   216→  const [endH, endM] = endTime.split(':').map(Number);
   217→  const startDecimal = startH + startM / 60;
   218→  const endDecimal = endH + endM / 60;
   219→
   220→  // Sort schedule by startHour
   221→  const sorted = [...schedule].sort((a, b) => a.startHour - b.startHour);
   222→
   223→  let total = 0;
   224→  let cursor = startDecimal;
   225→
   226→  for (const slot of sorted) {
   227→    const slotStart = slot.startHour;
   228→    const slotEnd = slot.endHour;
   229→
   230→    if (cursor >= slotEnd) continue; // cursor already past this slot
   231→
   232→    // Overlap start/end
   233→    const overlapStart = Math.max(cursor, slotStart);
   234→    const overlapEnd = Math.min(endDecimal, slotEnd);
   235→
   236→    if (overlapEnd > overlapStart) {
   237→      const hours = overlapEnd - overlapStart;
   238→      total += hours * slot.pricePerHour;
   239→      cursor = overlapEnd;
   240→    }
   241→  }
   242→
   243→  // If cursor didn't reach endDecimal, the remaining time is uncovered by schedule
   244→  // (fall back to 0 or caller's default)
   245→  return Math.round(total * 100) / 100;
   246→}

   248→export async function GET(request: NextRequest) {
   249→  const startTime = Date.now();
   250→  const firebaseOk = isFirebaseAvailable();
   251→
   252→  try {
   253→    const { searchParams } = new URL(request.url);
   254→    const courtId = searchParams.get('courtId');
   255→    const userId = searchParams.get('userId');
   256→    const date = searchParams.get('date');
   257→    const dateFrom = searchParams.get('dateFrom');
   258→    const dateTo = searchParams.get('dateTo');
   259→    const status = searchParams.get('status');
   260→
   261→    // Court availability check (courtId + date) is public — no auth required
   262→    if (courtId && date && !userId) {
   263→      if (!firebaseOk) {
   264→        return NextResponse.json({ error: 'Firebase no configurado', code: 'NO_FIREBASE' }, { status: 503 });
   265→      }
   266→      const bookings = await getBookings({ courtId, date });
   267→      // FIX P0-4: Filter out expired reserved bookings (ghost reservations)
   268→      const now = Date.now();
   269→      const active = bookings.filter((b: any) => {
   270→        if (migrateStatus(b.status || 'reserved') !== 'reserved') return true;
   271→        const exp = b.expires_at;
   272→        if (!exp) return true;
   273→        const expMs = exp.toMillis?.()?.() || new Date(exp).getTime();
   274→        return expMs > now;
   275→      });
   276→      return NextResponse.json(active.map(toCamelBooking));
   277→    }
   278→
   279→    // All other queries require authentication
   280→    const authResult = await requireAnyAuth(request);
   281→    if (authResult instanceof NextResponse) return authResult;
   282→    const authUser = authResult.user;
   283→
   284→    if (!firebaseOk) {
   285→      return NextResponse.json({ error: 'Firebase no configurado', code: 'NO_FIREBASE' }, { status: 503 });
   286→    }
   287→
   288→    let bookings: Record<string, unknown>[];
   289→
   290→    // Non-admin users: only their own bookings
   291→    if (authUser.role !== 'admin' && authUser.role !== 'super_admin') {
   292→      if (userId && userId !== authUser.id) {
   293→        return NextResponse.json({ error: 'No puedes ver reservas de otros usuarios.' }, { status: 403 });
   294→      }
   295→      const effectiveUserId = userId || authUser.id;
   296→      bookings = await searchBookingsForUser(effectiveUserId, authUser.email);
   297→    } else {
   298→      // Admin: fetch with optional filters (date range for performance)
   299→      bookings = (await getBookings({
   300→        courtId: courtId || undefined,
   301→        userId: userId || undefined,
   302→        date: date || undefined,
   303→        dateFrom: dateFrom || undefined,
   304→        dateTo: dateTo || undefined,
   305→        status: status || undefined,
   306→      })) as Record<string, unknown>[];
   307→      // DEBUG removed (P3-16): raw booking data no longer logged
   308→    }
   309→
   310→    // Enrich each booking with cached court/user lookups (avoid N+1 Firestore reads)
   311→    const courtCache = new Map<string, Record<string, unknown> | null>();
   312→    const userCache = new Map<string, Record<string, unknown> | null>();
   313→    const getCourtCached = async (cid: string) => {
   314→      if (courtCache.has(cid)) return courtCache.get(cid)!;
   315→      try {
   316→        const c = await getCourtById(cid);
   317→        courtCache.set(cid, c as Record<string, unknown> | null);
   318→        return c as Record<string, unknown> | null;
   319→      } catch { courtCache.set(cid, null); return null; }
   320→    };
   321→    const getUserCached = async (uid: string) => {
   322→      if (userCache.has(uid)) return userCache.get(uid)!;
   323→      try {
   324→        const u = await getUserById(uid);
   325→        userCache.set(uid, u as Record<string, unknown> | null);
   326→        return u as Record<string, unknown> | null;
   327→      } catch { userCache.set(uid, null); return null; }
   328→    };
   329→
   330→    let enriched: Record<string, unknown>[] = [];
   331→    // Process in batches of 10 to avoid overwhelming Firestore
   332→    const BATCH_SIZE = 10;
   333→    for (let i = 0; i < bookings.length; i += BATCH_SIZE) {
   334→      const batch = bookings.slice(i, i + BATCH_SIZE);
   335→      const results = await Promise.allSettled(
   336→        batch.map(async (b) => {
   337→          const allCourtIds: string[] = Array.isArray(b.court_ids)
   338→            ? b.court_ids as string[] : (b.court_id ? [b.court_id as string] : []);
   339→          let court: Record<string, unknown> | null = null;
   340→          let courts: Array<Record<string, unknown>> = [];
   341→          let user: Record<string, unknown> | null = null;
   342→          try {
   343→            if (allCourtIds.length > 0) court = await getCourtCached(allCourtIds[0]);
   344→          } catch { /* skip */ }
   345→          if (allCourtIds.length > 1) {
   346→            const enrichedCourts: Array<Record<string, unknown>> = [];
   347→            for (const cid of allCourtIds) {
   348→              try {
   349→                const c = await getCourtCached(cid);
   350→                if (c) enrichedCourts.push(c);
   351→              } catch { /* skip */ }
   352→            }
   353→            courts = enrichedCourts;
   354→          }
   355→          try {
   356→            if (b.user_id) user = await getUserCached(b.user_id as string);
   357→          } catch { /* skip */ }
   358→          return toCamelBooking({ ...b, _court: court, _courts: courts, _user: user });
   359→        })
   360→      );
   361→      for (const r of results) {
   362→        if (r.status === 'fulfilled') {
   363→          enriched.push(r.value);
   364→        } else {
   365→          console.error('[BOOKINGS] Batch enrichment error:', r.reason);
   366→        }
   367→      }
   368→    }
   369→
   370→    // Default server-side sort: today first, future ascending, past descending
   371→    // Within each date: chronological time
   372→    const today = new Date().toLocaleDateString('en-CA', { timeZone: 'America/Lima' });
   373→    enriched.sort((a, b) => {
   374→      const dA = String(a.date || ''), dB = String(b.date || '');
   375→      const aIsToday = dA === today, bIsToday = dB === today;
   376→      const aFuture = dA > today, bFuture = dB > today;
   377→      const aPast = dA < today, bPast = dB < today;
   378→      // 1) Today always first
   379→      if (aIsToday && !bIsToday) return -1;
   380→      if (!aIsToday && bIsToday) return 1;
   381→      // 2) Both on same side of today
   382→      if (aFuture && bFuture) {
   383→        // Future dates: ascending (correlativo)
   384→        const dc = dA.localeCompare(dB);
   385→        if (dc !== 0) return dc;
   386→      } else if (aPast && bPast) {
   387→        // Past dates: descending (most recent past first)
   388→        const dc = dB.localeCompare(dA);
   389→        if (dc !== 0) return dc;
   390→      }
   391→      // Same date: chronological by start time
   392→      return String(a.startTime || '').localeCompare(String(b.startTime || ''));
   393→    });
   394→
   395→    // FIX P0-4: Lazy-expire reserved bookings past their TTL
   396→    try {
   397→      const db = await getAdminDb();
   398→      const nowMs = Date.now();
   399→      const expiredIds: string[] = [];
   400→      for (const b of bookings) {
   401→        if (migrateStatus(b.status || 'reserved') !== 'reserved') continue;
   402→        const exp = b.expires_at;
   403→        if (!exp) continue;
   404→        const expMs = exp.toMillis?.()?.() || new Date(exp).getTime();
   405→        if (expMs <= nowMs) expiredIds.push(b.id as string);
   406→      }
   407→      if (expiredIds.length > 0) {
   408→        await Promise.all(expiredIds.map((id) =>
   409→          db.collection('bookings').doc(id).update({
   410→            status: 'cancelled',
   411→            slot_status: 'available',
   412→            updated_at: Timestamp.now(),
   413→          })
   414→        ));
   415→        // Remove expired from enriched results
   416→        const expiredSet = new Set(expiredIds);
   417→        enriched = enriched.filter((b) => !expiredSet.has(b.id as string));
   418→      }
   419→    } catch { /* non-critical: best-effort cleanup */ }
   420→
   421→    return NextResponse.json(enriched);
   422→  } catch (error) {
   423→    console.error('[BOOKINGS] GET error:', error);
   424→    const msg = error instanceof Error ? error.message : 'Unknown error';
   425→    return NextResponse.json({ error: 'Error al obtener reservas', detail: msg }, { status: 500 });
   426→  }
   427→}

   429→export async function POST(request: NextRequest) {
   430→  try {
   431→    const authResult = await requireAnyAuth(request);
   432→    if (authResult instanceof NextResponse) return authResult;
   433→    const authUser = authResult.user;
   434→
   435→    if (!isFirebaseAvailable()) {
   436→      return NextResponse.json({ error: 'Firebase no configurado' }, { status: 503 });
   437→    }
   438→
   439→    const body = await request.json();
   440→    const {
   441→      courtId,
   442→      courtIds: bodyCourtIds,
   443→      userId,
   444→      date,
   445→      startTime,
   446→      endTime,
   447→      totalPrice,
   448→      advanceAmount,
   449→      remainingAmount,
   450→      status,
   451→      paymentMethod,
   452→      paymentType,
   453→      notes,
   454→      equipmentItems,
   455→      selectedSlots,
   456→    } = body;
   457→
   458→    // ── Resolve court IDs (support both single courtId and multi-court courtIds) ──
   459→    const allCourtIds: string[] = Array.isArray(bodyCourtIds) && bodyCourtIds.length > 0
   460→      ? bodyCourtIds
   461→      : courtId
   462→        ? [courtId]
   463→        : [];
   464→
   465→    // ── Validate payment method (restricted set, UPPERCASE) ──
   466→    // FIX P1-7: Added 'CULQI', 'TARJETA', 'CARD' so online payments are stored correctly
   467→    const VALID_PAYMENT_METHODS = ['EFECTIVO', 'YAPE', 'PLIN', 'CULQI', 'TARJETA', 'CARD'];
   468→    const normalizedPaymentMethod = VALID_PAYMENT_METHODS.includes((paymentMethod || '').toUpperCase())
   469→      ? (paymentMethod || '').toUpperCase()
   470→      : paymentMethod?.toLowerCase() === 'culqi' ? 'CULQI'
   471→        : paymentMethod?.toLowerCase() === 'card' ? 'TARJETA'
   472→          : 'EFECTIVO';
   473→
   474→    if (allCourtIds.length === 0 || !userId || !date || !startTime || !endTime) {
   475→      return NextResponse.json(
   476→        { error: 'Faltan campos requeridos: courtId/courtIds, userId, date, startTime, endTime' },
   477→        { status: 400 }
   478→      );
   479→    }
   480→
   481→    // ── Minimum advance validation ──
   482→    // Admin/super_admin: NO time restrictions — can book any slot (walk-in, past, current)
   483→    // Regular users: require 30 min minimum advance
   484→    const isAdmin = authUser.role === 'admin' || authUser.role === 'super_admin';
   485→
   486→    if (!isAdmin) {
   487→      const now = new Date();
   488→      const slotDateTime = new Date(`${date}T${startTime}:00`);
   489→      const diffMs = slotDateTime.getTime() - now.getTime();
   490→      const thirtyMinMs = 30 * 60 * 1000;
   491→      if (diffMs < thirtyMinMs) {
   492→        const minsLeft = Math.floor(diffMs / 60000);
   493→        if (minsLeft < 0) {
   494→          return NextResponse.json(
   495→            { error: 'No se puede reservar un horario que ya ha pasado.' },
   496→            { status: 422 }
   497→          );
   498→        }
   499→        return NextResponse.json(
   500→          { error: `La reserva requiere al menos 30 minutos de anticipación. El horario seleccionado comienza en ${minsLeft} minuto${minsLeft === 1 ? '' : 's'}. Selecciona un horario a partir de las ${new Date(now.getTime() + thirtyMinMs).toLocaleTimeString('es-PE', { hour: '2-digit', minute: '2-digit', hour12: false })} o posterior.` },
   501→          { status: 422 }
   502→        );
   503→      }
   504→    }
   505→
   506→    // Non-admin users can only create bookings for themselves
   507→    if (!isAdmin) {
   508→      if (userId !== authUser.id) {
   509→        return NextResponse.json(
   510→          { error: 'No puedes crear reservas para otros usuarios.' },
   511→          { status: 403 }
   512→        );
   513→      }
   514→    }
   515→
   516→    // ── Calculate price (BEFORE transaction — needed for booking data) ──
   517→    const eqItems = Array.isArray(equipmentItems) ? equipmentItems : [];
   518→    let equipmentSubtotal = 0;
   519→    for (const eq of eqItems) {
   520→      equipmentSubtotal += (eq.quantity || 0) * (eq.unit_price || eq.unitPrice || 0);
   521→    }
   522→
   523→    // FIX P1-6: ALWAYS recalculate price server-side from court rates.
   524→    // Client-sent totalPrice is ignored for storage — used only as a UI hint.
   525→    let courtPriceTotal = 0;
   526→    for (const cId of allCourtIds) {
   527→      const court = await getCourtById(cId);
   528→      if (court) {
   529→        const schedule = court.pricing_schedule as Array<{ label: string; startHour: number; endHour: number; pricePerHour: number }> | undefined;

   699→      totalPrice: price,

   701→      remainingAmount: rem,
   702→      status: bookingStatus,
   703→      paymentMethod: normalizedPaymentMethod || paymentMethod || null,

   713→      error: 'Failed to create booking',
   714→      detail: err?.message || 'Unknown error'
   715→    }, { status: 500 });
   716→  }
   717→}

   982→            // FIX Bug A: Only create retained advance when admin explicitly chose an action

   984→            // incorrectly defaulting to 'refunded'
   985→            if (!advanceAction) {
   986→              console.warn('[BOOKINGS] Admin cancellation with advance but no advanceAction — skipping retained advance creation for booking', id);
   987→              warningFlags.push('no_advance_action');
   988→            } else {
   989→            const advAmount = (bookingForCancel.advance_amount as number) || 0;
   990→            const isRetained = advanceAction === 'retain';
   991→            const statusAdvance = isRetained ? 'retained' : 'refunded';
   992→
   993→            // Check for duplicate retained advance
   994→            const { getRetainedAdvances } = await import('@/lib/db');
   995→            const existing = await getRetainedAdvances({ bookingId: id });
   996→            if (existing.length > 0) {
   997→              console.warn('[BOOKINGS] Retained advance already exists for booking', id, '- skipping creation');
   998→            } else {
   999→              // Get court name for the record
  1000→              let courtName = '';
  1001→              try {
  1002→                const cId = (bookingForCancel.court_ids as string[])?.[0] || (bookingForCancel.court_id as string) || '';
  1003→                if (cId) {
  1004→                  const court = await getCourtById(cId);
  1005→                  courtName = (court?.name as string) || '';
  1006→                }
  1007→              } catch { /* non-critical */ }
  1008→
  1009→              // Get user name
  1010→              let userName = '';
  1011→              let userEmail = (bookingForCancel.user_email as string) || null;
  1012→              try {
  1013→                const uId = bookingForCancel.user_id as string;
  1014→                if (uId) {
  1015→                  const user = await getUserById(uId);
  1016→                  userName = (user?.name as string) || '';
  1017→                  // FIX Bug #8: Fallback userEmail from user doc if missing in booking
  1018→                  if (!userEmail) {
  1019→                    userEmail = (user?.email as string) || null;
  1020→                  }
  1021→                }
  1022→              } catch { /* non-critical */ }
  1023→
  1024→              // Create retained advance record
  1025→              try {
  1026→                await createRetainedAdvance({
  1027→                  booking_id: id,
  1028→                  user_id: (bookingForCancel.user_id as string) || '',
  1029→                  user_name: userName,
  1030→                  user_email: userEmail,
  1031→                  court_name: courtName,
  1032→                  booking_date: (bookingForCancel.date as string) || '',
  1033→                  amount: advAmount,
  1034→                  original_total: (bookingForCancel.total_price as number) || 0,

  1047→                  user_id: (bookingForCancel.user_id as string) || '',
  1048→                  amount: advAmount,
  1049→                  type: isRetained ? 'retained' : 'refund',

  1055→                warningFlags.push('payment_record_failed');

  1057→            }
  1058→            }
  1059→          }
  1060→        } catch (err) {
  1061→          console.error('[BOOKINGS] Warning: could not process retained advance on cancellation:', err);
  1062→          warningFlags.push('retained_advance_failed');
  1063→        }
  1064→      }
  1065→    }
  1066→
  1067→    // B11 FIX: Skip advance payment record when cancelling or editBooking (handled above / no record needed)
  1068→    if (typeof reqAdvance === 'number' && typeof reqRemaining === 'number' && !isCancelling && !editBooking) {
  1069→      try {

  1071→        if (booking) {
  1072→          const prevAdvance = (booking.advance_amount as number) || 0;
  1073→          const paymentIncrement = reqAdvance - prevAdvance;

  1076→              await createPayment(id, {
  1077→                user_id: booking.user_id as string,
  1078→                amount: paymentIncrement,
  1079→                type: reqRemaining <= 0 ? 'full_payment' : 'advance',
  1080→                method: (updateData.payment_method as string) || 'EFECTIVO',
  1081→                status: 'completed',

  1086→          }
  1087→        }
  1088→      } catch { /* non-critical: proceed with booking update */ }
  1089→    }
  1090→
  1091→    await updateBooking(id, updateData);
  1092→
  1093→    // FIX Bug #9: Return warnings if any
  1094→    if (warningFlags.length > 0) {
  1095→      return NextResponse.json({ success: true, warnings: warningFlags });
  1096→    }
  1097→    return NextResponse.json({ success: true });
  1098→  } catch (error) {
  1099→    console.error('[BOOKINGS] PUT error:', error);
  1100→    const msg = error instanceof Error ? error.message : 'Unknown error';
  1101→    return NextResponse.json({ error: 'Error al actualizar reserva', detail: msg }, { status: 500 });
  1102→  }
