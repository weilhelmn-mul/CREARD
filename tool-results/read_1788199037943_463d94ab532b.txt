    20→// Migrate old status values to the new 3-status system
    21→function migrateStatus(s: string): string {
    22→  switch (s) {
    23→    case 'confirmed':
    24→    case 'pending':
    25→    case 'partially_paid':
    26→      return 'reserved';
    27→    case 'fully_paid':
    28→    case 'completed':
    29→      return 'completed';
    30→    case 'no_show':
    31→    case 'expired':
    32→      return 'cancelled';
    33→    default:
    34→      return s; // 'reserved', 'completed', 'cancelled' pass through
    35→  }
    36→}

    38→// Transformar snake_case (Firestore) a camelCase (frontend)
    39→function toCamelBooking(b: Record<string, unknown>) {
    40→  // Primary court (backward compat)
    41→  const courtRaw = b._court as Record<string, unknown> | null;
    42→  const court = courtRaw ? {
    43→    id: courtRaw.id || courtRaw.court_id,
    44→    name: courtRaw.name || courtRaw.court_name,
    45→    sport: courtRaw.sport,
    46→    branch: (courtRaw.branch as Record<string, unknown>) || {
    47→      id: courtRaw.branch_id || 'branch-1',
    48→      name: 'CREARD',
    49→    },
    50→  } : null;
    51→
    52→  // Additional courts (multi-court bookings)
    53→  const courtsRaw = b._courts as Array<Record<string, unknown>> | null;
    54→  const courts = courtsRaw ? courtsRaw.map((cr) => ({
    55→    id: cr.id || cr.court_id,
    56→    name: cr.name || cr.court_name,
    57→    sport: cr.sport,
    58→    branch: (cr.branch as Record<string, unknown>) || { id: 'branch-1', name: 'CREARD' },
    59→  })) : [];
    60→
    61→  const userRaw = b._user as Record<string, unknown> | null;
    62→  const user = userRaw ? {
    63→    id: userRaw.id,
    64→    name: userRaw.name,
    65→    email: userRaw.email,
    66→    phone: userRaw.phone,
    67→  } : null;
    68→
    69→  // Court IDs array
    70→  const courtIds: string[] = Array.isArray(b.court_ids) ? b.court_ids : (b.court_id ? [b.court_id] : []);
    71→
    72→  // Equipment items
    73→  const equipmentItems = Array.isArray(b.equipment_items) ? b.equipment_items.map((eq: Record<string, unknown>) => ({
    74→    equipmentId: eq.equipment_id || eq.id,
    75→    name: eq.name,
    76→    quantity: eq.quantity || 1,
    77→    unitPrice: eq.unit_price || eq.price_per_rental || 0,
    78→    subtotal: (eq.quantity || 1) * (eq.unit_price || eq.price_per_rental || 0),
    79→  })) : [];
    80→
    81→  // Normalize: recalc equipment subtotal from items (more reliable than stored value)
    82→  const computedEquipSubtotal = equipmentItems.reduce((s: number, eq: { subtotal: number }) => s + eq.subtotal, 0);
    83→  const equipmentSubtotal = computedEquipSubtotal;
    84→
    85→  // Normalize court_subtotal: if stored value is wrong (old double-counting bug),
    86→  // derive it from total_price - equipment_subtotal
    87→  const totalPrice = b.total_price || 0;
    88→  const storedCourtSubtotal = b.court_subtotal || 0;
    89→  const courtSubtotal = (storedCourtSubtotal + equipmentSubtotal > totalPrice + 0.01)
    90→    ? Math.max(0, Math.round((totalPrice - equipmentSubtotal) * 100) / 100)
    91→    : storedCourtSubtotal;
    92→
    93→  // Normalize remaining_amount: ensure it equals total - advance
    94→  const advanceAmount = b.advance_amount || 0;
    95→  const storedRemaining = b.remaining_amount || 0;
    96→  const remainingAmount = (Math.abs(storedRemaining - (totalPrice - advanceAmount)) > 0.01)
    97→    ? Math.max(0, Math.round((totalPrice - advanceAmount) * 100) / 100)
    98→    : storedRemaining;
    99→
   100→  return {
   101→    id: b.id,
   102→    courtId: b.court_id,
   103→    courtIds,
   104→    userId: b.user_id,
   105→    date: typeof b.date === 'string' ? b.date : (b.date ? String(b.date) : ''),
   106→    startTime: b.start_time,
   107→    endTime: b.end_time,
   108→    totalPrice,
   109→    courtSubtotal,
   110→    equipmentSubtotal,
   111→    equipmentItems,
   112→    advanceAmount,
   113→    remainingAmount,
   114→    status: migrateStatus(b.status || 'reserved'),
   115→    slotStatus: b.slot_status,
   116→    paymentMethod: b.payment_method,
   117→    remainingPaymentStatus: b.remaining_payment_status || null,
   118→    notes: b.notes,
   119→    createdAt: b.created_at,
   120→    updatedAt: b.updated_at,
   121→    recurringGroupId: b.recurring_group_id,
   122→    recurringIndex: b.recurring_index,
   123→    equipmentDelivered: b.equipment_delivered || false,
   124→    equipmentReturned: b.equipment_returned || false,
   125→    court,
   126→    courts,
   127→    user,
   128→  };
   129→}

   131→// Enrich a single booking (all courts + user), safe — never throws
   132→async function safeEnrichBooking(b: Record<string, unknown>): Promise<Record<string, unknown>> {
   133→  let court: Record<string, unknown> | null = null;
   134→  let courts: Array<Record<string, unknown>> = [];
   135→  let user: Record<string, unknown> | null = null;
   136→
   137→  // Get all court IDs (from court_ids array or single court_id)
   138→  const allCourtIds: string[] = Array.isArray(b.court_ids)
   139→    ? b.court_ids as string[]
   140→    : (b.court_id ? [b.court_id as string] : []);
   141→
   142→  // Enrich primary court (first one)
   143→  try {
   144→    if (allCourtIds.length > 0) {
   145→      const c = await getCourtById(allCourtIds[0]);
   146→      if (c) court = c as Record<string, unknown>;
   147→    }
   148→  } catch { /* court enrichment failed */ }
   149→
   150→
   151→  // Enrich all courts for multi-court bookings
   152→  if (allCourtIds.length > 1) {
   153→    const enrichedCourts: Array<Record<string, unknown>> = [];
   154→    for (const cid of allCourtIds) {
   155→      try {
   156→        const c = await getCourtById(cid);
   157→        if (c) enrichedCourts.push(c as Record<string, unknown>);
   158→      } catch { /* skip failed court enrichment */ }
   159→    }
   160→    courts = enrichedCourts;
   161→  }
   162→
   163→  try {
   164→    if (b.user_id) {
   165→      const u = await getUserById(b.user_id as string);
   166→      if (u) user = u as Record<string, unknown>;
   167→    }
   168→  } catch { /* user enrichment failed */ }
   169→
   170→
   171→  return { ...b, _court: court, _courts: courts, _user: user };
   172→}

   174→// Search bookings by userId OR userEmail, deduplicate
   175→async function searchBookingsForUser(userId: string, userEmail: string | null) {
   176→  const results = new Map<string, Record<string, unknown>>();
   177→
   178→  // Always search by userId
   179→  try {
   180→    const byId = await getBookings({ userId });
   181→    for (const b of byId) {
   182→      if (b.id && !results.has(b.id)) {
   183→        results.set(b.id, b as Record<string, unknown>);
   184→      }
   185→    }
   186→  } catch (e) {
   187→    console.error('[BOOKINGS] Error searching by userId:', userId, e);
   188→  }
   189→
   190→  // Also search by email to handle demo/Firebase ID mismatches
   191→  if (userEmail && userEmail !== userId) {
   192→    try {
   193→      const byEmail = await getBookings({ userEmail });
   194→      for (const b of byEmail) {
   195→        if (b.id && !results.has(b.id)) {
   196→          results.set(b.id, b as Record<string, unknown>);
   197→        }
   198→      }
   199→    } catch (e) {
   200→      console.error('[BOOKINGS] Error searching by userEmail:', userEmail, e);
   201→    }
   202→  }
   203→
   204→  return Array.from(results.values());
   205→}

   207→/**
   208→ * Calculate the total price for a time slot using a pricing schedule.
   209→ * Supports slots that span multiple schedules (e.g., 16:00-19:00 = 1hr morning + 2hr night).
   210→ */
   211→function calculatePriceForTimeSlot(
   212→  schedule: Array<{ label: string; startHour: number; endHour: number; pricePerHour: number }>,
   213→  startTime: string,
   214→  endTime: string,
   215→): number {
   216→  const [startH, startM] = startTime.split(':').map(Number);
   217→  const [endH, endM] = endTime.split(':').map(Number);
   218→  const startDecimal = startH + startM / 60;
   219→  const endDecimal = endH + endM / 60;
   220→
   221→  // Sort schedule by startHour
   222→  const sorted = [...schedule].sort((a, b) => a.startHour - b.startHour);
   223→
   224→  let total = 0;
   225→  let cursor = startDecimal;
   226→
   227→  for (const slot of sorted) {
   228→    const slotStart = slot.startHour;
   229→    const slotEnd = slot.endHour;
   230→
   231→    if (cursor >= slotEnd) continue; // cursor already past this slot
   232→
   233→    // Overlap start/end
   234→    const overlapStart = Math.max(cursor, slotStart);
   235→    const overlapEnd = Math.min(endDecimal, slotEnd);
   236→
   237→    if (overlapEnd > overlapStart) {
   238→      const hours = overlapEnd - overlapStart;
   239→      total += hours * slot.pricePerHour;
   240→      cursor = overlapEnd;
   241→    }
   242→  }
   243→
   244→  // If cursor didn't reach endDecimal, the remaining time is uncovered by schedule
   245→  // (fall back to 0 or caller's default)
   246→  return Math.round(total * 100) / 100;
   247→}

   249→export async function GET(request: NextRequest) {
   250→  const startTime = Date.now();
   251→  const firebaseOk = isFirebaseAvailable();
   252→
   253→  try {
   254→    const { searchParams } = new URL(request.url);
   255→    const courtId = searchParams.get('courtId');
   256→    const userId = searchParams.get('userId');
   257→    const date = searchParams.get('date');
   258→    const dateFrom = searchParams.get('dateFrom');
   259→    const dateTo = searchParams.get('dateTo');
   260→    const status = searchParams.get('status');
   261→
   262→    // Court availability check (courtId + date) is public — no auth required
   263→    if (courtId && date && !userId) {
   264→      if (!firebaseOk) {
   265→        return NextResponse.json({ error: 'Firebase no configurado', code: 'NO_FIREBASE' }, { status: 503 });
   266→      }
   267→      const bookings = await getBookings({ courtId, date });
   268→      // FIX P0-4: Filter out expired reserved bookings (ghost reservations)
   269→      const now = Date.now();
   270→      const active = bookings.filter((b: any) => {
   271→        if (migrateStatus(b.status || 'reserved') !== 'reserved') return true;
   272→        const exp = b.expires_at;
   273→        if (!exp) return true;
   274→        const expMs = exp.toMillis?.()?.() || new Date(exp).getTime();
   275→        return expMs > now;
   276→      });
   277→      return NextResponse.json(active.map(toCamelBooking));
   278→    }
   279→
   280→    // All other queries require authentication
   281→    const authResult = await requireAnyAuth(request);
   282→    if (authResult instanceof NextResponse) return authResult;
   283→    const authUser = authResult.user;
   284→
   285→    if (!firebaseOk) {
   286→      return NextResponse.json({ error: 'Firebase no configurado', code: 'NO_FIREBASE' }, { status: 503 });
   287→    }
   288→
   289→    let bookings: Record<string, unknown>[];
   290→
   291→    // Non-admin users: only their own bookings
   292→    if (authUser.role !== 'admin' && authUser.role !== 'super_admin') {
   293→      if (userId && userId !== authUser.id) {
   294→        return NextResponse.json({ error: 'No puedes ver reservas de otros usuarios.' }, { status: 403 });
   295→      }
   296→      const effectiveUserId = userId || authUser.id;
   297→      bookings = await searchBookingsForUser(effectiveUserId, authUser.email);
   298→    } else {
   299→      // Admin: fetch with optional filters (date range for performance)
   300→      bookings = (await getBookings({
   301→        courtId: courtId || undefined,
   302→        userId: userId || undefined,
   303→        date: date || undefined,
   304→        dateFrom: dateFrom || undefined,
   305→        dateTo: dateTo || undefined,
   306→        status: status || undefined,
   307→      })) as Record<string, unknown>[];
   308→      // DEBUG removed (P3-16): raw booking data no longer logged
   309→    }
   310→
   311→    // Enrich each booking with cached court/user lookups (avoid N+1 Firestore reads)
   312→    const courtCache = new Map<string, Record<string, unknown> | null>();
   313→    const userCache = new Map<string, Record<string, unknown> | null>();
   314→    const getCourtCached = async (cid: string) => {
   315→      if (courtCache.has(cid)) return courtCache.get(cid)!;
   316→      try {
   317→        const c = await getCourtById(cid);
   318→        courtCache.set(cid, c as Record<string, unknown> | null);
   319→        return c as Record<string, unknown> | null;
   320→      } catch { courtCache.set(cid, null); return null; }
   321→    };
   322→    const getUserCached = async (uid: string) => {
   323→      if (userCache.has(uid)) return userCache.get(uid)!;
   324→      try {
   325→        const u = await getUserById(uid);
   326→        userCache.set(uid, u as Record<string, unknown> | null);
   327→        return u as Record<string, unknown> | null;
   328→      } catch { userCache.set(uid, null); return null; }
   329→    };
   330→
   331→    let enriched: Record<string, unknown>[] = [];
   332→    // Process in batches of 10 to avoid overwhelming Firestore
   333→    const BATCH_SIZE = 10;
   334→    for (let i = 0; i < bookings.length; i += BATCH_SIZE) {
   335→      const batch = bookings.slice(i, i + BATCH_SIZE);
   336→      const results = await Promise.allSettled(
   337→        batch.map(async (b) => {
   338→          const allCourtIds: string[] = Array.isArray(b.court_ids)
   339→            ? b.court_ids as string[] : (b.court_id ? [b.court_id as string] : []);
   340→          let court: Record<string, unknown> | null = null;
   341→          let courts: Array<Record<string, unknown>> = [];
   342→          let user: Record<string, unknown> | null = null;
   343→          try {
   344→            if (allCourtIds.length > 0) court = await getCourtCached(allCourtIds[0]);
   345→          } catch { /* skip */ }
   346→          if (allCourtIds.length > 1) {
   347→            const enrichedCourts: Array<Record<string, unknown>> = [];
   348→            for (const cid of allCourtIds) {
   349→              try {
   350→                const c = await getCourtCached(cid);
   351→                if (c) enrichedCourts.push(c);
   352→              } catch { /* skip */ }
   353→            }
   354→            courts = enrichedCourts;
   355→          }
   356→          try {
   357→            if (b.user_id) user = await getUserCached(b.user_id as string);
   358→          } catch { /* skip */ }
   359→          return toCamelBooking({ ...b, _court: court, _courts: courts, _user: user });
   360→        })
   361→      );
   362→      for (const r of results) {
   363→        if (r.status === 'fulfilled') {
   364→          enriched.push(r.value);
   365→        } else {
   366→          console.error('[BOOKINGS] Batch enrichment error:', r.reason);
   367→        }
   368→      }
   369→    }
   370→
   371→    // Default server-side sort: today first, future ascending, past descending
   372→    // Within each date: chronological time
   373→    const today = new Date().toLocaleDateString('en-CA', { timeZone: 'America/Lima' });
   374→    enriched.sort((a, b) => {
   375→      const dA = String(a.date || ''), dB = String(b.date || '');
   376→      const aIsToday = dA === today, bIsToday = dB === today;
   377→      const aFuture = dA > today, bFuture = dB > today;
   378→      const aPast = dA < today, bPast = dB < today;
   379→      // 1) Today always first
   380→      if (aIsToday && !bIsToday) return -1;
   381→      if (!aIsToday && bIsToday) return 1;
   382→      // 2) Both on same side of today
   383→      if (aFuture && bFuture) {
   384→        // Future dates: ascending (correlativo)
   385→        const dc = dA.localeCompare(dB);
   386→        if (dc !== 0) return dc;
   387→      } else if (aPast && bPast) {
   388→        // Past dates: descending (most recent past first)
   389→        const dc = dB.localeCompare(dA);
   390→        if (dc !== 0) return dc;
   391→      }
   392→      // Same date: chronological by start time
   393→      return String(a.startTime || '').localeCompare(String(b.startTime || ''));
   394→    });
   395→
   396→    // FIX P0-4: Lazy-expire reserved bookings past their TTL
   397→    try {
   398→      const db = await getAdminDb();
   399→      const nowMs = Date.now();
   400→      const expiredIds: string[] = [];
   401→      for (const b of bookings) {
   402→        if (migrateStatus(b.status || 'reserved') !== 'reserved') continue;
   403→        const exp = b.expires_at;
   404→        if (!exp) continue;
   405→        const expMs = exp.toMillis?.()?.() || new Date(exp).getTime();
   406→        if (expMs <= nowMs) expiredIds.push(b.id as string);
   407→      }
   408→      if (expiredIds.length > 0) {
   409→        await Promise.all(expiredIds.map((id) =>
   410→          db.collection('bookings').doc(id).update({
   411→            status: 'cancelled',
   412→            slot_status: 'available',
   413→            updated_at: Timestamp.now(),
   414→          })
   415→        ));
   416→        // Remove expired from enriched results
   417→        const expiredSet = new Set(expiredIds);
   418→        enriched = enriched.filter((b) => !expiredSet.has(b.id as string));
   419→      }
   420→    } catch { /* non-critical: best-effort cleanup */ }
   421→
   422→    return NextResponse.json(enriched);
   423→  } catch (error) {
   424→    console.error('[BOOKINGS] GET error:', error);
   425→    const msg = error instanceof Error ? error.message : 'Unknown error';
   426→    return NextResponse.json({ error: 'Error al obtener reservas', detail: msg }, { status: 500 });
   427→  }
   428→}

   430→export async function POST(request: NextRequest) {
   431→  try {
   432→    const authResult = await requireAnyAuth(request);
   433→    if (authResult instanceof NextResponse) return authResult;
   434→    const authUser = authResult.user;
   435→
   436→    if (!isFirebaseAvailable()) {
   437→      return NextResponse.json({ error: 'Firebase no configurado' }, { status: 503 });
   438→    }
   439→
   440→    const body = await request.json();
   441→    const {
   442→      courtId,
   443→      courtIds: bodyCourtIds,
   444→      userId,
   445→      date,
   446→      startTime,
   447→      endTime,
   448→      totalPrice,
   449→      advanceAmount,
   450→      remainingAmount,
   451→      status,
   452→      paymentMethod,
   453→      paymentType,
   454→      notes,
   455→      equipmentItems,
   456→      selectedSlots,
   457→    } = body;
   458→
   459→    // ── Resolve court IDs (support both single courtId and multi-court courtIds) ──
   460→    const allCourtIds: string[] = Array.isArray(bodyCourtIds) && bodyCourtIds.length > 0
   461→      ? bodyCourtIds
   462→      : courtId
   463→        ? [courtId]
   464→        : [];
   465→
   466→    // ── Validate payment method (restricted set, UPPERCASE) ──
   467→    // FIX P1-7: Added 'CULQI', 'TARJETA', 'CARD' so online payments are stored correctly
   468→    const VALID_PAYMENT_METHODS = ['EFECTIVO', 'YAPE', 'PLIN', 'CULQI', 'TARJETA', 'CARD'];
   469→    const normalizedPaymentMethod = VALID_PAYMENT_METHODS.includes((paymentMethod || '').toUpperCase())
   470→      ? (paymentMethod || '').toUpperCase()
   471→      : paymentMethod?.toLowerCase() === 'culqi' ? 'CULQI'
   472→        : paymentMethod?.toLowerCase() === 'card' ? 'TARJETA'
   473→          : 'EFECTIVO';
   474→
   475→    if (allCourtIds.length === 0 || !userId || !date || !startTime || !endTime) {
   476→      return NextResponse.json(
   477→        { error: 'Faltan campos requeridos: courtId/courtIds, userId, date, startTime, endTime' },
   478→        { status: 400 }
   479→      );
   480→    }
   481→
   482→    // ── Minimum advance validation ──
   483→    // Admin/super_admin: NO time restrictions — can book any slot (walk-in, past, current)
   484→    // Regular users: require 30 min minimum advance
   485→    const isAdmin = authUser.role === 'admin' || authUser.role === 'super_admin';
   486→
   487→    if (!isAdmin) {
   488→      const now = new Date();
   489→      const slotDateTime = new Date(`${date}T${startTime}:00`);
   490→      const diffMs = slotDateTime.getTime() - now.getTime();
   491→      const thirtyMinMs = 30 * 60 * 1000;
   492→      if (diffMs < thirtyMinMs) {
   493→        const minsLeft = Math.floor(diffMs / 60000);
   494→        if (minsLeft < 0) {
   495→          return NextResponse.json(
   496→            { error: 'No se puede reservar un horario que ya ha pasado.' },
   497→            { status: 422 }
   498→          );
   499→        }
   500→        return NextResponse.json(
   501→          { error: `La reserva requiere al menos 30 minutos de anticipación. El horario seleccionado comienza en ${minsLeft} minuto${minsLeft === 1 ? '' : 's'}. Selecciona un horario a partir de las ${new Date(now.getTime() + thirtyMinMs).toLocaleTimeString('es-PE', { hour: '2-digit', minute: '2-digit', hour12: false })} o posterior.` },
   502→          { status: 422 }
   503→        );
   504→      }
   505→    }
   506→
   507→    // Non-admin users can only create bookings for themselves
   508→    if (!isAdmin) {
   509→      if (userId !== authUser.id) {
   510→        return NextResponse.json(
   511→          { error: 'No puedes crear reservas para otros usuarios.' },
   512→          { status: 403 }
   513→        );
   514→      }
   515→    }
   516→
   517→    // ── Calculate price (BEFORE transaction — needed for booking data) ──
   518→    const eqItems = Array.isArray(equipmentItems) ? equipmentItems : [];
   519→    let equipmentSubtotal = 0;
   520→    for (const eq of eqItems) {
   521→      equipmentSubtotal += (eq.quantity || 0) * (eq.unit_price || eq.unitPrice || 0);
   522→    }
   523→
   524→    // FIX P1-6: ALWAYS recalculate price server-side from court rates.
   525→    // Client-sent totalPrice is ignored for storage — used only as a UI hint.
   526→    let courtPriceTotal = 0;
   527→    for (const cId of allCourtIds) {
   528→      const court = await getCourtById(cId);
   529→      if (court) {

   585→            }
   586→          }
   587→        }

   596→          user_email: clientEmail || null,

   598→          start_time: startTime,

   731→      error: 'Failed to create booking',
   732→      detail: err?.message || 'Unknown error'

   970→
   971→      // Skip the generic advance payment record creation below — we handle it here
   972→      // (prevent double payment records)
   973→    }
   974→
   975→    // B11 FIX: Track if this is a cancellation to skip advance payment record
   976→    const isCancelling = status === 'cancelled' || (status && migrateStatus(status) === 'cancelled');
   977→
   978→    // FIX Bug #9: Track warnings to return to frontend
   979→    const warningFlags: string[] = [];
   980→
   981→    // When cancelling a booking with advance, handle retained/refunded advance
   982→    // Only create retained advance records when an admin explicitly provides advanceAction

   984→    const isAdminCancellation = authUser.role === 'admin' || authUser.role === 'super_admin';
   985→    if (isCancelling) {
   986→      // B6 FIX: Fetch booking once, reuse for both checks
   987→      let bookingForCancel: Record<string, unknown> | null = null;
   988→      try {
   989→        bookingForCancel = await getBookingById(id);
   990→      } catch { /* non-blocking */ }
   991→
   992→      // Prevent cancelling completed bookings
   993→      if (bookingForCancel && (bookingForCancel.status === 'completed' || migrateStatus(bookingForCancel.status as string || '') === 'completed')) {
   994→        return NextResponse.json({ error: 'No se puede cancelar una reserva completada. Contacta soporte si es necesario.' }, { status: 400 });
   995→      }
   996→
   997→      if (isAdminCancellation) {
   998→        try {
   999→          if (bookingForCancel && ((bookingForCancel.advance_amount as number) || 0) > 0) {
  1000→            // FIX Bug A: Only create retained advance when admin explicitly chose an action
  1001→            // If advanceAction is missing (unexpected code path), skip creation to avoid
  1002→            // incorrectly defaulting to 'refunded'
  1003→            if (!advanceAction) {
  1004→              console.warn('[BOOKINGS] Admin cancellation with advance but no advanceAction — skipping retained advance creation for booking', id);
  1005→              warningFlags.push('no_advance_action');
  1006→            } else {
  1007→            const advAmount = (bookingForCancel.advance_amount as number) || 0;
  1008→            const isRetained = advanceAction === 'retain';
  1009→            const statusAdvance = isRetained ? 'retained' : 'refunded';
  1010→
  1011→            // Check for duplicate retained advance
  1012→            const { getRetainedAdvances } = await import('@/lib/db');
  1013→            const existing = await getRetainedAdvances({ bookingId: id });
  1014→            if (existing.length > 0) {
  1015→              console.warn('[BOOKINGS] Retained advance already exists for booking', id, '- skipping creation');
  1016→            } else {
  1017→              // Get court name for the record
  1018→              let courtName = '';
  1019→              try {
  1020→                const cId = (bookingForCancel.court_ids as string[])?.[0] || (bookingForCancel.court_id as string) || '';
  1021→                if (cId) {
  1022→                  const court = await getCourtById(cId);
  1023→                  courtName = (court?.name as string) || '';
  1024→                }
  1025→              } catch { /* non-critical */ }
  1026→
  1027→              // Get user name
  1028→              let userName = '';
  1029→              let userEmail = (bookingForCancel.user_email as string) || null;
  1030→              try {
  1031→                const uId = bookingForCancel.user_id as string;
  1032→                if (uId) {
  1033→                  const user = await getUserById(uId);
  1034→                  userName = (user?.name as string) || '';
  1035→                  // FIX Bug #8: Fallback userEmail from user doc if missing in booking
  1036→                  if (!userEmail) {
  1037→                    userEmail = (user?.email as string) || null;
  1038→                  }
  1039→                }
  1040→              } catch { /* non-critical */ }
  1041→
  1042→              // Create retained advance record
  1043→              try {
  1044→                await createRetainedAdvance({
  1045→                  booking_id: id,
  1046→                  user_id: (bookingForCancel.user_id as string) || '',
  1047→                  user_name: userName,
  1048→                  user_email: userEmail,
  1049→                  court_name: courtName,
  1050→                  booking_date: (bookingForCancel.date as string) || '',
  1051→                  amount: advAmount,
  1052→                  original_total: (bookingForCancel.total_price as number) || 0,
  1053→                  payment_method: (bookingForCancel.payment_method as string) || 'EFECTIVO',
  1054→                  reason: cancelReason || 'Cancelación de reserva',
  1055→                  status: statusAdvance,
  1056→                });
  1057→              } catch (raErr) {
  1058→                console.error('[BOOKINGS] Error: could not create retained advance record:', raErr);
  1059→                warningFlags.push('retained_advance_failed');
  1060→              }
  1061→
  1062→              // Create a payment record for the refund/retention
  1063→              try {
  1064→                await createPayment(id, {
  1065→                  user_id: (bookingForCancel.user_id as string) || '',
  1066→                  amount: advAmount,
  1067→                  type: isRetained ? 'retained' : 'refund',
  1068→                  method: (bookingForCancel.payment_method as string) || 'EFECTIVO',
  1069→                  status: 'completed',
  1070→                });
  1071→              } catch (payErr) {
  1072→                console.error('[BOOKINGS] Warning: could not create payment record for cancellation:', payErr);
  1073→                warningFlags.push('payment_record_failed');
  1074→              }
  1075→            }
  1076→            }
  1077→          }
  1078→        } catch (err) {
  1079→          console.error('[BOOKINGS] Warning: could not process retained advance on cancellation:', err);
  1080→          warningFlags.push('retained_advance_failed');
  1081→        }
  1082→      }
  1083→    }
  1084→
  1085→    // B11 FIX: Skip advance payment record when cancelling or editBooking (handled above / no record needed)
  1086→    if (typeof reqAdvance === 'number' && typeof reqRemaining === 'number' && !isCancelling && !editBooking) {
  1087→      try {
  1088→        const booking = await getBookingById(id);
  1089→        if (booking) {
  1090→          const prevAdvance = (booking.advance_amount as number) || 0;
  1091→          const paymentIncrement = reqAdvance - prevAdvance;
  1092→          if (paymentIncrement > 0.01) {
  1093→            try {
  1094→              await createPayment(id, {
  1095→                user_id: booking.user_id as string,
  1096→                amount: paymentIncrement,
  1097→                type: reqRemaining <= 0 ? 'full_payment' : 'advance',
  1098→                method: (updateData.payment_method as string) || 'EFECTIVO',
  1099→                status: 'completed',
  1100→              });
  1101→            } catch (payErr) {
  1102→              console.error('[BOOKINGS] Warning: could not create payment record for advance update:', payErr);
  1103→            }
  1104→          }
  1105→        }
  1106→      } catch { /* non-critical: proceed with booking update */ }
  1107→    }
  1108→
  1109→    await updateBooking(id, updateData);
  1110→
  1111→    // FIX Bug #9: Return warnings if any
  1112→    if (warningFlags.length > 0) {
  1113→      return NextResponse.json({ success: true, warnings: warningFlags });
  1114→    }
  1115→    return NextResponse.json({ success: true });
  1116→  } catch (error) {
  1117→    console.error('[BOOKINGS] PUT error:', error);
  1118→    const msg = error instanceof Error ? error.message : 'Unknown error';
  1119→    return NextResponse.json({ error: 'Error al actualizar reserva', detail: msg }, { status: 500 });
  1120→  }
  1121→}
