    18→// Migrate old status values to the new 3-status system
    19→function migrateStatus(s: string): string {
    20→  switch (s) {
    21→    case 'confirmed':
    22→    case 'pending':
    23→    case 'partially_paid':
    24→      return 'reserved';
    25→    case 'fully_paid':
    26→    case 'completed':
    27→      return 'completed';
    28→    case 'no_show':
    29→    case 'expired':
    30→      return 'cancelled';
    31→    default:
    32→      return s; // 'reserved', 'completed', 'cancelled' pass through
    33→  }
    34→}

    36→// Transformar snake_case (Firestore) a camelCase (frontend)
    37→function toCamelBooking(b: Record<string, unknown>) {
    38→  // Primary court (backward compat)
    39→  const courtRaw = b._court as Record<string, unknown> | null;
    40→  const court = courtRaw ? {
    41→    id: courtRaw.id || courtRaw.court_id,
    42→    name: courtRaw.name || courtRaw.court_name,
    43→    sport: courtRaw.sport,
    44→    branch: (courtRaw.branch as Record<string, unknown>) || {
    45→      id: courtRaw.branch_id || 'branch-1',
    46→      name: 'CREARD',
    47→    },
    48→  } : null;
    49→
    50→  // Additional courts (multi-court bookings)
    51→  const courtsRaw = b._courts as Array<Record<string, unknown>> | null;
    52→  const courts = courtsRaw ? courtsRaw.map((cr) => ({
    53→    id: cr.id || cr.court_id,
    54→    name: cr.name || cr.court_name,
    55→    sport: cr.sport,
    56→    branch: (cr.branch as Record<string, unknown>) || { id: 'branch-1', name: 'CREARD' },
    57→  })) : [];
    58→
    59→  const userRaw = b._user as Record<string, unknown> | null;
    60→  const user = userRaw ? {
    61→    id: userRaw.id,
    62→    name: userRaw.name,
    63→    email: userRaw.email,
    64→    phone: userRaw.phone,
    65→  } : null;
    66→
    67→  // Court IDs array
    68→  const courtIds: string[] = Array.isArray(b.court_ids) ? b.court_ids : (b.court_id ? [b.court_id] : []);
    69→
    70→  // Equipment items
    71→  const equipmentItems = Array.isArray(b.equipment_items) ? b.equipment_items.map((eq: Record<string, unknown>) => ({
    72→    equipmentId: eq.equipment_id || eq.id,
    73→    name: eq.name,
    74→    quantity: eq.quantity || 1,
    75→    unitPrice: eq.unit_price || eq.price_per_rental || 0,
    76→    subtotal: (eq.quantity || 1) * (eq.unit_price || eq.price_per_rental || 0),
    77→  })) : [];
    78→
    79→  // Normalize: recalc equipment subtotal from items (more reliable than stored value)
    80→  const computedEquipSubtotal = equipmentItems.reduce((s: number, eq: { subtotal: number }) => s + eq.subtotal, 0);
    81→  const equipmentSubtotal = computedEquipSubtotal;
    82→
    83→  // Normalize court_subtotal: if stored value is wrong (old double-counting bug),
    84→  // derive it from total_price - equipment_subtotal
    85→  const totalPrice = b.total_price || 0;
    86→  const storedCourtSubtotal = b.court_subtotal || 0;
    87→  const courtSubtotal = (storedCourtSubtotal + equipmentSubtotal > totalPrice + 0.01)
    88→    ? Math.max(0, Math.round((totalPrice - equipmentSubtotal) * 100) / 100)
    89→    : storedCourtSubtotal;
    90→
    91→  // Normalize remaining_amount: ensure it equals total - advance
    92→  const advanceAmount = b.advance_amount || 0;
    93→  const storedRemaining = b.remaining_amount || 0;
    94→  const remainingAmount = (Math.abs(storedRemaining - (totalPrice - advanceAmount)) > 0.01)
    95→    ? Math.max(0, Math.round((totalPrice - advanceAmount) * 100) / 100)
    96→    : storedRemaining;
    97→
    98→  return {
    99→    id: b.id,
   100→    courtId: b.court_id,
   101→    courtIds,
   102→    userId: b.user_id,
   103→    date: typeof b.date === 'string' ? b.date : (b.date ? String(b.date) : ''),
   104→    startTime: b.start_time,
   105→    endTime: b.end_time,
   106→    totalPrice,
   107→    courtSubtotal,
   108→    equipmentSubtotal,
   109→    equipmentItems,
   110→    advanceAmount,
   111→    remainingAmount,
   112→    status: migrateStatus(b.status || 'reserved'),
   113→    slotStatus: b.slot_status,
   114→    paymentMethod: b.payment_method,
   115→    remainingPaymentStatus: b.remaining_payment_status || null,
   116→    notes: b.notes,
   117→    createdAt: b.created_at,
   118→    updatedAt: b.updated_at,
   119→    recurringGroupId: b.recurring_group_id,
   120→    recurringIndex: b.recurring_index,
   121→    equipmentDelivered: b.equipment_delivered || false,
   122→    equipmentReturned: b.equipment_returned || false,
   123→    court,
   124→    courts,
   125→    user,
   126→  };
   127→}

   129→// Enrich a single booking (all courts + user), safe — never throws
   130→async function safeEnrichBooking(b: Record<string, unknown>): Promise<Record<string, unknown>> {
   131→  let court: Record<string, unknown> | null = null;
   132→  let courts: Array<Record<string, unknown>> = [];
   133→  let user: Record<string, unknown> | null = null;
   134→
   135→  // Get all court IDs (from court_ids array or single court_id)
   136→  const allCourtIds: string[] = Array.isArray(b.court_ids)
   137→    ? b.court_ids as string[]
   138→    : (b.court_id ? [b.court_id as string] : []);
   139→
   140→  // Enrich primary court (first one)
   141→  try {
   142→    if (allCourtIds.length > 0) {
   143→      const c = await getCourtById(allCourtIds[0]);
   144→      if (c) court = c as Record<string, unknown>;
   145→    }
   146→  } catch { /* court enrichment failed */ }
   147→
   148→
   149→  // Enrich all courts for multi-court bookings
   150→  if (allCourtIds.length > 1) {
   151→    const enrichedCourts: Array<Record<string, unknown>> = [];
   152→    for (const cid of allCourtIds) {
   153→      try {
   154→        const c = await getCourtById(cid);
   155→        if (c) enrichedCourts.push(c as Record<string, unknown>);
   156→      } catch { /* skip failed court enrichment */ }
   157→    }
   158→    courts = enrichedCourts;
   159→  }
   160→
   161→  try {
   162→    if (b.user_id) {
   163→      const u = await getUserById(b.user_id as string);
   164→      if (u) user = u as Record<string, unknown>;
   165→    }
   166→  } catch { /* user enrichment failed */ }
   167→
   168→
   169→  return { ...b, _court: court, _courts: courts, _user: user };
   170→}

   172→// Search bookings by userId OR userEmail, deduplicate
   173→async function searchBookingsForUser(userId: string, userEmail: string | null) {
   174→  const results = new Map<string, Record<string, unknown>>();
   175→
   176→  // Always search by userId
   177→  try {
   178→    const byId = await getBookings({ userId });
   179→    for (const b of byId) {
   180→      if (b.id && !results.has(b.id)) {
   181→        results.set(b.id, b as Record<string, unknown>);
   182→      }
   183→    }
   184→  } catch (e) {
   185→    console.error('[BOOKINGS] Error searching by userId:', userId, e);
   186→  }
   187→
   188→  // Also search by email to handle demo/Firebase ID mismatches
   189→  if (userEmail && userEmail !== userId) {
   190→    try {
   191→      const byEmail = await getBookings({ userEmail });
   192→      for (const b of byEmail) {
   193→        if (b.id && !results.has(b.id)) {
   194→          results.set(b.id, b as Record<string, unknown>);
   195→        }
   196→      }
   197→    } catch (e) {
   198→      console.error('[BOOKINGS] Error searching by userEmail:', userEmail, e);
   199→    }
   200→  }
   201→
   202→  return Array.from(results.values());
   203→}

   205→/**
   206→ * Calculate the total price for a time slot using a pricing schedule.
   207→ * Supports slots that span multiple schedules (e.g., 16:00-19:00 = 1hr morning + 2hr night).
   208→ */
   209→function calculatePriceForTimeSlot(
   210→  schedule: Array<{ label: string; startHour: number; endHour: number; pricePerHour: number }>,
   211→  startTime: string,
   212→  endTime: string,
   213→): number {
   214→  const [startH, startM] = startTime.split(':').map(Number);
   215→  const [endH, endM] = endTime.split(':').map(Number);
   216→  const startDecimal = startH + startM / 60;
   217→  const endDecimal = endH + endM / 60;
   218→
   219→  // Sort schedule by startHour
   220→  const sorted = [...schedule].sort((a, b) => a.startHour - b.startHour);
   221→
   222→  let total = 0;
   223→  let cursor = startDecimal;
   224→
   225→  for (const slot of sorted) {
   226→    const slotStart = slot.startHour;
   227→    const slotEnd = slot.endHour;
   228→
   229→    if (cursor >= slotEnd) continue; // cursor already past this slot
   230→
   231→    // Overlap start/end
   232→    const overlapStart = Math.max(cursor, slotStart);
   233→    const overlapEnd = Math.min(endDecimal, slotEnd);
   234→
   235→    if (overlapEnd > overlapStart) {
   236→      const hours = overlapEnd - overlapStart;
   237→      total += hours * slot.pricePerHour;
   238→      cursor = overlapEnd;
   239→    }
   240→  }
   241→
   242→  // If cursor didn't reach endDecimal, the remaining time is uncovered by schedule
   243→  // (fall back to 0 or caller's default)
   244→  return Math.round(total * 100) / 100;
   245→}

   247→export async function GET(request: NextRequest) {
   248→  const startTime = Date.now();
   249→  const firebaseOk = isFirebaseAvailable();
   250→
   251→  try {
   252→    const { searchParams } = new URL(request.url);
   253→    const courtId = searchParams.get('courtId');
   254→    const userId = searchParams.get('userId');
   255→    const date = searchParams.get('date');
   256→    const dateFrom = searchParams.get('dateFrom');
   257→    const dateTo = searchParams.get('dateTo');
   258→    const status = searchParams.get('status');
   259→
   260→    // Court availability check (courtId + date) is public — no auth required
   261→    if (courtId && date && !userId) {
   262→      if (!firebaseOk) {
   263→        return NextResponse.json({ error: 'Firebase no configurado', code: 'NO_FIREBASE' }, { status: 503 });
   264→      }
   265→      const bookings = await getBookings({ courtId, date });
   266→      // FIX P0-4: Filter out expired reserved bookings (ghost reservations)
   267→      const now = Date.now();
   268→      const active = bookings.filter((b: any) => {
   269→        if (migrateStatus(b.status || 'reserved') !== 'reserved') return true;
   270→        const exp = b.expires_at;
   271→        if (!exp) return true;
   272→        const expMs = exp.toMillis?.()?.() || new Date(exp).getTime();
   273→        return expMs > now;
   274→      });
   275→      return NextResponse.json(active.map(toCamelBooking));
   276→    }
   277→
   278→    // All other queries require authentication
   279→    const authResult = await requireAnyAuth(request);
   280→    if (authResult instanceof NextResponse) return authResult;
   281→    const authUser = authResult.user;
   282→
   283→    if (!firebaseOk) {
   284→      return NextResponse.json({ error: 'Firebase no configurado', code: 'NO_FIREBASE' }, { status: 503 });
   285→    }
   286→
   287→    let bookings: Record<string, unknown>[];
   288→
   289→    // Non-admin users: only their own bookings
   290→    if (authUser.role !== 'admin' && authUser.role !== 'super_admin') {
   291→      if (userId && userId !== authUser.id) {
   292→        return NextResponse.json({ error: 'No puedes ver reservas de otros usuarios.' }, { status: 403 });
   293→      }
   294→      const effectiveUserId = userId || authUser.id;
   295→      bookings = await searchBookingsForUser(effectiveUserId, authUser.email);
   296→    } else {
   297→      // Admin: fetch with optional filters (date range for performance)
   298→      bookings = (await getBookings({
   299→        courtId: courtId || undefined,
   300→        userId: userId || undefined,
   301→        date: date || undefined,
   302→        dateFrom: dateFrom || undefined,
   303→        dateTo: dateTo || undefined,
   304→        status: status || undefined,
   305→      })) as Record<string, unknown>[];
   306→      // DEBUG removed (P3-16): raw booking data no longer logged
   307→    }
   308→
   309→    // Enrich each booking with cached court/user lookups (avoid N+1 Firestore reads)
   310→    const courtCache = new Map<string, Record<string, unknown> | null>();
   311→    const userCache = new Map<string, Record<string, unknown> | null>();
   312→    const getCourtCached = async (cid: string) => {
   313→      if (courtCache.has(cid)) return courtCache.get(cid)!;
   314→      try {
   315→        const c = await getCourtById(cid);
   316→        courtCache.set(cid, c as Record<string, unknown> | null);
   317→        return c as Record<string, unknown> | null;
   318→      } catch { courtCache.set(cid, null); return null; }
   319→    };
   320→    const getUserCached = async (uid: string) => {
   321→      if (userCache.has(uid)) return userCache.get(uid)!;
   322→      try {
   323→        const u = await getUserById(uid);
   324→        userCache.set(uid, u as Record<string, unknown> | null);
   325→        return u as Record<string, unknown> | null;
   326→      } catch { userCache.set(uid, null); return null; }
   327→    };
   328→
   329→    let enriched: Record<string, unknown>[] = [];
   330→    // Process in batches of 10 to avoid overwhelming Firestore
   331→    const BATCH_SIZE = 10;
   332→    for (let i = 0; i < bookings.length; i += BATCH_SIZE) {
   333→      const batch = bookings.slice(i, i + BATCH_SIZE);
   334→      const results = await Promise.allSettled(
   335→        batch.map(async (b) => {
   336→          const allCourtIds: string[] = Array.isArray(b.court_ids)
   337→            ? b.court_ids as string[] : (b.court_id ? [b.court_id as string] : []);
   338→          let court: Record<string, unknown> | null = null;
   339→          let courts: Array<Record<string, unknown>> = [];
   340→          let user: Record<string, unknown> | null = null;
   341→          try {
   342→            if (allCourtIds.length > 0) court = await getCourtCached(allCourtIds[0]);
   343→          } catch { /* skip */ }
   344→          if (allCourtIds.length > 1) {
   345→            const enrichedCourts: Array<Record<string, unknown>> = [];
   346→            for (const cid of allCourtIds) {
   347→              try {
   348→                const c = await getCourtCached(cid);
   349→                if (c) enrichedCourts.push(c);
   350→              } catch { /* skip */ }
   351→            }
   352→            courts = enrichedCourts;
   353→          }
   354→          try {
   355→            if (b.user_id) user = await getUserCached(b.user_id as string);
   356→          } catch { /* skip */ }
   357→          return toCamelBooking({ ...b, _court: court, _courts: courts, _user: user });
   358→        })
   359→      );
   360→      for (const r of results) {
   361→        if (r.status === 'fulfilled') {
   362→          enriched.push(r.value);
   363→        } else {
   364→          console.error('[BOOKINGS] Batch enrichment error:', r.reason);
   365→        }
   366→      }
   367→    }
   368→
   369→    // Default server-side sort: today first, future ascending, past descending
   370→    // Within each date: chronological time
   371→    const today = new Date().toLocaleDateString('en-CA', { timeZone: 'America/Lima' });
   372→    enriched.sort((a, b) => {
   373→      const dA = String(a.date || ''), dB = String(b.date || '');
   374→      const aIsToday = dA === today, bIsToday = dB === today;
   375→      const aFuture = dA > today, bFuture = dB > today;
   376→      const aPast = dA < today, bPast = dB < today;
   377→      // 1) Today always first
   378→      if (aIsToday && !bIsToday) return -1;
   379→      if (!aIsToday && bIsToday) return 1;
   380→      // 2) Both on same side of today
   381→      if (aFuture && bFuture) {
   382→        // Future dates: ascending (correlativo)
   383→        const dc = dA.localeCompare(dB);
   384→        if (dc !== 0) return dc;
   385→      } else if (aPast && bPast) {
   386→        // Past dates: descending (most recent past first)
   387→        const dc = dB.localeCompare(dA);
   388→        if (dc !== 0) return dc;
   389→      }
   390→      // Same date: chronological by start time
   391→      return String(a.startTime || '').localeCompare(String(b.startTime || ''));
   392→    });
   393→
   394→    // FIX P0-4: Lazy-expire reserved bookings past their TTL
   395→    try {
   396→      const db = await getAdminDb();
   397→      const nowMs = Date.now();
   398→      const expiredIds: string[] = [];
   399→      for (const b of bookings) {
   400→        if (migrateStatus(b.status || 'reserved') !== 'reserved') continue;
   401→        const exp = b.expires_at;
   402→        if (!exp) continue;
   403→        const expMs = exp.toMillis?.()?.() || new Date(exp).getTime();
   404→        if (expMs <= nowMs) expiredIds.push(b.id as string);
   405→      }
   406→      if (expiredIds.length > 0) {
   407→        await Promise.all(expiredIds.map((id) =>
   408→          db.collection('bookings').doc(id).update({
   409→            status: 'cancelled',
   410→            slot_status: 'available',
   411→            updated_at: Timestamp.now(),
   412→          })
   413→        ));
   414→        // Remove expired from enriched results
   415→        const expiredSet = new Set(expiredIds);
   416→        enriched = enriched.filter((b) => !expiredSet.has(b.id as string));
   417→      }
   418→    } catch { /* non-critical: best-effort cleanup */ }
   419→
   420→    return NextResponse.json(enriched);
   421→  } catch (error) {
   422→    console.error('[BOOKINGS] GET error:', error);
   423→    const msg = error instanceof Error ? error.message : 'Unknown error';
   424→    return NextResponse.json({ error: 'Error al obtener reservas', detail: msg }, { status: 500 });
   425→  }
   426→}

   428→export async function POST(request: NextRequest) {
   429→  try {
   430→    const authResult = await requireAnyAuth(request);
   431→    if (authResult instanceof NextResponse) return authResult;
   432→    const authUser = authResult.user;
   433→
   434→    if (!isFirebaseAvailable()) {
   435→      return NextResponse.json({ error: 'Firebase no configurado' }, { status: 503 });
   436→    }
   437→
   438→    const body = await request.json();
   439→    const {
   440→      courtId,
   441→      courtIds: bodyCourtIds,
   442→      userId,
   443→      date,
   444→      startTime,
   445→      endTime,
   446→      totalPrice,
   447→      advanceAmount,
   448→      remainingAmount,
   449→      status,
   450→      paymentMethod,
   451→      notes,
   452→      equipmentItems,
   453→      selectedSlots,
   454→    } = body;
   455→
   456→    // ── Resolve court IDs (support both single courtId and multi-court courtIds) ──
   457→    const allCourtIds: string[] = Array.isArray(bodyCourtIds) && bodyCourtIds.length > 0
   458→      ? bodyCourtIds
   459→      : courtId
   460→        ? [courtId]
   461→        : [];
   462→
   463→    // ── Validate payment method (restricted set, UPPERCASE) ──
   464→    // FIX P1-7: Added 'CULQI', 'TARJETA', 'CARD' so online payments are stored correctly
   465→    const VALID_PAYMENT_METHODS = ['EFECTIVO', 'YAPE', 'PLIN', 'CULQI', 'TARJETA', 'CARD'];
   466→    const normalizedPaymentMethod = VALID_PAYMENT_METHODS.includes((paymentMethod || '').toUpperCase())
   467→      ? (paymentMethod || '').toUpperCase()
   468→      : paymentMethod?.toLowerCase() === 'culqi' ? 'CULQI'
   469→        : paymentMethod?.toLowerCase() === 'card' ? 'TARJETA'
   470→          : 'EFECTIVO';
   471→
   472→    if (allCourtIds.length === 0 || !userId || !date || !startTime || !endTime) {
   473→      return NextResponse.json(
   474→        { error: 'Faltan campos requeridos: courtId/courtIds, userId, date, startTime, endTime' },
   475→        { status: 400 }
   476→      );
   477→    }
   478→
   479→    // ── Minimum advance validation ──
   480→    // Admin/super_admin: NO time restrictions — can book any slot (walk-in, past, current)
   481→    // Regular users: require 30 min minimum advance
   482→    const isAdmin = authUser.role === 'admin' || authUser.role === 'super_admin';
   483→
   484→    if (!isAdmin) {
   485→      const now = new Date();
   486→      const slotDateTime = new Date(`${date}T${startTime}:00`);
   487→      const diffMs = slotDateTime.getTime() - now.getTime();
   488→      const thirtyMinMs = 30 * 60 * 1000;
   489→      if (diffMs < thirtyMinMs) {
   490→        const minsLeft = Math.floor(diffMs / 60000);
   491→        if (minsLeft < 0) {
   492→          return NextResponse.json(
   493→            { error: 'No se puede reservar un horario que ya ha pasado.' },
   494→            { status: 422 }
   495→          );
   496→        }
   497→        return NextResponse.json(
   498→          { error: `La reserva requiere al menos 30 minutos de anticipación. El horario seleccionado comienza en ${minsLeft} minuto${minsLeft === 1 ? '' : 's'}. Selecciona un horario a partir de las ${new Date(now.getTime() + thirtyMinMs).toLocaleTimeString('es-PE', { hour: '2-digit', minute: '2-digit', hour12: false })} o posterior.` },
   499→          { status: 422 }
   500→        );
   501→      }
   502→    }
   503→
   504→    // Non-admin users can only create bookings for themselves
   505→    if (!isAdmin) {
   506→      if (userId !== authUser.id) {
   507→        return NextResponse.json(
   508→          { error: 'No puedes crear reservas para otros usuarios.' },
   509→          { status: 403 }
   510→        );
   511→      }
   512→    }
   513→
   514→    // ── Calculate price (BEFORE transaction — needed for booking data) ──
   515→    const eqItems = Array.isArray(equipmentItems) ? equipmentItems : [];
   516→    let equipmentSubtotal = 0;
   517→    for (const eq of eqItems) {
   518→      equipmentSubtotal += (eq.quantity || 0) * (eq.unit_price || eq.unitPrice || 0);
   519→    }
   520→
   521→    // FIX P1-6: ALWAYS recalculate price server-side from court rates.
   522→    // Client-sent totalPrice is ignored for storage — used only as a UI hint.
   523→    let courtPriceTotal = 0;
   524→    for (const cId of allCourtIds) {
   525→      const court = await getCourtById(cId);
   526→      if (court) {
   527→        const schedule = court.pricing_schedule as Array<{ label: string; startHour: number; endHour: number; pricePerHour: number }> | undefined;
   528→        let courtPrice = 0;
   529→        if (schedule && schedule.length > 0) {
   530→          courtPrice = calculatePriceForTimeSlot(schedule, startTime, endTime);
   531→          if (courtPrice <= 0) courtPrice = (court.price_per_hour as number) || 0;
   532→        } else {
   533→          courtPrice = (court.price_per_hour as number) || 0;
   534→        }
   535→        courtPriceTotal += courtPrice;
   536→      }
   537→    }
   538→
   539→    const price = courtPriceTotal + equipmentSubtotal;
   540→
   541→    // B4 FIX: Validate advance + remaining === total consistency
   542→    const adv = parseFloat(advanceAmount) || price * 0.5;
   543→    let rem = parseFloat(remainingAmount) || price - adv;
   544→    // Force consistency: remaining must equal total - advance
   545→    rem = Math.max(0, Math.round((price - adv) * 100) / 100);
   546→    const bookingStatus = migrateStatus(status || 'reserved');
   547→
   548→    // Resolve client email for denormalized search
   549→    let clientEmail = authUser.email;
   550→    try {
   551→      const clientUser = await getUserById(userId);
   552→      if (clientUser?.email) clientEmail = clientUser.email;
   553→    } catch { /* fallback to admin email */ }
   554→
   555→    // ── FIX P1-5: Check overlap + create booking in Firestore transaction ──
   556→    const db = await getAdminDb();
   557→    let bookingId: string;
   558→
   559→    try {
   560→      bookingId = await db.runTransaction(async (transaction) => {
   561→        // Re-check overlap inside transaction for each court
   562→        for (const cId of allCourtIds) {
   563→          const bookingsRef = db.collection('bookings')
   564→            .where('date', '==', date)
   565→            .where('court_ids', 'array-contains', cId);
   566→          const snapshot = await transaction.get(bookingsRef);
   567→
   568→          for (const doc of snapshot.docs) {
   569→            const b = doc.data();
   570→            const bCourtIds: string[] = Array.isArray(b.court_ids) ? b.court_ids : [b.court_id];
   571→            if (!bCourtIds.includes(cId)) continue;
   572→            if (migrateStatus(b.status || '') === 'cancelled') continue;
   573→            if ((b.start_time || '') < endTime && (b.end_time || '') > startTime) {
   574→              let courtName = cId;
   575→              try {
   576→                const c = await getCourtById(cId);
   577→                if (c?.name) courtName = c.name as string;
   578→              } catch { /* use ID */ }
   579→              throw new Error(`Cancha no disponible: ${courtName}. Por favor selecciona otro horario o cancha.`);
   580→            }
   581→          }
   582→        }
   583→
   584→        // All clear — create the booking inside the transaction
   585→        const now = Timestamp.now();
   586→        const docRef = db.collection('bookings').doc();
   587→        transaction.set(docRef, {
   588→          court_id: allCourtIds[0],
   589→          court_ids: allCourtIds,
   590→          user_id: userId,
   591→          user_email: clientEmail || null,
   592→          date,
   593→          start_time: startTime,
   594→          end_time: endTime,
   595→          total_price: price,
   596→          court_subtotal: courtPriceTotal,
   597→          equipment_subtotal: equipmentSubtotal,
   598→          equipment_items: eqItems,
   599→          equipment_delivered: false,
   600→          equipment_returned: false,
   601→          advance_amount: adv,
   602→          remaining_amount: rem,
   603→          status: bookingStatus,
   604→          slot_status: 'available',
   605→          payment_method: normalizedPaymentMethod || null,
   606→          notes: notes || null,
   607→          recurring_group_id: null,
   608→          recurring_index: null,
   609→          selected_slots: Array.isArray(selectedSlots) ? selectedSlots : [],
   610→          created_at: now,
   611→          updated_at: now,
   612→          // FIX P0-4: Auto-expire unpaid bookings after 15 minutes
   613→          expires_at: Timestamp.fromMillis(now.toMillis() + 15 * 60 * 1000),
   614→        });
   615→        return docRef.id;
   616→      });
   617→    } catch (txError: any) {
   618→      if (txError?.message?.includes('Cancha no disponible')) {
   619→        return NextResponse.json({ error: txError.message }, { status: 409 });
   620→      }
   621→      throw txError;
   622→    }
   623→
   624→    // Booking was already created inside the transaction (P1-5)
   625→    const id = bookingId;
   626→
   627→    // B5 FIX: Only create payment record if advance > 0
   628→    if (adv > 0.01) {
   629→      try {
   630→        await createPayment(id, {
   631→          user_id: userId,
   632→          amount: adv,
   633→          type: 'advance',
   634→          method: normalizedPaymentMethod || 'EFECTIVO',
   635→          status: 'completed',
   636→        });
   637→      } catch (payErr) {
   638→        console.error('[BOOKINGS] Warning: could not create payment record:', payErr);
   639→      }
   640→    }
   641→
   642→    return NextResponse.json({
   643→      id,
   644→      courtId: allCourtIds[0],
   645→      courtIds: allCourtIds,
   646→      userId,
   647→      date,
   648→      startTime,
   649→      endTime,
   650→      totalPrice: price,
   651→      advanceAmount: adv,
   652→      remainingAmount: rem,
   653→      status: bookingStatus,
   654→      paymentMethod: normalizedPaymentMethod || paymentMethod || null,
   655→      success: true,
   656→    }, { status: 201 });
   657→  } catch (error: unknown) {
   658→    const err = error as { message?: string; code?: string; stack?: string };
   659→    console.error('[BOOKINGS] POST error:', err?.message || err);
   660→    console.error('[BOOKINGS] Stack:', err?.stack || 'no stack');
   661→    return NextResponse.json({
   662→      error: 'Failed to create booking',
   663→      detail: err?.message || 'Unknown error'
   664→    }, { status: 500 });
   665→  }
   666→}

   696→        }
   697→      } else {
   698→        return NextResponse.json(
   699→          { error: 'Solo puedes cancelar tus propias reservas.' },
   700→          { status: 403 }
   701→        );
   702→      }
   703→    }
   704→
   705→    const updateData: Record<string, unknown> = {};

   709→    if (slot_status) updateData.slot_status = slot_status;

   715→      const upper = (reqPaymentMethod as string).toUpperCase();
   716→      updateData.payment_method = VALID_PM.includes(upper) ? upper : 'EFECTIVO';
   717→    }
   718→    if (typeof equipmentDelivered === 'boolean') updateData.equipment_delivered = equipmentDelivered;
   719→    if (typeof equipmentReturned === 'boolean') updateData.equipment_returned = equipmentReturned;
   720→
   721→    // Extend time: update end_time and total_price, create payment record
   722→    if (extendTime && reqEndTime && reqTotalPrice) {
   723→      // Validate: new end must be after current end
   724→      const booking = await getBookingById(id);
   725→      if (booking) {
   726→        const currentEnd = booking.end_time as string || '';
   727→        if (reqEndTime <= currentEnd) {
   728→          return NextResponse.json({ error: 'La nueva hora de fin debe ser posterior a la actual' }, { status: 400 });
   729→        }
   730→        // Check for conflicts with other bookings (same court, same date, overlapping time)
   731→        const allCourtIds: string[] = Array.isArray(booking.court_ids) ? booking.court_ids : (booking.court_id ? [booking.court_id] : []);
   732→        const bookingDate = booking.date as string;

   743→            if (ob.date !== bookingDate) return false;
   744→            const obCourtIds: string[] = Array.isArray(ob.court_ids) ? ob.court_ids : (ob.court_id ? [ob.court_id] : []);

   746→            return (ob.start_time as string) < reqEndTime && (ob.end_time as string) > (booking.start_time as string);

   768→                user_id: booking.user_id as string,

   770→                type: 'extension',

   772→                status: 'completed',

   777→          }
   778→        }
   779→      }

   844→        // Recalc court_subtotal
   845→        const equipSub = (updateData.equipment_subtotal as number) || (booking.equipment_subtotal as number) || 0;
   846→        updateData.court_subtotal = Math.max(0, Math.round((reqTotalPrice - equipSub) * 100) / 100);

   849→        const currentTotal = (booking.total_price as number) || 0;
   850→        updateData.advance_amount = reqAdvance;
   851→        updateData.remaining_amount = Math.max(0, Math.round((currentTotal - reqAdvance) * 100) / 100);

   968→                    userEmail = (user?.email as string) || null;
   969→                  }
   970→                }
   971→              } catch { /* non-critical */ }
   972→
   973→              // Create retained advance record
   974→              try {

   982→                  amount: advAmount,

   995→                await createPayment(id, {
   996→                  user_id: (bookingForCancel.user_id as string) || '',
   997→                  amount: advAmount,
   998→                  type: isRetained ? 'retained' : 'refund',

  1000→                  status: 'completed',
  1001→                });
  1002→              } catch (payErr) {
  1003→                console.error('[BOOKINGS] Warning: could not create payment record for cancellation:', payErr);
  1004→                warningFlags.push('payment_record_failed');
  1005→              }
  1006→            }
  1007→            }
  1008→          }
  1009→        } catch (err) {
  1010→          console.error('[BOOKINGS] Warning: could not process retained advance on cancellation:', err);
  1011→          warningFlags.push('retained_advance_failed');
  1012→        }
  1013→      }
  1014→    }
  1015→
  1016→    // B11 FIX: Skip advance payment record when cancelling or editBooking (handled above / no record needed)
  1017→    if (typeof reqAdvance === 'number' && typeof reqRemaining === 'number' && !isCancelling && !editBooking) {
  1018→      try {

  1022→          const paymentIncrement = reqAdvance - prevAdvance;

  1025→              await createPayment(id, {
  1026→                user_id: booking.user_id as string,
  1027→                amount: paymentIncrement,
  1028→                type: reqRemaining <= 0 ? 'full_payment' : 'advance',
  1029→                method: (updateData.payment_method as string) || 'EFECTIVO',
  1030→                status: 'completed',

  1034→            }
  1035→          }
  1036→        }
  1037→      } catch { /* non-critical: proceed with booking update */ }
  1038→    }
  1039→
  1040→    await updateBooking(id, updateData);
  1041→
  1042→    // FIX Bug #9: Return warnings if any
  1043→    if (warningFlags.length > 0) {
  1044→      return NextResponse.json({ success: true, warnings: warningFlags });
  1045→    }
  1046→    return NextResponse.json({ success: true });
  1047→  } catch (error) {
  1048→    console.error('[BOOKINGS] PUT error:', error);
  1049→    const msg = error instanceof Error ? error.message : 'Unknown error';
  1050→    return NextResponse.json({ error: 'Error al actualizar reserva', detail: msg }, { status: 500 });
  1051→  }
