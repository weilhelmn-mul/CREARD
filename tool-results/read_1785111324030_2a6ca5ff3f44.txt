    11→    11→// --- Animated Counter Component ---
    12→    12→function AnimatedCounter({ target, duration = 2, suffix = '' }: { target: number; duration?: number; suffix?: string }) {
    13→    13→  const [count, setCount] = useState(0)
    14→    14→  const ref = useRef<HTMLSpanElement>(null)
    15→    15→  const isInView = useInView(ref, { once: true, margin: '-50px' })
    16→    16→  const hasStarted = useRef(false)
    17→    17→
    18→    18→  useEffect(() => {
    19→    19→    if (!isInView || hasStarted.current) return
    20→    20→    hasStarted.current = true
    21→    21→
    22→    22→    const startTime = performance.now()
    23→    23→    const from = 0
    24→    24→    const to = target
    25→    25→
    26→    26→    function update(currentTime: number) {
    27→    27→      const elapsed = currentTime - startTime
    28→    28→      const progress = Math.min(elapsed / (duration * 1000), 1)
    29→    29→      const eased = 1 - Math.pow(1 - progress, 3)
    30→    30→      const current = Math.round(from + (to - from) * eased)
    31→    31→      setCount(current)
    32→    32→      if (progress < 1) {
    33→    33→        requestAnimationFrame(update)
    34→    34→      }
    35→    35→    }
    36→    36→    requestAnimationFrame(update)
    37→    37→  }, [isInView, target, duration])
    38→    38→
    39→    39→  return (
    40→    40→    <span ref={ref}>
    41→    41→      {count}{suffix}
    42→    42→    </span>
    43→    43→  )
    44→    44→}

    46→    46→// --- Date Chips ---
    47→    47→function getDateChips() {
    48→    48→  const days = ['Dom', 'Lun', 'Mar', 'Mié', 'Jue', 'Vie', 'Sáb']
    49→    49→  const months = ['Ene', 'Feb', 'Mar', 'Abr', 'May', 'Jun', 'Jul', 'Ago', 'Sep', 'Oct', 'Nov', 'Dic']
    50→    50→  const chips: { date: Date; label: string; dayName: string; isToday: boolean }[] = []
    51→    51→
    52→    52→  for (let i = 0; i < 4; i++) {
    53→    53→    const d = new Date()
    54→    54→    d.setDate(d.getDate() + i)
    55→    55→    chips.push({
    56→    56→      date: d,
    57→    57→      label: `${d.getDate()} ${months[d.getMonth()]}`,
    58→    58→      dayName: i === 0 ? 'Hoy' : days[d.getDay()],
    59→    59→      isToday: i === 0,
    60→    60→    })
    61→    61→  }
    62→    62→  return chips
    63→    63→}
