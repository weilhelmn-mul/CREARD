    59→function formatDateISO(date: Date): string {
    60→  const y = date.getFullYear()
    61→  const m = String(date.getMonth() + 1).padStart(2, '0')
    62→  const d = String(date.getDate()).padStart(2, '0')
    63→  return `${y}-${m}-${d}`
    64→}
    65→
    66→function getNext14Days(): Date[] {
    67→  const days: Date[] = []
    68→  const now = new Date()
    69→  for (let i = 0; i < 14; i++) {
    70→    days.push(new Date(now.getFullYear(), now.getMonth(), now.getDate() + i))
    71→  }
    72→  return days
    73→}
    74→
    75→function isToday(date: Date): boolean {
    76→  const t = new Date()
    77→  return date.getFullYear() === t.getFullYear() &&
    78→    date.getMonth() === t.getMonth() &&
    79→    date.getDate() === t.getDate()
    80→}
    81→
    82→/** Returns the next full hour boundary. E.g. if now is 14:35, returns 15 */
    83→function getNextFullHour(): number {
    84→  const now = new Date()
    85→  const h = now.getHours()
    86→  const m = now.getMinutes()
    87→  return m >= 1 ? h + 1 : h
    88→}
    89→
    90→function getMinSlotHour(date: Date): number {
    91→  if (isToday(date)) {
    92→    return Math.max(SLOT_START, getNextFullHour())
    93→  }
    94→  return SLOT_START
    95→}
    96→
    97→function generateHourSlots(): string[] {
    98→  const slots: string[] = []
    99→  for (let h = SLOT_START; h < SLOT_END; h++) {
   100→    slots.push(`${String(h).padStart(2, '0')}:00`)
   101→  }
   102→  return slots
   103→}
   104→
   105→function formatHour(slot: string): string {
   106→  const [h] = slot.split(':').map(Number)
   107→  if (h === 0 || h === 12) return `${h === 0 ? 12 : 12}:00 ${h === 0 ? 'AM' : 'PM'}`
   108→  if (h < 12) return `${h}:00 AM`
   109→  return `${h - 12}:00 PM`
   110→}
   111→
   112→function getRefCode(bookingId: string): string {
   113→  const hash = bookingId.slice(-8).toUpperCase()
   114→  return `CRE-${hash.slice(0, 4)}-${hash.slice(4)}`
   115→}
   116→
   117→/** Calculate price for a 1-hour slot using a pricing schedule */
   118→function calcSlotPrice(schedule: PricingScheduleItem[], slot: string): number {
   119→  const hour = parseInt(slot.split(':')[0], 10)
   120→  if (schedule.length > 0) {
   121→    for (const tier of schedule) {
   122→      if (hour >= tier.startHour && hour < tier.endHour) return tier.pricePerHour
   123→    }
   124→  }
   125→  return 0
   126→}
