   169→/** Compare Firestore Timestamps (seconds) or ISO strings for sorting */
   170→function compareTimestamps(a: unknown, b: unknown): number {
   171→  const toMs = (t: unknown): number => {
   172→    if (!t) return 0
   173→    if (typeof t === 'number') return t * 1000 // Firestore seconds
   174→    if (typeof t === 'object' && t !== null && '_seconds' in (t as Record<string, unknown>)) return (t as Record<string, unknown>)._seconds as number * 1000
   175→    if (typeof t === 'string') return new Date(t).getTime()
   176→    return 0
   177→  }
   178→  return toMs(a) - toMs(b)
   179→}

   185→/* ─── Sortable Section Card ─── */
   186→function SortableSectionCard({
   187→  id,
   188→  label,
   189→  icon,
   190→  color,
   191→  isCustom,
   192→  visible,
   193→  onToggle,
   194→  onEdit,
   195→  onDelete,
   196→}: {
   197→  id: string
   198→  label: string
   199→  icon: string
   200→  color: string
   201→  isCustom: boolean
   202→  visible: boolean
   203→  onToggle: () => void
   204→  onEdit: () => void
   205→  onDelete?: () => void
   206→}) {
   207→  const { attributes, listeners, setNodeRef, transform, transition, isDragging } = useSortable({ id })
   208→  const style = { transform: CSS.Transform.toString(transform), transition }
   209→
   210→  return (
   211→    <div
   212→      ref={setNodeRef}
   213→      style={style}
   214→      className={`glass-card rounded-xl p-3 flex items-center gap-3 transition-all ${isDragging ? 'opacity-60 shadow-xl z-50' : ''} ${!visible ? 'opacity-40' : ''}`}
   215→    >
   216→      {/* Drag handle */}
   217→      <button type="button" {...attributes} {...listeners} className="cursor-grab active:cursor-grabbing text-cm-on-surface-variant/40 hover:text-cm-on-surface-variant p-1">
   218→        <span className="material-symbols-outlined text-[18px]">drag_indicator</span>
   219→      </button>
   220→
   221→      {/* Icon */}
   222→      <div className={`w-9 h-9 rounded-lg flex items-center justify-center flex-shrink-0 ${color}`}>
   223→        <span className="material-symbols-outlined text-[18px] text-white" style={{ fontVariationSettings: '"FILL" 1' }}>
   224→          {icon}
   225→        </span>
   226→      </div>
   227→
   228→      {/* Label */}
   229→      <span className="flex-1 text-sm font-semibold text-cm-on-surface truncate font-[family-name:var(--font-sora)]">
   230→        {label}
   231→      </span>
   232→
   233→      {/* Actions */}
   234→      <div className="flex items-center gap-1.5 flex-shrink-0">
   235→        <button type="button" onClick={onToggle} className={`p-1.5 rounded-lg transition-colors ${visible ? 'text-cm-primary hover:bg-cm-primary/10' : 'text-cm-on-surface-variant/40 hover:bg-white/5'}`} title={visible ? 'Ocultar' : 'Mostrar'}>
   236→          <span className="material-symbols-outlined text-[16px]">{visible ? 'visibility' : 'visibility_off'}</span>
   237→        </button>
   238→        <button type="button" onClick={onEdit} className="p-1.5 rounded-lg text-cm-on-surface-variant hover:text-cm-primary hover:bg-cm-primary/10 transition-colors" title="Editar">
   239→          <span className="material-symbols-outlined text-[16px]">edit</span>
   240→        </button>
   241→        {isCustom && onDelete && (
   242→          <button type="button" onClick={onDelete} className="p-1.5 rounded-lg text-cm-on-surface-variant hover:text-red-400 hover:bg-red-500/10 transition-colors" title="Eliminar">
   243→            <span className="material-symbols-outlined text-[16px]">delete</span>
   244→          </button>
   245→        )}
   246→      </div>
   247→    </div>
   248→  )
   249→}
