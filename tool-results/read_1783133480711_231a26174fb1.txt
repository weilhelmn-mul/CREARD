   501→   501→   501→          if (c?.name) courtName = c.name as string;
   502→   502→   502→        } catch { /* use ID */ }
   503→   503→   503→        conflicts.push({ courtId: cId, courtName });
   504→   504→   504→      }
   505→   505→   505→    }
   506→   506→   506→
   507→   507→   507→    if (conflicts.length > 0) {
   508→   508→   508→      const conflictNames = conflicts.map((c) => c.courtName).join(', ');

   512→   512→   512→          conflicts,
   513→   513→   513→        },
   514→   514→   514→        { status: 409 }
   515→   515→   515→      );
   516→   516→   516→    }
   517→   517→   517→
   518→   518→   518→    // B15 FIX: Re-check conflicts right before creation (optimistic concurrency)
   519→   519→   519→    for (const cId of allCourtIds) {
   520→   520→   520→      const recheck = await getBookings({ courtId: cId, date });
   521→   521→   521→      const stillConflicts = recheck.filter(
   522→   522→   522→        (b) => {
   523→   523→   523→          const bCourtIds: string[] = Array.isArray(b.court_ids)
   524→   524→   524→            ? b.court_ids as string[]
   525→   525→   525→            : [b.court_id as string];
   526→   526→   526→          if (!bCourtIds.includes(cId)) return false;
   527→   527→   527→          return (
   528→   528→   528→            !['cancelled'].includes(migrateStatus(b.status || '')) &&
   529→   529→   529→            (b.start_time || '') < endTime &&
   530→   530→   530→            (b.end_time || '') > startTime
   531→   531→   531→          );
   532→   532→   532→        }
   533→   533→   533→      );
   534→   534→   534→      if (stillConflicts.length > 0) {
   535→   535→   535→        let courtName = cId;
   536→   536→   536→        try {
   537→   537→   537→          const c = await getCourtById(cId);
   538→   538→   538→          if (c?.name) courtName = c.name as string;
   539→   539→   539→        } catch { /* use ID */ }
   540→   540→   540→        return NextResponse.json(
   541→   541→   541→          { error: `Horario ocupado en ${courtName}. Alguien reservó mientras completabas el formulario. Intenta de nuevo.`, conflicts: [{ courtId: cId, courtName }] },
   542→   542→   542→          { status: 409 }
   543→   543→   543→        );
   544→   544→   544→      }
   545→   545→   545→    }
   546→   546→   546→
   547→   547→   547→    // ── Calculate price ──
   548→   548→   548→    // Frontend sends totalPrice as grand total (court + equipment combined).
   549→   549→   549→    // We compute courtSubtotal and equipmentSubtotal separately for storage/display.
   550→   550→   550→    const eqItems = Array.isArray(equipmentItems) ? equipmentItems : [];
   551→   551→   551→    let equipmentSubtotal = 0;
   552→   552→   552→    for (const eq of eqItems) {
   553→   553→   553→      equipmentSubtotal += (eq.quantity || 0) * (eq.unit_price || eq.unitPrice || 0);
   554→   554→   554→    }
   555→   555→   555→
   556→   556→   556→    let courtPriceTotal: number;
   557→   557→   557→    const providedTotal = parseFloat(totalPrice) || 0;
   558→   558→   558→    if (providedTotal > 0) {
   559→   559→   559→      // Frontend already included equipment in totalPrice — extract court portion
   560→   560→   560→      courtPriceTotal = Math.max(0, providedTotal - equipmentSubtotal);
   561→   561→   561→    } else {
   562→   562→   562→      // No price from frontend — calculate from court rates
   563→   563→   563→      courtPriceTotal = 0;
   564→   564→   564→      for (const cId of allCourtIds) {
   565→   565→   565→        const court = await getCourtById(cId);
   566→   566→   566→        if (court) {
   567→   567→   567→          const schedule = court.pricing_schedule as Array<{ label: string; startHour: number; endHour: number; pricePerHour: number }> | undefined;
   568→   568→   568→          let courtPrice = 0;
   569→   569→   569→          if (schedule && schedule.length > 0) {
   570→   570→   570→            courtPrice = calculatePriceForTimeSlot(schedule, startTime, endTime);
   571→   571→   571→            if (courtPrice <= 0) courtPrice = (court.price_per_hour as number) || 0;
   572→   572→   572→          } else {
   573→   573→   573→            courtPrice = (court.price_per_hour as number) || 0;
   574→   574→   574→          }
   575→   575→   575→          courtPriceTotal += courtPrice;
   576→   576→   576→        }
   577→   577→   577→      }
   578→   578→   578→    }
   579→   579→   579→
   580→   580→   580→    const price = courtPriceTotal + equipmentSubtotal;
   581→   581→   581→
   582→   582→   582→    // B4 FIX: Validate advance + remaining === total consistency
   583→   583→   583→    const adv = parseFloat(advanceAmount) || price * 0.5;
   584→   584→   584→    let rem = parseFloat(remainingAmount) || price - adv;
   585→   585→   585→    // Force consistency: remaining must equal total - advance
   586→   586→   586→    rem = Math.max(0, Math.round((price - adv) * 100) / 100);
   587→   587→   587→    const bookingStatus = migrateStatus(status || 'reserved');
   588→   588→   588→
   589→   589→   589→    // Resolve client email for denormalized search
   590→   590→   590→    let clientEmail = authUser.email;

   594→   594→   594→    } catch { /* fallback to admin email */ }
   595→   595→   595→
   596→   596→   596→    // Save to Firestore — single booking with all courts

   598→   598→   598→      court_ids: allCourtIds,
   599→   599→   599→      user_id: userId,
   600→   600→   600→      user_email: clientEmail,
   601→   601→   601→      date,
   602→   602→   602→      start_time: startTime,
   603→   603→   603→      end_time: endTime,
   604→   604→   604→      total_price: price,
   605→   605→   605→      court_subtotal: courtPriceTotal,
   606→   606→   606→      equipment_subtotal: equipmentSubtotal,
   607→   607→   607→      equipment_items: eqItems,
   608→   608→   608→      advance_amount: adv,
   609→   609→   609→      remaining_amount: rem,
   610→   610→   610→      status: bookingStatus,
   611→   611→   611→      slot_status: 'available',
   612→   612→   612→      payment_method: normalizedPaymentMethod || null,
   613→   613→   613→      notes: notes || null,
   614→   614→   614→    });
   615→   615→   615→
   616→   616→   616→    console.log(`[BOOKINGS] POST: Created booking ${id} for user=${userId} date=${date} ${startTime}-${endTime} courts=${allCourtIds.join(',')} total=${price} advance=${adv} remaining=${rem} method=${normalizedPaymentMethod} email=${clientEmail}`);
   617→   617→   617→
   618→   618→   618→    // B5 FIX: Only create payment record if advance > 0
   619→   619→   619→    if (adv > 0.01) {
   620→   620→   620→      try {
   621→   621→   621→        await createPayment(id, {
   622→   622→   622→          user_id: userId,
   623→   623→   623→          amount: adv,
   624→   624→   624→          type: 'advance',
   625→   625→   625→          method: normalizedPaymentMethod || 'EFECTIVO',
   626→   626→   626→          status: 'completed',
   627→   627→   627→        });
   628→   628→   628→      } catch (payErr) {
   629→   629→   629→        console.error('[BOOKINGS] Warning: could not create payment record:', payErr);
   630→   630→   630→      }
   631→   631→   631→    }
   632→   632→   632→
   633→   633→   633→    return NextResponse.json({
   634→   634→   634→      id,
   635→   635→   635→      courtId: allCourtIds[0],
   636→   636→   636→      courtIds: allCourtIds,
   637→   637→   637→      userId,
   638→   638→   638→      date,
   639→   639→   639→      startTime,
   640→   640→   640→      endTime,
   641→   641→   641→      totalPrice: price,
   642→   642→   642→      advanceAmount: adv,
   643→   643→   643→      remainingAmount: rem,
   644→   644→   644→      status: bookingStatus,
   645→   645→   645→      paymentMethod: paymentMethod || null,
   646→   646→   646→      success: true,
   647→   647→   647→    }, { status: 201 });
   648→   648→   648→  } catch (error: unknown) {
   649→   649→   649→    const err = error as { message?: string; code?: string; stack?: string };
   650→   650→   650→    console.error('[BOOKINGS] POST error:', err?.message || err);
   651→   651→   651→    console.error('[BOOKINGS] Stack:', err?.stack || 'no stack');
   652→   652→   652→    return NextResponse.json({
   653→   653→   653→      error: 'Failed to create booking',
   654→   654→   654→      detail: err?.message || 'Unknown error'
   655→   655→   655→    }, { status: 500 });
   656→   656→   656→  }
   657→   657→   657→}

   704→   704→   704→    }

   710→   710→   710→      // Validate: new end must be after current end
   711→   711→   711→      const booking = await getBookingById(id);

   713→   713→   713→        const currentEnd = booking.end_time as string || '';
   714→   714→   714→        if (reqEndTime <= currentEnd) {

   737→   737→   737→            return (ob.start_time as string) < reqEndTime && (ob.end_time as string) > currentEnd;

   739→   739→   739→          if (realConflicts.length > 0) {

   741→   741→   741→          }
   742→   742→   742→        }

   751→   751→   751→          const paymentIncrement = reqAdvance - prevAdvance;

   754→   754→   754→              await createPayment(id, {
   755→   755→   755→                user_id: booking.user_id as string,
   756→   756→   756→                amount: paymentIncrement,
   757→   757→   757→                type: 'extension',
   758→   758→   758→                method: (reqPaymentMethod as string) || 'EFECTIVO',
   759→   759→   759→                status: 'completed',
   760→   760→   760→              });
   761→   761→   761→            } catch (payErr) {
   762→   762→   762→              console.error('[BOOKINGS] Warning: could not create payment record for extension:', payErr);
   763→   763→   763→            }
   764→   764→   764→          }
   765→   765→   765→        }
   766→   766→   766→      }
   767→   767→   767→    }
   768→   768→   768→

   772→   772→   772→        return NextResponse.json({ error: 'Solo el super administrador puede editar reservas.' }, { status: 403 });
   773→   773→   773→      }
   774→   774→   774→

   776→   776→   776→      if (!booking) {
   777→   777→   777→        return NextResponse.json({ error: 'Reserva no encontrada.' }, { status: 404 });
   778→   778→   778→      }
   779→   779→   779→

   782→   782→   782→      const hasTimeOrCourtChange = reqStartTime || reqEndTime || reqDate || (reqCourtIds && reqCourtIds.length > 0);
   783→   783→   783→      if ((bStatus === 'completed' || bStatus === 'cancelled') && hasTimeOrCourtChange) {
   784→   784→   784→        return NextResponse.json({ error: 'No se puede editar el horario/cancha de una reserva completada o cancelada.' }, { status: 400 });
   785→   785→   785→      }
   786→   786→   786→
   787→   787→   787→      // Validate time if provided
   788→   788→   788→      if (reqStartTime && reqEndTime) {
   789→   789→   789→        if (reqStartTime >= reqEndTime) {
   790→   790→   790→          return NextResponse.json({ error: 'La hora de inicio debe ser anterior a la hora de fin.' }, { status: 400 });

   792→   792→   792→      }

   802→   802→   802→        updateData.court_id = reqCourtIds[0]; // backward compat

   818→   818→   818→        updateData.equipment_items = reqEquipmentItems;

   833→   833→   833→        const currentTotal = (booking.total_price as number) || 0;
   834→   834→   834→        updateData.advance_amount = reqAdvance;
   835→   835→   835→        updateData.remaining_amount = Math.max(0, Math.round((currentTotal - reqAdvance) * 100) / 100);

   837→   837→   837→

   839→   839→   839→      if (reqPaymentMethod) {

   842→   842→   842→      }
   843→   843→   843→
   844→   844→   844→      // Conflict check (only if time or court or date changed)
   845→   845→   845→      const finalCourtIds: string[] = (updateData.court_ids as string[]) || (Array.isArray(booking.court_ids) ? booking.court_ids : (booking.court_id ? [booking.court_id] : []));
   846→   846→   846→      const finalDate = (updateData.date as string) || (booking.date as string);
   847→   847→   847→      const finalStart = (updateData.start_time as string) || (booking.start_time as string);

   849→   849→   849→
   850→   850→   850→      if (finalCourtIds.length > 0 && finalDate && finalStart && finalEnd) {
