   192→/** Compare Firestore Timestamps (seconds) or ISO strings for sorting */
   193→function compareTimestamps(a: unknown, b: unknown): number {
   194→  const toMs = (t: unknown): number => {
   195→    if (!t) return 0
   196→    if (typeof t === 'number') return t * 1000 // Firestore seconds
   197→    if (typeof t === 'object' && t !== null && '_seconds' in (t as Record<string, unknown>)) return (t as Record<string, unknown>)._seconds as number * 1000
   198→    if (typeof t === 'string') return new Date(t).getTime()
   199→    return 0
   200→  }
   201→  return toMs(a) - toMs(b)
   202→}

   208→/* ─── Sortable Section Card ─── */
   209→function SortableSectionCard({
   210→  id,
   211→  label,
   212→  icon,
   213→  color,
   214→  isCustom,
   215→  visible,
   216→  onToggle,
   217→  onEdit,
   218→  onDelete,
   219→}: {
   220→  id: string
   221→  label: string
   222→  icon: string
   223→  color: string
   224→  isCustom: boolean
   225→  visible: boolean
   226→  onToggle: () => void
   227→  onEdit: () => void
   228→  onDelete?: () => void
   229→}) {
   230→  const { attributes, listeners, setNodeRef, transform, transition, isDragging } = useSortable({ id })
   231→  const style = { transform: CSS.Transform.toString(transform), transition }
   232→
   233→  return (
   234→    <div
   235→      ref={setNodeRef}
   236→      style={style}
   237→      className={`glass-card rounded-xl p-3 flex items-center gap-3 transition-all ${isDragging ? 'opacity-60 shadow-xl z-50' : ''} ${!visible ? 'opacity-40' : ''}`}
   238→    >
   239→      {/* Drag handle */}
   240→      <button type="button" {...attributes} {...listeners} className="cursor-grab active:cursor-grabbing text-cm-on-surface-variant/40 hover:text-cm-on-surface-variant p-1">
   241→        <span className="material-symbols-outlined text-[18px]">drag_indicator</span>
   242→      </button>
   243→
   244→      {/* Icon */}
   245→      <div className={`w-9 h-9 rounded-lg flex items-center justify-center flex-shrink-0 ${color}`}>
   246→        <span className="material-symbols-outlined text-[18px] text-white" style={{ fontVariationSettings: '"FILL" 1' }}>
   247→          {icon}
   248→        </span>
   249→      </div>
   250→
   251→      {/* Label */}
   252→      <span className="flex-1 text-sm font-semibold text-cm-on-surface truncate font-[family-name:var(--font-sora)]">
   253→        {label}
   254→      </span>
   255→
   256→      {/* Actions */}
   257→      <div className="flex items-center gap-1.5 flex-shrink-0">
   258→        <button type="button" onClick={onToggle} className={`p-1.5 rounded-lg transition-colors ${visible ? 'text-cm-primary hover:bg-cm-primary/10' : 'text-cm-on-surface-variant/40 hover:bg-white/5'}`} title={visible ? 'Ocultar' : 'Mostrar'}>
   259→          <span className="material-symbols-outlined text-[16px]">{visible ? 'visibility' : 'visibility_off'}</span>
   260→        </button>
   261→        <button type="button" onClick={onEdit} className="p-1.5 rounded-lg text-cm-on-surface-variant hover:text-cm-primary hover:bg-cm-primary/10 transition-colors" title="Editar">
   262→          <span className="material-symbols-outlined text-[16px]">edit</span>
   263→        </button>
   264→        {isCustom && onDelete && (
   265→          <button type="button" onClick={onDelete} className="p-1.5 rounded-lg text-cm-on-surface-variant hover:text-red-400 hover:bg-red-500/10 transition-colors" title="Eliminar">
   266→            <span className="material-symbols-outlined text-[16px]">delete</span>
   267→          </button>
   268→        )}
   269→      </div>
   270→    </div>
   271→  )
   272→}

   532→      if (!target[keys[i]] || typeof target[keys[i]] !== 'object') target[keys[i]] = {}
   533→      target = target[keys[i]] as Record<string, unknown>
   534→    }
   535→    target[keys[keys.length - 1]] = value
   536→    setEditForm(copy)

   546→    }
