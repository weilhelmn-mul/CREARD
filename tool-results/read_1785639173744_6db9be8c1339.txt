    87→    87→// Helpers
    88→    88→const MONTHS_ES = ['Ene', 'Feb', 'Mar', 'Abr', 'May', 'Jun', 'Jul', 'Ago', 'Sep', 'Oct', 'Nov', 'Dic']
    89→    89→const fmtDate = (d: string) => {
    90→    90→  if (!d) return '-'
    91→    91→  if (d.includes('-')) {
    92→    92→    const p = d.split('-').map(Number)
    93→    93→    return `${p[2]} ${MONTHS_ES[p[1] - 1]} ${p[0]}`
    94→    94→  }
    95→    95→  return d
    96→    96→}
    97→    97→const fmtDateShort = (d: string) => {
    98→    98→  if (!d) return '-'
    99→    99→  if (d.includes('-')) {
   100→   100→    const p = d.split('-').map(Number)
   101→   101→    return `${p[2]} ${MONTHS_ES[p[1] - 1]}`
   102→   102→  }
   103→   103→  return d
   104→   104→}
   105→   105→const fmtTimestamp = (t: any) => {
   106→   106→  if (!t) return '-'
   107→   107→  const ms = t._seconds ? t._seconds * 1000 : (typeof t === 'number' ? t : new Date(t).getTime())
   108→   108→  return new Date(ms).toLocaleString('es-PE', { timeZone: 'America/Lima', day: '2-digit', month: '2-digit', year: 'numeric', hour: '2-digit', minute: '2-digit', second: '2-digit', hour12: false })
   109→   109→}
   110→   110→const fmtMoney = (n: number) => `S/ ${(n ?? 0).toFixed(2)}`
   111→   111→const fmtPayDate = (d: string) => {
   112→   112→  if (!d) return '-'
   113→   113→  if (d.includes('/')) {
   114→   114→    const parts = d.split('/')
   115→   115→    if (parts.length === 3) return `${parts[0].padStart(2,'0')}/${parts[1].padStart(2,'0')}/${parts[2]}`
   116→   116→  }
   117→   117→  if (d.includes('-')) {
   118→   118→    const [y, m, day] = d.split('-')
   119→   119→    return `${day}/${m}/${y}`
   120→   120→  }
   121→   121→  return d
   122→   122→}

   159→   159→const calcDuration = (start?: string, end?: string) => {
   160→   160→  if (!start || !end) return '-'
   161→   161→  const [sh] = start.split(':').map(Number)
   162→   162→  const [eh] = end.split(':').map(Number)
   163→   163→  const diff = eh - sh
   164→   164→  return diff > 0 ? `${diff}h` : '-'
   165→   165→}

   167→   167→const getFinancialStatus = (p: PaymentRecord) => {
   168→   168→  const total = p.total_price || 0
   169→   169→  const paid = p.amount_paid || 0
   170→   170→  const remaining = p.remaining_balance || 0
   171→   171→  if (p.payment_status === 'rejected') return { label: 'Rechazado', color: 'text-red-400', bg: 'bg-red-500/10' }
   172→   172→  if (total > 0 && remaining <= 0.01) return { label: 'Pagado totalmente', color: 'text-green-400', bg: 'bg-green-500/10' }
   173→   173→  if (paid > 0 && remaining > 0) return { label: 'Pagado parcialmente', color: 'text-amber-400', bg: 'bg-amber-500/10' }
   174→   174→  if (p.payment_type === 'advance') return { label: 'Adelanto registrado', color: 'text-blue-400', bg: 'bg-blue-500/10' }
   175→   175→  return { label: 'Pendiente', color: 'text-cm-on-surface-variant', bg: 'bg-white/5' }
   176→   176→}
