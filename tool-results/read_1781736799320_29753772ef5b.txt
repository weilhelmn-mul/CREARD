    15→// Migrate old status values to the new 3-status system
    16→function migrateStatus(s: string): string {
    17→  switch (s) {
    18→    case 'confirmed':
    19→    case 'pending':
    20→    case 'partially_paid':
    21→      return 'reserved';
    22→    case 'fully_paid':
    23→    case 'completed':
    24→      return 'completed';
    25→    case 'no_show':
    26→    case 'expired':
    27→      return 'cancelled';
    28→    default:
    29→      return s; // 'reserved', 'completed', 'cancelled' pass through
    30→  }
    31→}

    33→// Transformar snake_case (Firestore) a camelCase (frontend)
    34→function toCamelBooking(b: Record<string, unknown>) {
    35→  // Primary court (backward compat)
    36→  const courtRaw = b._court as Record<string, unknown> | null;
    37→  const court = courtRaw ? {
    38→    id: courtRaw.id || courtRaw.court_id,
    39→    name: courtRaw.name || courtRaw.court_name,
    40→    sport: courtRaw.sport,
    41→    branch: (courtRaw.branch as Record<string, unknown>) || {
    42→      id: courtRaw.branch_id || 'branch-1',
    43→      name: 'CREARD',
    44→    },
    45→  } : null;
    46→
    47→  // Additional courts (multi-court bookings)
    48→  const courtsRaw = b._courts as Array<Record<string, unknown>> | null;
    49→  const courts = courtsRaw ? courtsRaw.map((cr) => ({
    50→    id: cr.id || cr.court_id,
    51→    name: cr.name || cr.court_name,
    52→    sport: cr.sport,
    53→    branch: (cr.branch as Record<string, unknown>) || { id: 'branch-1', name: 'CREARD' },
    54→  })) : [];
    55→
    56→  const userRaw = b._user as Record<string, unknown> | null;
    57→  const user = userRaw ? {
    58→    id: userRaw.id,
    59→    name: userRaw.name,
    60→    email: userRaw.email,
    61→    phone: userRaw.phone,
    62→  } : null;
    63→
    64→  // Court IDs array
    65→  const courtIds: string[] = Array.isArray(b.court_ids) ? b.court_ids : (b.court_id ? [b.court_id] : []);
    66→
    67→  // Equipment items
    68→  const equipmentItems = Array.isArray(b.equipment_items) ? b.equipment_items.map((eq: Record<string, unknown>) => ({
    69→    equipmentId: eq.equipment_id || eq.id,
    70→    name: eq.name,
    71→    quantity: eq.quantity || 1,
    72→    unitPrice: eq.unit_price || eq.price_per_rental || 0,
    73→    subtotal: (eq.quantity || 1) * (eq.unit_price || eq.price_per_rental || 0),
    74→  })) : [];
    75→
    76→  // Normalize: recalc equipment subtotal from items (more reliable than stored value)
    77→  const computedEquipSubtotal = equipmentItems.reduce((s: number, eq: { subtotal: number }) => s + eq.subtotal, 0);
    78→  const equipmentSubtotal = computedEquipSubtotal;
    79→
    80→  // Normalize court_subtotal: if stored value is wrong (old double-counting bug),
    81→  // derive it from total_price - equipment_subtotal
    82→  const totalPrice = b.total_price || 0;
    83→  const storedCourtSubtotal = b.court_subtotal || 0;
    84→  const courtSubtotal = (storedCourtSubtotal + equipmentSubtotal > totalPrice + 0.01)
    85→    ? Math.max(0, Math.round((totalPrice - equipmentSubtotal) * 100) / 100)
    86→    : storedCourtSubtotal;
    87→
    88→  // Normalize remaining_amount: ensure it equals total - advance
    89→  const advanceAmount = b.advance_amount || 0;
    90→  const storedRemaining = b.remaining_amount || 0;
    91→  const remainingAmount = (Math.abs(storedRemaining - (totalPrice - advanceAmount)) > 0.01)
    92→    ? Math.max(0, Math.round((totalPrice - advanceAmount) * 100) / 100)
    93→    : storedRemaining;
    94→
    95→  return {
    96→    id: b.id,
    97→    courtId: b.court_id,
    98→    courtIds,
    99→    userId: b.user_id,
   100→    date: typeof b.date === 'string' ? b.date : (b.date ? String(b.date) : ''),
   101→    startTime: b.start_time,
   102→    endTime: b.end_time,
   103→    totalPrice,
   104→    courtSubtotal,
   105→    equipmentSubtotal,
   106→    equipmentItems,
   107→    advanceAmount,
   108→    remainingAmount,
   109→    status: migrateStatus(b.status || 'reserved'),
   110→    slotStatus: b.slot_status,
   111→    paymentMethod: b.payment_method,
   112→    notes: b.notes,
   113→    createdAt: b.created_at,
   114→    updatedAt: b.updated_at,
   115→    recurringGroupId: b.recurring_group_id,
   116→    recurringIndex: b.recurring_index,
   117→    equipmentDelivered: b.equipment_delivered || false,
   118→    equipmentReturned: b.equipment_returned || false,
   119→    court,
   120→    courts,
   121→    user,
   122→  };
   123→}

   125→// Enrich a single booking (all courts + user), safe — never throws
   126→async function safeEnrichBooking(b: Record<string, unknown>): Promise<Record<string, unknown>> {
   127→  let court: Record<string, unknown> | null = null;
   128→  let courts: Array<Record<string, unknown>> = [];
   129→  let user: Record<string, unknown> | null = null;
   130→
   131→  // Get all court IDs (from court_ids array or single court_id)
   132→  const allCourtIds: string[] = Array.isArray(b.court_ids)
   133→    ? b.court_ids as string[]
   134→    : (b.court_id ? [b.court_id as string] : []);
   135→
   136→  // Enrich primary court (first one)
   137→  try {
   138→    if (allCourtIds.length > 0) {
   139→      const c = await getCourtById(allCourtIds[0]);
   140→      if (c) court = c as Record<string, unknown>;
   141→    }
   142→  } catch (e) {
   143→    console.warn('[BOOKINGS] Failed to load primary court for booking:', b.id, e);
   144→  }
   145→
   146→  // Enrich all courts for multi-court bookings
   147→  if (allCourtIds.length > 1) {
   148→    const enrichedCourts: Array<Record<string, unknown>> = [];
   149→    for (const cid of allCourtIds) {
   150→      try {
   151→        const c = await getCourtById(cid);
   152→        if (c) enrichedCourts.push(c as Record<string, unknown>);
   153→      } catch { /* skip failed court enrichment */ }
   154→    }
   155→    courts = enrichedCourts;
   156→  }
   157→
   158→  try {
   159→    if (b.user_id) {
   160→      const u = await getUserById(b.user_id as string);
   161→      if (u) user = u as Record<string, unknown>;
   162→    }
   163→  } catch (e) {
   164→    console.warn('[BOOKINGS] Failed to load user for booking:', b.id, e);
   165→  }
   166→
   167→  return { ...b, _court: court, _courts: courts, _user: user };
   168→}

   170→// Search bookings by userId OR userEmail, deduplicate
   171→async function searchBookingsForUser(userId: string, userEmail: string | null) {
   172→  const results = new Map<string, Record<string, unknown>>();
   173→
   174→  // Always search by userId
   175→  try {
   176→    const byId = await getBookings({ userId });
   177→    for (const b of byId) {
   178→      if (b.id && !results.has(b.id)) {
   179→        results.set(b.id, b as Record<string, unknown>);
   180→      }
   181→    }
   182→  } catch (e) {
   183→    console.error('[BOOKINGS] Error searching by userId:', userId, e);
   184→  }
   185→
   186→  // Also search by email to handle demo/Firebase ID mismatches
   187→  if (userEmail && userEmail !== userId) {
   188→    try {
   189→      const byEmail = await getBookings({ userEmail });
   190→      for (const b of byEmail) {
   191→        if (b.id && !results.has(b.id)) {
   192→          results.set(b.id, b as Record<string, unknown>);
   193→        }
   194→      }
   195→    } catch (e) {
   196→      console.error('[BOOKINGS] Error searching by userEmail:', userEmail, e);
   197→    }
   198→  }
   199→
   200→  return Array.from(results.values());
   201→}

   203→/**
   204→ * Calculate the total price for a time slot using a pricing schedule.
   205→ * Supports slots that span multiple schedules (e.g., 16:00-19:00 = 1hr morning + 2hr night).
   206→ */
   207→function calculatePriceForTimeSlot(
   208→  schedule: Array<{ label: string; startHour: number; endHour: number; pricePerHour: number }>,
   209→  startTime: string,
   210→  endTime: string,
   211→): number {
   212→  const [startH, startM] = startTime.split(':').map(Number);
   213→  const [endH, endM] = endTime.split(':').map(Number);
   214→  const startDecimal = startH + startM / 60;
   215→  const endDecimal = endH + endM / 60;
   216→
   217→  // Sort schedule by startHour
   218→  const sorted = [...schedule].sort((a, b) => a.startHour - b.startHour);
   219→
   220→  let total = 0;
   221→  let cursor = startDecimal;
   222→
   223→  for (const slot of sorted) {
   224→    const slotStart = slot.startHour;
   225→    const slotEnd = slot.endHour;
   226→
   227→    if (cursor >= slotEnd) continue; // cursor already past this slot
   228→
   229→    // Overlap start/end
   230→    const overlapStart = Math.max(cursor, slotStart);
   231→    const overlapEnd = Math.min(endDecimal, slotEnd);
   232→
   233→    if (overlapEnd > overlapStart) {
   234→      const hours = overlapEnd - overlapStart;
   235→      total += hours * slot.pricePerHour;
   236→      cursor = overlapEnd;
   237→    }
   238→  }
   239→
   240→  // If cursor didn't reach endDecimal, the remaining time is uncovered by schedule
   241→  // (fall back to 0 or caller's default)
   242→  return Math.round(total * 100) / 100;
   243→}

   245→export async function GET(request: NextRequest) {
   246→  const startTime = Date.now();
   247→  const firebaseOk = isFirebaseAvailable();
   248→
   249→  try {
   250→    const { searchParams } = new URL(request.url);
   251→    const courtId = searchParams.get('courtId');
   252→    const userId = searchParams.get('userId');
   253→    const date = searchParams.get('date');
   254→    const dateFrom = searchParams.get('dateFrom');
   255→    const dateTo = searchParams.get('dateTo');
   256→    const status = searchParams.get('status');
   257→
   258→    // Court availability check (courtId + date) is public — no auth required
   259→    if (courtId && date && !userId) {
   260→      if (!firebaseOk) {
   261→        return NextResponse.json({ error: 'Firebase no configurado', code: 'NO_FIREBASE' }, { status: 503 });
   262→      }
   263→      const bookings = await getBookings({ courtId, date });
   264→      return NextResponse.json(bookings.map(toCamelBooking));
   265→    }
   266→
   267→    // All other queries require authentication
   268→    const authResult = await requireAnyAuth(request);
   269→    if (authResult instanceof NextResponse) return authResult;
   270→    const authUser = authResult.user;
   271→
   272→    if (!firebaseOk) {
   273→      return NextResponse.json({ error: 'Firebase no configurado', code: 'NO_FIREBASE' }, { status: 503 });
   274→    }
   275→
   276→    let bookings: Record<string, unknown>[];
   277→
   278→    // Non-admin users: only their own bookings
   279→    if (authUser.role !== 'admin' && authUser.role !== 'super_admin') {
   280→      if (userId && userId !== authUser.id) {
   281→        return NextResponse.json({ error: 'No puedes ver reservas de otros usuarios.' }, { status: 403 });
   282→      }
   283→      const effectiveUserId = userId || authUser.id;
   284→      bookings = await searchBookingsForUser(effectiveUserId, authUser.email);
   285→    } else {
   286→      // Admin: fetch with optional filters (date range for performance)
   287→      bookings = (await getBookings({
   288→        courtId: courtId || undefined,
   289→        userId: userId || undefined,
   290→        date: date || undefined,
   291→        dateFrom: dateFrom || undefined,
   292→        dateTo: dateTo || undefined,
   293→        status: status || undefined,
   294→      })) as Record<string, unknown>[];
   295→      console.log(`[BOOKINGS] Firestore returned ${bookings.length} raw docs (dateFrom=${dateFrom}, dateTo=${dateTo})`);
   296→      if (bookings.length > 0) {
   297→        console.log(`[BOOKINGS] Sample raw date field: id=${bookings[0].id}, date=${JSON.stringify(bookings[0].date)}, type=${typeof bookings[0].date}`);
   298→      }
   299→    }
   300→
   301→    // Enrich each booking with cached court/user lookups (avoid N+1 Firestore reads)
   302→    const courtCache = new Map<string, Record<string, unknown> | null>();
   303→    const userCache = new Map<string, Record<string, unknown> | null>();
   304→    const getCourtCached = async (cid: string) => {
   305→      if (courtCache.has(cid)) return courtCache.get(cid)!;
   306→      try {
   307→        const c = await getCourtById(cid);
   308→        courtCache.set(cid, c as Record<string, unknown> | null);
   309→        return c as Record<string, unknown> | null;
   310→      } catch { courtCache.set(cid, null); return null; }
   311→    };
   312→    const getUserCached = async (uid: string) => {
   313→      if (userCache.has(uid)) return userCache.get(uid)!;
   314→      try {
   315→        const u = await getUserById(uid);
   316→        userCache.set(uid, u as Record<string, unknown> | null);
   317→        return u as Record<string, unknown> | null;
   318→      } catch { userCache.set(uid, null); return null; }
   319→    };
   320→
   321→    const enriched: Record<string, unknown>[] = [];
   322→    // Process in batches of 10 to avoid overwhelming Firestore
   323→    const BATCH_SIZE = 10;
   324→    for (let i = 0; i < bookings.length; i += BATCH_SIZE) {
   325→      const batch = bookings.slice(i, i + BATCH_SIZE);
   326→      const results = await Promise.allSettled(
   327→        batch.map(async (b) => {
   328→          const allCourtIds: string[] = Array.isArray(b.court_ids)
   329→            ? b.court_ids as string[] : (b.court_id ? [b.court_id as string] : []);
   330→          let court: Record<string, unknown> | null = null;
   331→          let courts: Array<Record<string, unknown>> = [];
   332→          let user: Record<string, unknown> | null = null;
   333→          try {
   334→            if (allCourtIds.length > 0) court = await getCourtCached(allCourtIds[0]);
   335→          } catch { /* skip */ }
   336→          if (allCourtIds.length > 1) {
   337→            const enrichedCourts: Array<Record<string, unknown>> = [];
   338→            for (const cid of allCourtIds) {
   339→              try {
   340→                const c = await getCourtCached(cid);
   341→                if (c) enrichedCourts.push(c);
   342→              } catch { /* skip */ }
   343→            }
   344→            courts = enrichedCourts;
   345→          }
   346→          try {
   347→            if (b.user_id) user = await getUserCached(b.user_id as string);
   348→          } catch { /* skip */ }
   349→          return toCamelBooking({ ...b, _court: court, _courts: courts, _user: user });
   350→        })
   351→      );
   352→      for (const r of results) {
   353→        if (r.status === 'fulfilled') {
   354→          enriched.push(r.value);
   355→        } else {
   356→          console.error('[BOOKINGS] Batch enrichment error:', r.reason);
   357→        }
   358→      }
   359→    }
   360→
   361→    // Default server-side sort: today first, future ascending, past descending
   362→    // Within each date: chronological time
   363→    const today = new Date().toLocaleDateString('en-CA', { timeZone: 'America/Lima' });
   364→    enriched.sort((a, b) => {
   365→      const dA = String(a.date || ''), dB = String(b.date || '');
   366→      const aIsToday = dA === today, bIsToday = dB === today;
   367→      const aFuture = dA > today, bFuture = dB > today;
   368→      const aPast = dA < today, bPast = dB < today;
   369→      // 1) Today always first
   370→      if (aIsToday && !bIsToday) return -1;
   371→      if (!aIsToday && bIsToday) return 1;
   372→      // 2) Both on same side of today
   373→      if (aFuture && bFuture) {
   374→        // Future dates: ascending (correlativo)
   375→        const dc = dA.localeCompare(dB);
   376→        if (dc !== 0) return dc;
   377→      } else if (aPast && bPast) {
   378→        // Past dates: descending (most recent past first)
   379→        const dc = dB.localeCompare(dA);
   380→        if (dc !== 0) return dc;
   381→      }
   382→      // Same date: chronological by start time
   383→      return String(a.startTime || '').localeCompare(String(b.startTime || ''));
   384→    });
   385→
   386→    console.log(`[BOOKINGS] GET: ${enriched.length} bookings returned in ${Date.now() - startTime}ms (auth=${authUser.role}, filters: courtId=${courtId||'all'}, date=${date||'all'}, status=${status||'all'})`);
   387→
   388→    return NextResponse.json(enriched);
   389→  } catch (error) {
   390→    console.error('[BOOKINGS] GET error:', error);
   391→    const msg = error instanceof Error ? error.message : 'Unknown error';
   392→    return NextResponse.json({ error: 'Error al obtener reservas', detail: msg }, { status: 500 });
   393→  }
   394→}

   396→export async function POST(request: NextRequest) {
   397→  try {
   398→    const authResult = await requireAnyAuth(request);
   399→    if (authResult instanceof NextResponse) return authResult;
   400→    const authUser = authResult.user;
   401→
   402→    if (!isFirebaseAvailable()) {
   403→      return NextResponse.json({ error: 'Firebase no configurado' }, { status: 503 });
   404→    }
   405→
   406→    const body = await request.json();
   407→    const {
   408→      courtId,
   409→      courtIds: bodyCourtIds,
   410→      userId,
   411→      date,
   412→      startTime,
   413→      endTime,
   414→      totalPrice,
   415→      advanceAmount,
   416→      remainingAmount,
   417→      status,
   418→      paymentMethod,
   419→      notes,
   420→      equipmentItems,
   421→    } = body;
   422→
   423→    // ── Resolve court IDs (support both single courtId and multi-court courtIds) ──
   424→    const allCourtIds: string[] = Array.isArray(bodyCourtIds) && bodyCourtIds.length > 0
   425→      ? bodyCourtIds
   426→      : courtId
   427→        ? [courtId]
   428→        : [];
   429→
   430→    // ── Validate payment method (restricted set, UPPERCASE) ──
   431→    const VALID_PAYMENT_METHODS = ['EFECTIVO', 'YAPE', 'PLIN'];
   432→    const normalizedPaymentMethod = VALID_PAYMENT_METHODS.includes(paymentMethod)
   433→      ? paymentMethod
   434→      : 'EFECTIVO';
   435→
   436→    if (allCourtIds.length === 0 || !userId || !date || !startTime || !endTime) {
   437→      return NextResponse.json(
   438→        { error: 'Faltan campos requeridos: courtId/courtIds, userId, date, startTime, endTime' },
   439→        { status: 400 }
   440→      );
   441→    }
   442→
   443→    // ── Minimum advance validation ──
   444→    // Admin/super_admin: NO time restrictions — can book any slot (walk-in, past, current)
   445→    // Regular users: require 30 min minimum advance
   446→    const isAdmin = authUser.role === 'admin' || authUser.role === 'super_admin';
   447→
   448→    if (!isAdmin) {
   449→      const now = new Date();
   450→      const slotDateTime = new Date(`${date}T${startTime}:00`);
   451→      const diffMs = slotDateTime.getTime() - now.getTime();
   452→      const thirtyMinMs = 30 * 60 * 1000;
   453→      if (diffMs < thirtyMinMs) {
   454→        const minsLeft = Math.floor(diffMs / 60000);
   455→        if (minsLeft < 0) {
   456→          return NextResponse.json(
   457→            { error: 'No se puede reservar un horario que ya ha pasado.' },
   458→            { status: 422 }
   459→          );
   460→        }
   461→        return NextResponse.json(
   462→          { error: `La reserva requiere al menos 30 minutos de anticipación. El horario seleccionado comienza en ${minsLeft} minuto${minsLeft === 1 ? '' : 's'}. Selecciona un horario a partir de las ${new Date(now.getTime() + thirtyMinMs).toLocaleTimeString('es-PE', { hour: '2-digit', minute: '2-digit', hour12: false })} o posterior.` },
   463→          { status: 422 }
   464→        );
   465→      }
   466→    }
   467→
   468→    // Non-admin users can only create bookings for themselves
   469→    if (!isAdmin) {
   470→      if (userId !== authUser.id) {
   471→        return NextResponse.json(
   472→          { error: 'No puedes crear reservas para otros usuarios.' },
   473→          { status: 403 }
   474→        );
   475→      }
   476→    }
   477→
   478→    // ── Check overlap for ALL courts ──
   479→    const conflicts: Array<{ courtId: string; courtName: string }> = [];
   480→    for (const cId of allCourtIds) {
   481→      const existing = await getBookings({ courtId: cId, date });
   482→      const overlapping = existing.filter(
   483→        (b) => {
   484→          const bCourtIds: string[] = Array.isArray(b.court_ids)
   485→            ? b.court_ids as string[]
   486→            : [b.court_id as string];
   487→          // Skip if this existing booking doesn't include the court we're checking
   488→          if (!bCourtIds.includes(cId)) return false;
   489→          return (
   490→            !['cancelled'].includes(migrateStatus(b.status || '')) &&
   491→            (b.start_time || '') < endTime &&
   492→            (b.end_time || '') > startTime
   493→          );
   494→        }
   495→      );
   496→      if (overlapping.length > 0) {
   497→        let courtName = cId;
   498→        try {
   499→          const c = await getCourtById(cId);
   500→          if (c?.name) courtName = c.name as string;
   501→        } catch { /* use ID */ }
   502→        conflicts.push({ courtId: cId, courtName });
   503→      }
   504→    }
   505→
   506→    if (conflicts.length > 0) {
   507→      const conflictNames = conflicts.map((c) => c.courtName).join(', ');
   508→      return NextResponse.json(
   509→        {
   510→          error: `Cancha${conflicts.length > 1 ? 's' : ''} no disponible${conflicts.length > 1 ? 's' : ''}: ${conflictNames}. Por favor selecciona otro horario o cancha.`,
   511→          conflicts,
   512→        },
   513→        { status: 409 }
   514→      );
   515→    }
   516→
   517→    // ── Calculate price ──
   518→    // Frontend sends totalPrice as grand total (court + equipment combined).
   519→    // We compute courtSubtotal and equipmentSubtotal separately for storage/display.
   520→    const eqItems = Array.isArray(equipmentItems) ? equipmentItems : [];
   521→    let equipmentSubtotal = 0;
   522→    for (const eq of eqItems) {
   523→      equipmentSubtotal += (eq.quantity || 0) * (eq.unit_price || eq.unitPrice || 0);
   524→    }
   525→
   526→    let courtPriceTotal: number;
   527→    const providedTotal = parseFloat(totalPrice) || 0;
   528→    if (providedTotal > 0) {
   529→      // Frontend already included equipment in totalPrice — extract court portion
   530→      courtPriceTotal = Math.max(0, providedTotal - equipmentSubtotal);
   531→    } else {
   532→      // No price from frontend — calculate from court rates
   533→      courtPriceTotal = 0;
   534→      for (const cId of allCourtIds) {
   535→        const court = await getCourtById(cId);
   536→        if (court) {
   537→          const schedule = court.pricing_schedule as Array<{ label: string; startHour: number; endHour: number; pricePerHour: number }> | undefined;
   538→          let courtPrice = 0;
   539→          if (schedule && schedule.length > 0) {
   540→            courtPrice = calculatePriceForTimeSlot(schedule, startTime, endTime);
   541→            if (courtPrice <= 0) courtPrice = (court.price_per_hour as number) || 0;
   542→          } else {
   543→            courtPrice = (court.price_per_hour as number) || 0;
   544→          }
   545→          courtPriceTotal += courtPrice;
   546→        }
   547→      }
   548→    }
   549→
   550→    const price = courtPriceTotal + equipmentSubtotal;
   551→
   552→    const adv = parseFloat(advanceAmount) || price * 0.5;
   553→    const rem = parseFloat(remainingAmount) || price - adv;
   554→    const bookingStatus = migrateStatus(status || 'reserved');
   555→
   556→    // Resolve client email for denormalized search
   557→    let clientEmail = authUser.email;
   558→    try {
   559→      const clientUser = await getUserById(userId);
   560→      if (clientUser?.email) clientEmail = clientUser.email;
   561→    } catch { /* fallback to admin email */ }
   562→
   563→    // Save to Firestore — single booking with all courts
   564→    const id = await createBooking({
   565→      court_ids: allCourtIds,
   566→      user_id: userId,
   567→      user_email: clientEmail,
   568→      date,
   569→      start_time: startTime,
   570→      end_time: endTime,
   571→      total_price: price,
   572→      court_subtotal: courtPriceTotal,
   573→      equipment_subtotal: equipmentSubtotal,
   574→      equipment_items: eqItems,
   575→      advance_amount: adv,
   576→      remaining_amount: rem,
   577→      status: bookingStatus,
   578→      slot_status: 'available',
   579→      payment_method: normalizedPaymentMethod || null,
   580→      notes: notes || null,
   581→    });
   582→
   583→    console.log(`[BOOKINGS] POST: Created booking ${id} for user=${userId} date=${date} ${startTime}-${endTime} courts=${allCourtIds.join(',')} total=${price} advance=${adv} remaining=${rem} method=${normalizedPaymentMethod} email=${clientEmail}`);
   584→
   585→    // Create advance payment record
   586→    try {
   587→      await createPayment(id, {
   588→        user_id: userId,
   589→        amount: adv,
   590→        type: 'advance',
   591→        method: normalizedPaymentMethod || 'EFECTIVO',
   592→        status: 'completed',
   593→      });
   594→    } catch (payErr) {
   595→      console.error('[BOOKINGS] Warning: could not create payment record:', payErr);
   596→    }
   597→
   598→    return NextResponse.json({
   599→      id,
   600→      courtId: allCourtIds[0],
   601→      courtIds: allCourtIds,
   602→      userId,
   603→      date,
   604→      startTime,
   605→      endTime,
   606→      totalPrice: price,
   607→      advanceAmount: adv,
   608→      remainingAmount: rem,
   609→      status: bookingStatus,
   610→      paymentMethod: paymentMethod || null,
   611→      success: true,
   612→    }, { status: 201 });
   613→  } catch (error: unknown) {
   614→    const err = error as { message?: string; code?: string; stack?: string };
   615→    console.error('[BOOKINGS] POST error:', err?.message || err);
   616→    console.error('[BOOKINGS] Stack:', err?.stack || 'no stack');
   617→    return NextResponse.json({
   618→      error: 'Failed to create booking',
   619→      detail: err?.message || 'Unknown error'
   620→    }, { status: 500 });
   621→  }
   622→}
   623→
   624→export async function PUT(request: NextRequest) {
   625→  try {
   626→    const authResult = await requireAnyAuth(request);
   627→    if (authResult instanceof NextResponse) return authResult;
   628→    const authUser = authResult.user;
   629→
   630→    if (!isFirebaseAvailable()) {
   631→      return NextResponse.json({ error: 'Firebase no configurado' }, { status: 503 });
   632→    }
   633→
   634→    const body = await request.json();
   635→    const { id, status, slot_status, advanceAmount: reqAdvance, remainingAmount: reqRemaining, paymentMethod: reqPaymentMethod, equipmentDelivered, equipmentReturned, advanceAction, cancelReason } = body;
   636→
   637→    if (!id) {
   638→      return NextResponse.json({ error: 'Booking ID is required' }, { status: 400 });
   639→    }
   640→
   641→    // Non-admin users can only cancel their own bookings
   642→    if (authUser.role !== 'admin' && authUser.role !== 'super_admin') {
   643→      if (status === 'cancelled') {
   644→        // Verify the booking belongs to the user (by both userId and email)
   645→        const bookings = await searchBookingsForUser(authUser.id, authUser.email);
   646→        const booking = bookings.find((b) => b.id === id);
   647→        if (!booking) {
   648→          return NextResponse.json(
   649→            { error: 'No puedes modificar reservas de otros usuarios.' },
   650→            { status: 403 }
   651→          );
   652→        }
   653→      } else {
   654→        return NextResponse.json(
   655→          { error: 'Solo puedes cancelar tus propias reservas.' },
   656→          { status: 403 }
   657→        );
   658→      }
   659→    }
   660→
   661→    const updateData: Record<string, unknown> = {};
   662→    if (status) updateData.status = migrateStatus(status);
   663→    if (slot_status) updateData.slot_status = slot_status;
   664→    if (typeof reqAdvance === 'number') updateData.advance_amount = reqAdvance;
   665→    if (typeof reqRemaining === 'number') updateData.remaining_amount = reqRemaining;
   666→    if (reqPaymentMethod) {
   667→      const VALID_PM = ['EFECTIVO', 'YAPE', 'PLIN'];
   668→      updateData.payment_method = VALID_PM.includes(reqPaymentMethod) ? reqPaymentMethod : 'EFECTIVO';
   669→    }
   670→    if (typeof equipmentDelivered === 'boolean') updateData.equipment_delivered = equipmentDelivered;
   671→    if (typeof equipmentReturned === 'boolean') updateData.equipment_returned = equipmentReturned;
   672→
   673→    // When cancelling a booking with advance, handle retained/refunded advance
   674→    if (status === 'cancelled' || (status && migrateStatus(status) === 'cancelled')) {
   675→      try {
   676→        const booking = await getBookingById(id);
   677→        if (booking && ((booking.advance_amount as number) || 0) > 0) {
   678→          const advAmount = (booking.advance_amount as number) || 0;
   679→          const isRetained = advanceAction === 'retain';
   680→          const statusAdvance = isRetained ? 'retained' : 'refunded';
   681→
   682→          // Get court name for the record
   683→          let courtName = '';
   684→          try {
   685→            const cId = (booking.court_ids as string[])?.[0] || (booking.court_id as string) || '';
   686→            if (cId) {
   687→              const court = await getCourtById(cId);
   688→              courtName = (court?.name as string) || '';
   689→            }
   690→          } catch { /* non-critical */ }
   691→
   692→          // Get user name
   693→          let userName = '';
   694→          try {
   695→            const uId = booking.user_id as string;
   696→            if (uId) {
   697→              const user = await getUserById(uId);
   698→              userName = (user?.name as string) || '';
   699→            }
   700→          } catch { /* non-critical */ }
   701→
   702→          await createRetainedAdvance({
   703→            booking_id: id,
   704→            user_id: (booking.user_id as string) || '',
   705→            user_name: userName,
   706→            user_email: (booking.user_email as string) || null,
   707→            court_name: courtName,
   708→            booking_date: (booking.date as string) || '',
   709→            amount: advAmount,
   710→            original_total: (booking.total_price as number) || 0,
   711→            payment_method: (booking.payment_method as string) || 'EFECTIVO',
   712→            reason: cancelReason || 'Cancelación de reserva',
   713→            status: statusAdvance,
   714→          });
   715→
   716→          // Create a payment record for the refund/retention
   717→          try {
   718→            await createPayment(id, {
   719→              user_id: (booking.user_id as string) || '',
   720→              amount: advAmount,
   721→              type: isRetained ? 'retained' : 'refund',
   722→              method: (booking.payment_method as string) || 'EFECTIVO',
   723→              status: 'completed',
   724→            });
   725→          } catch (payErr) {
   726→            console.error('[BOOKINGS] Warning: could not create payment record for cancellation:', payErr);
   727→          }
   728→        }
   729→      } catch (err) {
   730→        console.error('[BOOKINGS] Warning: could not create retained advance record:', err);
   731→      }
   732→    }
   733→
   734→    // If advance amount is being updated, create a payment record for the increment
   735→    if (typeof reqAdvance === 'number' && typeof reqRemaining === 'number') {
   736→      try {
   737→        const booking = await getBookingById(id);
   738→        if (booking) {
   739→          const prevAdvance = (booking.advance_amount as number) || 0;
   740→          const paymentIncrement = reqAdvance - prevAdvance;
   741→          if (paymentIncrement > 0.01) {
   742→            try {
   743→              await createPayment(id, {
   744→                user_id: booking.user_id as string,
   745→                amount: paymentIncrement,
   746→                type: reqRemaining <= 0 ? 'full_payment' : 'advance',
   747→                method: (updateData.payment_method as string) || 'EFECTIVO',
   748→                status: 'completed',
   749→              });
   750→            } catch (payErr) {
   751→              console.error('[BOOKINGS] Warning: could not create payment record for advance update:', payErr);
   752→            }
   753→          }
   754→        }
   755→      } catch { /* non-critical: proceed with booking update */ }
   756→    }
   757→
   758→    await updateBooking(id, updateData);
   759→
   760→    return NextResponse.json({ success: true });
   761→  } catch (error) {
   762→    console.error('[BOOKINGS] PUT error:', error);
   763→    const msg = error instanceof Error ? error.message : 'Unknown error';
   764→    return NextResponse.json({ error: 'Error al actualizar reserva', detail: msg }, { status: 500 });
   765→  }
   766→}
