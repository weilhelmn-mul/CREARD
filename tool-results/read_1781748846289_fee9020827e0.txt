    17→    17→function migrateStatus(s: string): string {
    18→    18→  switch (s) {
    19→    19→    case 'confirmed':
    20→    20→    case 'pending':
    21→    21→    case 'partially_paid':
    22→    22→      return 'reserved';
    23→    23→    case 'fully_paid':
    24→    24→    case 'completed':
    25→    25→      return 'completed';
    26→    26→    case 'no_show':
    27→    27→    case 'expired':
    28→    28→      return 'cancelled';
    29→    29→    default:
    30→    30→      return s; // 'reserved', 'completed', 'cancelled' pass through
    31→    31→  }

    34→    34→// Transformar snake_case (Firestore) a camelCase (frontend)
    35→    35→function toCamelBooking(b: Record<string, unknown>) {
    36→    36→  // Primary court (backward compat)
    37→    37→  const courtRaw = b._court as Record<string, unknown> | null;
    38→    38→  const court = courtRaw ? {
    39→    39→    id: courtRaw.id || courtRaw.court_id,
    40→    40→    name: courtRaw.name || courtRaw.court_name,
    41→    41→    sport: courtRaw.sport,
    42→    42→    branch: (courtRaw.branch as Record<string, unknown>) || {
    43→    43→      id: courtRaw.branch_id || 'branch-1',
    44→    44→      name: 'CREARD',
    45→    45→    },
    46→    46→  } : null;
    47→    47→
    48→    48→  // Additional courts (multi-court bookings)
    49→    49→  const courtsRaw = b._courts as Array<Record<string, unknown>> | null;
    50→    50→  const courts = courtsRaw ? courtsRaw.map((cr) => ({
    51→    51→    id: cr.id || cr.court_id,
    52→    52→    name: cr.name || cr.court_name,
    53→    53→    sport: cr.sport,
    54→    54→    branch: (cr.branch as Record<string, unknown>) || { id: 'branch-1', name: 'CREARD' },
    55→    55→  })) : [];
    56→    56→
    57→    57→  const userRaw = b._user as Record<string, unknown> | null;
    58→    58→  const user = userRaw ? {
    59→    59→    id: userRaw.id,
    60→    60→    name: userRaw.name,
    61→    61→    email: userRaw.email,
    62→    62→    phone: userRaw.phone,
    63→    63→  } : null;
    64→    64→
    65→    65→  // Court IDs array
    66→    66→  const courtIds: string[] = Array.isArray(b.court_ids) ? b.court_ids : (b.court_id ? [b.court_id] : []);
    67→    67→
    68→    68→  // Equipment items
    69→    69→  const equipmentItems = Array.isArray(b.equipment_items) ? b.equipment_items.map((eq: Record<string, unknown>) => ({
    70→    70→    equipmentId: eq.equipment_id || eq.id,
    71→    71→    name: eq.name,
    72→    72→    quantity: eq.quantity || 1,
    73→    73→    unitPrice: eq.unit_price || eq.price_per_rental || 0,
    74→    74→    subtotal: (eq.quantity || 1) * (eq.unit_price || eq.price_per_rental || 0),
    75→    75→  })) : [];
    76→    76→
    77→    77→  // Normalize: recalc equipment subtotal from items (more reliable than stored value)
    78→    78→  const computedEquipSubtotal = equipmentItems.reduce((s: number, eq: { subtotal: number }) => s + eq.subtotal, 0);
    79→    79→  const equipmentSubtotal = computedEquipSubtotal;
    80→    80→
    81→    81→  // Normalize court_subtotal: if stored value is wrong (old double-counting bug),
    82→    82→  // derive it from total_price - equipment_subtotal
    83→    83→  const totalPrice = b.total_price || 0;
    84→    84→  const storedCourtSubtotal = b.court_subtotal || 0;
    85→    85→  const courtSubtotal = (storedCourtSubtotal + equipmentSubtotal > totalPrice + 0.01)
    86→    86→    ? Math.max(0, Math.round((totalPrice - equipmentSubtotal) * 100) / 100)
    87→    87→    : storedCourtSubtotal;
    88→    88→
    89→    89→  // Normalize remaining_amount: ensure it equals total - advance
    90→    90→  const advanceAmount = b.advance_amount || 0;
    91→    91→  const storedRemaining = b.remaining_amount || 0;
    92→    92→  const remainingAmount = (Math.abs(storedRemaining - (totalPrice - advanceAmount)) > 0.01)
    93→    93→    ? Math.max(0, Math.round((totalPrice - advanceAmount) * 100) / 100)
    94→    94→    : storedRemaining;
    95→    95→
    96→    96→  return {
    97→    97→    id: b.id,
    98→    98→    courtId: b.court_id,
    99→    99→    courtIds,
   100→   100→    userId: b.user_id,
   101→   101→    date: typeof b.date === 'string' ? b.date : (b.date ? String(b.date) : ''),
   102→   102→    startTime: b.start_time,
   103→   103→    endTime: b.end_time,
   104→   104→    totalPrice,
   105→   105→    courtSubtotal,
   106→   106→    equipmentSubtotal,
   107→   107→    equipmentItems,
   108→   108→    advanceAmount,
   109→   109→    remainingAmount,
   110→   110→    status: migrateStatus(b.status || 'reserved'),
   111→   111→    slotStatus: b.slot_status,
   112→   112→    paymentMethod: b.payment_method,
   113→   113→    notes: b.notes,
   114→   114→    createdAt: b.created_at,
   115→   115→    updatedAt: b.updated_at,
   116→   116→    recurringGroupId: b.recurring_group_id,
   117→   117→    recurringIndex: b.recurring_index,
   118→   118→    equipmentDelivered: b.equipment_delivered || false,
   119→   119→    equipmentReturned: b.equipment_returned || false,
   120→   120→    court,
   121→   121→    courts,
   122→   122→    user,
   123→   123→  };
   124→   124→}

   126→   126→// Enrich a single booking (all courts + user), safe — never throws
   127→   127→async function safeEnrichBooking(b: Record<string, unknown>): Promise<Record<string, unknown>> {
   128→   128→  let court: Record<string, unknown> | null = null;
   129→   129→  let courts: Array<Record<string, unknown>> = [];
   130→   130→  let user: Record<string, unknown> | null = null;
   131→   131→
   132→   132→  // Get all court IDs (from court_ids array or single court_id)
   133→   133→  const allCourtIds: string[] = Array.isArray(b.court_ids)
   134→   134→    ? b.court_ids as string[]
   135→   135→    : (b.court_id ? [b.court_id as string] : []);
   136→   136→
   137→   137→  // Enrich primary court (first one)
   138→   138→  try {
   139→   139→    if (allCourtIds.length > 0) {
   140→   140→      const c = await getCourtById(allCourtIds[0]);
   141→   141→      if (c) court = c as Record<string, unknown>;
   142→   142→    }
   143→   143→  } catch (e) {
   144→   144→    console.warn('[BOOKINGS] Failed to load primary court for booking:', b.id, e);
   145→   145→  }
   146→   146→
   147→   147→  // Enrich all courts for multi-court bookings
   148→   148→  if (allCourtIds.length > 1) {
   149→   149→    const enrichedCourts: Array<Record<string, unknown>> = [];
   150→   150→    for (const cid of allCourtIds) {
   151→   151→      try {
   152→   152→        const c = await getCourtById(cid);
   153→   153→        if (c) enrichedCourts.push(c as Record<string, unknown>);
   154→   154→      } catch { /* skip failed court enrichment */ }
   155→   155→    }
   156→   156→    courts = enrichedCourts;
   157→   157→  }
   158→   158→
   159→   159→  try {
   160→   160→    if (b.user_id) {
   161→   161→      const u = await getUserById(b.user_id as string);
   162→   162→      if (u) user = u as Record<string, unknown>;
   163→   163→    }
   164→   164→  } catch (e) {
   165→   165→    console.warn('[BOOKINGS] Failed to load user for booking:', b.id, e);
   166→   166→  }
   167→   167→
   168→   168→  return { ...b, _court: court, _courts: courts, _user: user };
   169→   169→}

   191→   191→      for (const b of byEmail) {
   192→   192→        if (b.id && !results.has(b.id)) {
   193→   193→          results.set(b.id, b as Record<string, unknown>);

   204→   204→/**
   205→   205→ * Calculate the total price for a time slot using a pricing schedule.
   206→   206→ * Supports slots that span multiple schedules (e.g., 16:00-19:00 = 1hr morning + 2hr night).
   207→   207→ */
   208→   208→function calculatePriceForTimeSlot(
   209→   209→  schedule: Array<{ label: string; startHour: number; endHour: number; pricePerHour: number }>,
   210→   210→  startTime: string,
   211→   211→  endTime: string,
   212→   212→): number {
   213→   213→  const [startH, startM] = startTime.split(':').map(Number);
   214→   214→  const [endH, endM] = endTime.split(':').map(Number);
   215→   215→  const startDecimal = startH + startM / 60;
   216→   216→  const endDecimal = endH + endM / 60;
   217→   217→
   218→   218→  // Sort schedule by startHour
   219→   219→  const sorted = [...schedule].sort((a, b) => a.startHour - b.startHour);
   220→   220→
   221→   221→  let total = 0;
   222→   222→  let cursor = startDecimal;
   223→   223→
   224→   224→  for (const slot of sorted) {
   225→   225→    const slotStart = slot.startHour;
   226→   226→    const slotEnd = slot.endHour;
   227→   227→
   228→   228→    if (cursor >= slotEnd) continue; // cursor already past this slot
   229→   229→
   230→   230→    // Overlap start/end
   231→   231→    const overlapStart = Math.max(cursor, slotStart);
   232→   232→    const overlapEnd = Math.min(endDecimal, slotEnd);
   233→   233→
   234→   234→    if (overlapEnd > overlapStart) {
   235→   235→      const hours = overlapEnd - overlapStart;
   236→   236→      total += hours * slot.pricePerHour;
   237→   237→      cursor = overlapEnd;
   238→   238→    }
   239→   239→  }
   240→   240→
   241→   241→  // If cursor didn't reach endDecimal, the remaining time is uncovered by schedule
   242→   242→  // (fall back to 0 or caller's default)
   243→   243→  return Math.round(total * 100) / 100;
   244→   244→}

   246→   246→export async function GET(request: NextRequest) {
   247→   247→  const startTime = Date.now();
   248→   248→  const firebaseOk = isFirebaseAvailable();
   249→   249→
   250→   250→  try {
   251→   251→    const { searchParams } = new URL(request.url);
   252→   252→    const courtId = searchParams.get('courtId');
   253→   253→    const userId = searchParams.get('userId');
   254→   254→    const date = searchParams.get('date');
   255→   255→    const dateFrom = searchParams.get('dateFrom');
   256→   256→    const dateTo = searchParams.get('dateTo');
   257→   257→    const status = searchParams.get('status');
   258→   258→
   259→   259→    // Court availability check (courtId + date) is public — no auth required
   260→   260→    if (courtId && date && !userId) {
   261→   261→      if (!firebaseOk) {
   262→   262→        return NextResponse.json({ error: 'Firebase no configurado', code: 'NO_FIREBASE' }, { status: 503 });
   263→   263→      }
   264→   264→      const bookings = await getBookings({ courtId, date });
   265→   265→      return NextResponse.json(bookings.map(toCamelBooking));
   266→   266→    }
   267→   267→
   268→   268→    // All other queries require authentication
   269→   269→    const authResult = await requireAnyAuth(request);
   270→   270→    if (authResult instanceof NextResponse) return authResult;
   271→   271→    const authUser = authResult.user;
   272→   272→
   273→   273→    if (!firebaseOk) {
   274→   274→      return NextResponse.json({ error: 'Firebase no configurado', code: 'NO_FIREBASE' }, { status: 503 });
   275→   275→    }
   276→   276→
   277→   277→    let bookings: Record<string, unknown>[];
   278→   278→
   279→   279→    // Non-admin users: only their own bookings
   280→   280→    if (authUser.role !== 'admin' && authUser.role !== 'super_admin') {
   281→   281→      if (userId && userId !== authUser.id) {
   282→   282→        return NextResponse.json({ error: 'No puedes ver reservas de otros usuarios.' }, { status: 403 });
   283→   283→      }
   284→   284→      const effectiveUserId = userId || authUser.id;
   285→   285→      bookings = await searchBookingsForUser(effectiveUserId, authUser.email);
   286→   286→    } else {
   287→   287→      // Admin: fetch with optional filters (date range for performance)
   288→   288→      bookings = (await getBookings({
   289→   289→        courtId: courtId || undefined,
   290→   290→        userId: userId || undefined,
   291→   291→        date: date || undefined,
   292→   292→        dateFrom: dateFrom || undefined,
   293→   293→        dateTo: dateTo || undefined,
   294→   294→        status: status || undefined,
   295→   295→      })) as Record<string, unknown>[];
   296→   296→      console.log(`[BOOKINGS] Firestore returned ${bookings.length} raw docs (dateFrom=${dateFrom}, dateTo=${dateTo})`);
   297→   297→      if (bookings.length > 0) {
   298→   298→        console.log(`[BOOKINGS] Sample raw date field: id=${bookings[0].id}, date=${JSON.stringify(bookings[0].date)}, type=${typeof bookings[0].date}`);
   299→   299→      }
   300→   300→    }
   301→   301→
   302→   302→    // Enrich each booking with cached court/user lookups (avoid N+1 Firestore reads)
   303→   303→    const courtCache = new Map<string, Record<string, unknown> | null>();
   304→   304→    const userCache = new Map<string, Record<string, unknown> | null>();
   305→   305→    const getCourtCached = async (cid: string) => {
   306→   306→      if (courtCache.has(cid)) return courtCache.get(cid)!;
   307→   307→      try {
   308→   308→        const c = await getCourtById(cid);
   309→   309→        courtCache.set(cid, c as Record<string, unknown> | null);
   310→   310→        return c as Record<string, unknown> | null;
   311→   311→      } catch { courtCache.set(cid, null); return null; }
   312→   312→    };
   313→   313→    const getUserCached = async (uid: string) => {
   314→   314→      if (userCache.has(uid)) return userCache.get(uid)!;
   315→   315→      try {
   316→   316→        const u = await getUserById(uid);
   317→   317→        userCache.set(uid, u as Record<string, unknown> | null);
   318→   318→        return u as Record<string, unknown> | null;
   319→   319→      } catch { userCache.set(uid, null); return null; }
   320→   320→    };
   321→   321→
   322→   322→    const enriched: Record<string, unknown>[] = [];
   323→   323→    // Process in batches of 10 to avoid overwhelming Firestore
   324→   324→    const BATCH_SIZE = 10;
   325→   325→    for (let i = 0; i < bookings.length; i += BATCH_SIZE) {
   326→   326→      const batch = bookings.slice(i, i + BATCH_SIZE);
   327→   327→      const results = await Promise.allSettled(
   328→   328→        batch.map(async (b) => {
   329→   329→          const allCourtIds: string[] = Array.isArray(b.court_ids)
   330→   330→            ? b.court_ids as string[] : (b.court_id ? [b.court_id as string] : []);
   331→   331→          let court: Record<string, unknown> | null = null;
   332→   332→          let courts: Array<Record<string, unknown>> = [];
   333→   333→          let user: Record<string, unknown> | null = null;
   334→   334→          try {
   335→   335→            if (allCourtIds.length > 0) court = await getCourtCached(allCourtIds[0]);
   336→   336→          } catch { /* skip */ }
   337→   337→          if (allCourtIds.length > 1) {
   338→   338→            const enrichedCourts: Array<Record<string, unknown>> = [];
   339→   339→            for (const cid of allCourtIds) {
   340→   340→              try {
   341→   341→                const c = await getCourtCached(cid);
   342→   342→                if (c) enrichedCourts.push(c);
   343→   343→              } catch { /* skip */ }
   344→   344→            }
   345→   345→            courts = enrichedCourts;
   346→   346→          }
   347→   347→          try {
   348→   348→            if (b.user_id) user = await getUserCached(b.user_id as string);
   349→   349→          } catch { /* skip */ }
   350→   350→          return toCamelBooking({ ...b, _court: court, _courts: courts, _user: user });
   351→   351→        })
   352→   352→      );
   353→   353→      for (const r of results) {
   354→   354→        if (r.status === 'fulfilled') {
   355→   355→          enriched.push(r.value);
   356→   356→        } else {
   357→   357→          console.error('[BOOKINGS] Batch enrichment error:', r.reason);
   358→   358→        }
   359→   359→      }
   360→   360→    }
   361→   361→
   362→   362→    // Default server-side sort: today first, future ascending, past descending
   363→   363→    // Within each date: chronological time
   364→   364→    const today = new Date().toLocaleDateString('en-CA', { timeZone: 'America/Lima' });
   365→   365→    enriched.sort((a, b) => {
   366→   366→      const dA = String(a.date || ''), dB = String(b.date || '');
   367→   367→      const aIsToday = dA === today, bIsToday = dB === today;
   368→   368→      const aFuture = dA > today, bFuture = dB > today;
   369→   369→      const aPast = dA < today, bPast = dB < today;
   370→   370→      // 1) Today always first
   371→   371→      if (aIsToday && !bIsToday) return -1;
   372→   372→      if (!aIsToday && bIsToday) return 1;
   373→   373→      // 2) Both on same side of today
   374→   374→      if (aFuture && bFuture) {
   375→   375→        // Future dates: ascending (correlativo)
   376→   376→        const dc = dA.localeCompare(dB);
   377→   377→        if (dc !== 0) return dc;
   378→   378→      } else if (aPast && bPast) {
   379→   379→        // Past dates: descending (most recent past first)
   380→   380→        const dc = dB.localeCompare(dA);
   381→   381→        if (dc !== 0) return dc;
   382→   382→      }
   383→   383→      // Same date: chronological by start time
   384→   384→      return String(a.startTime || '').localeCompare(String(b.startTime || ''));
   385→   385→    });
   386→   386→
   387→   387→    console.log(`[BOOKINGS] GET: ${enriched.length} bookings returned in ${Date.now() - startTime}ms (auth=${authUser.role}, filters: courtId=${courtId||'all'}, date=${date||'all'}, status=${status||'all'})`);
   388→   388→
   389→   389→    return NextResponse.json(enriched);
   390→   390→  } catch (error) {
   391→   391→    console.error('[BOOKINGS] GET error:', error);
   392→   392→    const msg = error instanceof Error ? error.message : 'Unknown error';
   393→   393→    return NextResponse.json({ error: 'Error al obtener reservas', detail: msg }, { status: 500 });
   394→   394→  }
   395→   395→}

   397→   397→export async function POST(request: NextRequest) {
   398→   398→  try {
   399→   399→    const authResult = await requireAnyAuth(request);
   400→   400→    if (authResult instanceof NextResponse) return authResult;

   405→   405→    }
   406→   406→
   407→   407→    const body = await request.json();
   408→   408→    const {
   409→   409→      courtId,
   410→   410→      courtIds: bodyCourtIds,
   411→   411→      userId,

   414→   414→      endTime,
   415→   415→      totalPrice,
   416→   416→      advanceAmount,

   418→   418→      status,

   420→   420→      notes,
   421→   421→      equipmentItems,
   422→   422→    } = body;
   423→   423→
   424→   424→    // ── Resolve court IDs (support both single courtId and multi-court courtIds) ──
   425→   425→    const allCourtIds: string[] = Array.isArray(bodyCourtIds) && bodyCourtIds.length > 0
   426→   426→      ? bodyCourtIds
   427→   427→      : courtId
   428→   428→        ? [courtId]
   429→   429→        : [];
   430→   430→
   431→   431→    // ── Validate payment method (restricted set, UPPERCASE) ──
   432→   432→    const VALID_PAYMENT_METHODS = ['EFECTIVO', 'YAPE', 'PLIN'];
   433→   433→    const normalizedPaymentMethod = VALID_PAYMENT_METHODS.includes(paymentMethod)
   434→   434→      ? paymentMethod
   435→   435→      : 'EFECTIVO';
   436→   436→
   437→   437→    if (allCourtIds.length === 0 || !userId || !date || !startTime || !endTime) {
   438→   438→      return NextResponse.json(
   439→   439→        { error: 'Faltan campos requeridos: courtId/courtIds, userId, date, startTime, endTime' },
   440→   440→        { status: 400 }
   441→   441→      );
   442→   442→    }
   443→   443→
   444→   444→    // ── Minimum advance validation ──
   445→   445→    // Admin/super_admin: NO time restrictions — can book any slot (walk-in, past, current)
   446→   446→    // Regular users: require 30 min minimum advance
   447→   447→    const isAdmin = authUser.role === 'admin' || authUser.role === 'super_admin';
   448→   448→
   449→   449→    if (!isAdmin) {
   450→   450→      const now = new Date();
   451→   451→      const slotDateTime = new Date(`${date}T${startTime}:00`);
   452→   452→      const diffMs = slotDateTime.getTime() - now.getTime();
   453→   453→      const thirtyMinMs = 30 * 60 * 1000;
   454→   454→      if (diffMs < thirtyMinMs) {
   455→   455→        const minsLeft = Math.floor(diffMs / 60000);
   456→   456→        if (minsLeft < 0) {
   457→   457→          return NextResponse.json(
   458→   458→            { error: 'No se puede reservar un horario que ya ha pasado.' },
   459→   459→            { status: 422 }
   460→   460→          );
   461→   461→        }
   462→   462→        return NextResponse.json(
   463→   463→          { error: `La reserva requiere al menos 30 minutos de anticipación. El horario seleccionado comienza en ${minsLeft} minuto${minsLeft === 1 ? '' : 's'}. Selecciona un horario a partir de las ${new Date(now.getTime() + thirtyMinMs).toLocaleTimeString('es-PE', { hour: '2-digit', minute: '2-digit', hour12: false })} o posterior.` },
   464→   464→          { status: 422 }
   465→   465→        );
   466→   466→      }
   467→   467→    }
   468→   468→
   469→   469→    // Non-admin users can only create bookings for themselves
   470→   470→    if (!isAdmin) {
   471→   471→      if (userId !== authUser.id) {
   472→   472→        return NextResponse.json(
   473→   473→          { error: 'No puedes crear reservas para otros usuarios.' },
   474→   474→          { status: 403 }
   475→   475→        );
   476→   476→      }
   477→   477→    }
   478→   478→
   479→   479→    // ── Check overlap for ALL courts ──
   480→   480→    const conflicts: Array<{ courtId: string; courtName: string }> = [];
   481→   481→    for (const cId of allCourtIds) {
   482→   482→      const existing = await getBookings({ courtId: cId, date });
   483→   483→      const overlapping = existing.filter(
   484→   484→        (b) => {
   485→   485→          const bCourtIds: string[] = Array.isArray(b.court_ids)
   486→   486→            ? b.court_ids as string[]
   487→   487→            : [b.court_id as string];
   488→   488→          // Skip if this existing booking doesn't include the court we're checking
   489→   489→          if (!bCourtIds.includes(cId)) return false;
   490→   490→          return (
   491→   491→            !['cancelled'].includes(migrateStatus(b.status || '')) &&
   492→   492→            (b.start_time || '') < endTime &&
   493→   493→            (b.end_time || '') > startTime
   494→   494→          );
   495→   495→        }
   496→   496→      );
   497→   497→      if (overlapping.length > 0) {
   498→   498→        let courtName = cId;
   499→   499→        try {
   500→   500→          const c = await getCourtById(cId);
   501→   501→          if (c?.name) courtName = c.name as string;
   502→   502→        } catch { /* use ID */ }
   503→   503→        conflicts.push({ courtId: cId, courtName });
   504→   504→      }
   505→   505→    }
   506→   506→
   507→   507→    if (conflicts.length > 0) {
   508→   508→      const conflictNames = conflicts.map((c) => c.courtName).join(', ');
   509→   509→      return NextResponse.json(
   510→   510→        {
   511→   511→          error: `Cancha${conflicts.length > 1 ? 's' : ''} no disponible${conflicts.length > 1 ? 's' : ''}: ${conflictNames}. Por favor selecciona otro horario o cancha.`,
   512→   512→          conflicts,
   513→   513→        },
   514→   514→        { status: 409 }
   515→   515→      );
   516→   516→    }
   517→   517→
   518→   518→    // ── Calculate price ──
   519→   519→    // Frontend sends totalPrice as grand total (court + equipment combined).
   520→   520→    // We compute courtSubtotal and equipmentSubtotal separately for storage/display.
   521→   521→    const eqItems = Array.isArray(equipmentItems) ? equipmentItems : [];
   522→   522→    let equipmentSubtotal = 0;
   523→   523→    for (const eq of eqItems) {
   524→   524→      equipmentSubtotal += (eq.quantity || 0) * (eq.unit_price || eq.unitPrice || 0);
   525→   525→    }
   526→   526→

   528→   528→    const providedTotal = parseFloat(totalPrice) || 0;
   529→   529→    if (providedTotal > 0) {
   530→   530→      // Frontend already included equipment in totalPrice — extract court portion
   531→   531→      courtPriceTotal = Math.max(0, providedTotal - equipmentSubtotal);
   532→   532→    } else {
   533→   533→      // No price from frontend — calculate from court rates
   534→   534→      courtPriceTotal = 0;
   535→   535→      for (const cId of allCourtIds) {
   536→   536→        const court = await getCourtById(cId);
   537→   537→        if (court) {
   538→   538→          const schedule = court.pricing_schedule as Array<{ label: string; startHour: number; endHour: number; pricePerHour: number }> | undefined;
   539→   539→          let courtPrice = 0;
   540→   540→          if (schedule && schedule.length > 0) {
   541→   541→            courtPrice = calculatePriceForTimeSlot(schedule, startTime, endTime);
   542→   542→            if (courtPrice <= 0) courtPrice = (court.price_per_hour as number) || 0;
   543→   543→          } else {
   544→   544→            courtPrice = (court.price_per_hour as number) || 0;
   545→   545→          }
   546→   546→          courtPriceTotal += courtPrice;
   547→   547→        }
   548→   548→      }
   549→   549→    }
   550→   550→
   551→   551→    const price = courtPriceTotal + equipmentSubtotal;
   552→   552→
   553→   553→    const adv = parseFloat(advanceAmount) || price * 0.5;
   554→   554→    const rem = parseFloat(remainingAmount) || price - adv;
   555→   555→    const bookingStatus = migrateStatus(status || 'reserved');
   556→   556→
   557→   557→    // Resolve client email for denormalized search
   558→   558→    let clientEmail = authUser.email;
   559→   559→    try {
   560→   560→      const clientUser = await getUserById(userId);
   561→   561→      if (clientUser?.email) clientEmail = clientUser.email;
   562→   562→    } catch { /* fallback to admin email */ }
   563→   563→
   564→   564→    // Save to Firestore — single booking with all courts
   565→   565→    const id = await createBooking({
   566→   566→      court_ids: allCourtIds,
   567→   567→      user_id: userId,
   568→   568→      user_email: clientEmail,
   569→   569→      date,
   570→   570→      start_time: startTime,
   571→   571→      end_time: endTime,
   572→   572→      total_price: price,
   573→   573→      court_subtotal: courtPriceTotal,
   574→   574→      equipment_subtotal: equipmentSubtotal,
   575→   575→      equipment_items: eqItems,
   576→   576→      advance_amount: adv,
   577→   577→      remaining_amount: rem,
   578→   578→      status: bookingStatus,
   579→   579→      slot_status: 'available',
   580→   580→      payment_method: normalizedPaymentMethod || null,
   581→   581→      notes: notes || null,
   582→   582→    });
   583→   583→
   584→   584→    console.log(`[BOOKINGS] POST: Created booking ${id} for user=${userId} date=${date} ${startTime}-${endTime} courts=${allCourtIds.join(',')} total=${price} advance=${adv} remaining=${rem} method=${normalizedPaymentMethod} email=${clientEmail}`);
   585→   585→
   586→   586→    // Create advance payment record
   587→   587→    try {
   588→   588→      await createPayment(id, {
   589→   589→        user_id: userId,
   590→   590→        amount: adv,
   591→   591→        type: 'advance',
   592→   592→        method: normalizedPaymentMethod || 'EFECTIVO',
   593→   593→        status: 'completed',
   594→   594→      });
   595→   595→    } catch (payErr) {
   596→   596→      console.error('[BOOKINGS] Warning: could not create payment record:', payErr);
   597→   597→    }
   598→   598→
   599→   599→    return NextResponse.json({
   600→   600→      id,
   601→   601→      courtId: allCourtIds[0],
   602→   602→      courtIds: allCourtIds,
   603→   603→      userId,
   604→   604→      date,
   605→   605→      startTime,
   606→   606→      endTime,
   607→   607→      totalPrice: price,
   608→   608→      advanceAmount: adv,
   609→   609→      remainingAmount: rem,
   610→   610→      status: bookingStatus,
   611→   611→      paymentMethod: paymentMethod || null,
   612→   612→      success: true,
   613→   613→    }, { status: 201 });
   614→   614→  } catch (error: unknown) {
   615→   615→    const err = error as { message?: string; code?: string; stack?: string };
   616→   616→    console.error('[BOOKINGS] POST error:', err?.message || err);
   617→   617→    console.error('[BOOKINGS] Stack:', err?.stack || 'no stack');
   618→   618→    return NextResponse.json({
   619→   619→      error: 'Failed to create booking',
   620→   620→      detail: err?.message || 'Unknown error'
   621→   621→    }, { status: 500 });
   622→   622→  }

   625→   625→export async function PUT(request: NextRequest) {
   626→   626→  try {
   627→   627→    const authResult = await requireAnyAuth(request);
   628→   628→    if (authResult instanceof NextResponse) return authResult;
   629→   629→    const authUser = authResult.user;
   630→   630→
   631→   631→    if (!isFirebaseAvailable()) {
   632→   632→      return NextResponse.json({ error: 'Firebase no configurado' }, { status: 503 });
   633→   633→    }
   634→   634→
   635→   635→    const body = await request.json();
   636→   636→    const { id, status, slot_status, advanceAmount: reqAdvance, remainingAmount: reqRemaining, paymentMethod: reqPaymentMethod, equipmentDelivered, equipmentReturned, advanceAction, cancelReason, endTime: reqEndTime, totalPrice: reqTotalPrice, extendTime } = body;
   637→   637→
   638→   638→    if (!id) {
   639→   639→      return NextResponse.json({ error: 'Booking ID is required' }, { status: 400 });
   640→   640→    }
   641→   641→
   642→   642→    // Non-admin users can only cancel their own bookings
   643→   643→    if (authUser.role !== 'admin' && authUser.role !== 'super_admin') {
   644→   644→      if (status === 'cancelled') {
   645→   645→        // Verify the booking belongs to the user (by both userId and email)
   646→   646→        const bookings = await searchBookingsForUser(authUser.id, authUser.email);
   647→   647→        const booking = bookings.find((b) => b.id === id);
   648→   648→        if (!booking) {
   649→   649→          return NextResponse.json(
   650→   650→            { error: 'No puedes modificar reservas de otros usuarios.' },
   651→   651→            { status: 403 }
   652→   652→          );
   653→   653→        }
   654→   654→      } else {
   655→   655→        return NextResponse.json(
   656→   656→          { error: 'Solo puedes cancelar tus propias reservas.' },
   657→   657→          { status: 403 }
   658→   658→        );
   659→   659→      }
   660→   660→    }
   661→   661→
   662→   662→    const updateData: Record<string, unknown> = {};
   663→   663→    if (status) updateData.status = migrateStatus(status);
   664→   664→    if (slot_status) updateData.slot_status = slot_status;
   665→   665→    if (typeof reqAdvance === 'number') updateData.advance_amount = reqAdvance;
   666→   666→    if (typeof reqRemaining === 'number') updateData.remaining_amount = reqRemaining;
   667→   667→    if (reqPaymentMethod) {
   668→   668→      const VALID_PM = ['EFECTIVO', 'YAPE', 'PLIN'];
   669→   669→      updateData.payment_method = VALID_PM.includes(reqPaymentMethod) ? reqPaymentMethod : 'EFECTIVO';
   670→   670→    }
   671→   671→    if (typeof equipmentDelivered === 'boolean') updateData.equipment_delivered = equipmentDelivered;
   672→   672→    if (typeof equipmentReturned === 'boolean') updateData.equipment_returned = equipmentReturned;
   673→   673→
   674→   674→    // Extend time: update end_time and total_price, create payment record
   675→   675→    if (extendTime && reqEndTime && reqTotalPrice) {
   676→   676→      // Validate: new end must be after current end
   677→   677→      const booking = await getBookingById(id);
   678→   678→      if (booking) {
   679→   679→        const currentEnd = booking.end_time as string || '';
   680→   680→        if (reqEndTime <= currentEnd) {
   681→   681→          return NextResponse.json({ error: 'La nueva hora de fin debe ser posterior a la actual' }, { status: 400 });
   682→   682→        }
   683→   683→        // Check for conflicts with other bookings (same court, same date, overlapping time)
   684→   684→        const allCourtIds: string[] = Array.isArray(booking.court_ids) ? booking.court_ids : (booking.court_id ? [booking.court_id] : []);
   685→   685→        const bookingDate = booking.date as string;
   686→   686→        if (allCourtIds.length > 0 && bookingDate) {
   687→   687→          const allBookings = await getBookings();
   688→   688→          const conflicts = allBookings.filter((ob: Record<string, unknown>) => {
   689→   689→            const obId = ob.id as string;
   690→   690→            if (obId === id) return false;
   691→   691→            if (ob.status === 'cancelled') return false;
   692→   692→            if (ob.date !== bookingDate) return false;
   693→   693→            const obCourtIds: string[] = Array.isArray(ob.court_ids) ? ob.court_ids : (ob.court_id ? [ob.court_id] : []);
   694→   694→            if (!obCourtIds.some((cid: string) => allCourtIds.includes(cid))) return false;
   695→   695→            return (ob.start_time as string) < reqEndTime && (ob.end_time as string) > (booking.start_time as string);
   696→   696→          });
   697→   697→          // Filter to only real conflicts (overlapping with the NEW extended period)
   698→   698→          const realConflicts = conflicts.filter((ob: Record<string, unknown>) => {
   699→   699→            return (ob.start_time as string) < reqEndTime && (ob.end_time as string) > currentEnd;
   700→   700→          });
   701→   701→          if (realConflicts.length > 0) {
   702→   702→            return NextResponse.json({ error: 'Hay una reserva que ocupa ese horario. No se puede extender.' }, { status: 409 });
   703→   703→          }
   704→   704→        }
   705→   705→
   706→   706→        const extraCost = Math.max(0, (reqTotalPrice - ((booking.total_price as number) || 0)));
   707→   707→        updateData.end_time = reqEndTime;
   708→   708→        updateData.total_price = reqTotalPrice;
   709→   709→
   710→   710→        // If admin collected payment for the extension, create a payment record
   711→   711→        if (typeof reqAdvance === 'number' && extraCost > 0.01) {
   712→   712→          const prevAdvance = (booking.advance_amount as number) || 0;
   713→   713→          const paymentIncrement = reqAdvance - prevAdvance;
   714→   714→          if (paymentIncrement > 0.01) {
   715→   715→            try {
   716→   716→              await createPayment(id, {
   717→   717→                user_id: booking.user_id as string,
   718→   718→                amount: paymentIncrement,
   719→   719→                type: 'extension',
   720→   720→                method: (reqPaymentMethod as string) || 'EFECTIVO',
   721→   721→                status: 'completed',
   722→   722→              });
   723→   723→            } catch (payErr) {
   724→   724→              console.error('[BOOKINGS] Warning: could not create payment record for extension:', payErr);
   725→   725→            }
   726→   726→          }
   727→   727→        }
   728→   728→      }
   729→   729→    }
   730→   730→
   731→   731→    // When cancelling a booking with advance, handle retained/refunded advance
   732→   732→    // Only create retained advance records when an admin explicitly provides advanceAction
   733→   733→    // (user-side cancellations should NOT auto-create records — the admin decides later)
   734→   734→    const isAdminCancellation = authUser.role === 'admin' || authUser.role === 'super_admin';
   735→   735→    if (status === 'cancelled' || (status && migrateStatus(status) === 'cancelled')) {
   736→   736→      // Bug #13: Prevent cancelling completed bookings
   737→   737→      try {
   738→   738→        const booking = await getBookingById(id);
   739→   739→        if (booking && booking.status === 'completed') {
   740→   740→          return NextResponse.json({ error: 'No se puede cancelar una reserva completada. Contacta soporte si es necesario.' }, { status: 400 });
   741→   741→        }
   742→   742→      } catch { /* non-blocking */ }
   743→   743→
   744→   744→      if (isAdminCancellation) {
   745→   745→      try {
   746→   746→        const booking = await getBookingById(id);
   747→   747→        if (booking && ((booking.advance_amount as number) || 0) > 0) {
   748→   748→          // Bug #4: Check for duplicate retained advance
   749→   749→          const { getRetainedAdvances } = await import('@/lib/db');
   750→   750→          const existing = await getRetainedAdvances({ bookingId: id });
   751→   751→          if (existing.length > 0) {
   752→   752→            console.warn('[BOOKINGS] Retained advance already exists for booking', id, '- skipping creation');
   753→   753→          } else {
   754→   754→          const advAmount = (booking.advance_amount as number) || 0;
   755→   755→          const isRetained = advanceAction === 'retain';
   756→   756→          const statusAdvance = isRetained ? 'retained' : 'refunded';
   757→   757→
   758→   758→          // Get court name for the record
   759→   759→          let courtName = '';
   760→   760→          try {
   761→   761→            const cId = (booking.court_ids as string[])?.[0] || (booking.court_id as string) || '';
   762→   762→            if (cId) {
   763→   763→              const court = await getCourtById(cId);
   764→   764→              courtName = (court?.name as string) || '';
   765→   765→            }
   766→   766→          } catch { /* non-critical */ }
   767→   767→
   768→   768→          // Get user name
   769→   769→          let userName = '';
   770→   770→          try {
   771→   771→            const uId = booking.user_id as string;
   772→   772→            if (uId) {
   773→   773→              const user = await getUserById(uId);
   774→   774→              userName = (user?.name as string) || '';
   775→   775→            }
   776→   776→          } catch { /* non-critical */ }
   777→   777→
   778→   778→          await createRetainedAdvance({
   779→   779→            booking_id: id,
   780→   780→            user_id: (booking.user_id as string) || '',
   781→   781→            user_name: userName,
   782→   782→            user_email: (booking.user_email as string) || null,
   783→   783→            court_name: courtName,
   784→   784→            booking_date: (booking.date as string) || '',
   785→   785→            amount: advAmount,
   786→   786→            original_total: (booking.total_price as number) || 0,
   787→   787→            payment_method: (booking.payment_method as string) || 'EFECTIVO',
   788→   788→            reason: cancelReason || 'Cancelación de reserva',
   789→   789→            status: statusAdvance,
   790→   790→          });
   791→   791→
   792→   792→          // Create a payment record for the refund/retention
   793→   793→          try {
   794→   794→            await createPayment(id, {
   795→   795→              user_id: (booking.user_id as string) || '',
   796→   796→              amount: advAmount,
   797→   797→              type: isRetained ? 'retained' : 'refund',
   798→   798→              method: (booking.payment_method as string) || 'EFECTIVO',
   799→   799→              status: 'completed',
   800→   800→            });
   801→   801→          } catch (payErr) {
   802→   802→            console.error('[BOOKINGS] Warning: could not create payment record for cancellation:', payErr);
   803→   803→          }
   804→   804→          }
   805→   805→        }
   806→   806→      } catch (err) {
   807→   807→        console.error('[BOOKINGS] Warning: could not process retained advance on cancellation:', err);
   808→   808→      }
   809→   809→      } // end isAdminCancellation
   810→   810→    }
   811→   811→
   812→   812→    // If advance amount is being updated, create a payment record for the increment
   813→   813→    if (typeof reqAdvance === 'number' && typeof reqRemaining === 'number') {
   814→   814→      try {
   815→   815→        const booking = await getBookingById(id);
   816→   816→        if (booking) {
   817→   817→          const prevAdvance = (booking.advance_amount as number) || 0;
   818→   818→          const paymentIncrement = reqAdvance - prevAdvance;
   819→   819→          if (paymentIncrement > 0.01) {
   820→   820→            try {
   821→   821→              await createPayment(id, {
   822→   822→                user_id: booking.user_id as string,
   823→   823→                amount: paymentIncrement,
   824→   824→                type: reqRemaining <= 0 ? 'full_payment' : 'advance',
   825→   825→                method: (updateData.payment_method as string) || 'EFECTIVO',
   826→   826→                status: 'completed',
   827→   827→              });
   828→   828→            } catch (payErr) {
   829→   829→              console.error('[BOOKINGS] Warning: could not create payment record for advance update:', payErr);
   830→   830→            }
   831→   831→          }
   832→   832→        }
   833→   833→      } catch { /* non-critical: proceed with booking update */ }
   834→   834→    }
   835→   835→
   836→   836→    await updateBooking(id, updateData);
   837→   837→
   838→   838→    return NextResponse.json({ success: true });
   839→   839→  } catch (error) {
   840→   840→    console.error('[BOOKINGS] PUT error:', error);
   841→   841→    const msg = error instanceof Error ? error.message : 'Unknown error';
   842→   842→    return NextResponse.json({ error: 'Error al actualizar reserva', detail: msg }, { status: 500 });
   843→   843→  }
   844→   844→}
