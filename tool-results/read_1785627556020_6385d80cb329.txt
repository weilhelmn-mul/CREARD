   254→function buildQuery(
   255→  collectionName: string,
   256→  constraints: Array<{ field: string; op: string; value: unknown }>
   257→): Query {
   258→  let q: Query = getAdminDb().collection(collectionName);
   259→  for (const c of constraints) {
   260→    switch (c.op) {
   261→      case '==': q = q.where(c.field, '==', c.value); break;
   262→      case '!=': q = q.where(c.field, '!=', c.value); break;
   263→      case '<': q = q.where(c.field, '<', c.value); break;
   264→      case '<=': q = q.where(c.field, '<=', c.value); break;
   265→      case '>': q = q.where(c.field, '>', c.value); break;
   266→      case '>=': q = q.where(c.field, '>=', c.value); break;
   267→      case 'in': q = q.where(c.field, 'in', c.value as unknown[]); break;
   268→      case 'not-in': q = q.where(c.field, 'not-in', c.value as unknown[]); break;
   269→      case 'array-contains': q = q.where(c.field, 'array-contains', c.value); break;
   270→    }
   271→  }
   272→  return q;
   273→}

   275→async function queryDocs(
   276→  collectionName: string,
   277→  constraints: Array<{ field: string; op: string; value: unknown }> = [],
   278→  orderField?: string,
   279→  orderDir: 'asc' | 'desc' = 'asc'
   280→): Promise<DocumentData[]> {
   281→  let q = buildQuery(collectionName, constraints);
   282→  // NEVER use Firestore orderBy — always sort client-side
   283→  // This avoids needing composite indexes (user_id+start_time, user_email+start_time, etc.)
   284→  const snapshot: QuerySnapshot = await q.get();
   285→  let docs = snapshot.docs.map((d) => ({ id: d.id, ...d.data() }));
   286→  // Client-side sort for all queries
   287→  if (orderField) {
   288→    docs.sort((a, b) => {
   289→      const va = a[orderField] ?? '';
   290→      const vb = b[orderField] ?? '';
   291→      if (va < vb) return orderDir === 'asc' ? -1 : 1;
   292→      if (va > vb) return orderDir === 'asc' ? 1 : -1;
   293→      return 0;
   294→    });
   295→  }
   296→  return docs;
   297→}

   299→async function getDocById(collectionName: string, id: string): Promise<DocumentData | null> {
   300→  const docSnap = await getAdminDb().collection(collectionName).doc(id).get();
   301→  if (!docSnap.exists) return null;
   302→  return { id: docSnap.id, ...docSnap.data() };
   303→}

   356→export async function getCourts(filters?: {
   357→  sport?: string;
   358→  branchId?: string;
   359→  active?: boolean;
   360→}): Promise<Partial<Court>[]> {
   361→  const constraints: Array<{ field: string; op: string; value: unknown }> = [];
   362→  if (filters?.sport && filters.sport !== 'todos') {
   363→    constraints.push({ field: 'sport', op: '==', value: filters.sport });
   364→  }
   365→  if (filters?.branchId) {
   366→    constraints.push({ field: 'branch_id', op: '==', value: filters.branchId });
   367→  }
   368→  if (filters?.active !== undefined) {
   369→    constraints.push({ field: 'is_active', op: '==', value: filters.active });
   370→  }
   371→
   372→  const docs = await queryDocs('courts', constraints, 'name');
   373→  return docs.map((d) => ({
   374→    ...d,
   375→    images: Array.isArray(d.images) ? d.images : [],
   376→    amenities: Array.isArray(d.amenities) ? d.amenities : [],
   377→  }));
   378→}

   386→    amenities: Array.isArray(d.amenities) ? d.amenities : [],

   405→export async function getBookings(filters?: {
   406→  courtId?: string;
   407→  userId?: string;
   408→  userEmail?: string;
   409→  date?: string;
   410→  dateFrom?: string;
   411→  dateTo?: string;
   412→  status?: string;
   413→}): Promise<Partial<Booking>[]> {
   414→  // Build non-court constraints (shared by both queries)
   415→  const baseConstraints: Array<{ field: string; op: string; value: unknown }> = [];
   416→  if (filters?.userId) baseConstraints.push({ field: 'user_id', op: '==', value: filters.userId });
   417→  if (filters?.userEmail) baseConstraints.push({ field: 'user_email', op: '==', value: filters.userEmail });
   418→  if (filters?.date) baseConstraints.push({ field: 'date', op: '==', value: filters.date });
   419→  if (filters?.dateFrom) baseConstraints.push({ field: 'date', op: '>=', value: filters.dateFrom });
   420→  if (filters?.dateTo) baseConstraints.push({ field: 'date', op: '<=', value: filters.dateTo });
   421→  if (filters?.status) baseConstraints.push({ field: 'status', op: '==', value: filters.status });
   422→
   423→  // No court filter — single query
   424→  if (!filters?.courtId) {
   425→    return queryDocs('bookings', baseConstraints, 'start_time');
   426→  }
   427→
   428→  // Court filter present — need TWO queries:
   429→  //   1) court_id == X  (primary court)
   430→  //   2) court_ids array-contains X  (secondary court in multi-court booking)
   431→  const [byPrimary, byArray] = await Promise.all([
   432→    queryDocs('bookings', [
   433→      ...baseConstraints,
   434→      { field: 'court_id', op: '==', value: filters.courtId },
   435→    ], 'start_time'),
   436→    queryDocs('bookings', [
   437→      ...baseConstraints,
   438→      { field: 'court_ids', op: 'array-contains', value: filters.courtId },
   439→    ], 'start_time'),
   440→  ]);
   441→
   442→  // Merge & deduplicate by document ID
   443→  const seen = new Set<string>();
   444→  const merged: DocumentData[] = [];
   445→  for (const doc of [...byPrimary, ...byArray]) {
   446→    if (!seen.has(doc.id)) {
   447→      seen.add(doc.id);
   448→      merged.push(doc);
   449→    }
   450→  }
   451→
   452→  // Client-side sort (already sorted individually, but merge breaks order)
   453→  merged.sort((a, b) => {
   454→    const va = a.start_time ?? '';
   455→    const vb = b.start_time ?? '';
   456→    if (va < vb) return -1;
   457→    if (va > vb) return 1;
   458→    return 0;
   459→  });
   460→
   461→  return merged as Partial<Booking>[];
   462→}

   505→  bookingId: string,
   506→  data: Partial<Payment> & {
   507→    payment_id?: string;
   508→    payment_code?: string;
   509→    booking_code?: string;
   510→    user_name?: string;

   527→): Promise<string> {

   529→  const db = await getAdminDb();
   530→

   532→  const paymentData: Record<string, unknown> = {
   533→    booking_id: bookingId,
   534→    user_id: data.user_id,
   535→    amount: data.amount || 0,
   536→    type: data.type || 'advance',

   538→    status: data.status || 'completed',
   539→    external_ref: data.external_ref || null,
   540→    created_at: now,
   541→    updated_at: now,
   542→  };
   543→
   544→  // Add new extra fields if provided
   545→  if (data.payment_id !== undefined) paymentData.payment_id = data.payment_id;
   546→  if (data.payment_code !== undefined) paymentData.payment_code = data.payment_code;
   547→  if (data.booking_code !== undefined) paymentData.booking_code = data.booking_code;
   548→  if (data.user_name !== undefined) paymentData.user_name = data.user_name;
   549→  if (data.user_email !== undefined) paymentData.user_email = data.user_email;
   550→  if (data.user_phone !== undefined) paymentData.user_phone = data.user_phone;
   551→  if (data.user_document !== undefined) paymentData.user_document = data.user_document;
   552→  if (data.court_name !== undefined) paymentData.court_name = data.court_name;
   553→  if (data.sport !== undefined) paymentData.sport = data.sport;
   554→  if (data.booking_date !== undefined) paymentData.booking_date = data.booking_date;
   555→  if (data.booking_start_time !== undefined) paymentData.booking_start_time = data.booking_start_time;
   556→  if (data.booking_end_time !== undefined) paymentData.booking_end_time = data.booking_end_time;
   557→  if (data.payment_type !== undefined) paymentData.payment_type = data.payment_type;
   558→  if (data.amount_paid !== undefined) paymentData.amount_paid = data.amount_paid;
   559→  if (data.remaining_balance !== undefined) paymentData.remaining_balance = data.remaining_balance;
   560→  if (data.payment_method_display !== undefined) paymentData.payment_method_display = data.payment_method_display;
   561→  if (data.payment_status !== undefined) paymentData.payment_status = data.payment_status;
   562→  if (data.payment_date !== undefined) paymentData.payment_date = data.payment_date;
   563→  if (data.payment_time !== undefined) paymentData.payment_time = data.payment_time;
   564→
   565→  // Write to subcollection (bookings/{id}/payments)
   566→  const docRef = await db
   567→    .collection('bookings')
   568→    .doc(bookingId)

   592→/**
   593→ * Get the next payment number from the counters collection (atomic increment via transaction)
   594→ */
   595→export async function getNextPaymentNumber(): Promise<number> {
   596→  const db = await getAdminDb();
   597→  const counterRef = db.collection('counters').doc('payments');
   598→
   599→  const result = await db.runTransaction(async (transaction) => {
   600→    const counterSnap = await transaction.get(counterRef);
   601→    if (!counterSnap.exists) {
   602→      // Counter doesn't exist yet, create it starting at 1
   603→      const newCounter = { value: 1, updated_at: Timestamp.now() };
   604→      transaction.set(counterRef, newCounter);
   605→      return 1;
   606→    }
   607→    const current = counterSnap.data()?.value ?? 0;
   608→    const next = current + 1;
   609→    transaction.update(counterRef, {
   610→      value: next,
   611→      updated_at: Timestamp.now(),
   612→    });
   613→    return next;
   614→  });
   615→
   616→  return result;
   617→}
