    11→// --- Animated Counter Component ---
    12→function AnimatedCounter({ target, duration = 2, suffix = '' }: { target: number; duration?: number; suffix?: string }) {
    13→  const [count, setCount] = useState(0)
    14→  const ref = useRef<HTMLSpanElement>(null)
    15→  const isInView = useInView(ref, { once: true, margin: '-50px' })
    16→  const hasStarted = useRef(false)
    17→
    18→  useEffect(() => {
    19→    if (!isInView || hasStarted.current) return
    20→    hasStarted.current = true
    21→
    22→    const startTime = performance.now()
    23→    const from = 0
    24→    const to = target
    25→
    26→    function update(currentTime: number) {
    27→      const elapsed = currentTime - startTime
    28→      const progress = Math.min(elapsed / (duration * 1000), 1)
    29→      const eased = 1 - Math.pow(1 - progress, 3)
    30→      const current = Math.round(from + (to - from) * eased)
    31→      setCount(current)
    32→      if (progress < 1) {
    33→        requestAnimationFrame(update)
    34→      }
    35→    }
    36→    requestAnimationFrame(update)
    37→  }, [isInView, target, duration])
    38→
    39→  return (
    40→    <span ref={ref}>
    41→      {count}{suffix}
    42→    </span>
    43→  )
    44→}
    45→
    46→// --- Paginated Date Chips ---
    47→const DAYS_SHORT = ['Dom', 'Lun', 'Mar', 'Mié', 'Jue', 'Vie', 'Sáb']
    48→const MONTHS_SHORT = ['Ene', 'Feb', 'Mar', 'Abr', 'May', 'Jun', 'Jul', 'Ago', 'Sep', 'Oct', 'Nov', 'Dic']
    49→const VISIBLE_DATES = 4
    50→
    51→function getDateChipsForPage(page: number) {
    52→  const chips: { date: Date; label: string; dayName: string; isToday: boolean }[] = []
    53→  const today = new Date()
    54→  today.setHours(0, 0, 0, 0)
    55→  for (let i = 0; i < VISIBLE_DATES; i++) {
    56→    const d = new Date(today)
    57→    d.setDate(d.getDate() + page * VISIBLE_DATES + i)
    58→    chips.push({
    59→      date: d,
    60→      label: `${d.getDate()} ${MONTHS_SHORT[d.getMonth()]}`,
    61→      dayName: i === 0 && page === 0 ? 'Hoy' : DAYS_SHORT[d.getDay()],
    62→      isToday: page === 0 && i === 0,
    63→    })
    64→  }
    65→  return chips
    66→}
