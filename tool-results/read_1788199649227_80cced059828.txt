    20→    20→// Migrate old status values to the new 3-status system
    21→    21→function migrateStatus(s: string): string {
    22→    22→  switch (s) {
    23→    23→    case 'confirmed':
    24→    24→    case 'pending':
    25→    25→    case 'partially_paid':
    26→    26→      return 'reserved';
    27→    27→    case 'fully_paid':
    28→    28→    case 'completed':
    29→    29→      return 'completed';
    30→    30→    case 'no_show':
    31→    31→    case 'expired':
    32→    32→      return 'cancelled';
    33→    33→    default:
    34→    34→      return s; // 'reserved', 'completed', 'cancelled' pass through
    35→    35→  }
    36→    36→}

    38→    38→// Transformar snake_case (Firestore) a camelCase (frontend)
    39→    39→function toCamelBooking(b: Record<string, unknown>) {
    40→    40→  // Primary court (backward compat)
    41→    41→  const courtRaw = b._court as Record<string, unknown> | null;
    42→    42→  const court = courtRaw ? {
    43→    43→    id: courtRaw.id || courtRaw.court_id,
    44→    44→    name: courtRaw.name || courtRaw.court_name,
    45→    45→    sport: courtRaw.sport,
    46→    46→    branch: (courtRaw.branch as Record<string, unknown>) || {
    47→    47→      id: courtRaw.branch_id || 'branch-1',
    48→    48→      name: 'CREARD',
    49→    49→    },
    50→    50→  } : null;
    51→    51→
    52→    52→  // Additional courts (multi-court bookings)
    53→    53→  const courtsRaw = b._courts as Array<Record<string, unknown>> | null;
    54→    54→  const courts = courtsRaw ? courtsRaw.map((cr) => ({
    55→    55→    id: cr.id || cr.court_id,
    56→    56→    name: cr.name || cr.court_name,
    57→    57→    sport: cr.sport,
    58→    58→    branch: (cr.branch as Record<string, unknown>) || { id: 'branch-1', name: 'CREARD' },
    59→    59→  })) : [];
    60→    60→
    61→    61→  const userRaw = b._user as Record<string, unknown> | null;
    62→    62→  const user = userRaw ? {
    63→    63→    id: userRaw.id,
    64→    64→    name: userRaw.name,
    65→    65→    email: userRaw.email,
    66→    66→    phone: userRaw.phone,
    67→    67→  } : null;
    68→    68→
    69→    69→  // Court IDs array
    70→    70→  const courtIds: string[] = Array.isArray(b.court_ids) ? b.court_ids : (b.court_id ? [b.court_id] : []);
    71→    71→
    72→    72→  // Equipment items
    73→    73→  const equipmentItems = Array.isArray(b.equipment_items) ? b.equipment_items.map((eq: Record<string, unknown>) => ({
    74→    74→    equipmentId: eq.equipment_id || eq.id,
    75→    75→    name: eq.name,
    76→    76→    quantity: eq.quantity || 1,
    77→    77→    unitPrice: eq.unit_price || eq.price_per_rental || 0,
    78→    78→    subtotal: (eq.quantity || 1) * (eq.unit_price || eq.price_per_rental || 0),
    79→    79→  })) : [];
    80→    80→
    81→    81→  // Normalize: recalc equipment subtotal from items (more reliable than stored value)
    82→    82→  const computedEquipSubtotal = equipmentItems.reduce((s: number, eq: { subtotal: number }) => s + eq.subtotal, 0);
    83→    83→  const equipmentSubtotal = computedEquipSubtotal;
    84→    84→
    85→    85→  // Normalize court_subtotal: if stored value is wrong (old double-counting bug),
    86→    86→  // derive it from total_price - equipment_subtotal
    87→    87→  const totalPrice = b.total_price || 0;
    88→    88→  const storedCourtSubtotal = b.court_subtotal || 0;
    89→    89→  const courtSubtotal = (storedCourtSubtotal + equipmentSubtotal > totalPrice + 0.01)
    90→    90→    ? Math.max(0, Math.round((totalPrice - equipmentSubtotal) * 100) / 100)
    91→    91→    : storedCourtSubtotal;
    92→    92→
    93→    93→  // Normalize remaining_amount: ensure it equals total - advance
    94→    94→  const advanceAmount = b.advance_amount || 0;
    95→    95→  const storedRemaining = b.remaining_amount || 0;
    96→    96→  const remainingAmount = (Math.abs(storedRemaining - (totalPrice - advanceAmount)) > 0.01)
    97→    97→    ? Math.max(0, Math.round((totalPrice - advanceAmount) * 100) / 100)
    98→    98→    : storedRemaining;
    99→    99→
   100→   100→  return {
   101→   101→    id: b.id,
   102→   102→    courtId: b.court_id,
   103→   103→    courtIds,
   104→   104→    userId: b.user_id,
   105→   105→    date: typeof b.date === 'string' ? b.date : (b.date ? String(b.date) : ''),
   106→   106→    startTime: b.start_time,
   107→   107→    endTime: b.end_time,
   108→   108→    totalPrice,
   109→   109→    courtSubtotal,
   110→   110→    equipmentSubtotal,
   111→   111→    equipmentItems,
   112→   112→    advanceAmount,
   113→   113→    remainingAmount,
   114→   114→    status: migrateStatus(b.status || 'reserved'),
   115→   115→    slotStatus: b.slot_status,
   116→   116→    paymentMethod: b.payment_method,
   117→   117→    remainingPaymentStatus: b.remaining_payment_status || null,
   118→   118→    notes: b.notes,
   119→   119→    createdAt: b.created_at,
   120→   120→    updatedAt: b.updated_at,
   121→   121→    recurringGroupId: b.recurring_group_id,
   122→   122→    recurringIndex: b.recurring_index,
   123→   123→    equipmentDelivered: b.equipment_delivered || false,
   124→   124→    equipmentReturned: b.equipment_returned || false,
   125→   125→    court,
   126→   126→    courts,
   127→   127→    user,
   128→   128→  };
   129→   129→}

   131→   131→// Enrich a single booking (all courts + user), safe — never throws
   132→   132→async function safeEnrichBooking(b: Record<string, unknown>): Promise<Record<string, unknown>> {
   133→   133→  let court: Record<string, unknown> | null = null;
   134→   134→  let courts: Array<Record<string, unknown>> = [];
   135→   135→  let user: Record<string, unknown> | null = null;
   136→   136→
   137→   137→  // Get all court IDs (from court_ids array or single court_id)
   138→   138→  const allCourtIds: string[] = Array.isArray(b.court_ids)
   139→   139→    ? b.court_ids as string[]
   140→   140→    : (b.court_id ? [b.court_id as string] : []);
   141→   141→
   142→   142→  // Enrich primary court (first one)
   143→   143→  try {
   144→   144→    if (allCourtIds.length > 0) {
   145→   145→      const c = await getCourtById(allCourtIds[0]);
   146→   146→      if (c) court = c as Record<string, unknown>;
   147→   147→    }
   148→   148→  } catch { /* court enrichment failed */ }
   149→   149→
   150→   150→
   151→   151→  // Enrich all courts for multi-court bookings
   152→   152→  if (allCourtIds.length > 1) {
   153→   153→    const enrichedCourts: Array<Record<string, unknown>> = [];
   154→   154→    for (const cid of allCourtIds) {
   155→   155→      try {
   156→   156→        const c = await getCourtById(cid);
   157→   157→        if (c) enrichedCourts.push(c as Record<string, unknown>);
   158→   158→      } catch { /* skip failed court enrichment */ }
   159→   159→    }
   160→   160→    courts = enrichedCourts;
   161→   161→  }
   162→   162→
   163→   163→  try {
   164→   164→    if (b.user_id) {
   165→   165→      const u = await getUserById(b.user_id as string);
   166→   166→      if (u) user = u as Record<string, unknown>;
   167→   167→    }
   168→   168→  } catch { /* user enrichment failed */ }
   169→   169→
   170→   170→
   171→   171→  return { ...b, _court: court, _courts: courts, _user: user };
   172→   172→}

   175→   175→async function searchBookingsForUser(userId: string, userEmail: string | null) {
   176→   176→  const results = new Map<string, Record<string, unknown>>();

   183→   183→        results.set(b.id, b as Record<string, unknown>);
   184→   184→      }
   185→   185→    }

   189→   189→

   191→   191→  if (userEmail && userEmail !== userId) {
   192→   192→    try {
   193→   193→      const byEmail = await getBookings({ userEmail });
   194→   194→      for (const b of byEmail) {
   195→   195→        if (b.id && !results.has(b.id)) {
   196→   196→          results.set(b.id, b as Record<string, unknown>);
   197→   197→        }
   198→   198→      }
   199→   199→    } catch (e) {
   200→   200→      console.error('[BOOKINGS] Error searching by userEmail:', userEmail, e);

   202→   202→  }
   203→   203→
   204→   204→  return Array.from(results.values());

   207→   207→/**
   208→   208→ * Calculate the total price for a time slot using a pricing schedule.
   209→   209→ * Supports slots that span multiple schedules (e.g., 16:00-19:00 = 1hr morning + 2hr night).
   210→   210→ */
   211→   211→function calculatePriceForTimeSlot(
   212→   212→  schedule: Array<{ label: string; startHour: number; endHour: number; pricePerHour: number }>,
   213→   213→  startTime: string,
   214→   214→  endTime: string,
   215→   215→): number {
   216→   216→  const [startH, startM] = startTime.split(':').map(Number);
   217→   217→  const [endH, endM] = endTime.split(':').map(Number);
   218→   218→  const startDecimal = startH + startM / 60;
   219→   219→  const endDecimal = endH + endM / 60;
   220→   220→
   221→   221→  // Sort schedule by startHour
   222→   222→  const sorted = [...schedule].sort((a, b) => a.startHour - b.startHour);
   223→   223→
   224→   224→  let total = 0;
   225→   225→  let cursor = startDecimal;
   226→   226→
   227→   227→  for (const slot of sorted) {
   228→   228→    const slotStart = slot.startHour;
   229→   229→    const slotEnd = slot.endHour;
   230→   230→
   231→   231→    if (cursor >= slotEnd) continue; // cursor already past this slot
   232→   232→
   233→   233→    // Overlap start/end
   234→   234→    const overlapStart = Math.max(cursor, slotStart);
   235→   235→    const overlapEnd = Math.min(endDecimal, slotEnd);
   236→   236→
   237→   237→    if (overlapEnd > overlapStart) {
   238→   238→      const hours = overlapEnd - overlapStart;
   239→   239→      total += hours * slot.pricePerHour;
   240→   240→      cursor = overlapEnd;
   241→   241→    }
   242→   242→  }
   243→   243→
   244→   244→  // If cursor didn't reach endDecimal, the remaining time is uncovered by schedule
   245→   245→  // (fall back to 0 or caller's default)
   246→   246→  return Math.round(total * 100) / 100;
   247→   247→}

   249→   249→export async function GET(request: NextRequest) {
   250→   250→  const startTime = Date.now();
   251→   251→  const firebaseOk = isFirebaseAvailable();
   252→   252→
   253→   253→  try {
   254→   254→    const { searchParams } = new URL(request.url);
   255→   255→    const courtId = searchParams.get('courtId');
   256→   256→    const userId = searchParams.get('userId');
   257→   257→    const date = searchParams.get('date');
   258→   258→    const dateFrom = searchParams.get('dateFrom');
   259→   259→    const dateTo = searchParams.get('dateTo');
   260→   260→    const status = searchParams.get('status');
   261→   261→
   262→   262→    // Court availability check (courtId + date) is public — no auth required
   263→   263→    if (courtId && date && !userId) {
   264→   264→      if (!firebaseOk) {
   265→   265→        return NextResponse.json({ error: 'Firebase no configurado', code: 'NO_FIREBASE' }, { status: 503 });
   266→   266→      }
   267→   267→      const bookings = await getBookings({ courtId, date });
   268→   268→      // FIX P0-4: Filter out expired reserved bookings (ghost reservations)
   269→   269→      const now = Date.now();
   270→   270→      const active = bookings.filter((b: any) => {
   271→   271→        if (migrateStatus(b.status || 'reserved') !== 'reserved') return true;
   272→   272→        const exp = b.expires_at;
   273→   273→        if (!exp) return true;
   274→   274→        const expMs = exp.toMillis?.()?.() || new Date(exp).getTime();
   275→   275→        return expMs > now;
   276→   276→      });
   277→   277→      return NextResponse.json(active.map(toCamelBooking));
   278→   278→    }
   279→   279→
   280→   280→    // All other queries require authentication
   281→   281→    const authResult = await requireAnyAuth(request);
   282→   282→    if (authResult instanceof NextResponse) return authResult;
   283→   283→    const authUser = authResult.user;
   284→   284→
   285→   285→    if (!firebaseOk) {
   286→   286→      return NextResponse.json({ error: 'Firebase no configurado', code: 'NO_FIREBASE' }, { status: 503 });
   287→   287→    }
   288→   288→
   289→   289→    let bookings: Record<string, unknown>[];
   290→   290→
   291→   291→    // Non-admin users: only their own bookings
   292→   292→    if (authUser.role !== 'admin' && authUser.role !== 'super_admin') {
   293→   293→      if (userId && userId !== authUser.id) {
   294→   294→        return NextResponse.json({ error: 'No puedes ver reservas de otros usuarios.' }, { status: 403 });
   295→   295→      }
   296→   296→      const effectiveUserId = userId || authUser.id;
   297→   297→      bookings = await searchBookingsForUser(effectiveUserId, authUser.email);
   298→   298→    } else {
   299→   299→      // Admin: fetch with optional filters (date range for performance)
   300→   300→      bookings = (await getBookings({
   301→   301→        courtId: courtId || undefined,
   302→   302→        userId: userId || undefined,
   303→   303→        date: date || undefined,
   304→   304→        dateFrom: dateFrom || undefined,
   305→   305→        dateTo: dateTo || undefined,
   306→   306→        status: status || undefined,
   307→   307→      })) as Record<string, unknown>[];
   308→   308→      // DEBUG removed (P3-16): raw booking data no longer logged
   309→   309→    }
   310→   310→
   311→   311→    // Enrich each booking with cached court/user lookups (avoid N+1 Firestore reads)
   312→   312→    const courtCache = new Map<string, Record<string, unknown> | null>();
   313→   313→    const userCache = new Map<string, Record<string, unknown> | null>();
   314→   314→    const getCourtCached = async (cid: string) => {
   315→   315→      if (courtCache.has(cid)) return courtCache.get(cid)!;
   316→   316→      try {
   317→   317→        const c = await getCourtById(cid);
   318→   318→        courtCache.set(cid, c as Record<string, unknown> | null);
   319→   319→        return c as Record<string, unknown> | null;
   320→   320→      } catch { courtCache.set(cid, null); return null; }
   321→   321→    };
   322→   322→    const getUserCached = async (uid: string) => {
   323→   323→      if (userCache.has(uid)) return userCache.get(uid)!;
   324→   324→      try {
   325→   325→        const u = await getUserById(uid);
   326→   326→        userCache.set(uid, u as Record<string, unknown> | null);
   327→   327→        return u as Record<string, unknown> | null;
   328→   328→      } catch { userCache.set(uid, null); return null; }
   329→   329→    };
   330→   330→
   331→   331→    let enriched: Record<string, unknown>[] = [];
   332→   332→    // Process in batches of 10 to avoid overwhelming Firestore
   333→   333→    const BATCH_SIZE = 10;
   334→   334→    for (let i = 0; i < bookings.length; i += BATCH_SIZE) {
   335→   335→      const batch = bookings.slice(i, i + BATCH_SIZE);
   336→   336→      const results = await Promise.allSettled(
   337→   337→        batch.map(async (b) => {
   338→   338→          const allCourtIds: string[] = Array.isArray(b.court_ids)
   339→   339→            ? b.court_ids as string[] : (b.court_id ? [b.court_id as string] : []);
   340→   340→          let court: Record<string, unknown> | null = null;
   341→   341→          let courts: Array<Record<string, unknown>> = [];
   342→   342→          let user: Record<string, unknown> | null = null;
   343→   343→          try {
   344→   344→            if (allCourtIds.length > 0) court = await getCourtCached(allCourtIds[0]);
   345→   345→          } catch { /* skip */ }
   346→   346→          if (allCourtIds.length > 1) {
   347→   347→            const enrichedCourts: Array<Record<string, unknown>> = [];
   348→   348→            for (const cid of allCourtIds) {
   349→   349→              try {
   350→   350→                const c = await getCourtCached(cid);
   351→   351→                if (c) enrichedCourts.push(c);
   352→   352→              } catch { /* skip */ }
   353→   353→            }
   354→   354→            courts = enrichedCourts;
   355→   355→          }
   356→   356→          try {
   357→   357→            if (b.user_id) user = await getUserCached(b.user_id as string);
   358→   358→          } catch { /* skip */ }
   359→   359→          return toCamelBooking({ ...b, _court: court, _courts: courts, _user: user });
   360→   360→        })
   361→   361→      );
   362→   362→      for (const r of results) {
   363→   363→        if (r.status === 'fulfilled') {
   364→   364→          enriched.push(r.value);
   365→   365→        } else {
   366→   366→          console.error('[BOOKINGS] Batch enrichment error:', r.reason);
   367→   367→        }
   368→   368→      }
   369→   369→    }
   370→   370→
   371→   371→    // Default server-side sort: today first, future ascending, past descending
   372→   372→    // Within each date: chronological time
   373→   373→    const today = new Date().toLocaleDateString('en-CA', { timeZone: 'America/Lima' });
   374→   374→    enriched.sort((a, b) => {
   375→   375→      const dA = String(a.date || ''), dB = String(b.date || '');
   376→   376→      const aIsToday = dA === today, bIsToday = dB === today;
   377→   377→      const aFuture = dA > today, bFuture = dB > today;
   378→   378→      const aPast = dA < today, bPast = dB < today;
   379→   379→      // 1) Today always first
   380→   380→      if (aIsToday && !bIsToday) return -1;
   381→   381→      if (!aIsToday && bIsToday) return 1;
   382→   382→      // 2) Both on same side of today
   383→   383→      if (aFuture && bFuture) {
   384→   384→        // Future dates: ascending (correlativo)
   385→   385→        const dc = dA.localeCompare(dB);
   386→   386→        if (dc !== 0) return dc;
   387→   387→      } else if (aPast && bPast) {
   388→   388→        // Past dates: descending (most recent past first)
   389→   389→        const dc = dB.localeCompare(dA);
   390→   390→        if (dc !== 0) return dc;
   391→   391→      }
   392→   392→      // Same date: chronological by start time
   393→   393→      return String(a.startTime || '').localeCompare(String(b.startTime || ''));
   394→   394→    });
   395→   395→
   396→   396→    // FIX P0-4: Lazy-expire reserved bookings past their TTL
   397→   397→    try {
   398→   398→      const db = await getAdminDb();
   399→   399→      const nowMs = Date.now();
   400→   400→      const expiredIds: string[] = [];
   401→   401→      for (const b of bookings) {
   402→   402→        if (migrateStatus(b.status || 'reserved') !== 'reserved') continue;
   403→   403→        const exp = b.expires_at;
   404→   404→        if (!exp) continue;
   405→   405→        const expMs = exp.toMillis?.()?.() || new Date(exp).getTime();
   406→   406→        if (expMs <= nowMs) expiredIds.push(b.id as string);
   407→   407→      }
   408→   408→      if (expiredIds.length > 0) {
   409→   409→        await Promise.all(expiredIds.map((id) =>
   410→   410→          db.collection('bookings').doc(id).update({
   411→   411→            status: 'cancelled',
   412→   412→            slot_status: 'available',
   413→   413→            updated_at: Timestamp.now(),
   414→   414→          })
   415→   415→        ));
   416→   416→        // Remove expired from enriched results
   417→   417→        const expiredSet = new Set(expiredIds);
   418→   418→        enriched = enriched.filter((b) => !expiredSet.has(b.id as string));
   419→   419→      }
   420→   420→    } catch { /* non-critical: best-effort cleanup */ }
   421→   421→
   422→   422→    return NextResponse.json(enriched);
   423→   423→  } catch (error) {
   424→   424→    console.error('[BOOKINGS] GET error:', error);
   425→   425→    const msg = error instanceof Error ? error.message : 'Unknown error';
   426→   426→    return NextResponse.json({ error: 'Error al obtener reservas', detail: msg }, { status: 500 });
   427→   427→  }
   428→   428→}

   904→   904→        updateData.equipment_subtotal = Math.round(equipSub * 100) / 100;

   913→   913→        // Recalc court_subtotal

   916→   916→      } else if (typeof reqAdvance === 'number') {
   917→   917→        // Only advance changed, not total

   919→   919→        updateData.advance_amount = reqAdvance;
   920→   920→        updateData.remaining_amount = Math.max(0, Math.round((currentTotal - reqAdvance) * 100) / 100);

   922→   922→

   924→   924→      if (reqPaymentMethod) {
   925→   925→        // FIX P1-7: Include online payment methods

   927→   927→        const upper = (reqPaymentMethod as string).toUpperCase();
   928→   928→        updateData.payment_method = VALID_PM.includes(upper) ? upper : 'EFECTIVO';
   929→   929→      }
   930→   930→
   931→   931→      // Status change (with guard: cancelled → other is blocked, completed → cancelled is blocked)
   932→   932→      if (status) {
   933→   933→        const newStatus = migrateStatus(status);
   934→   934→        // Enforce same rules as generic handler
   935→   935→        if (bStatus === 'cancelled' && newStatus !== 'cancelled') {
   936→   936→          return NextResponse.json({ error: 'No se puede reactivar una reserva cancelada.' }, { status: 400 });
   937→   937→        }
   938→   938→        if (bStatus === 'completed' && newStatus === 'cancelled') {
   939→   939→          return NextResponse.json({ error: 'No se puede cancelar una reserva completada.' }, { status: 400 });
   940→   940→        }
   941→   941→        updateData.status = newStatus;
   942→   942→      }
   943→   943→
   944→   944→      // Conflict check (only if time or court or date changed)
   945→   945→      const finalCourtIds: string[] = (updateData.court_ids as string[]) || (Array.isArray(booking.court_ids) ? booking.court_ids : (booking.court_id ? [booking.court_id] : []));
   946→   946→      const finalDate = (updateData.date as string) || (booking.date as string);
   947→   947→      const finalStart = (updateData.start_time as string) || (booking.start_time as string);
   948→   948→      const finalEnd = (updateData.end_time as string) || (booking.end_time as string);
   949→   949→
   950→   950→      if (finalCourtIds.length > 0 && finalDate && finalStart && finalEnd) {
   951→   951→        const conflictQueries = finalCourtIds.map(cid => getBookings({ courtId: cid, date: finalDate }));
   952→   952→        const conflictResults = await Promise.all(conflictQueries);
   953→   953→        const allConflicting = conflictResults.flat();
   954→   954→        const conflicts = allConflicting.filter((ob: Record<string, unknown>) => {
   955→   955→          const obId = ob.id as string;
   956→   956→          if (obId === id) return false;
   957→   957→          if (migrateStatus(ob.status || '') === 'cancelled') return false;
   958→   958→          const obCourtIds: string[] = Array.isArray(ob.court_ids) ? ob.court_ids : (ob.court_id ? [ob.court_id] : []);
   959→   959→          if (!obCourtIds.some((cid: string) => finalCourtIds.includes(cid))) return false;
   960→   960→          return (ob.start_time as string) < finalEnd && (ob.end_time as string) > finalStart;
   961→   961→        });
   962→   962→        if (conflicts.length > 0) {
   963→   963→          const c = conflicts[0];
   964→   964→          return NextResponse.json({
   965→   965→            error: 'Hay una reserva que ocupa ese horario.',
   966→   966→            detail: `Conflicto con reserva: ${(c.start_time as string)} - ${(c.end_time as string)}`,
   967→   967→          }, { status: 409 });
   968→   968→        }
   969→   969→      }
   970→   970→
   971→   971→      // Skip the generic advance payment record creation below — we handle it here
   972→   972→      // (prevent double payment records)
   973→   973→    }
   974→   974→
   975→   975→    // B11 FIX: Track if this is a cancellation to skip advance payment record
   976→   976→    const isCancelling = status === 'cancelled' || (status && migrateStatus(status) === 'cancelled');
   977→   977→
   978→   978→    // FIX Bug #9: Track warnings to return to frontend
   979→   979→    const warningFlags: string[] = [];
   980→   980→
   981→   981→    // When cancelling a booking with advance, handle retained/refunded advance
   982→   982→    // Only create retained advance records when an admin explicitly provides advanceAction
   983→   983→    // (user-side cancellations should NOT auto-create records — the admin decides later)
   984→   984→    const isAdminCancellation = authUser.role === 'admin' || authUser.role === 'super_admin';
   985→   985→    if (isCancelling) {
   986→   986→      // B6 FIX: Fetch booking once, reuse for both checks
   987→   987→      let bookingForCancel: Record<string, unknown> | null = null;
   988→   988→      try {
   989→   989→        bookingForCancel = await getBookingById(id);
   990→   990→      } catch { /* non-blocking */ }
   991→   991→
   992→   992→      // Prevent cancelling completed bookings
   993→   993→      if (bookingForCancel && (bookingForCancel.status === 'completed' || migrateStatus(bookingForCancel.status as string || '') === 'completed')) {
   994→   994→        return NextResponse.json({ error: 'No se puede cancelar una reserva completada. Contacta soporte si es necesario.' }, { status: 400 });
   995→   995→      }
   996→   996→
   997→   997→      if (isAdminCancellation) {
   998→   998→        try {
   999→   999→          if (bookingForCancel && ((bookingForCancel.advance_amount as number) || 0) > 0) {
  1000→  1000→            // FIX Bug A: Only create retained advance when admin explicitly chose an action

  1002→  1002→            // incorrectly defaulting to 'refunded'

  1004→  1004→              console.warn('[BOOKINGS] Admin cancellation with advance but no advanceAction — skipping retained advance creation for booking', id);
  1005→  1005→              warningFlags.push('no_advance_action');
  1006→  1006→            } else {
  1007→  1007→            const advAmount = (bookingForCancel.advance_amount as number) || 0;
  1008→  1008→            const isRetained = advanceAction === 'retain';
  1009→  1009→            const statusAdvance = isRetained ? 'retained' : 'refunded';
  1010→  1010→
  1011→  1011→            // Check for duplicate retained advance
  1012→  1012→            const { getRetainedAdvances } = await import('@/lib/db');
  1013→  1013→            const existing = await getRetainedAdvances({ bookingId: id });
  1014→  1014→            if (existing.length > 0) {

  1016→  1016→            } else {

  1041→  1041→

  1043→  1043→              try {

  1109→  1109→    await updateBooking(id, updateData);
  1110→  1110→
  1111→  1111→    // FIX Bug #9: Return warnings if any

  1116→  1116→  } catch (error) {

  1118→  1118→    const msg = error instanceof Error ? error.message : 'Unknown error';

  1120→  1120→  }
