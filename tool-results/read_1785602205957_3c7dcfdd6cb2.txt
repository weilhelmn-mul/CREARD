    60→    60→function formatDateISO(date: Date): string {
    61→    61→  const y = date.getFullYear()
    62→    62→  const m = String(date.getMonth() + 1).padStart(2, '0')
    63→    63→  const d = String(date.getDate()).padStart(2, '0')
    64→    64→  return `${y}-${m}-${d}`
    65→    65→}
    66→    66→
    67→    67→function getNext14Days(): Date[] {
    68→    68→  const days: Date[] = []
    69→    69→  const now = new Date()
    70→    70→  for (let i = 0; i < 14; i++) {
    71→    71→    days.push(new Date(now.getFullYear(), now.getMonth(), now.getDate() + i))
    72→    72→  }
    73→    73→  return days
    74→    74→}
    75→    75→
    76→    76→function isToday(date: Date): boolean {
    77→    77→  const t = new Date()
    78→    78→  return date.getFullYear() === t.getFullYear() &&
    79→    79→    date.getMonth() === t.getMonth() &&
    80→    80→    date.getDate() === t.getDate()
    81→    81→}
    82→    82→
    83→    83→/** Returns the next full hour boundary. E.g. if now is 14:35, returns 15 */
    84→    84→function getNextFullHour(): number {
    85→    85→  const now = new Date()
    86→    86→  const h = now.getHours()
    87→    87→  const m = now.getMinutes()
    88→    88→  return m >= 1 ? h + 1 : h
    89→    89→}
    90→    90→
    91→    91→function getMinSlotHour(date: Date): number {
    92→    92→  if (isToday(date)) {
    93→    93→    return Math.max(SLOT_START, getNextFullHour())
    94→    94→  }
    95→    95→  return SLOT_START
    96→    96→}

    98→    98→function generateHourSlots(): string[] {
    99→    99→  const slots: string[] = []
   100→   100→  for (let h = SLOT_START; h < SLOT_END; h++) {
   101→   101→    slots.push(`${String(h).padStart(2, '0')}:00`)
   102→   102→  }
   103→   103→  return slots
   104→   104→}
   105→   105→
   106→   106→function formatHour(slot: string): string {
   107→   107→  const [h] = slot.split(':').map(Number)
   108→   108→  if (h === 0 || h === 12) return `${h === 0 ? 12 : 12}:00 ${h === 0 ? 'AM' : 'PM'}`
   109→   109→  if (h < 12) return `${h}:00 AM`
   110→   110→  return `${h - 12}:00 PM`
   111→   111→}

   113→   113→function getRefCode(bookingId: string): string {
   114→   114→  const hash = bookingId.slice(-8).toUpperCase()
   115→   115→  return `CRE-${hash.slice(0, 4)}-${hash.slice(4)}`
   116→   116→}
   117→   117→
   118→   118→/** Calculate price for a 1-hour slot using a pricing schedule */
   119→   119→function calcSlotPrice(schedule: PricingScheduleItem[], slot: string): number {
   120→   120→  const hour = parseInt(slot.split(':')[0], 10)
   121→   121→  if (schedule.length > 0) {
   122→   122→    for (const tier of schedule) {
   123→   123→      if (hour >= tier.startHour && hour < tier.endHour) return tier.pricePerHour
   124→   124→    }
   125→   125→  }
   126→   126→  return 0
   127→   127→}
