   193→/** Compare Firestore Timestamps (seconds) or ISO strings for sorting */
   194→function compareTimestamps(a: unknown, b: unknown): number {
   195→  const toMs = (t: unknown): number => {
   196→    if (!t) return 0
   197→    if (typeof t === 'number') return t * 1000 // Firestore seconds
   198→    if (typeof t === 'object' && t !== null && '_seconds' in (t as Record<string, unknown>)) return (t as Record<string, unknown>)._seconds as number * 1000
   199→    if (typeof t === 'string') return new Date(t).getTime()
   200→    return 0
   201→  }
   202→  return toMs(a) - toMs(b)
   203→}

   210→function SortableSectionCard({
   211→  id,
   212→  label,
   213→  icon,
   214→  color,
   215→  isCustom,
   216→  visible,
   217→  onToggle,
   218→  onEdit,
   219→  onDelete,
   220→}: {
   221→  id: string
   222→  label: string
   223→  icon: string
   224→  color: string
   225→  isCustom: boolean
   226→  visible: boolean
   227→  onToggle: () => void
   228→  onEdit: () => void
   229→  onDelete?: () => void
   230→}) {
   231→  const { attributes, listeners, setNodeRef, transform, transition, isDragging } = useSortable({ id })
   232→  const style = { transform: CSS.Transform.toString(transform), transition }
   233→
   234→  return (
   235→    <div
   236→      ref={setNodeRef}
   237→      style={style}
   238→      className={`glass-card rounded-xl p-3 flex items-center gap-3 transition-all ${isDragging ? 'opacity-60 shadow-xl z-50' : ''} ${!visible ? 'opacity-40' : ''}`}
   239→    >
   240→      {/* Drag handle */}
   241→      <button type="button" {...attributes} {...listeners} className="cursor-grab active:cursor-grabbing text-cm-on-surface-variant/40 hover:text-cm-on-surface-variant p-1">
   242→        <span className="material-symbols-outlined text-[18px]">drag_indicator</span>
   243→      </button>
   244→
   245→      {/* Icon */}
   246→      <div className={`w-9 h-9 rounded-lg flex items-center justify-center flex-shrink-0 ${color}`}>
   247→        <span className="material-symbols-outlined text-[18px] text-white" style={{ fontVariationSettings: '"FILL" 1' }}>
   248→          {icon}
   249→        </span>
   250→      </div>
   251→
   252→      {/* Label */}
   253→      <span className="flex-1 text-sm font-semibold text-cm-on-surface truncate font-[family-name:var(--font-sora)]">
   254→        {label}
   255→      </span>
   256→
   257→      {/* Actions */}
   258→      <div className="flex items-center gap-1.5 flex-shrink-0">
   259→        <button type="button" onClick={onToggle} className={`p-1.5 rounded-lg transition-colors ${visible ? 'text-cm-primary hover:bg-cm-primary/10' : 'text-cm-on-surface-variant/40 hover:bg-white/5'}`} title={visible ? 'Ocultar' : 'Mostrar'}>

   261→        </button>
   262→        <button type="button" onClick={onEdit} className="p-1.5 rounded-lg text-cm-on-surface-variant hover:text-cm-primary hover:bg-cm-primary/10 transition-colors" title="Editar">
   263→          <span className="material-symbols-outlined text-[16px]">edit</span>
   264→        </button>
   265→        {isCustom && onDelete && (
   266→          <button type="button" onClick={onDelete} className="p-1.5 rounded-lg text-cm-on-surface-variant hover:text-red-400 hover:bg-red-500/10 transition-colors" title="Eliminar">
   267→            <span className="material-symbols-outlined text-[16px]">delete</span>
   268→          </button>
   269→        )}
   270→      </div>
   271→    </div>
   272→  )

   533→      if (!target[keys[i]] || typeof target[keys[i]] !== 'object') target[keys[i]] = {}
   534→      target = target[keys[i]] as Record<string, unknown>
   535→    }
   536→    target[keys[keys.length - 1]] = value
   537→    setEditForm(copy)

   547→    }
