   187→/** Compare Firestore Timestamps (seconds) or ISO strings for sorting */
   188→function compareTimestamps(a: unknown, b: unknown): number {
   189→  const toMs = (t: unknown): number => {
   190→    if (!t) return 0
   191→    if (typeof t === 'number') return t * 1000 // Firestore seconds
   192→    if (typeof t === 'object' && t !== null && '_seconds' in (t as Record<string, unknown>)) return (t as Record<string, unknown>)._seconds as number * 1000
   193→    if (typeof t === 'string') return new Date(t).getTime()
   194→    return 0
   195→  }
   196→  return toMs(a) - toMs(b)
   197→}

   203→/* ─── Sortable Section Card ─── */
   204→function SortableSectionCard({
   205→  id,
   206→  label,
   207→  icon,
   208→  color,
   209→  isCustom,
   210→  visible,
   211→  onToggle,
   212→  onEdit,
   213→  onDelete,
   214→}: {
   215→  id: string
   216→  label: string
   217→  icon: string
   218→  color: string
   219→  isCustom: boolean
   220→  visible: boolean
   221→  onToggle: () => void
   222→  onEdit: () => void
   223→  onDelete?: () => void
   224→}) {
   225→  const { attributes, listeners, setNodeRef, transform, transition, isDragging } = useSortable({ id })
   226→  const style = { transform: CSS.Transform.toString(transform), transition }
   227→
   228→  return (
   229→    <div
   230→      ref={setNodeRef}
   231→      style={style}
   232→      className={`glass-card rounded-xl p-3 flex items-center gap-3 transition-all ${isDragging ? 'opacity-60 shadow-xl z-50' : ''} ${!visible ? 'opacity-40' : ''}`}
   233→    >
   234→      {/* Drag handle */}
   235→      <button type="button" {...attributes} {...listeners} className="cursor-grab active:cursor-grabbing text-cm-on-surface-variant/40 hover:text-cm-on-surface-variant p-1">
   236→        <span className="material-symbols-outlined text-[18px]">drag_indicator</span>
   237→      </button>
   238→
   239→      {/* Icon */}
   240→      <div className={`w-9 h-9 rounded-lg flex items-center justify-center flex-shrink-0 ${color}`}>
   241→        <span className="material-symbols-outlined text-[18px] text-white" style={{ fontVariationSettings: '"FILL" 1' }}>
   242→          {icon}
   243→        </span>
   244→      </div>
   245→
   246→      {/* Label */}
   247→      <span className="flex-1 text-sm font-semibold text-cm-on-surface truncate font-[family-name:var(--font-sora)]">
   248→        {label}
   249→      </span>
   250→
   251→      {/* Actions */}
   252→      <div className="flex items-center gap-1.5 flex-shrink-0">
   253→        <button type="button" onClick={onToggle} className={`p-1.5 rounded-lg transition-colors ${visible ? 'text-cm-primary hover:bg-cm-primary/10' : 'text-cm-on-surface-variant/40 hover:bg-white/5'}`} title={visible ? 'Ocultar' : 'Mostrar'}>
   254→          <span className="material-symbols-outlined text-[16px]">{visible ? 'visibility' : 'visibility_off'}</span>
   255→        </button>
   256→        <button type="button" onClick={onEdit} className="p-1.5 rounded-lg text-cm-on-surface-variant hover:text-cm-primary hover:bg-cm-primary/10 transition-colors" title="Editar">
   257→          <span className="material-symbols-outlined text-[16px]">edit</span>
   258→        </button>
   259→        {isCustom && onDelete && (
   260→          <button type="button" onClick={onDelete} className="p-1.5 rounded-lg text-cm-on-surface-variant hover:text-red-400 hover:bg-red-500/10 transition-colors" title="Eliminar">
   261→            <span className="material-symbols-outlined text-[16px]">delete</span>
   262→          </button>
   263→        )}
   264→      </div>
   265→    </div>
   266→  )
   267→}
