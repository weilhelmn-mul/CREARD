   206→/** Compare Firestore Timestamps (seconds) or ISO strings for sorting */
   207→function compareTimestamps(a: unknown, b: unknown): number {
   208→  const toMs = (t: unknown): number => {
   209→    if (!t) return 0
   210→    if (typeof t === 'number') return t * 1000 // Firestore seconds
   211→    if (typeof t === 'object' && t !== null && '_seconds' in (t as Record<string, unknown>)) return (t as Record<string, unknown>)._seconds as number * 1000
   212→    if (typeof t === 'string') return new Date(t).getTime()
   213→    return 0
   214→  }
   215→  return toMs(a) - toMs(b)
   216→}

   223→function SortableSectionCard({
   224→  id,
   225→  label,
   226→  icon,
   227→  color,
   228→  isCustom,
   229→  visible,
   230→  onToggle,
   231→  onEdit,

   233→}: {
   234→  id: string
   235→  label: string
   236→  icon: string
   237→  color: string
   238→  isCustom: boolean
   239→  visible: boolean
   240→  onToggle: () => void
   241→  onEdit: () => void
   242→  onDelete?: () => void

   244→  const { attributes, listeners, setNodeRef, transform, transition, isDragging } = useSortable({ id })

   251→      className={`glass-card rounded-xl p-3 flex items-center gap-3 transition-all ${isDragging ? 'opacity-60 shadow-xl z-50' : ''} ${!visible ? 'opacity-40' : ''}`}
   252→    >
   253→      {/* Drag handle */}
   254→      <button type="button" {...attributes} {...listeners} className="cursor-grab active:cursor-grabbing text-cm-on-surface-variant/40 hover:text-cm-on-surface-variant p-1">

   270→      {/* Actions */}

   272→        <button type="button" onClick={onToggle} className={`p-1.5 rounded-lg transition-colors ${visible ? 'text-cm-primary hover:bg-cm-primary/10' : 'text-cm-on-surface-variant/40 hover:bg-white/5'}`} title={visible ? 'Ocultar' : 'Mostrar'}>

   281→          </button>
   282→        )}
   283→      </div>

   548→    }

   550→    setEditForm(copy)

   560→    }
