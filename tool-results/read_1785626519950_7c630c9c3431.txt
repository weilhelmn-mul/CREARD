    60→function formatDateISO(date: Date): string {
    61→  const y = date.getFullYear()
    62→  const m = String(date.getMonth() + 1).padStart(2, '0')
    63→  const d = String(date.getDate()).padStart(2, '0')
    64→  return `${y}-${m}-${d}`
    65→}
    66→
    67→function getNext14Days(): Date[] {
    68→  const days: Date[] = []
    69→  const now = new Date()
    70→  for (let i = 0; i < 14; i++) {
    71→    days.push(new Date(now.getFullYear(), now.getMonth(), now.getDate() + i))
    72→  }
    73→  return days
    74→}
    75→
    76→function isToday(date: Date): boolean {
    77→  const t = new Date()
    78→  return date.getFullYear() === t.getFullYear() &&
    79→    date.getMonth() === t.getMonth() &&
    80→    date.getDate() === t.getDate()
    81→}
    82→
    83→/** Returns the next full hour boundary. E.g. if now is 14:35, returns 15 */
    84→function getNextFullHour(): number {
    85→  const now = new Date()
    86→  const h = now.getHours()
    87→  const m = now.getMinutes()
    88→  return m >= 1 ? h + 1 : h
    89→}
    90→
    91→function getMinSlotHour(date: Date): number {
    92→  if (isToday(date)) {
    93→    return Math.max(SLOT_START, getNextFullHour())
    94→  }
    95→  return SLOT_START
    96→}
    97→
    98→function generateHourSlots(): string[] {
    99→  const slots: string[] = []
   100→  for (let h = SLOT_START; h < SLOT_END; h++) {
   101→    slots.push(`${String(h).padStart(2, '0')}:00`)
   102→  }
   103→  return slots
   104→}
   105→
   106→function formatHour(slot: string): string {
   107→  const [h] = slot.split(':').map(Number)
   108→  if (h === 0 || h === 12) return `${h === 0 ? 12 : 12}:00 ${h === 0 ? 'AM' : 'PM'}`
   109→  if (h < 12) return `${h}:00 AM`
   110→  return `${h - 12}:00 PM`
   111→}
   112→
   113→function getRefCode(bookingId: string): string {
   114→  const hash = bookingId.slice(-8).toUpperCase()
   115→  return `CRE-${hash.slice(0, 4)}-${hash.slice(4)}`
   116→}
   117→
   118→/** Calculate price for a 1-hour slot using a pricing schedule */
   119→function calcSlotPrice(schedule: PricingScheduleItem[], slot: string): number {
   120→  const hour = parseInt(slot.split(':')[0], 10)
   121→  if (schedule.length > 0) {
   122→    for (const tier of schedule) {
   123→      if (hour >= tier.startHour && hour < tier.endHour) return tier.pricePerHour
   124→    }
   125→  }
   126→  return 0
   127→}
