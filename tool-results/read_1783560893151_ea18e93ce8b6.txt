    16→// Migrate old status values to the new 3-status system
    17→function migrateStatus(s: string): string {
    18→  switch (s) {
    19→    case 'confirmed':
    20→    case 'pending':
    21→    case 'partially_paid':
    22→      return 'reserved';
    23→    case 'fully_paid':
    24→    case 'completed':
    25→      return 'completed';
    26→    case 'no_show':
    27→    case 'expired':
    28→      return 'cancelled';
    29→    default:
    30→      return s; // 'reserved', 'completed', 'cancelled' pass through
    31→  }
    32→}

    34→// Transformar snake_case (Firestore) a camelCase (frontend)
    35→function toCamelBooking(b: Record<string, unknown>) {
    36→  // Primary court (backward compat)
    37→  const courtRaw = b._court as Record<string, unknown> | null;
    38→  const court = courtRaw ? {
    39→    id: courtRaw.id || courtRaw.court_id,
    40→    name: courtRaw.name || courtRaw.court_name,
    41→    sport: courtRaw.sport,
    42→    branch: (courtRaw.branch as Record<string, unknown>) || {
    43→      id: courtRaw.branch_id || 'branch-1',
    44→      name: 'CREARD',
    45→    },
    46→  } : null;
    47→
    48→  // Additional courts (multi-court bookings)
    49→  const courtsRaw = b._courts as Array<Record<string, unknown>> | null;
    50→  const courts = courtsRaw ? courtsRaw.map((cr) => ({
    51→    id: cr.id || cr.court_id,
    52→    name: cr.name || cr.court_name,
    53→    sport: cr.sport,
    54→    branch: (cr.branch as Record<string, unknown>) || { id: 'branch-1', name: 'CREARD' },
    55→  })) : [];
    56→
    57→  const userRaw = b._user as Record<string, unknown> | null;
    58→  const user = userRaw ? {
    59→    id: userRaw.id,
    60→    name: userRaw.name,
    61→    email: userRaw.email,
    62→    phone: userRaw.phone,
    63→  } : null;
    64→
    65→  // Court IDs array
    66→  const courtIds: string[] = Array.isArray(b.court_ids) ? b.court_ids : (b.court_id ? [b.court_id] : []);
    67→
    68→  // Equipment items
    69→  const equipmentItems = Array.isArray(b.equipment_items) ? b.equipment_items.map((eq: Record<string, unknown>) => ({
    70→    equipmentId: eq.equipment_id || eq.id,
    71→    name: eq.name,
    72→    quantity: eq.quantity || 1,
    73→    unitPrice: eq.unit_price || eq.price_per_rental || 0,
    74→    subtotal: (eq.quantity || 1) * (eq.unit_price || eq.price_per_rental || 0),
    75→  })) : [];
    76→
    77→  // Normalize: recalc equipment subtotal from items (more reliable than stored value)
    78→  const computedEquipSubtotal = equipmentItems.reduce((s: number, eq: { subtotal: number }) => s + eq.subtotal, 0);
    79→  const equipmentSubtotal = computedEquipSubtotal;
    80→
    81→  // Normalize court_subtotal: if stored value is wrong (old double-counting bug),
    82→  // derive it from total_price - equipment_subtotal
    83→  const totalPrice = b.total_price || 0;
    84→  const storedCourtSubtotal = b.court_subtotal || 0;
    85→  const courtSubtotal = (storedCourtSubtotal + equipmentSubtotal > totalPrice + 0.01)
    86→    ? Math.max(0, Math.round((totalPrice - equipmentSubtotal) * 100) / 100)
    87→    : storedCourtSubtotal;
    88→
    89→  // Normalize remaining_amount: ensure it equals total - advance
    90→  const advanceAmount = b.advance_amount || 0;
    91→  const storedRemaining = b.remaining_amount || 0;
    92→  const remainingAmount = (Math.abs(storedRemaining - (totalPrice - advanceAmount)) > 0.01)
    93→    ? Math.max(0, Math.round((totalPrice - advanceAmount) * 100) / 100)
    94→    : storedRemaining;
    95→
    96→  return {
    97→    id: b.id,
    98→    courtId: b.court_id,
    99→    courtIds,
   100→    userId: b.user_id,
   101→    date: typeof b.date === 'string' ? b.date : (b.date ? String(b.date) : ''),
   102→    startTime: b.start_time,
   103→    endTime: b.end_time,
   104→    totalPrice,
   105→    courtSubtotal,
   106→    equipmentSubtotal,
   107→    equipmentItems,
   108→    advanceAmount,
   109→    remainingAmount,
   110→    status: migrateStatus(b.status || 'reserved'),
   111→    slotStatus: b.slot_status,
   112→    paymentMethod: b.payment_method,
   113→    notes: b.notes,
   114→    createdAt: b.created_at,
   115→    updatedAt: b.updated_at,
   116→    recurringGroupId: b.recurring_group_id,
   117→    recurringIndex: b.recurring_index,
   118→    equipmentDelivered: b.equipment_delivered || false,
   119→    equipmentReturned: b.equipment_returned || false,
   120→    court,
   121→    courts,
   122→    user,
   123→  };
   124→}

   126→// Enrich a single booking (all courts + user), safe — never throws
   127→async function safeEnrichBooking(b: Record<string, unknown>): Promise<Record<string, unknown>> {
   128→  let court: Record<string, unknown> | null = null;
   129→  let courts: Array<Record<string, unknown>> = [];
   130→  let user: Record<string, unknown> | null = null;
   131→
   132→  // Get all court IDs (from court_ids array or single court_id)
   133→  const allCourtIds: string[] = Array.isArray(b.court_ids)
   134→    ? b.court_ids as string[]
   135→    : (b.court_id ? [b.court_id as string] : []);
   136→
   137→  // Enrich primary court (first one)
   138→  try {
   139→    if (allCourtIds.length > 0) {
   140→      const c = await getCourtById(allCourtIds[0]);
   141→      if (c) court = c as Record<string, unknown>;
   142→    }
   143→  } catch (e) {
   144→    console.warn('[BOOKINGS] Failed to load primary court for booking:', b.id, e);
   145→  }
   146→
   147→  // Enrich all courts for multi-court bookings
   148→  if (allCourtIds.length > 1) {
   149→    const enrichedCourts: Array<Record<string, unknown>> = [];
   150→    for (const cid of allCourtIds) {
   151→      try {
   152→        const c = await getCourtById(cid);
   153→        if (c) enrichedCourts.push(c as Record<string, unknown>);
   154→      } catch { /* skip failed court enrichment */ }
   155→    }
   156→    courts = enrichedCourts;
   157→  }
   158→
   159→  try {
   160→    if (b.user_id) {
   161→      const u = await getUserById(b.user_id as string);
   162→      if (u) user = u as Record<string, unknown>;
   163→    }
   164→  } catch (e) {
   165→    console.warn('[BOOKINGS] Failed to load user for booking:', b.id, e);
   166→  }
   167→
   168→  return { ...b, _court: court, _courts: courts, _user: user };
   169→}

   171→// Search bookings by userId OR userEmail, deduplicate
   172→async function searchBookingsForUser(userId: string, userEmail: string | null) {
   173→  const results = new Map<string, Record<string, unknown>>();
   174→
   175→  // Always search by userId
   176→  try {
   177→    const byId = await getBookings({ userId });
   178→    for (const b of byId) {
   179→      if (b.id && !results.has(b.id)) {
   180→        results.set(b.id, b as Record<string, unknown>);
   181→      }
   182→    }
   183→  } catch (e) {
   184→    console.error('[BOOKINGS] Error searching by userId:', userId, e);
   185→  }
   186→
   187→  // Also search by email to handle demo/Firebase ID mismatches
   188→  if (userEmail && userEmail !== userId) {
   189→    try {
   190→      const byEmail = await getBookings({ userEmail });
   191→      for (const b of byEmail) {
   192→        if (b.id && !results.has(b.id)) {
   193→          results.set(b.id, b as Record<string, unknown>);
   194→        }
   195→      }
   196→    } catch (e) {
   197→      console.error('[BOOKINGS] Error searching by userEmail:', userEmail, e);
   198→    }
   199→  }
   200→
   201→  return Array.from(results.values());
   202→}

   204→/**
   205→ * Calculate the total price for a time slot using a pricing schedule.
   206→ * Supports slots that span multiple schedules (e.g., 16:00-19:00 = 1hr morning + 2hr night).
   207→ */
   208→function calculatePriceForTimeSlot(
   209→  schedule: Array<{ label: string; startHour: number; endHour: number; pricePerHour: number }>,
   210→  startTime: string,
   211→  endTime: string,
   212→): number {
   213→  const [startH, startM] = startTime.split(':').map(Number);
   214→  const [endH, endM] = endTime.split(':').map(Number);
   215→  const startDecimal = startH + startM / 60;
   216→  const endDecimal = endH + endM / 60;
   217→
   218→  // Sort schedule by startHour
   219→  const sorted = [...schedule].sort((a, b) => a.startHour - b.startHour);
   220→
   221→  let total = 0;
   222→  let cursor = startDecimal;
   223→
   224→  for (const slot of sorted) {
   225→    const slotStart = slot.startHour;
   226→    const slotEnd = slot.endHour;
   227→
   228→    if (cursor >= slotEnd) continue; // cursor already past this slot
   229→
   230→    // Overlap start/end
   231→    const overlapStart = Math.max(cursor, slotStart);
   232→    const overlapEnd = Math.min(endDecimal, slotEnd);
   233→
   234→    if (overlapEnd > overlapStart) {
   235→      const hours = overlapEnd - overlapStart;
   236→      total += hours * slot.pricePerHour;
   237→      cursor = overlapEnd;
   238→    }
   239→  }
   240→
   241→  // If cursor didn't reach endDecimal, the remaining time is uncovered by schedule
   242→  // (fall back to 0 or caller's default)
   243→  return Math.round(total * 100) / 100;
   244→}

   246→export async function GET(request: NextRequest) {
   247→  const startTime = Date.now();
   248→  const firebaseOk = isFirebaseAvailable();
   249→
   250→  try {
   251→    const { searchParams } = new URL(request.url);
   252→    const courtId = searchParams.get('courtId');
   253→    const userId = searchParams.get('userId');
   254→    const date = searchParams.get('date');
   255→    const dateFrom = searchParams.get('dateFrom');
   256→    const dateTo = searchParams.get('dateTo');
   257→    const status = searchParams.get('status');
   258→
   259→    // Court availability check (courtId + date) is public — no auth required
   260→    if (courtId && date && !userId) {
   261→      if (!firebaseOk) {
   262→        return NextResponse.json({ error: 'Firebase no configurado', code: 'NO_FIREBASE' }, { status: 503 });
   263→      }
   264→      const bookings = await getBookings({ courtId, date });
   265→      return NextResponse.json(bookings.map(toCamelBooking));
   266→    }
   267→
   268→    // All other queries require authentication
   269→    const authResult = await requireAnyAuth(request);
   270→    if (authResult instanceof NextResponse) return authResult;
   271→    const authUser = authResult.user;
   272→
   273→    if (!firebaseOk) {
   274→      return NextResponse.json({ error: 'Firebase no configurado', code: 'NO_FIREBASE' }, { status: 503 });
   275→    }
   276→
   277→    let bookings: Record<string, unknown>[];
   278→
   279→    // Non-admin users: only their own bookings
   280→    if (authUser.role !== 'admin' && authUser.role !== 'super_admin') {
   281→      if (userId && userId !== authUser.id) {
   282→        return NextResponse.json({ error: 'No puedes ver reservas de otros usuarios.' }, { status: 403 });
   283→      }
   284→      const effectiveUserId = userId || authUser.id;
   285→      bookings = await searchBookingsForUser(effectiveUserId, authUser.email);
   286→    } else {
   287→      // Admin: fetch with optional filters (date range for performance)
   288→      bookings = (await getBookings({
   289→        courtId: courtId || undefined,
   290→        userId: userId || undefined,
   291→        date: date || undefined,
   292→        dateFrom: dateFrom || undefined,
   293→        dateTo: dateTo || undefined,
   294→        status: status || undefined,
   295→      })) as Record<string, unknown>[];
   296→      console.log(`[BOOKINGS] Firestore returned ${bookings.length} raw docs (dateFrom=${dateFrom}, dateTo=${dateTo})`);
   297→      if (bookings.length > 0) {
   298→        console.log(`[BOOKINGS] Sample raw date field: id=${bookings[0].id}, date=${JSON.stringify(bookings[0].date)}, type=${typeof bookings[0].date}`);
   299→      }
   300→    }
   301→
   302→    // Enrich each booking with cached court/user lookups (avoid N+1 Firestore reads)
   303→    const courtCache = new Map<string, Record<string, unknown> | null>();
   304→    const userCache = new Map<string, Record<string, unknown> | null>();
   305→    const getCourtCached = async (cid: string) => {
   306→      if (courtCache.has(cid)) return courtCache.get(cid)!;
   307→      try {
   308→        const c = await getCourtById(cid);
   309→        courtCache.set(cid, c as Record<string, unknown> | null);
   310→        return c as Record<string, unknown> | null;
   311→      } catch { courtCache.set(cid, null); return null; }
   312→    };
   313→    const getUserCached = async (uid: string) => {
   314→      if (userCache.has(uid)) return userCache.get(uid)!;
   315→      try {
   316→        const u = await getUserById(uid);
   317→        userCache.set(uid, u as Record<string, unknown> | null);
   318→        return u as Record<string, unknown> | null;
   319→      } catch { userCache.set(uid, null); return null; }
   320→    };
   321→
   322→    const enriched: Record<string, unknown>[] = [];
   323→    // Process in batches of 10 to avoid overwhelming Firestore
   324→    const BATCH_SIZE = 10;
   325→    for (let i = 0; i < bookings.length; i += BATCH_SIZE) {
   326→      const batch = bookings.slice(i, i + BATCH_SIZE);
   327→      const results = await Promise.allSettled(
   328→        batch.map(async (b) => {
   329→          const allCourtIds: string[] = Array.isArray(b.court_ids)
   330→            ? b.court_ids as string[] : (b.court_id ? [b.court_id as string] : []);
   331→          let court: Record<string, unknown> | null = null;
   332→          let courts: Array<Record<string, unknown>> = [];
   333→          let user: Record<string, unknown> | null = null;
   334→          try {
   335→            if (allCourtIds.length > 0) court = await getCourtCached(allCourtIds[0]);
   336→          } catch { /* skip */ }
   337→          if (allCourtIds.length > 1) {
   338→            const enrichedCourts: Array<Record<string, unknown>> = [];
   339→            for (const cid of allCourtIds) {
   340→              try {
   341→                const c = await getCourtCached(cid);
   342→                if (c) enrichedCourts.push(c);
   343→              } catch { /* skip */ }
   344→            }
   345→            courts = enrichedCourts;
   346→          }
   347→          try {
   348→            if (b.user_id) user = await getUserCached(b.user_id as string);
   349→          } catch { /* skip */ }
   350→          return toCamelBooking({ ...b, _court: court, _courts: courts, _user: user });
   351→        })
   352→      );
   353→      for (const r of results) {
   354→        if (r.status === 'fulfilled') {
   355→          enriched.push(r.value);
   356→        } else {
   357→          console.error('[BOOKINGS] Batch enrichment error:', r.reason);
   358→        }
   359→      }
   360→    }
   361→
   362→    // Default server-side sort: today first, future ascending, past descending
   363→    // Within each date: chronological time
   364→    const today = new Date().toLocaleDateString('en-CA', { timeZone: 'America/Lima' });
   365→    enriched.sort((a, b) => {
   366→      const dA = String(a.date || ''), dB = String(b.date || '');
   367→      const aIsToday = dA === today, bIsToday = dB === today;
   368→      const aFuture = dA > today, bFuture = dB > today;
   369→      const aPast = dA < today, bPast = dB < today;
   370→      // 1) Today always first
   371→      if (aIsToday && !bIsToday) return -1;
   372→      if (!aIsToday && bIsToday) return 1;
   373→      // 2) Both on same side of today
   374→      if (aFuture && bFuture) {
   375→        // Future dates: ascending (correlativo)
   376→        const dc = dA.localeCompare(dB);
   377→        if (dc !== 0) return dc;
   378→      } else if (aPast && bPast) {
   379→        // Past dates: descending (most recent past first)
   380→        const dc = dB.localeCompare(dA);
   381→        if (dc !== 0) return dc;
   382→      }
   383→      // Same date: chronological by start time
   384→      return String(a.startTime || '').localeCompare(String(b.startTime || ''));
   385→    });
   386→
   387→    console.log(`[BOOKINGS] GET: ${enriched.length} bookings returned in ${Date.now() - startTime}ms (auth=${authUser.role}, filters: courtId=${courtId||'all'}, date=${date||'all'}, status=${status||'all'})`);
   388→
   389→    return NextResponse.json(enriched);
   390→  } catch (error) {
   391→    console.error('[BOOKINGS] GET error:', error);
   392→    const msg = error instanceof Error ? error.message : 'Unknown error';
   393→    return NextResponse.json({ error: 'Error al obtener reservas', detail: msg }, { status: 500 });
   394→  }
   395→}

   397→export async function POST(request: NextRequest) {
   398→  try {
   399→    const authResult = await requireAnyAuth(request);
   400→    if (authResult instanceof NextResponse) return authResult;
   401→    const authUser = authResult.user;
   402→
   403→    if (!isFirebaseAvailable()) {
   404→      return NextResponse.json({ error: 'Firebase no configurado' }, { status: 503 });
   405→    }
   406→
   407→    const body = await request.json();
   408→    const {
   409→      courtId,
   410→      courtIds: bodyCourtIds,
   411→      userId,
   412→      date,
   413→      startTime,
   414→      endTime,
   415→      totalPrice,
   416→      advanceAmount,
   417→      remainingAmount,
   418→      status,
   419→      paymentMethod,
   420→      notes,
   421→      equipmentItems,
   422→      selectedSlots,
   423→    } = body;
   424→
   425→    // ── Resolve court IDs (support both single courtId and multi-court courtIds) ──
   426→    const allCourtIds: string[] = Array.isArray(bodyCourtIds) && bodyCourtIds.length > 0
   427→      ? bodyCourtIds
   428→      : courtId
   429→        ? [courtId]
   430→        : [];
   431→
   432→    // ── Validate payment method (restricted set, UPPERCASE) ──
   433→    const VALID_PAYMENT_METHODS = ['EFECTIVO', 'YAPE', 'PLIN'];
   434→    const normalizedPaymentMethod = VALID_PAYMENT_METHODS.includes(paymentMethod)
   435→      ? paymentMethod
   436→      : 'EFECTIVO';
   437→
   438→    if (allCourtIds.length === 0 || !userId || !date || !startTime || !endTime) {
   439→      return NextResponse.json(
   440→        { error: 'Faltan campos requeridos: courtId/courtIds, userId, date, startTime, endTime' },
   441→        { status: 400 }
   442→      );
   443→    }
   444→
   445→    // ── Minimum advance validation ──
   446→    // Admin/super_admin: NO time restrictions — can book any slot (walk-in, past, current)
   447→    // Regular users: require 30 min minimum advance
   448→    const isAdmin = authUser.role === 'admin' || authUser.role === 'super_admin';
   449→
   450→    if (!isAdmin) {
   451→      const now = new Date();
   452→      const slotDateTime = new Date(`${date}T${startTime}:00`);
   453→      const diffMs = slotDateTime.getTime() - now.getTime();
   454→      const thirtyMinMs = 30 * 60 * 1000;
   455→      if (diffMs < thirtyMinMs) {
   456→        const minsLeft = Math.floor(diffMs / 60000);
   457→        if (minsLeft < 0) {
   458→          return NextResponse.json(
   459→            { error: 'No se puede reservar un horario que ya ha pasado.' },
   460→            { status: 422 }
   461→          );
   462→        }
   463→        return NextResponse.json(
   464→          { error: `La reserva requiere al menos 30 minutos de anticipación. El horario seleccionado comienza en ${minsLeft} minuto${minsLeft === 1 ? '' : 's'}. Selecciona un horario a partir de las ${new Date(now.getTime() + thirtyMinMs).toLocaleTimeString('es-PE', { hour: '2-digit', minute: '2-digit', hour12: false })} o posterior.` },
   465→          { status: 422 }
   466→        );
   467→      }
   468→    }
   469→
   470→    // Non-admin users can only create bookings for themselves
   471→    if (!isAdmin) {
   472→      if (userId !== authUser.id) {
   473→        return NextResponse.json(
   474→          { error: 'No puedes crear reservas para otros usuarios.' },
   475→          { status: 403 }
   476→        );
   477→      }
   478→    }
   479→
   480→    // ── Check overlap for ALL courts ──
   481→    const conflicts: Array<{ courtId: string; courtName: string }> = [];
   482→    for (const cId of allCourtIds) {
   483→      const existing = await getBookings({ courtId: cId, date });
   484→      const overlapping = existing.filter(
   485→        (b) => {
   486→          const bCourtIds: string[] = Array.isArray(b.court_ids)
   487→            ? b.court_ids as string[]
   488→            : [b.court_id as string];
   489→          // Skip if this existing booking doesn't include the court we're checking
   490→          if (!bCourtIds.includes(cId)) return false;
   491→          return (
   492→            !['cancelled'].includes(migrateStatus(b.status || '')) &&
   493→            (b.start_time || '') < endTime &&
   494→            (b.end_time || '') > startTime
   495→          );
   496→        }
   497→      );
   498→      if (overlapping.length > 0) {
   499→        let courtName = cId;
   500→        try {
   501→          const c = await getCourtById(cId);
   502→          if (c?.name) courtName = c.name as string;
   503→        } catch { /* use ID */ }
   504→        conflicts.push({ courtId: cId, courtName });
   505→      }
   506→    }
   507→
   508→    if (conflicts.length > 0) {
   509→      const conflictNames = conflicts.map((c) => c.courtName).join(', ');
   510→      return NextResponse.json(
   511→        {
   512→          error: `Cancha${conflicts.length > 1 ? 's' : ''} no disponible${conflicts.length > 1 ? 's' : ''}: ${conflictNames}. Por favor selecciona otro horario o cancha.`,
   513→          conflicts,
   514→        },
   515→        { status: 409 }
   516→      );
   517→    }
   518→
   519→    // ── Calculate price ──
   520→    // Frontend sends totalPrice as grand total (court + equipment combined).
   521→    // We compute courtSubtotal and equipmentSubtotal separately for storage/display.
   522→    const eqItems = Array.isArray(equipmentItems) ? equipmentItems : [];
   523→    let equipmentSubtotal = 0;
   524→    for (const eq of eqItems) {
   525→      equipmentSubtotal += (eq.quantity || 0) * (eq.unit_price || eq.unitPrice || 0);
   526→    }
   527→
   528→    let courtPriceTotal: number;
   529→    const providedTotal = parseFloat(totalPrice) || 0;
   530→    if (providedTotal > 0) {
   531→      // Frontend already included equipment in totalPrice — extract court portion
   532→      courtPriceTotal = Math.max(0, providedTotal - equipmentSubtotal);
   533→    } else {
   534→      // No price from frontend — calculate from court rates
   535→      courtPriceTotal = 0;
   536→      for (const cId of allCourtIds) {
   537→        const court = await getCourtById(cId);
   538→        if (court) {
   539→          const schedule = court.pricing_schedule as Array<{ label: string; startHour: number; endHour: number; pricePerHour: number }> | undefined;
   540→          let courtPrice = 0;
   541→          if (schedule && schedule.length > 0) {
   542→            courtPrice = calculatePriceForTimeSlot(schedule, startTime, endTime);
   543→            if (courtPrice <= 0) courtPrice = (court.price_per_hour as number) || 0;
   544→          } else {
   545→            courtPrice = (court.price_per_hour as number) || 0;
   546→          }
   547→          courtPriceTotal += courtPrice;
   548→        }
   549→      }
   550→    }
   551→
   552→    const price = courtPriceTotal + equipmentSubtotal;
   553→
   554→    // B4 FIX: Validate advance + remaining === total consistency
   555→    const adv = parseFloat(advanceAmount) || price * 0.5;
   556→    let rem = parseFloat(remainingAmount) || price - adv;
   557→    // Force consistency: remaining must equal total - advance
   558→    rem = Math.max(0, Math.round((price - adv) * 100) / 100);
   559→    const bookingStatus = migrateStatus(status || 'reserved');
   560→
   561→    // Resolve client email for denormalized search
   562→    let clientEmail = authUser.email;
   563→    try {
   564→      const clientUser = await getUserById(userId);
   565→      if (clientUser?.email) clientEmail = clientUser.email;
   566→    } catch { /* fallback to admin email */ }
   567→
   568→    // Save to Firestore — single booking with all courts
   569→    const id = await createBooking({
   570→      court_ids: allCourtIds,
   571→      user_id: userId,
   572→      user_email: clientEmail,
   573→      date,
   574→      start_time: startTime,
   575→      end_time: endTime,
   576→      total_price: price,
   577→      court_subtotal: courtPriceTotal,
   578→      equipment_subtotal: equipmentSubtotal,
   579→      equipment_items: eqItems,
   580→      advance_amount: adv,
   581→      remaining_amount: rem,
   582→      status: bookingStatus,
   583→      slot_status: 'available',
   584→      payment_method: normalizedPaymentMethod || null,
   585→      notes: notes || null,
   586→      selected_slots: Array.isArray(selectedSlots) ? selectedSlots : [],
   587→    });
   588→
   589→    console.log(`[BOOKINGS] POST: Created booking ${id} for user=${userId} date=${date} ${startTime}-${endTime} courts=${allCourtIds.join(',')} total=${price} advance=${adv} remaining=${rem} method=${normalizedPaymentMethod} email=${clientEmail}`);
   590→
   591→    // B5 FIX: Only create payment record if advance > 0
   592→    if (adv > 0.01) {
   593→      try {
   594→        await createPayment(id, {
   595→          user_id: userId,
   596→          amount: adv,
   597→          type: 'advance',
   598→          method: normalizedPaymentMethod || 'EFECTIVO',
   599→          status: 'completed',
   600→        });
   601→      } catch (payErr) {
   602→        console.error('[BOOKINGS] Warning: could not create payment record:', payErr);
   603→      }
   604→    }
   605→
   606→    return NextResponse.json({
   607→      id,
   608→      courtId: allCourtIds[0],
   609→      courtIds: allCourtIds,
   610→      userId,
   611→      date,
   612→      startTime,
   613→      endTime,
   614→      totalPrice: price,
   615→      advanceAmount: adv,
   616→      remainingAmount: rem,
   617→      status: bookingStatus,
   618→      paymentMethod: paymentMethod || null,
   619→      success: true,
   620→    }, { status: 201 });
   621→  } catch (error: unknown) {
   622→    const err = error as { message?: string; code?: string; stack?: string };
   623→    console.error('[BOOKINGS] POST error:', err?.message || err);
   624→    console.error('[BOOKINGS] Stack:', err?.stack || 'no stack');
   625→    return NextResponse.json({
   626→      error: 'Failed to create booking',
   627→      detail: err?.message || 'Unknown error'
   628→    }, { status: 500 });
   629→  }
   630→}

   632→export async function PUT(request: NextRequest) {
   633→  try {
   634→    const authResult = await requireAnyAuth(request);
   635→    if (authResult instanceof NextResponse) return authResult;
   636→    const authUser = authResult.user;
   637→
   638→    if (!isFirebaseAvailable()) {
   639→      return NextResponse.json({ error: 'Firebase no configurado' }, { status: 503 });
   640→    }
   641→
   642→    const body = await request.json();
   643→    const { id, status, slot_status, advanceAmount: reqAdvance, remainingAmount: reqRemaining, paymentMethod: reqPaymentMethod, equipmentDelivered, equipmentReturned, advanceAction, cancelReason, endTime: reqEndTime, totalPrice: reqTotalPrice, extendTime, editTime, editBooking, startTime: reqStartTime, date: reqDate, courtIds: reqCourtIds, userId: reqUserId, notes: reqNotes, equipmentItems: reqEquipmentItems } = body;
   644→
   645→    if (!id) {
   646→      return NextResponse.json({ error: 'Booking ID is required' }, { status: 400 });
   647→    }
   648→
   649→    // Non-admin users can only cancel their own bookings
   650→    if (authUser.role !== 'admin' && authUser.role !== 'super_admin') {
   651→      if (status === 'cancelled') {
   652→        // Verify the booking belongs to the user (by both userId and email)
   653→        const bookings = await searchBookingsForUser(authUser.id, authUser.email);
   654→        const booking = bookings.find((b) => b.id === id);
   655→        if (!booking) {
   656→          return NextResponse.json(
   657→            { error: 'No puedes modificar reservas de otros usuarios.' },
   658→            { status: 403 }
   659→          );
   660→        }
   661→      } else {
   662→        return NextResponse.json(
   663→          { error: 'Solo puedes cancelar tus propias reservas.' },
   664→          { status: 403 }
   665→        );
   666→      }
   667→    }
   668→
   669→    const updateData: Record<string, unknown> = {};
   670→    // Bug fix #7: Only set status from generic handler if NOT in editBooking mode
   671→    // (editBooking handles status via its own branch to avoid double-processing)
   672→    if (status && !editBooking) updateData.status = migrateStatus(status);
   673→    if (slot_status) updateData.slot_status = slot_status;
   674→    if (typeof reqAdvance === 'number') updateData.advance_amount = reqAdvance;
   675→    if (typeof reqRemaining === 'number') updateData.remaining_amount = reqRemaining;
   676→    if (reqPaymentMethod) {
   677→      const VALID_PM = ['EFECTIVO', 'YAPE', 'PLIN'];
   678→      updateData.payment_method = VALID_PM.includes(reqPaymentMethod) ? reqPaymentMethod : 'EFECTIVO';
   679→    }
   680→    if (typeof equipmentDelivered === 'boolean') updateData.equipment_delivered = equipmentDelivered;
   681→    if (typeof equipmentReturned === 'boolean') updateData.equipment_returned = equipmentReturned;
   682→
   683→    // Extend time: update end_time and total_price, create payment record
   684→    if (extendTime && reqEndTime && reqTotalPrice) {
   685→      // Validate: new end must be after current end
   686→      const booking = await getBookingById(id);
   687→      if (booking) {
   688→        const currentEnd = booking.end_time as string || '';
   689→        if (reqEndTime <= currentEnd) {
   690→          return NextResponse.json({ error: 'La nueva hora de fin debe ser posterior a la actual' }, { status: 400 });
   691→        }
   692→        // Check for conflicts with other bookings (same court, same date, overlapping time)
   693→        const allCourtIds: string[] = Array.isArray(booking.court_ids) ? booking.court_ids : (booking.court_id ? [booking.court_id] : []);
   694→        const bookingDate = booking.date as string;
   695→        if (allCourtIds.length > 0 && bookingDate) {
   696→          // B10 FIX: Filter by date and courts instead of fetching ALL bookings
   697→          const conflictQueries = allCourtIds.map(cid => getBookings({ courtId: cid, date: bookingDate }));
   698→          const conflictResults = await Promise.all(conflictQueries);
   699→          const allConflicting = conflictResults.flat();
   700→          const conflicts = allConflicting.filter((ob: Record<string, unknown>) => {
   701→            const obId = ob.id as string;
   702→            if (obId === id) return false;
   703→            // B9 FIX: Use migrateStatus for legacy status values
   704→            if (migrateStatus(ob.status || '') === 'cancelled') return false;
   705→            if (ob.date !== bookingDate) return false;
   706→            const obCourtIds: string[] = Array.isArray(ob.court_ids) ? ob.court_ids : (ob.court_id ? [ob.court_id] : []);
   707→            if (!obCourtIds.some((cid: string) => allCourtIds.includes(cid))) return false;
   708→            return (ob.start_time as string) < reqEndTime && (ob.end_time as string) > (booking.start_time as string);
   709→          });
   710→          // Filter to only real conflicts (overlapping with the NEW extended period)
   711→          const realConflicts = conflicts.filter((ob: Record<string, unknown>) => {
   712→            return (ob.start_time as string) < reqEndTime && (ob.end_time as string) > currentEnd;
   713→          });
   714→          if (realConflicts.length > 0) {
   715→            return NextResponse.json({ error: 'Hay una reserva que ocupa ese horario. No se puede extender.' }, { status: 409 });
   716→          }
   717→        }
   718→
   719→        const extraCost = Math.max(0, (reqTotalPrice - ((booking.total_price as number) || 0)));
   720→        updateData.end_time = reqEndTime;
   721→        updateData.total_price = reqTotalPrice;
   722→
   723→        // If admin collected payment for the extension, create a payment record
   724→        if (typeof reqAdvance === 'number' && extraCost > 0.01) {
   725→          const prevAdvance = (booking.advance_amount as number) || 0;
   726→          const paymentIncrement = reqAdvance - prevAdvance;
   727→          if (paymentIncrement > 0.01) {
   728→            try {
   729→              await createPayment(id, {
   730→                user_id: booking.user_id as string,
   731→                amount: paymentIncrement,
   732→                type: 'extension',
   733→                method: (reqPaymentMethod as string) || 'EFECTIVO',
   734→                status: 'completed',
   735→              });
   736→            } catch (payErr) {
   737→              console.error('[BOOKINGS] Warning: could not create payment record for extension:', payErr);
   738→            }
   739→          }
   740→        }
   741→      }
   742→    }
   743→
   744→    // Edit booking (super_admin only): change any field
   745→    if (editBooking) {
   746→      if (authUser.role !== 'super_admin') {
   747→        return NextResponse.json({ error: 'Solo el super administrador puede editar reservas.' }, { status: 403 });
   748→      }
   749→
   750→      const booking = await getBookingById(id);
   751→      if (!booking) {
   752→        return NextResponse.json({ error: 'Reserva no encontrada.' }, { status: 404 });
   753→      }
   754→
   755→      // Don't allow editing completed or cancelled bookings (unless only changing notes/status)
   756→      const bStatus = migrateStatus(booking.status || 'reserved');
   757→      const hasTimeOrCourtChange = reqStartTime || reqEndTime || reqDate || (reqCourtIds && reqCourtIds.length > 0);
   758→      if ((bStatus === 'completed' || bStatus === 'cancelled') && hasTimeOrCourtChange) {
   759→        return NextResponse.json({ error: 'No se puede editar el horario/cancha de una reserva completada o cancelada.' }, { status: 400 });
   760→      }
   761→
   762→      // Validate time if provided
   763→      if (reqStartTime && reqEndTime) {
   764→        if (reqStartTime >= reqEndTime) {
   765→          return NextResponse.json({ error: 'La hora de inicio debe ser anterior a la hora de fin.' }, { status: 400 });
   766→        }
   767→      }
   768→
   769→      // Date & time
   770→      if (reqDate) updateData.date = reqDate;
   771→      if (reqStartTime) updateData.start_time = reqStartTime;
   772→      if (reqEndTime) updateData.end_time = reqEndTime;
   773→
   774→      // Court change
   775→      if (reqCourtIds && Array.isArray(reqCourtIds) && reqCourtIds.length > 0) {
   776→        updateData.court_ids = reqCourtIds;
   777→        updateData.court_id = reqCourtIds[0]; // backward compat
   778→      }
   779→
   780→      // Client change
   781→      if (reqUserId) {
   782→        updateData.user_id = reqUserId;
   783→      }
   784→
   785→      // Notes
   786→      if (reqNotes !== undefined) {
   787→        updateData.notes = reqNotes;
   788→      }
   789→
   790→      // Equipment items
   791→      if (reqEquipmentItems && Array.isArray(reqEquipmentItems)) {
   792→        const equipSub = reqEquipmentItems.reduce((s: number, eq: Record<string, unknown>) => s + ((eq.quantity as number) || 0) * ((eq.unit_price as number) || 0), 0);
   793→        updateData.equipment_items = reqEquipmentItems;
   794→        updateData.equipment_subtotal = Math.round(equipSub * 100) / 100;
   795→      }
   796→
   797→      // Price fields
   798→      if (typeof reqTotalPrice === 'number' && reqTotalPrice > 0) {
   799→        updateData.total_price = reqTotalPrice;
   800→        const currentAdvance = typeof reqAdvance === 'number' ? reqAdvance : (booking.advance_amount as number) || 0;
   801→        updateData.advance_amount = Math.min(currentAdvance, reqTotalPrice);
   802→        updateData.remaining_amount = Math.max(0, Math.round((reqTotalPrice - currentAdvance) * 100) / 100);
   803→        // Recalc court_subtotal
   804→        const equipSub = (updateData.equipment_subtotal as number) || (booking.equipment_subtotal as number) || 0;
   805→        updateData.court_subtotal = Math.max(0, Math.round((reqTotalPrice - equipSub) * 100) / 100);
   806→      } else if (typeof reqAdvance === 'number') {
   807→        // Only advance changed, not total
   808→        const currentTotal = (booking.total_price as number) || 0;
   809→        updateData.advance_amount = reqAdvance;
   810→        updateData.remaining_amount = Math.max(0, Math.round((currentTotal - reqAdvance) * 100) / 100);
   811→      }
   812→
   813→      // Payment method
   814→      if (reqPaymentMethod) {
   815→        const VALID_PM = ['EFECTIVO', 'YAPE', 'PLIN'];
   816→        updateData.payment_method = VALID_PM.includes(reqPaymentMethod) ? reqPaymentMethod : 'EFECTIVO';
   817→      }
   818→
   819→      // Status change (with guard: cancelled → other is blocked, completed → cancelled is blocked)
   820→      if (status) {
   821→        const newStatus = migrateStatus(status);
   822→        // Enforce same rules as generic handler
   823→        if (bStatus === 'cancelled' && newStatus !== 'cancelled') {
   824→          return NextResponse.json({ error: 'No se puede reactivar una reserva cancelada.' }, { status: 400 });
   825→        }
   826→        if (bStatus === 'completed' && newStatus === 'cancelled') {
   827→          return NextResponse.json({ error: 'No se puede cancelar una reserva completada.' }, { status: 400 });
   828→        }
   829→        updateData.status = newStatus;
   830→      }
   831→
   832→      // Conflict check (only if time or court or date changed)
   833→      const finalCourtIds: string[] = (updateData.court_ids as string[]) || (Array.isArray(booking.court_ids) ? booking.court_ids : (booking.court_id ? [booking.court_id] : []));
   834→      const finalDate = (updateData.date as string) || (booking.date as string);
   835→      const finalStart = (updateData.start_time as string) || (booking.start_time as string);
   836→      const finalEnd = (updateData.end_time as string) || (booking.end_time as string);
   837→

   839→        const conflictQueries = finalCourtIds.map(cid => getBookings({ courtId: cid, date: finalDate }));
   840→        const conflictResults = await Promise.all(conflictQueries);
   841→        const allConflicting = conflictResults.flat();

   843→          const obId = ob.id as string;
   844→          if (obId === id) return false;
   845→          if (migrateStatus(ob.status || '') === 'cancelled') return false;
   846→          const obCourtIds: string[] = Array.isArray(ob.court_ids) ? ob.court_ids : (ob.court_id ? [ob.court_id] : []);
   847→          if (!obCourtIds.some((cid: string) => finalCourtIds.includes(cid))) return false;

   849→        });
   850→        if (conflicts.length > 0) {
   851→          const c = conflicts[0];
   852→          return NextResponse.json({
   853→            error: 'Hay una reserva que ocupa ese horario.',
   854→            detail: `Conflicto con reserva: ${(c.start_time as string)} - ${(c.end_time as string)}`,
   855→          }, { status: 409 });
   856→        }
   857→      }
   858→
   859→      // Skip the generic advance payment record creation below — we handle it here
   860→      // (prevent double payment records)
   861→    }
   862→
   863→    // B11 FIX: Track if this is a cancellation to skip advance payment record
   864→    const isCancelling = status === 'cancelled' || (status && migrateStatus(status) === 'cancelled');
   865→
   866→    // When cancelling a booking with advance, handle retained/refunded advance
   867→    // Only create retained advance records when an admin explicitly provides advanceAction
   868→    // (user-side cancellations should NOT auto-create records — the admin decides later)
   869→    const isAdminCancellation = authUser.role === 'admin' || authUser.role === 'super_admin';
   870→    if (isCancelling) {
   871→      // B6 FIX: Fetch booking once, reuse for both checks
   872→      let bookingForCancel: Record<string, unknown> | null = null;
   873→      try {
   874→        bookingForCancel = await getBookingById(id);
   875→      } catch { /* non-blocking */ }
   876→
   877→      // Prevent cancelling completed bookings
   878→      if (bookingForCancel && (bookingForCancel.status === 'completed' || migrateStatus(bookingForCancel.status as string || '') === 'completed')) {
   879→        return NextResponse.json({ error: 'No se puede cancelar una reserva completada. Contacta soporte si es necesario.' }, { status: 400 });
   880→      }
   881→
   882→      if (isAdminCancellation) {
   883→      try {
   884→        if (bookingForCancel && ((bookingForCancel.advance_amount as number) || 0) > 0) {
   885→          // Bug #4: Check for duplicate retained advance
   886→          const { getRetainedAdvances } = await import('@/lib/db');
   887→          const existing = await getRetainedAdvances({ bookingId: id });
   888→          if (existing.length > 0) {
   889→            console.warn('[BOOKINGS] Retained advance already exists for booking', id, '- skipping creation');
   890→          } else {
   891→          const advAmount = (bookingForCancel.advance_amount as number) || 0;
   892→          const isRetained = advanceAction === 'retain';
   893→          const statusAdvance = isRetained ? 'retained' : 'refunded';
   894→
   895→          // Get court name for the record
   896→          let courtName = '';
   897→          try {
   898→            const cId = (bookingForCancel.court_ids as string[])?.[0] || (bookingForCancel.court_id as string) || '';
   899→            if (cId) {
   900→              const court = await getCourtById(cId);
   901→              courtName = (court?.name as string) || '';
   902→            }
   903→          } catch { /* non-critical */ }
   904→
   905→          // Get user name
   906→          let userName = '';
   907→          try {
   908→            const uId = bookingForCancel.user_id as string;
   909→            if (uId) {
   910→              const user = await getUserById(uId);
   911→              userName = (user?.name as string) || '';
   912→            }
   913→          } catch { /* non-critical */ }
   914→
   915→          await createRetainedAdvance({
   916→            booking_id: id,
   917→            user_id: (bookingForCancel.user_id as string) || '',
   918→            user_name: userName,
   919→            user_email: (bookingForCancel.user_email as string) || null,
   920→            court_name: courtName,
   921→            booking_date: (bookingForCancel.date as string) || '',
   922→            amount: advAmount,
   923→            original_total: (bookingForCancel.total_price as number) || 0,
   924→            payment_method: (bookingForCancel.payment_method as string) || 'EFECTIVO',
   925→            reason: cancelReason || 'Cancelación de reserva',
   926→            status: statusAdvance,
   927→          });
   928→
   929→          // Create a payment record for the refund/retention
   930→          try {
   931→            await createPayment(id, {
   932→              user_id: (bookingForCancel.user_id as string) || '',
   933→              amount: advAmount,
   934→              type: isRetained ? 'retained' : 'refund',
   935→              method: (bookingForCancel.payment_method as string) || 'EFECTIVO',
   936→              status: 'completed',
   937→            });
   938→          } catch (payErr) {
   939→            console.error('[BOOKINGS] Warning: could not create payment record for cancellation:', payErr);
   940→          }
   941→          }
   942→        }
   943→      } catch (err) {
   944→        console.error('[BOOKINGS] Warning: could not process retained advance on cancellation:', err);
   945→      }
   946→      } // end isAdminCancellation
   947→    }
   948→
   949→    // B11 FIX: Skip advance payment record when cancelling or editBooking (handled above / no record needed)
   950→    if (typeof reqAdvance === 'number' && typeof reqRemaining === 'number' && !isCancelling && !editBooking) {
   951→      try {
   952→        const booking = await getBookingById(id);
   953→        if (booking) {
   954→          const prevAdvance = (booking.advance_amount as number) || 0;
   955→          const paymentIncrement = reqAdvance - prevAdvance;
   956→          if (paymentIncrement > 0.01) {
   957→            try {
   958→              await createPayment(id, {
   959→                user_id: booking.user_id as string,
   960→                amount: paymentIncrement,
   961→                type: reqRemaining <= 0 ? 'full_payment' : 'advance',
   962→                method: (updateData.payment_method as string) || 'EFECTIVO',
   963→                status: 'completed',
   964→              });
   965→            } catch (payErr) {
   966→              console.error('[BOOKINGS] Warning: could not create payment record for advance update:', payErr);
   967→            }
   968→          }
   969→        }
   970→      } catch { /* non-critical: proceed with booking update */ }
   971→    }
   972→
   973→    await updateBooking(id, updateData);
   974→
   975→    return NextResponse.json({ success: true });
   976→  } catch (error) {
   977→    console.error('[BOOKINGS] PUT error:', error);
   978→    const msg = error instanceof Error ? error.message : 'Unknown error';
   979→    return NextResponse.json({ error: 'Error al actualizar reserva', detail: msg }, { status: 500 });
   980→  }

   989→    if (!isFirebaseAvailable()) {
   990→      return NextResponse.json({ error: 'Firebase no configurado' }, { status: 503 });

   992→
   993→    const { searchParams } = new URL(request.url);
   994→    const id = searchParams.get('id');
   995→
   996→    if (!id) {
   997→      return NextResponse.json({ error: 'Booking ID is required' }, { status: 400 });
   998→    }
   999→
  1000→    // Delete the booking document and its payments subcollection

  1002→    const db = await getAdminDb();

  1004→    // Delete payments subcollection

  1007→      for (const doc of paymentsSnapshot.docs) {
