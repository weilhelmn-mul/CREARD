     7→     7→// Helper: add auth headers to fetch calls
     8→     8→function authFetch(url: string, options: RequestInit = {}): Promise<Response> {
     9→     9→  const headers = new Headers(options.headers || {})
    10→    10→  const authHeaders = getAuthHeaders()
    11→    11→  for (const [key, value] of Object.entries(authHeaders)) {
    12→    12→    if (value) headers.set(key, value)
    13→    13→  }
    14→    14→  if (!headers.has('Content-Type')) {
    15→    15→    headers.set('Content-Type', 'application/json')
    16→    16→  }
    17→    17→  return fetch(url, { ...options, headers })
    18→    18→}
