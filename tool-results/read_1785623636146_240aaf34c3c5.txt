   200→/** Compare Firestore Timestamps (seconds) or ISO strings for sorting */
   201→function compareTimestamps(a: unknown, b: unknown): number {
   202→  const toMs = (t: unknown): number => {
   203→    if (!t) return 0
   204→    if (typeof t === 'number') return t * 1000 // Firestore seconds
   205→    if (typeof t === 'object' && t !== null && '_seconds' in (t as Record<string, unknown>)) return (t as Record<string, unknown>)._seconds as number * 1000
   206→    if (typeof t === 'string') return new Date(t).getTime()
   207→    return 0
   208→  }
   209→  return toMs(a) - toMs(b)
   210→}

   217→function SortableSectionCard({
   218→  id,
   219→  label,
   220→  icon,
   221→  color,
   222→  isCustom,
   223→  visible,
   224→  onToggle,
   225→  onEdit,
   226→  onDelete,
   227→}: {
   228→  id: string
   229→  label: string
   230→  icon: string
   231→  color: string
   232→  isCustom: boolean
   233→  visible: boolean
   234→  onToggle: () => void
   235→  onEdit: () => void
   236→  onDelete?: () => void
   237→}) {
   238→  const { attributes, listeners, setNodeRef, transform, transition, isDragging } = useSortable({ id })
   239→  const style = { transform: CSS.Transform.toString(transform), transition }
   240→
   241→  return (
   242→    <div
   243→      ref={setNodeRef}
   244→      style={style}
   245→      className={`glass-card rounded-xl p-3 flex items-center gap-3 transition-all ${isDragging ? 'opacity-60 shadow-xl z-50' : ''} ${!visible ? 'opacity-40' : ''}`}
   246→    >
   247→      {/* Drag handle */}
   248→      <button type="button" {...attributes} {...listeners} className="cursor-grab active:cursor-grabbing text-cm-on-surface-variant/40 hover:text-cm-on-surface-variant p-1">
   249→        <span className="material-symbols-outlined text-[18px]">drag_indicator</span>
   250→      </button>
   251→
   252→      {/* Icon */}
   253→      <div className={`w-9 h-9 rounded-lg flex items-center justify-center flex-shrink-0 ${color}`}>
   254→        <span className="material-symbols-outlined text-[18px] text-white" style={{ fontVariationSettings: '"FILL" 1' }}>
   255→          {icon}
   256→        </span>
   257→      </div>
   258→
   259→      {/* Label */}
   260→      <span className="flex-1 text-sm font-semibold text-cm-on-surface truncate font-[family-name:var(--font-sora)]">
   261→        {label}
   262→      </span>
   263→
   264→      {/* Actions */}
   265→      <div className="flex items-center gap-1.5 flex-shrink-0">
   266→        <button type="button" onClick={onToggle} className={`p-1.5 rounded-lg transition-colors ${visible ? 'text-cm-primary hover:bg-cm-primary/10' : 'text-cm-on-surface-variant/40 hover:bg-white/5'}`} title={visible ? 'Ocultar' : 'Mostrar'}>
   267→          <span className="material-symbols-outlined text-[16px]">{visible ? 'visibility' : 'visibility_off'}</span>
   268→        </button>
   269→        <button type="button" onClick={onEdit} className="p-1.5 rounded-lg text-cm-on-surface-variant hover:text-cm-primary hover:bg-cm-primary/10 transition-colors" title="Editar">
   270→          <span className="material-symbols-outlined text-[16px]">edit</span>
   271→        </button>
   272→        {isCustom && onDelete && (
   273→          <button type="button" onClick={onDelete} className="p-1.5 rounded-lg text-cm-on-surface-variant hover:text-red-400 hover:bg-red-500/10 transition-colors" title="Eliminar">
   274→            <span className="material-symbols-outlined text-[16px]">delete</span>
   275→          </button>
   276→        )}
   277→      </div>
   278→    </div>
   279→  )

   541→      target = target[keys[i]] as Record<string, unknown>
   542→    }

   554→    }
