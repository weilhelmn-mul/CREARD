   194→function parseLocalDate(dateStr: string): Date {
   195→  const parts = dateStr.split('-')
   196→  if (parts.length === 3) {
   197→    return new Date(parseInt(parts[0], 10), parseInt(parts[1], 10) - 1, parseInt(parts[2], 10))
   198→  }
   199→  return new Date(dateStr + 'T00:00:00')
   200→}
   201→
   202→const fmtDate = (dateStr: string) => {
   203→  const d = parseLocalDate(dateStr)
   204→  return d.toLocaleDateString('es-PE', { day: 'numeric', month: 'short', year: 'numeric' })
   205→}
   206→
   207→const fmtDateTime = (dateStr: string) => {
   208→  const d = new Date(dateStr)
   209→  return d.toLocaleDateString('es-PE', {
   210→    day: 'numeric',
   211→    month: 'short',
   212→    year: 'numeric',
   213→    hour: '2-digit',
   214→    minute: '2-digit',
   215→  })
   216→}
   217→
   218→function daysRemaining(deadlineDate: string): number {
   219→  const deadline = new Date(deadlineDate)
   220→  const now = new Date()
   221→  const diff = deadline.getTime() - now.getTime()
   222→  return Math.ceil(diff / (1000 * 60 * 60 * 24))
   223→}
