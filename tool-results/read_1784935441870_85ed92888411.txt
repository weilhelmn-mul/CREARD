   150→   150→
   151→   151→/** Returns the latest hour that is still within the 30-min restriction.
   152→   152→ *  For example, if it's 14:35, hours 14 and earlier are too soon.
   153→   153→ *  Returns -1 if no hours are restricted (i.e. more than 30 min until SLOT_START). */
   154→   154→function getRestrictedHour(): number {
   155→   155→  const now = new Date()
   156→   156→  const curH = now.getHours()
   157→   157→  const curM = now.getMinutes()
   158→   158→  // If current minute >= 30, then the current hour + 1 is still within 30 min
   159→   159→  // e.g. 14:45 → 15:00 is in 15 min → 15 is restricted too
   160→   160→  // e.g. 14:20 → 15:00 is in 40 min → 15 is NOT restricted, but 14 IS restricted
   161→   161→  if (curM >= 30) {
   162→   162→    return curH + 1
   163→   163→  }
   164→   164→  return curH
   165→   165→}

   167→   167→function getAdminSlotInfo(
   168→   168→  hour: number,
   169→   169→  bookings: BookingInfo[],
   170→   170→  isTodayDate: boolean,
   171→   171→  currentHour: number
   172→   172→): SlotInfo {
   173→   173→  const booking = bookings.find((b) => {
   174→   174→    const sh = parseInt(b.startTime.split(':')[0], 10)
   175→   175→    const eh = parseInt(b.endTime.split(':')[0], 10)
   176→   176→    return hour >= sh && hour < eh
   177→   177→  })
   178→   178→
   179→   179→  if (!booking) {
   180→   180→    // Admin: NO time restrictions — all unbooked slots are available (including past hours)
   181→   181→    // This allows admin to register walk-in bookings for in-progress or past slots
   182→   182→    return { status: 'available', booking: null }
   183→   183→  }
   184→   184→
   185→   185→  if (booking.status === 'cancelled') return { status: 'blocked', booking }
   186→   186→  if (booking.status === 'completed') return { status: 'completed', booking }
   187→   187→
   188→   188→  // Active: reserved
   189→   189→  if (isTodayDate && hour === currentHour) return { status: 'in_play', booking }
   190→   190→  return { status: 'reserved', booking }
   191→   191→}

   247→   247→function normalizeSchedule(schedule: Court['pricingSchedule']): PricingScheduleItem[] {
   248→   248→  if (!schedule) return []
   249→   249→  if (Array.isArray(schedule)) return schedule
   250→   250→  // Legacy { morning, night } object format
   251→   251→  const { morning, night } = schedule as PricingSchedule
   252→   252→  return [
   253→   253→    { label: 'Mañana', startHour: morning.startHour, endHour: morning.endHour, pricePerHour: morning.price },
   254→   254→    { label: 'Noche', startHour: night.startHour, endHour: night.endHour, pricePerHour: night.price },
   255→   255→  ]
   256→   256→}
   257→   257→
   258→   258→function getMinPrice(court: Court): number {
   259→   259→  const items = normalizeSchedule(court.pricingSchedule)
   260→   260→  if (items.length > 0) {
   261→   261→    return Math.min(...items.map(s => s.pricePerHour))
   262→   262→  }
   263→   263→  return court.pricePerHour
   264→   264→}
   265→   265→
   266→   266→function getPriceForHour(court: Court, hour: number): number {
   267→   267→  const items = normalizeSchedule(court.pricingSchedule)
   268→   268→  for (const slot of items) {
   269→   269→    if (hour >= slot.startHour && hour < slot.endHour) return slot.pricePerHour
   270→   270→  }
   271→   271→  return court.pricePerHour
   272→   272→}
