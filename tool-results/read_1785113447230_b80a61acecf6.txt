    46→// --- Date List Generator ---
    47→function generateDateList(count: number = 14) {
    48→  const days = ['Dom', 'Lun', 'Mar', 'Mié', 'Jue', 'Vie', 'Sáb']
    49→  const months = ['Ene', 'Feb', 'Mar', 'Abr', 'May', 'Jun', 'Jul', 'Ago', 'Sep', 'Oct', 'Nov', 'Dic']
    50→  const dates: { date: Date; label: string; dayName: string; isToday: boolean }[] = []
    51→
    52→  for (let i = 0; i < count; i++) {
    53→    const d = new Date()
    54→    d.setDate(d.getDate() + i)
    55→    dates.push({
    56→      date: d,
    57→      label: `${d.getDate()} ${months[d.getMonth()]}`,
    58→      dayName: i === 0 ? 'Hoy' : days[d.getDay()],
    59→      isToday: i === 0,
    60→    })
    61→  }
    62→  return dates
    63→}
