    61→function formatDateISO(date: Date): string {
    62→  const y = date.getFullYear()
    63→  const m = String(date.getMonth() + 1).padStart(2, '0')
    64→  const d = String(date.getDate()).padStart(2, '0')
    65→  return `${y}-${m}-${d}`
    66→}
    67→
    68→function getNext14Days(): Date[] {
    69→  const days: Date[] = []
    70→  const now = new Date()
    71→  for (let i = 0; i < 14; i++) {
    72→    days.push(new Date(now.getFullYear(), now.getMonth(), now.getDate() + i))
    73→  }
    74→  return days
    75→}
    76→
    77→function isToday(date: Date): boolean {
    78→  const t = new Date()
    79→  return date.getFullYear() === t.getFullYear() &&
    80→    date.getMonth() === t.getMonth() &&
    81→    date.getDate() === t.getDate()
    82→}
    83→
    84→/** Returns the next full hour boundary. E.g. if now is 14:35, returns 15 */
    85→function getNextFullHour(): number {
    86→  const now = new Date()
    87→  const h = now.getHours()
    88→  const m = now.getMinutes()
    89→  return m >= 1 ? h + 1 : h
    90→}
    91→
    92→function getMinSlotHour(date: Date): number {
    93→  if (isToday(date)) {
    94→    return Math.max(SLOT_START, getNextFullHour())
    95→  }
    96→  return SLOT_START
    97→}
    98→
    99→function generateHourSlots(): string[] {
   100→  const slots: string[] = []
   101→  for (let h = SLOT_START; h < SLOT_END; h++) {
   102→    slots.push(`${String(h).padStart(2, '0')}:00`)
   103→  }
   104→  return slots
   105→}
   106→
   107→function formatHour(slot: string): string {
   108→  const [h] = slot.split(':').map(Number)
   109→  if (h === 0 || h === 12) return `${h === 0 ? 12 : 12}:00 ${h === 0 ? 'AM' : 'PM'}`
   110→  if (h < 12) return `${h}:00 AM`
   111→  return `${h - 12}:00 PM`
   112→}
   113→
   114→function getRefCode(bookingId: string): string {
   115→  const hash = bookingId.slice(-8).toUpperCase()
   116→  return `CRE-${hash.slice(0, 4)}-${hash.slice(4)}`
   117→}
   118→
   119→/** Calculate price for a 1-hour slot using a pricing schedule */
   120→function calcSlotPrice(schedule: PricingScheduleItem[], slot: string): number {
   121→  const hour = parseInt(slot.split(':')[0], 10)
   122→  if (schedule.length > 0) {
   123→    for (const tier of schedule) {
   124→      if (hour >= tier.startHour && hour < tier.endHour) return tier.pricePerHour
   125→    }
   126→  }
   127→  return 0
   128→}
