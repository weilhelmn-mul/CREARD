    88→    88→// Parse date string as local date (avoids UTC timezone issues for Peru, UTC-5)
    89→    89→function parseLocalDate(dateStr: string): Date {
    90→    90→  const parts = dateStr.split('-')
    91→    91→  if (parts.length === 3) {
    92→    92→    return new Date(parseInt(parts[0], 10), parseInt(parts[1], 10) - 1, parseInt(parts[2], 10))
    93→    93→  }
    94→    94→  return new Date(dateStr + 'T00:00:00')
    95→    95→}
