    17→    17→function migrateStatus(s: string): string {
    18→    18→  switch (s) {
    19→    19→    case 'confirmed':
    20→    20→    case 'pending':
    21→    21→    case 'partially_paid':
    22→    22→      return 'reserved';
    23→    23→    case 'fully_paid':
    24→    24→    case 'completed':
    25→    25→      return 'completed';
    26→    26→    case 'no_show':
    27→    27→    case 'expired':
    28→    28→      return 'cancelled';
    29→    29→    default:
    30→    30→      return s; // 'reserved', 'completed', 'cancelled' pass through
    31→    31→  }

    34→    34→// Transformar snake_case (Firestore) a camelCase (frontend)
    35→    35→function toCamelBooking(b: Record<string, unknown>) {
    36→    36→  // Primary court (backward compat)
    37→    37→  const courtRaw = b._court as Record<string, unknown> | null;
    38→    38→  const court = courtRaw ? {
    39→    39→    id: courtRaw.id || courtRaw.court_id,
    40→    40→    name: courtRaw.name || courtRaw.court_name,
    41→    41→    sport: courtRaw.sport,
    42→    42→    branch: (courtRaw.branch as Record<string, unknown>) || {
    43→    43→      id: courtRaw.branch_id || 'branch-1',
    44→    44→      name: 'CREARD',
    45→    45→    },
    46→    46→  } : null;
    47→    47→
    48→    48→  // Additional courts (multi-court bookings)
    49→    49→  const courtsRaw = b._courts as Array<Record<string, unknown>> | null;
    50→    50→  const courts = courtsRaw ? courtsRaw.map((cr) => ({
    51→    51→    id: cr.id || cr.court_id,
    52→    52→    name: cr.name || cr.court_name,
    53→    53→    sport: cr.sport,
    54→    54→    branch: (cr.branch as Record<string, unknown>) || { id: 'branch-1', name: 'CREARD' },
    55→    55→  })) : [];
    56→    56→
    57→    57→  const userRaw = b._user as Record<string, unknown> | null;
    58→    58→  const user = userRaw ? {
    59→    59→    id: userRaw.id,
    60→    60→    name: userRaw.name,
    61→    61→    email: userRaw.email,
    62→    62→    phone: userRaw.phone,
    63→    63→  } : null;
    64→    64→
    65→    65→  // Court IDs array
    66→    66→  const courtIds: string[] = Array.isArray(b.court_ids) ? b.court_ids : (b.court_id ? [b.court_id] : []);
    67→    67→
    68→    68→  // Equipment items
    69→    69→  const equipmentItems = Array.isArray(b.equipment_items) ? b.equipment_items.map((eq: Record<string, unknown>) => ({
    70→    70→    equipmentId: eq.equipment_id || eq.id,
    71→    71→    name: eq.name,
    72→    72→    quantity: eq.quantity || 1,
    73→    73→    unitPrice: eq.unit_price || eq.price_per_rental || 0,
    74→    74→    subtotal: (eq.quantity || 1) * (eq.unit_price || eq.price_per_rental || 0),
    75→    75→  })) : [];
    76→    76→
    77→    77→  // Normalize: recalc equipment subtotal from items (more reliable than stored value)
    78→    78→  const computedEquipSubtotal = equipmentItems.reduce((s: number, eq: { subtotal: number }) => s + eq.subtotal, 0);
    79→    79→  const equipmentSubtotal = computedEquipSubtotal;
    80→    80→
    81→    81→  // Normalize court_subtotal: if stored value is wrong (old double-counting bug),
    82→    82→  // derive it from total_price - equipment_subtotal
    83→    83→  const totalPrice = b.total_price || 0;
    84→    84→  const storedCourtSubtotal = b.court_subtotal || 0;
    85→    85→  const courtSubtotal = (storedCourtSubtotal + equipmentSubtotal > totalPrice + 0.01)
    86→    86→    ? Math.max(0, Math.round((totalPrice - equipmentSubtotal) * 100) / 100)
    87→    87→    : storedCourtSubtotal;
    88→    88→
    89→    89→  // Normalize remaining_amount: ensure it equals total - advance
    90→    90→  const advanceAmount = b.advance_amount || 0;
    91→    91→  const storedRemaining = b.remaining_amount || 0;
    92→    92→  const remainingAmount = (Math.abs(storedRemaining - (totalPrice - advanceAmount)) > 0.01)
    93→    93→    ? Math.max(0, Math.round((totalPrice - advanceAmount) * 100) / 100)
    94→    94→    : storedRemaining;
    95→    95→
    96→    96→  return {
    97→    97→    id: b.id,
    98→    98→    courtId: b.court_id,
    99→    99→    courtIds,
   100→   100→    userId: b.user_id,
   101→   101→    date: typeof b.date === 'string' ? b.date : (b.date ? String(b.date) : ''),
   102→   102→    startTime: b.start_time,
   103→   103→    endTime: b.end_time,
   104→   104→    totalPrice,
   105→   105→    courtSubtotal,
   106→   106→    equipmentSubtotal,
   107→   107→    equipmentItems,
   108→   108→    advanceAmount,
   109→   109→    remainingAmount,
   110→   110→    status: migrateStatus(b.status || 'reserved'),
   111→   111→    slotStatus: b.slot_status,
   112→   112→    paymentMethod: b.payment_method,
   113→   113→    notes: b.notes,
   114→   114→    createdAt: b.created_at,
   115→   115→    updatedAt: b.updated_at,
   116→   116→    recurringGroupId: b.recurring_group_id,
   117→   117→    recurringIndex: b.recurring_index,
   118→   118→    equipmentDelivered: b.equipment_delivered || false,
   119→   119→    equipmentReturned: b.equipment_returned || false,
   120→   120→    court,
   121→   121→    courts,
   122→   122→    user,
   123→   123→  };
   124→   124→}

   126→   126→// Enrich a single booking (all courts + user), safe — never throws
   127→   127→async function safeEnrichBooking(b: Record<string, unknown>): Promise<Record<string, unknown>> {
   128→   128→  let court: Record<string, unknown> | null = null;
   129→   129→  let courts: Array<Record<string, unknown>> = [];
   130→   130→  let user: Record<string, unknown> | null = null;
   131→   131→
   132→   132→  // Get all court IDs (from court_ids array or single court_id)
   133→   133→  const allCourtIds: string[] = Array.isArray(b.court_ids)
   134→   134→    ? b.court_ids as string[]
   135→   135→    : (b.court_id ? [b.court_id as string] : []);
   136→   136→
   137→   137→  // Enrich primary court (first one)
   138→   138→  try {
   139→   139→    if (allCourtIds.length > 0) {
   140→   140→      const c = await getCourtById(allCourtIds[0]);
   141→   141→      if (c) court = c as Record<string, unknown>;
   142→   142→    }
   143→   143→  } catch (e) {
   144→   144→    console.warn('[BOOKINGS] Failed to load primary court for booking:', b.id, e);
   145→   145→  }
   146→   146→
   147→   147→  // Enrich all courts for multi-court bookings
   148→   148→  if (allCourtIds.length > 1) {
   149→   149→    const enrichedCourts: Array<Record<string, unknown>> = [];
   150→   150→    for (const cid of allCourtIds) {
   151→   151→      try {
   152→   152→        const c = await getCourtById(cid);
   153→   153→        if (c) enrichedCourts.push(c as Record<string, unknown>);
   154→   154→      } catch { /* skip failed court enrichment */ }
   155→   155→    }
   156→   156→    courts = enrichedCourts;
   157→   157→  }
   158→   158→
   159→   159→  try {
   160→   160→    if (b.user_id) {
   161→   161→      const u = await getUserById(b.user_id as string);
   162→   162→      if (u) user = u as Record<string, unknown>;
   163→   163→    }
   164→   164→  } catch (e) {
   165→   165→    console.warn('[BOOKINGS] Failed to load user for booking:', b.id, e);
   166→   166→  }
   167→   167→
   168→   168→  return { ...b, _court: court, _courts: courts, _user: user };
   169→   169→}

   191→   191→      for (const b of byEmail) {
   192→   192→        if (b.id && !results.has(b.id)) {
   193→   193→          results.set(b.id, b as Record<string, unknown>);

   204→   204→/**
   205→   205→ * Calculate the total price for a time slot using a pricing schedule.
   206→   206→ * Supports slots that span multiple schedules (e.g., 16:00-19:00 = 1hr morning + 2hr night).
   207→   207→ */
   208→   208→function calculatePriceForTimeSlot(
   209→   209→  schedule: Array<{ label: string; startHour: number; endHour: number; pricePerHour: number }>,
   210→   210→  startTime: string,
   211→   211→  endTime: string,
   212→   212→): number {
   213→   213→  const [startH, startM] = startTime.split(':').map(Number);
   214→   214→  const [endH, endM] = endTime.split(':').map(Number);
   215→   215→  const startDecimal = startH + startM / 60;
   216→   216→  const endDecimal = endH + endM / 60;
   217→   217→
   218→   218→  // Sort schedule by startHour
   219→   219→  const sorted = [...schedule].sort((a, b) => a.startHour - b.startHour);
   220→   220→
   221→   221→  let total = 0;
   222→   222→  let cursor = startDecimal;
   223→   223→
   224→   224→  for (const slot of sorted) {
   225→   225→    const slotStart = slot.startHour;
   226→   226→    const slotEnd = slot.endHour;
   227→   227→
   228→   228→    if (cursor >= slotEnd) continue; // cursor already past this slot
   229→   229→
   230→   230→    // Overlap start/end
   231→   231→    const overlapStart = Math.max(cursor, slotStart);
   232→   232→    const overlapEnd = Math.min(endDecimal, slotEnd);
   233→   233→
   234→   234→    if (overlapEnd > overlapStart) {
   235→   235→      const hours = overlapEnd - overlapStart;
   236→   236→      total += hours * slot.pricePerHour;
   237→   237→      cursor = overlapEnd;
   238→   238→    }
   239→   239→  }
   240→   240→
   241→   241→  // If cursor didn't reach endDecimal, the remaining time is uncovered by schedule
   242→   242→  // (fall back to 0 or caller's default)
   243→   243→  return Math.round(total * 100) / 100;
   244→   244→}

   246→   246→export async function GET(request: NextRequest) {
   247→   247→  const startTime = Date.now();
   248→   248→  const firebaseOk = isFirebaseAvailable();
   249→   249→
   250→   250→  try {
   251→   251→    const { searchParams } = new URL(request.url);
   252→   252→    const courtId = searchParams.get('courtId');
   253→   253→    const userId = searchParams.get('userId');
   254→   254→    const date = searchParams.get('date');
   255→   255→    const dateFrom = searchParams.get('dateFrom');
   256→   256→    const dateTo = searchParams.get('dateTo');
   257→   257→    const status = searchParams.get('status');
   258→   258→
   259→   259→    // Court availability check (courtId + date) is public — no auth required
   260→   260→    if (courtId && date && !userId) {
   261→   261→      if (!firebaseOk) {
   262→   262→        return NextResponse.json({ error: 'Firebase no configurado', code: 'NO_FIREBASE' }, { status: 503 });
   263→   263→      }
   264→   264→      const bookings = await getBookings({ courtId, date });
   265→   265→      return NextResponse.json(bookings.map(toCamelBooking));
   266→   266→    }
   267→   267→
   268→   268→    // All other queries require authentication
   269→   269→    const authResult = await requireAnyAuth(request);
   270→   270→    if (authResult instanceof NextResponse) return authResult;
   271→   271→    const authUser = authResult.user;
   272→   272→
   273→   273→    if (!firebaseOk) {
   274→   274→      return NextResponse.json({ error: 'Firebase no configurado', code: 'NO_FIREBASE' }, { status: 503 });
   275→   275→    }
   276→   276→
   277→   277→    let bookings: Record<string, unknown>[];
   278→   278→
   279→   279→    // Non-admin users: only their own bookings
   280→   280→    if (authUser.role !== 'admin' && authUser.role !== 'super_admin') {
   281→   281→      if (userId && userId !== authUser.id) {
   282→   282→        return NextResponse.json({ error: 'No puedes ver reservas de otros usuarios.' }, { status: 403 });
   283→   283→      }
   284→   284→      const effectiveUserId = userId || authUser.id;
   285→   285→      bookings = await searchBookingsForUser(effectiveUserId, authUser.email);
   286→   286→    } else {
   287→   287→      // Admin: fetch with optional filters (date range for performance)
   288→   288→      bookings = (await getBookings({
   289→   289→        courtId: courtId || undefined,
   290→   290→        userId: userId || undefined,
   291→   291→        date: date || undefined,
   292→   292→        dateFrom: dateFrom || undefined,
   293→   293→        dateTo: dateTo || undefined,
   294→   294→        status: status || undefined,
   295→   295→      })) as Record<string, unknown>[];
   296→   296→      console.log(`[BOOKINGS] Firestore returned ${bookings.length} raw docs (dateFrom=${dateFrom}, dateTo=${dateTo})`);
   297→   297→      if (bookings.length > 0) {
   298→   298→        console.log(`[BOOKINGS] Sample raw date field: id=${bookings[0].id}, date=${JSON.stringify(bookings[0].date)}, type=${typeof bookings[0].date}`);
   299→   299→      }
   300→   300→    }
   301→   301→
   302→   302→    // Enrich each booking with cached court/user lookups (avoid N+1 Firestore reads)
   303→   303→    const courtCache = new Map<string, Record<string, unknown> | null>();
   304→   304→    const userCache = new Map<string, Record<string, unknown> | null>();
   305→   305→    const getCourtCached = async (cid: string) => {
   306→   306→      if (courtCache.has(cid)) return courtCache.get(cid)!;
   307→   307→      try {
   308→   308→        const c = await getCourtById(cid);
   309→   309→        courtCache.set(cid, c as Record<string, unknown> | null);
   310→   310→        return c as Record<string, unknown> | null;
   311→   311→      } catch { courtCache.set(cid, null); return null; }
   312→   312→    };
   313→   313→    const getUserCached = async (uid: string) => {
   314→   314→      if (userCache.has(uid)) return userCache.get(uid)!;
   315→   315→      try {
   316→   316→        const u = await getUserById(uid);
   317→   317→        userCache.set(uid, u as Record<string, unknown> | null);
   318→   318→        return u as Record<string, unknown> | null;
   319→   319→      } catch { userCache.set(uid, null); return null; }
   320→   320→    };
   321→   321→
   322→   322→    const enriched: Record<string, unknown>[] = [];
   323→   323→    // Process in batches of 10 to avoid overwhelming Firestore
   324→   324→    const BATCH_SIZE = 10;
   325→   325→    for (let i = 0; i < bookings.length; i += BATCH_SIZE) {
   326→   326→      const batch = bookings.slice(i, i + BATCH_SIZE);
   327→   327→      const results = await Promise.allSettled(
   328→   328→        batch.map(async (b) => {
   329→   329→          const allCourtIds: string[] = Array.isArray(b.court_ids)
   330→   330→            ? b.court_ids as string[] : (b.court_id ? [b.court_id as string] : []);
   331→   331→          let court: Record<string, unknown> | null = null;
   332→   332→          let courts: Array<Record<string, unknown>> = [];
   333→   333→          let user: Record<string, unknown> | null = null;
   334→   334→          try {
   335→   335→            if (allCourtIds.length > 0) court = await getCourtCached(allCourtIds[0]);
   336→   336→          } catch { /* skip */ }
   337→   337→          if (allCourtIds.length > 1) {
   338→   338→            const enrichedCourts: Array<Record<string, unknown>> = [];
   339→   339→            for (const cid of allCourtIds) {
   340→   340→              try {
   341→   341→                const c = await getCourtCached(cid);
   342→   342→                if (c) enrichedCourts.push(c);
   343→   343→              } catch { /* skip */ }
   344→   344→            }
   345→   345→            courts = enrichedCourts;
   346→   346→          }
   347→   347→          try {
   348→   348→            if (b.user_id) user = await getUserCached(b.user_id as string);
   349→   349→          } catch { /* skip */ }
   350→   350→          return toCamelBooking({ ...b, _court: court, _courts: courts, _user: user });
   351→   351→        })
   352→   352→      );
   353→   353→      for (const r of results) {
   354→   354→        if (r.status === 'fulfilled') {
   355→   355→          enriched.push(r.value);
   356→   356→        } else {
   357→   357→          console.error('[BOOKINGS] Batch enrichment error:', r.reason);
   358→   358→        }
   359→   359→      }
   360→   360→    }
   361→   361→
   362→   362→    // Default server-side sort: today first, future ascending, past descending
   363→   363→    // Within each date: chronological time
   364→   364→    const today = new Date().toLocaleDateString('en-CA', { timeZone: 'America/Lima' });
   365→   365→    enriched.sort((a, b) => {
   366→   366→      const dA = String(a.date || ''), dB = String(b.date || '');
   367→   367→      const aIsToday = dA === today, bIsToday = dB === today;
   368→   368→      const aFuture = dA > today, bFuture = dB > today;
   369→   369→      const aPast = dA < today, bPast = dB < today;
   370→   370→      // 1) Today always first
   371→   371→      if (aIsToday && !bIsToday) return -1;
   372→   372→      if (!aIsToday && bIsToday) return 1;
   373→   373→      // 2) Both on same side of today
   374→   374→      if (aFuture && bFuture) {
   375→   375→        // Future dates: ascending (correlativo)
   376→   376→        const dc = dA.localeCompare(dB);
   377→   377→        if (dc !== 0) return dc;
   378→   378→      } else if (aPast && bPast) {
   379→   379→        // Past dates: descending (most recent past first)
   380→   380→        const dc = dB.localeCompare(dA);
   381→   381→        if (dc !== 0) return dc;
   382→   382→      }
   383→   383→      // Same date: chronological by start time
   384→   384→      return String(a.startTime || '').localeCompare(String(b.startTime || ''));
   385→   385→    });
   386→   386→
   387→   387→    console.log(`[BOOKINGS] GET: ${enriched.length} bookings returned in ${Date.now() - startTime}ms (auth=${authUser.role}, filters: courtId=${courtId||'all'}, date=${date||'all'}, status=${status||'all'})`);
   388→   388→
   389→   389→    return NextResponse.json(enriched);
   390→   390→  } catch (error) {
   391→   391→    console.error('[BOOKINGS] GET error:', error);
   392→   392→    const msg = error instanceof Error ? error.message : 'Unknown error';
   393→   393→    return NextResponse.json({ error: 'Error al obtener reservas', detail: msg }, { status: 500 });
   394→   394→  }
   395→   395→}

   397→   397→export async function POST(request: NextRequest) {
   398→   398→  try {
   399→   399→    const authResult = await requireAnyAuth(request);
   400→   400→    if (authResult instanceof NextResponse) return authResult;

   405→   405→    }
   406→   406→
   407→   407→    const body = await request.json();
   408→   408→    const {
   409→   409→      courtId,
   410→   410→      courtIds: bodyCourtIds,
   411→   411→      userId,

   414→   414→      endTime,
   415→   415→      totalPrice,
   416→   416→      advanceAmount,

   418→   418→      status,

   420→   420→      notes,
   421→   421→      equipmentItems,
   422→   422→    } = body;
   423→   423→
   424→   424→    // ── Resolve court IDs (support both single courtId and multi-court courtIds) ──
   425→   425→    const allCourtIds: string[] = Array.isArray(bodyCourtIds) && bodyCourtIds.length > 0
   426→   426→      ? bodyCourtIds
   427→   427→      : courtId
   428→   428→        ? [courtId]
   429→   429→        : [];
   430→   430→
   431→   431→    // ── Validate payment method (restricted set, UPPERCASE) ──
   432→   432→    const VALID_PAYMENT_METHODS = ['EFECTIVO', 'YAPE', 'PLIN'];
   433→   433→    const normalizedPaymentMethod = VALID_PAYMENT_METHODS.includes(paymentMethod)
   434→   434→      ? paymentMethod
   435→   435→      : 'EFECTIVO';
   436→   436→
   437→   437→    if (allCourtIds.length === 0 || !userId || !date || !startTime || !endTime) {
   438→   438→      return NextResponse.json(
   439→   439→        { error: 'Faltan campos requeridos: courtId/courtIds, userId, date, startTime, endTime' },
   440→   440→        { status: 400 }
   441→   441→      );
   442→   442→    }
   443→   443→
   444→   444→    // ── Minimum advance validation ──
   445→   445→    // Admin/super_admin: NO time restrictions — can book any slot (walk-in, past, current)
   446→   446→    // Regular users: require 30 min minimum advance
   447→   447→    const isAdmin = authUser.role === 'admin' || authUser.role === 'super_admin';
   448→   448→
   449→   449→    if (!isAdmin) {
   450→   450→      const now = new Date();
   451→   451→      const slotDateTime = new Date(`${date}T${startTime}:00`);
   452→   452→      const diffMs = slotDateTime.getTime() - now.getTime();
   453→   453→      const thirtyMinMs = 30 * 60 * 1000;
   454→   454→      if (diffMs < thirtyMinMs) {
   455→   455→        const minsLeft = Math.floor(diffMs / 60000);
   456→   456→        if (minsLeft < 0) {
   457→   457→          return NextResponse.json(
   458→   458→            { error: 'No se puede reservar un horario que ya ha pasado.' },
   459→   459→            { status: 422 }
   460→   460→          );
   461→   461→        }
   462→   462→        return NextResponse.json(
   463→   463→          { error: `La reserva requiere al menos 30 minutos de anticipación. El horario seleccionado comienza en ${minsLeft} minuto${minsLeft === 1 ? '' : 's'}. Selecciona un horario a partir de las ${new Date(now.getTime() + thirtyMinMs).toLocaleTimeString('es-PE', { hour: '2-digit', minute: '2-digit', hour12: false })} o posterior.` },
   464→   464→          { status: 422 }
   465→   465→        );
   466→   466→      }
   467→   467→    }
   468→   468→
   469→   469→    // Non-admin users can only create bookings for themselves
   470→   470→    if (!isAdmin) {
   471→   471→      if (userId !== authUser.id) {
   472→   472→        return NextResponse.json(
   473→   473→          { error: 'No puedes crear reservas para otros usuarios.' },
   474→   474→          { status: 403 }
   475→   475→        );
   476→   476→      }
   477→   477→    }
   478→   478→
   479→   479→    // ── Check overlap for ALL courts ──
   480→   480→    const conflicts: Array<{ courtId: string; courtName: string }> = [];
   481→   481→    for (const cId of allCourtIds) {
   482→   482→      const existing = await getBookings({ courtId: cId, date });
   483→   483→      const overlapping = existing.filter(
   484→   484→        (b) => {
   485→   485→          const bCourtIds: string[] = Array.isArray(b.court_ids)
   486→   486→            ? b.court_ids as string[]
   487→   487→            : [b.court_id as string];
   488→   488→          // Skip if this existing booking doesn't include the court we're checking
   489→   489→          if (!bCourtIds.includes(cId)) return false;
   490→   490→          return (
   491→   491→            !['cancelled'].includes(migrateStatus(b.status || '')) &&
   492→   492→            (b.start_time || '') < endTime &&
   493→   493→            (b.end_time || '') > startTime
   494→   494→          );
   495→   495→        }
   496→   496→      );
   497→   497→      if (overlapping.length > 0) {
   498→   498→        let courtName = cId;
   499→   499→        try {
   500→   500→          const c = await getCourtById(cId);
   501→   501→          if (c?.name) courtName = c.name as string;
   502→   502→        } catch { /* use ID */ }
   503→   503→        conflicts.push({ courtId: cId, courtName });
   504→   504→      }
   505→   505→    }
   506→   506→
   507→   507→    if (conflicts.length > 0) {
   508→   508→      const conflictNames = conflicts.map((c) => c.courtName).join(', ');
   509→   509→      return NextResponse.json(
   510→   510→        {
   511→   511→          error: `Cancha${conflicts.length > 1 ? 's' : ''} no disponible${conflicts.length > 1 ? 's' : ''}: ${conflictNames}. Por favor selecciona otro horario o cancha.`,
   512→   512→          conflicts,
   513→   513→        },
   514→   514→        { status: 409 }
   515→   515→      );
   516→   516→    }
   517→   517→
   518→   518→    // B15 FIX: Re-check conflicts right before creation (optimistic concurrency)
   519→   519→    for (const cId of allCourtIds) {
   520→   520→      const recheck = await getBookings({ courtId: cId, date });
   521→   521→      const stillConflicts = recheck.filter(
   522→   522→        (b) => {
   523→   523→          const bCourtIds: string[] = Array.isArray(b.court_ids)
   524→   524→            ? b.court_ids as string[]
   525→   525→            : [b.court_id as string];
   526→   526→          if (!bCourtIds.includes(cId)) return false;
   527→   527→          return (
   528→   528→            !['cancelled'].includes(migrateStatus(b.status || '')) &&
   529→   529→            (b.start_time || '') < endTime &&
   530→   530→            (b.end_time || '') > startTime
   531→   531→          );
   532→   532→        }
   533→   533→      );
   534→   534→      if (stillConflicts.length > 0) {
   535→   535→        let courtName = cId;
   536→   536→        try {
   537→   537→          const c = await getCourtById(cId);
   538→   538→          if (c?.name) courtName = c.name as string;
   539→   539→        } catch { /* use ID */ }
   540→   540→        return NextResponse.json(
   541→   541→          { error: `Horario ocupado en ${courtName}. Alguien reservó mientras completabas el formulario. Intenta de nuevo.`, conflicts: [{ courtId: cId, courtName }] },
   542→   542→          { status: 409 }
   543→   543→        );
   544→   544→      }
   545→   545→    }
   546→   546→
   547→   547→    // ── Calculate price ──
   548→   548→    // Frontend sends totalPrice as grand total (court + equipment combined).
   549→   549→    // We compute courtSubtotal and equipmentSubtotal separately for storage/display.
   550→   550→    const eqItems = Array.isArray(equipmentItems) ? equipmentItems : [];
   551→   551→    let equipmentSubtotal = 0;
   552→   552→    for (const eq of eqItems) {
   553→   553→      equipmentSubtotal += (eq.quantity || 0) * (eq.unit_price || eq.unitPrice || 0);
   554→   554→    }
   555→   555→
   556→   556→    let courtPriceTotal: number;
   557→   557→    const providedTotal = parseFloat(totalPrice) || 0;
   558→   558→    if (providedTotal > 0) {
   559→   559→      // Frontend already included equipment in totalPrice — extract court portion
   560→   560→      courtPriceTotal = Math.max(0, providedTotal - equipmentSubtotal);
   561→   561→    } else {
   562→   562→      // No price from frontend — calculate from court rates
   563→   563→      courtPriceTotal = 0;
   564→   564→      for (const cId of allCourtIds) {
   565→   565→        const court = await getCourtById(cId);
   566→   566→        if (court) {
   567→   567→          const schedule = court.pricing_schedule as Array<{ label: string; startHour: number; endHour: number; pricePerHour: number }> | undefined;
   568→   568→          let courtPrice = 0;
   569→   569→          if (schedule && schedule.length > 0) {
   570→   570→            courtPrice = calculatePriceForTimeSlot(schedule, startTime, endTime);
   571→   571→            if (courtPrice <= 0) courtPrice = (court.price_per_hour as number) || 0;
   572→   572→          } else {
   573→   573→            courtPrice = (court.price_per_hour as number) || 0;
   574→   574→          }
   575→   575→          courtPriceTotal += courtPrice;
   576→   576→        }
   577→   577→      }
   578→   578→    }
   579→   579→
   580→   580→    const price = courtPriceTotal + equipmentSubtotal;
   581→   581→
   582→   582→    // B4 FIX: Validate advance + remaining === total consistency
   583→   583→    const adv = parseFloat(advanceAmount) || price * 0.5;
   584→   584→    let rem = parseFloat(remainingAmount) || price - adv;
   585→   585→    // Force consistency: remaining must equal total - advance
   586→   586→    rem = Math.max(0, Math.round((price - adv) * 100) / 100);
   587→   587→    const bookingStatus = migrateStatus(status || 'reserved');
   588→   588→
   589→   589→    // Resolve client email for denormalized search
   590→   590→    let clientEmail = authUser.email;
   591→   591→    try {
   592→   592→      const clientUser = await getUserById(userId);
   593→   593→      if (clientUser?.email) clientEmail = clientUser.email;
   594→   594→    } catch { /* fallback to admin email */ }
   595→   595→
   596→   596→    // Save to Firestore — single booking with all courts
   597→   597→    const id = await createBooking({
   598→   598→      court_ids: allCourtIds,
   599→   599→      user_id: userId,
   600→   600→      user_email: clientEmail,
   601→   601→      date,
   602→   602→      start_time: startTime,
   603→   603→      end_time: endTime,
   604→   604→      total_price: price,
   605→   605→      court_subtotal: courtPriceTotal,
   606→   606→      equipment_subtotal: equipmentSubtotal,
   607→   607→      equipment_items: eqItems,
   608→   608→      advance_amount: adv,
   609→   609→      remaining_amount: rem,
   610→   610→      status: bookingStatus,
   611→   611→      slot_status: 'available',
   612→   612→      payment_method: normalizedPaymentMethod || null,
   613→   613→      notes: notes || null,
   614→   614→    });
   615→   615→
   616→   616→    console.log(`[BOOKINGS] POST: Created booking ${id} for user=${userId} date=${date} ${startTime}-${endTime} courts=${allCourtIds.join(',')} total=${price} advance=${adv} remaining=${rem} method=${normalizedPaymentMethod} email=${clientEmail}`);
   617→   617→
   618→   618→    // B5 FIX: Only create payment record if advance > 0
   619→   619→    if (adv > 0.01) {
   620→   620→      try {
   621→   621→        await createPayment(id, {
   622→   622→          user_id: userId,
   623→   623→          amount: adv,
   624→   624→          type: 'advance',
   625→   625→          method: normalizedPaymentMethod || 'EFECTIVO',
   626→   626→          status: 'completed',
   627→   627→        });
   628→   628→      } catch (payErr) {
   629→   629→        console.error('[BOOKINGS] Warning: could not create payment record:', payErr);
   630→   630→      }
   631→   631→    }
   632→   632→
   633→   633→    return NextResponse.json({
   634→   634→      id,
   635→   635→      courtId: allCourtIds[0],
   636→   636→      courtIds: allCourtIds,
   637→   637→      userId,
   638→   638→      date,
   639→   639→      startTime,
   640→   640→      endTime,
   641→   641→      totalPrice: price,
   642→   642→      advanceAmount: adv,
   643→   643→      remainingAmount: rem,
   644→   644→      status: bookingStatus,
   645→   645→      paymentMethod: paymentMethod || null,
   646→   646→      success: true,
   647→   647→    }, { status: 201 });
   648→   648→  } catch (error: unknown) {
   649→   649→    const err = error as { message?: string; code?: string; stack?: string };
   650→   650→    console.error('[BOOKINGS] POST error:', err?.message || err);
   651→   651→    console.error('[BOOKINGS] Stack:', err?.stack || 'no stack');
   652→   652→    return NextResponse.json({
   653→   653→      error: 'Failed to create booking',
   654→   654→      detail: err?.message || 'Unknown error'
   655→   655→    }, { status: 500 });
   656→   656→  }
   657→   657→}

   659→   659→export async function PUT(request: NextRequest) {
   660→   660→  try {
   661→   661→    const authResult = await requireAnyAuth(request);
   662→   662→    if (authResult instanceof NextResponse) return authResult;

   664→   664→
   665→   665→    if (!isFirebaseAvailable()) {
   666→   666→      return NextResponse.json({ error: 'Firebase no configurado' }, { status: 503 });
   667→   667→    }

   685→   685→            { status: 403 }
   686→   686→          );
   687→   687→        }
   688→   688→      } else {
   689→   689→        return NextResponse.json(
   690→   690→          { error: 'Solo puedes cancelar tus propias reservas.' },

   701→   701→    if (reqPaymentMethod) {
   702→   702→      const VALID_PM = ['EFECTIVO', 'YAPE', 'PLIN'];

   706→   706→    if (typeof equipmentReturned === 'boolean') updateData.equipment_returned = equipmentReturned;
   707→   707→

   710→   710→      // Validate: new end must be after current end
   711→   711→      const booking = await getBookingById(id);
   712→   712→      if (booking) {

   714→   714→        if (reqEndTime <= currentEnd) {

   716→   716→        }

   719→   719→        const bookingDate = booking.date as string;

   721→   721→          // B10 FIX: Filter by date and courts instead of fetching ALL bookings

   723→   723→          const conflictResults = await Promise.all(conflictQueries);
   724→   724→          const allConflicting = conflictResults.flat();
   725→   725→          const conflicts = allConflicting.filter((ob: Record<string, unknown>) => {
   726→   726→            const obId = ob.id as string;
   727→   727→            if (obId === id) return false;
   728→   728→            // B9 FIX: Use migrateStatus for legacy status values
   729→   729→            if (migrateStatus(ob.status || '') === 'cancelled') return false;

   731→   731→            const obCourtIds: string[] = Array.isArray(ob.court_ids) ? ob.court_ids : (ob.court_id ? [ob.court_id] : []);
   732→   732→            if (!obCourtIds.some((cid: string) => allCourtIds.includes(cid))) return false;
   733→   733→            return (ob.start_time as string) < reqEndTime && (ob.end_time as string) > (booking.start_time as string);
   734→   734→          });
   735→   735→          // Filter to only real conflicts (overlapping with the NEW extended period)
   736→   736→          const realConflicts = conflicts.filter((ob: Record<string, unknown>) => {
   737→   737→            return (ob.start_time as string) < reqEndTime && (ob.end_time as string) > currentEnd;
   738→   738→          });
   739→   739→          if (realConflicts.length > 0) {

   741→   741→          }
   742→   742→        }
   743→   743→
   744→   744→        const extraCost = Math.max(0, (reqTotalPrice - ((booking.total_price as number) || 0)));
   745→   745→        updateData.end_time = reqEndTime;
   746→   746→        updateData.total_price = reqTotalPrice;
   747→   747→
   748→   748→        // If admin collected payment for the extension, create a payment record
   749→   749→        if (typeof reqAdvance === 'number' && extraCost > 0.01) {

   751→   751→          const paymentIncrement = reqAdvance - prevAdvance;
   752→   752→          if (paymentIncrement > 0.01) {
   753→   753→            try {
   754→   754→              await createPayment(id, {
   755→   755→                user_id: booking.user_id as string,
   756→   756→                amount: paymentIncrement,
   757→   757→                type: 'extension',
   758→   758→                method: (reqPaymentMethod as string) || 'EFECTIVO',
   759→   759→                status: 'completed',
   760→   760→              });
   761→   761→            } catch (payErr) {
   762→   762→              console.error('[BOOKINGS] Warning: could not create payment record for extension:', payErr);
   763→   763→            }
   764→   764→          }
   765→   765→        }
   766→   766→      }
   767→   767→    }
   768→   768→
   769→   769→    // Edit booking (super_admin only): change any field
   770→   770→    if (editBooking) {
   771→   771→      if (authUser.role !== 'super_admin') {
   772→   772→        return NextResponse.json({ error: 'Solo el super administrador puede editar reservas.' }, { status: 403 });
   773→   773→      }
   774→   774→
   775→   775→      const booking = await getBookingById(id);
   776→   776→      if (!booking) {
   777→   777→        return NextResponse.json({ error: 'Reserva no encontrada.' }, { status: 404 });
   778→   778→      }
   779→   779→
   780→   780→      // Don't allow editing completed or cancelled bookings (unless only changing notes/status)
   781→   781→      const bStatus = migrateStatus(booking.status || 'reserved');
   782→   782→      const hasTimeOrCourtChange = reqStartTime || reqEndTime || reqDate || (reqCourtIds && reqCourtIds.length > 0);
   783→   783→      if ((bStatus === 'completed' || bStatus === 'cancelled') && hasTimeOrCourtChange) {
   784→   784→        return NextResponse.json({ error: 'No se puede editar el horario/cancha de una reserva completada o cancelada.' }, { status: 400 });
   785→   785→      }
   786→   786→
   787→   787→      // Validate time if provided
   788→   788→      if (reqStartTime && reqEndTime) {
   789→   789→        if (reqStartTime >= reqEndTime) {
   790→   790→          return NextResponse.json({ error: 'La hora de inicio debe ser anterior a la hora de fin.' }, { status: 400 });
   791→   791→        }
   792→   792→      }
   793→   793→
   794→   794→      // Date & time
   795→   795→      if (reqDate) updateData.date = reqDate;
   796→   796→      if (reqStartTime) updateData.start_time = reqStartTime;
   797→   797→      if (reqEndTime) updateData.end_time = reqEndTime;
   798→   798→
   799→   799→      // Court change
   800→   800→      if (reqCourtIds && Array.isArray(reqCourtIds) && reqCourtIds.length > 0) {
   801→   801→        updateData.court_ids = reqCourtIds;
   802→   802→        updateData.court_id = reqCourtIds[0]; // backward compat
   803→   803→      }
   804→   804→
   805→   805→      // Client change
   806→   806→      if (reqUserId) {
   807→   807→        updateData.user_id = reqUserId;
   808→   808→      }
   809→   809→
   810→   810→      // Notes
   811→   811→      if (reqNotes !== undefined) {
   812→   812→        updateData.notes = reqNotes;
   813→   813→      }
   814→   814→
   815→   815→      // Equipment items
   816→   816→      if (reqEquipmentItems && Array.isArray(reqEquipmentItems)) {
   817→   817→        const equipSub = reqEquipmentItems.reduce((s: number, eq: Record<string, unknown>) => s + ((eq.quantity as number) || 0) * ((eq.unit_price as number) || 0), 0);
   818→   818→        updateData.equipment_items = reqEquipmentItems;
   819→   819→        updateData.equipment_subtotal = Math.round(equipSub * 100) / 100;
   820→   820→      }
   821→   821→
   822→   822→      // Price fields
   823→   823→      if (typeof reqTotalPrice === 'number' && reqTotalPrice > 0) {
   824→   824→        updateData.total_price = reqTotalPrice;
   825→   825→        const currentAdvance = typeof reqAdvance === 'number' ? reqAdvance : (booking.advance_amount as number) || 0;
   826→   826→        updateData.advance_amount = Math.min(currentAdvance, reqTotalPrice);
   827→   827→        updateData.remaining_amount = Math.max(0, Math.round((reqTotalPrice - currentAdvance) * 100) / 100);
   828→   828→        // Recalc court_subtotal
   829→   829→        const equipSub = (updateData.equipment_subtotal as number) || (booking.equipment_subtotal as number) || 0;
   830→   830→        updateData.court_subtotal = Math.max(0, Math.round((reqTotalPrice - equipSub) * 100) / 100);
   831→   831→      } else if (typeof reqAdvance === 'number') {
   832→   832→        // Only advance changed, not total
   833→   833→        const currentTotal = (booking.total_price as number) || 0;
   834→   834→        updateData.advance_amount = reqAdvance;
   835→   835→        updateData.remaining_amount = Math.max(0, Math.round((currentTotal - reqAdvance) * 100) / 100);
   836→   836→      }
   837→   837→
   838→   838→      // Payment method
   839→   839→      if (reqPaymentMethod) {
   840→   840→        const VALID_PM = ['EFECTIVO', 'YAPE', 'PLIN'];
   841→   841→        updateData.payment_method = VALID_PM.includes(reqPaymentMethod) ? reqPaymentMethod : 'EFECTIVO';
   842→   842→      }
   843→   843→
   844→   844→      // Conflict check (only if time or court or date changed)
   845→   845→      const finalCourtIds: string[] = (updateData.court_ids as string[]) || (Array.isArray(booking.court_ids) ? booking.court_ids : (booking.court_id ? [booking.court_id] : []));
   846→   846→      const finalDate = (updateData.date as string) || (booking.date as string);
   847→   847→      const finalStart = (updateData.start_time as string) || (booking.start_time as string);
   848→   848→      const finalEnd = (updateData.end_time as string) || (booking.end_time as string);
   849→   849→
   850→   850→      if (finalCourtIds.length > 0 && finalDate && finalStart && finalEnd) {
   851→   851→        const conflictQueries = finalCourtIds.map(cid => getBookings({ courtId: cid, date: finalDate }));
   852→   852→        const conflictResults = await Promise.all(conflictQueries);
   853→   853→        const allConflicting = conflictResults.flat();
   854→   854→        const conflicts = allConflicting.filter((ob: Record<string, unknown>) => {
   855→   855→          const obId = ob.id as string;
   856→   856→          if (obId === id) return false;
   857→   857→          if (migrateStatus(ob.status || '') === 'cancelled') return false;
   858→   858→          const obCourtIds: string[] = Array.isArray(ob.court_ids) ? ob.court_ids : (ob.court_id ? [ob.court_id] : []);
   859→   859→          if (!obCourtIds.some((cid: string) => finalCourtIds.includes(cid))) return false;
   860→   860→          return (ob.start_time as string) < finalEnd && (ob.end_time as string) > finalStart;
   861→   861→        });
   862→   862→        if (conflicts.length > 0) {
   863→   863→          const c = conflicts[0];
   864→   864→          return NextResponse.json({
   865→   865→            error: 'Hay una reserva que ocupa ese horario.',
   866→   866→            detail: `Conflicto con reserva: ${(c.start_time as string)} - ${(c.end_time as string)}`,
   867→   867→          }, { status: 409 });
   868→   868→        }
   869→   869→      }
   870→   870→
   871→   871→      // Skip the generic advance payment record creation below — we handle it here
   872→   872→      // (prevent double payment records)
   873→   873→    }
   874→   874→
   875→   875→    // B11 FIX: Track if this is a cancellation to skip advance payment record
   876→   876→    const isCancelling = status === 'cancelled' || (status && migrateStatus(status) === 'cancelled');
   877→   877→
   878→   878→    // When cancelling a booking with advance, handle retained/refunded advance
   879→   879→    // Only create retained advance records when an admin explicitly provides advanceAction
   880→   880→    // (user-side cancellations should NOT auto-create records — the admin decides later)
   881→   881→    const isAdminCancellation = authUser.role === 'admin' || authUser.role === 'super_admin';
   882→   882→    if (isCancelling) {
   883→   883→      // B6 FIX: Fetch booking once, reuse for both checks
   884→   884→      let bookingForCancel: Record<string, unknown> | null = null;
   885→   885→      try {
   886→   886→        bookingForCancel = await getBookingById(id);
   887→   887→      } catch { /* non-blocking */ }
   888→   888→
   889→   889→      // Prevent cancelling completed bookings
   890→   890→      if (bookingForCancel && (bookingForCancel.status === 'completed' || migrateStatus(bookingForCancel.status as string || '') === 'completed')) {
   891→   891→        return NextResponse.json({ error: 'No se puede cancelar una reserva completada. Contacta soporte si es necesario.' }, { status: 400 });
   892→   892→      }
   893→   893→
   894→   894→      if (isAdminCancellation) {
   895→   895→      try {
   896→   896→        if (bookingForCancel && ((bookingForCancel.advance_amount as number) || 0) > 0) {
   897→   897→          // Bug #4: Check for duplicate retained advance
   898→   898→          const { getRetainedAdvances } = await import('@/lib/db');
   899→   899→          const existing = await getRetainedAdvances({ bookingId: id });
   900→   900→          if (existing.length > 0) {
   901→   901→            console.warn('[BOOKINGS] Retained advance already exists for booking', id, '- skipping creation');
   902→   902→          } else {
   903→   903→          const advAmount = (bookingForCancel.advance_amount as number) || 0;
   904→   904→          const isRetained = advanceAction === 'retain';
   905→   905→          const statusAdvance = isRetained ? 'retained' : 'refunded';
   906→   906→
   907→   907→          // Get court name for the record
   908→   908→          let courtName = '';
   909→   909→          try {
   910→   910→            const cId = (bookingForCancel.court_ids as string[])?.[0] || (bookingForCancel.court_id as string) || '';
   911→   911→            if (cId) {
   912→   912→              const court = await getCourtById(cId);
   913→   913→              courtName = (court?.name as string) || '';
   914→   914→            }
   915→   915→          } catch { /* non-critical */ }
   916→   916→
   917→   917→          // Get user name
   918→   918→          let userName = '';
   919→   919→          try {
   920→   920→            const uId = bookingForCancel.user_id as string;
   921→   921→            if (uId) {
   922→   922→              const user = await getUserById(uId);
   923→   923→              userName = (user?.name as string) || '';
   924→   924→            }
   925→   925→          } catch { /* non-critical */ }
   926→   926→
   927→   927→          await createRetainedAdvance({
   928→   928→            booking_id: id,
   929→   929→            user_id: (bookingForCancel.user_id as string) || '',
   930→   930→            user_name: userName,
   931→   931→            user_email: (bookingForCancel.user_email as string) || null,
   932→   932→            court_name: courtName,
   933→   933→            booking_date: (bookingForCancel.date as string) || '',
   934→   934→            amount: advAmount,
   935→   935→            original_total: (bookingForCancel.total_price as number) || 0,
   936→   936→            payment_method: (bookingForCancel.payment_method as string) || 'EFECTIVO',
   937→   937→            reason: cancelReason || 'Cancelación de reserva',
   938→   938→            status: statusAdvance,
   939→   939→          });
   940→   940→
   941→   941→          // Create a payment record for the refund/retention
   942→   942→          try {
   943→   943→            await createPayment(id, {
   944→   944→              user_id: (bookingForCancel.user_id as string) || '',
   945→   945→              amount: advAmount,
   946→   946→              type: isRetained ? 'retained' : 'refund',
   947→   947→              method: (bookingForCancel.payment_method as string) || 'EFECTIVO',
   948→   948→              status: 'completed',
   949→   949→            });
   950→   950→          } catch (payErr) {

   952→   952→          }
   953→   953→          }
   954→   954→        }
   955→   955→      } catch (err) {
   956→   956→        console.error('[BOOKINGS] Warning: could not process retained advance on cancellation:', err);
   957→   957→      }
   958→   958→      } // end isAdminCancellation
   959→   959→    }
   960→   960→
   961→   961→    // B11 FIX: Skip advance payment record when cancelling or editBooking (handled above / no record needed)
   962→   962→    if (typeof reqAdvance === 'number' && typeof reqRemaining === 'number' && !isCancelling && !editBooking) {
   963→   963→      try {
   964→   964→        const booking = await getBookingById(id);
   965→   965→        if (booking) {
   966→   966→          const prevAdvance = (booking.advance_amount as number) || 0;
   967→   967→          const paymentIncrement = reqAdvance - prevAdvance;
   968→   968→          if (paymentIncrement > 0.01) {
   969→   969→            try {
   970→   970→              await createPayment(id, {
   971→   971→                user_id: booking.user_id as string,
   972→   972→                amount: paymentIncrement,
   973→   973→                type: reqRemaining <= 0 ? 'full_payment' : 'advance',
   974→   974→                method: (updateData.payment_method as string) || 'EFECTIVO',
   975→   975→                status: 'completed',
   976→   976→              });
   977→   977→            } catch (payErr) {
   978→   978→              console.error('[BOOKINGS] Warning: could not create payment record for advance update:', payErr);
   979→   979→            }
   980→   980→          }
   981→   981→        }
   982→   982→      } catch { /* non-critical: proceed with booking update */ }
   983→   983→    }
   984→   984→
   985→   985→    await updateBooking(id, updateData);
   986→   986→
   987→   987→    return NextResponse.json({ success: true });
   988→   988→  } catch (error) {
   989→   989→    console.error('[BOOKINGS] PUT error:', error);
   990→   990→    const msg = error instanceof Error ? error.message : 'Unknown error';
   991→   991→    return NextResponse.json({ error: 'Error al actualizar reserva', detail: msg }, { status: 500 });
   992→   992→  }
