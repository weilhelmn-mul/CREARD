    76→/* ─── helpers ─── */
    77→// Parse date string as local date (avoids UTC timezone issues for Peru, UTC-5)
    78→function parseLocalDate(dateStr: string): Date {
    79→  const parts = dateStr.split('-')
    80→  if (parts.length === 3) {
    81→    return new Date(parseInt(parts[0], 10), parseInt(parts[1], 10) - 1, parseInt(parts[2], 10))
    82→  }
    83→  return new Date(dateStr + 'T00:00:00')
    84→}
    85→
    86→const fmt = (dateStr: string) => {
    87→  const d = parseLocalDate(dateStr)
    88→  return d.toLocaleDateString('es-PE', { day: 'numeric', month: 'short', year: 'numeric' })
    89→}

    91→const fmtCurrency = (n: number) => `S/ ${n.toFixed(2)}`

   337→                                  {sportLabels[booking.court.sport] || booking.court.sport}
